let run = 0, passed = 0;
const failures = [];
function test(name, fn) {
  run++;
  try { fn(); passed++; console.log('ok   ' + name); }
  catch (e) { failures.push(name); console.log('FAIL ' + name + '\n     ' + (e && e.message)); }

function assert(cond, msg) { if (!cond) throw new Error(msg || 'assertion failed'); }
function near(a, b, dp, msg) {
  const tol = 0.5 * Math.pow(10, -dp);
  if (!(Math.abs(a - b) <= tol)) throw new Error((msg || 'values differ') + ': got ' + a + ', expected ' + b);
}
function withP(over) { return Object.assign({}, best, over); }
function sum(o) { return Object.values(o).reduce((a, b) => a + b, 0); }
const NEW_IDS = ['s_exact', 's_similar', 's_comp', 's_part', 'ok_coverage_factor', 'fix_on_found'];

}
function randomStocks(random) { return { E_good: random() * 2000, E_bad: random() * 800 }; }

function funnelStages(row, fid) {

// ----------------------------------------------------------------- expression whitelist

test('expressions are evaluated against a whitelist only', () => {
  const ok = E.compileExpr('1 - exp(-a / b) + safediv(a, 0)', new Set(['a', 'b']));
  near(ok.fn({ a: 2, b: 2 }), 1 - Math.exp(-1), 9, 'parser arithmetic');
  const bad = ['constructor', 'a.b', 'a[0]', '"x"', 'process', 'alert(1)', 'a; b', 'a ** b', 'this'];
  bad.forEach(src => {
    let threw = false;
    try { E.compileExpr(src, new Set(['a', 'b'])); } catch (e) { threw = true; }
    assert(threw, 'accepted: ' + src);
  });
  near(E.compileExpr('-2 * 3 + 4 / (1 + 1)', new Set()).fn({}), -4, 12, 'precedence');
});

// ----------------------------------------------------------------- conservation under 1,000 draws

test('conservation over 1,000 draws (untied and tied): branch chances out of every state lie in 0 to 1 and sum to 1, outcomes sum to 1 for every class and policy, class chances sum to 1; stocks never go negative; funnel stages never rise', () => {
  const random = E.rng(20261112);
  const shared = E.rng(77);
  let worst = 0, checks = 0, pathsChecked = 0;
  const wide = JSON.parse(JSON.stringify(table));
  // last part: every chance and fraction drawn across the whole of 0 to 1 (retire_bad included)
  map.parameters.forEach(p => { if (p.kind === 'chance' || p.kind === 'fraction') { wide.parameters[p.id].low = 0; wide.parameters[p.id].high = 1; } });
  for (let i = 0; i < 1000; i++) {
    const tb = i < 350 ? table : i < 700 ? real : wide;
    const tie = [0, 0.5, 1][i % 3];
    const p = E.drawParams(map, tb, random, { tie, level: shared() });
    const stocks = randomStocks(random);
    if (i % 50 === 0) { stocks.E_good = 0; stocks.E_bad = 0; }
    if (i % 50 === 25) { stocks.E_good = 0; }
    const r = E.conservation(map, p, { stocks });
    assert(r.ok, 'draw ' + i + ': ' + r.problems.slice(0, 3).join('; '));
    worst = Math.max(worst, r.maxError);
    checks += r.checks;
    if (i % 10 === 0) {
      const tp = E.timePath(map, p, { lean: true });
      tp.periods.forEach(row => {
        assert(row.stocks.E_good >= -1e-9 && row.stocks.E_bad >= -1e-9, 'draw ' + i + ' period ' + row.period + ': a stock went negative');
        map.funnels.forEach(f => {
          const v = f.monotone || f.stages;
          for (let k = 1; k < v.length; k++) assert(row.library.measures[v[k]] <= row.library.measures[v[k - 1]] + 1e-9, 'draw ' + i + ': funnel ' + f.id + ' rose at ' + v[k]);
        });
      });
      assert(tp.finalStocks.E_good >= -1e-9 && tp.finalStocks.E_bad >= -1e-9, 'draw ' + i + ': final stock negative');
      pathsChecked++;
    }
  }
  console.log('     ' + checks + ' sums checked over 1000 draws, largest error ' + worst.toExponential(2) + '; ' + pathsChecked + ' time paths checked for negative stocks and rising funnels');
});

// Case 4 (revision 3, R11 and R12). Stocks start at E_good 30 and E_bad 10 (a copy of the map with
// those starting stocks). k_fill 40, delta_stale 0.1, retire_bad 0.2, w_write 0 (nothing is written).
//   fill = 1 - exp(-(30 + 10) / 40) = 1 - exp(-1) = 0.632120558828558. (Revision 2: 1 - exp(-0.75).)
//   wrong_share = 10 / 40 = 0.25.
//   Next good = 30 - 0.1 * 30 = 27. Next bad = 10 - 0.2 * 10 + 0.1 * 30 = 11.
//   One period later: good 27 - 2.7 = 24.3; bad 11 - 2.2 + 2.7 = 11.5.
//   With retire_bad 1 instead: next bad = 10 - 10 + 3 = 3; then 3 - 3 + 2.7 = 2.7.
test('hand case 4: fill counts both stocks, and a share retire_bad of the bad stock leaves each period', () => {
  const m = withStocks(map, { E_good: 30, E_bad: 10 });
  const p = withP({ k_fill: 40, delta_stale: 0.1, retire_bad: 0.2, w_write: 0, periods: 3 });
  const tp = E.timePath(m, p);
  near(tp.periods[0].stockValues.fill, 1 - Math.exp(-1), 12, 'fill');
  near(tp.periods[0].stockValues.wrong_share, 0.25, 12, 'bad share');
  near(tp.periods[0].next.E_good, 27, 12, 'good after one period');
  near(tp.periods[0].next.E_bad, 11, 12, 'bad after one period');
  near(tp.periods[1].next.E_good, 24.3, 12, 'good after two periods');
  near(tp.periods[1].next.E_bad, 11.5, 12, 'bad after two periods');
  const tp1 = E.timePath(m, Object.assign({}, p, { retire_bad: 1 }));
  near(tp1.periods[0].next.E_bad, 3, 12, 'all bad retired, then the new stale ones');
  near(tp1.periods[1].next.E_bad, 2.7, 12, 'all bad retired again');
});

test('R4: repair moves found strong-tier escapes from bad to good at fix_on_found; the stock equations match the brief\'s formulas (same_as) and the written-branch flows', () => {

    movedSoFar = s.moved_so_far;
  });
  function moveFav(ids) {
    const p = Object.assign({}, realBest);
    ids.forEach(id => { const row = E.tableRows(realTable)[id]; p[id] = row.good_end === 'high' ? row.high : row.low; });
    return p;
  }
  // the last cumulative change equals moving every library-side unknown named by any limitation
  const all = new Set();
  lims.items.forEach(l => l.params.forEach(id => { if (realRows[id].side === 'library') all.add(id); }));
  const last = W.sequence[W.sequence.length - 1];
  assert(last.moved_so_far.length === all.size, 'not every unknown moved at the end');
  const t = E.timePath(map, moveFav([...all])).last;
  near(last.values.ratios.cost_vs_ladder, t.cost_vs_ladder, 12, 'all steps together');
});

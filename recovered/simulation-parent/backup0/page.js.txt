  function fmt(v, kind) {
    if (typeof v === 'string') return v;
    if (v === null || v === undefined || !isFinite(v)) return 'none';
    var a = Math.abs(v);
    if (kind === 'pct') { var x = v * 100; return (x !== 0 && Math.abs(x) < 0.1 ? x.toFixed(2) : x.toFixed(1)) + '%'; }
    if (kind === 'ratio') return a >= 10 ? v.toFixed(1) : v.toFixed(2);
    if (kind === 'cost') return a >= 100 ? v.toFixed(0) : v.toFixed(1);
    if (kind === 'int') return Math.round(v).toLocaleString('en');
    if (kind === 'rho') return (v > 0 ? '+' : v < 0 ? '−' : '') + Math.abs(v).toFixed(2);
    if (v === 0) return '0';
    if (a >= 100) return v.toFixed(0);
    if (a >= 10) return v.toFixed(1);
    if (a >= 1) return v.toFixed(2);
    return v.toFixed(3).replace(/0+$/, '').replace(/\.$/, '.0');
  }

  function geom(mini) {

      : { cw: LY.col_w, rh: LY.row_h, x0: LY.x0, y0: LY.y0, nw: LY.node_w, nh: LY.node_h, head: [17, 29, 43] };

  function pos(g, sid) { var p = LY.states[sid]; return { c: p[0], r: p[1], x: g.x0 + p[0] * g.cw, y: g.y0 + p[1] * g.rh }; }
  function roundPath(pts, rad) {
    var clean = [pts[0]];
    for (var i = 1; i < pts.length; i++) { var q = clean[clean.length - 1]; if (Math.abs(pts[i][0] - q[0]) > 0.01 || Math.abs(pts[i][1] - q[1]) > 0.01) clean.push(pts[i]); }
    var d = 'M' + clean[0][0].toFixed(1) + ' ' + clean[0][1].toFixed(1);
    for (var k = 1; k < clean.length - 1; k++) {
      var p0 = clean[k - 1], p1 = clean[k], p2 = clean[k + 1];
      var d1x = p1[0] - p0[0], d1y = p1[1] - p0[1], d2x = p2[0] - p1[0], d2y = p2[1] - p1[1];
      var l1 = Math.hypot(d1x, d1y), l2 = Math.hypot(d2x, d2y);
      var r = Math.min(rad, l1 / 2, l2 / 2);
      var a = [p1[0] - d1x / l1 * r, p1[1] - d1y / l1 * r], b = [p1[0] + d2x / l2 * r, p1[1] + d2y / l2 * r];
      d += 'L' + a[0].toFixed(1) + ' ' + a[1].toFixed(1) + 'Q' + p1[0].toFixed(1) + ' ' + p1[1].toFixed(1) + ' ' + b[0].toFixed(1) + ' ' + b[1].toFixed(1);
    }
    var z = clean[clean.length - 1];
    return d + 'L' + z[0].toFixed(1) + ' ' + z[1].toFixed(1);
  }

  function edgePath(g, b, mini) {

    var x1 = a.x + g.nw / 2, y1 = a.y + o, x2 = z.x - g.nw / 2 - 1, y2 = z.y + o;
    if (z.c - a.c === 1) { var mx = (x1 + x2) / 2; return 'M' + x1 + ' ' + y1.toFixed(1) + 'C' + mx + ' ' + y1.toFixed(1) + ',' + mx + ' ' + y2.toFixed(1) + ',' + x2 + ' ' + y2.toFixed(1); }
    var lane = LY.lanes[b.id] !== undefined ? LY.lanes[b.id] : z.r;
    var yl = g.y0 + lane * g.rh + o, gap = g.cw - g.nw;
    var xb1 = x1 + gap / 2, xb2 = x2 - gap / 2 + 1;
    return roundPath([[x1, y1], [xb1, y1], [xb1, yl], [xb2, yl], [xb2, y2], [x2, y2]], mini ? 4 : 6);
  }
  var graphUid = 0;
  /**
   * opts: { mini, flow: {branch: share} or null, bv: branch values or null, stateFlow, on: Set of branches,
   *         onStates: Set, cur: branch id, bands: {stage: 1|2}, start: state id }
   */
  function drawGraph(el, opts) {
    var mini = !!opts.mini, g = geom(mini), uid = 'g' + (++graphUid);
    var W = g.x0 + 15 * g.cw + g.nw / 2 + 14, H = g.y0 + 10.6 * g.rh + 12;
    var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="The map: states and branches of the library design">';
    s += '<defs><marker id="' + uid + 'a" viewBox="0 0 10 10" refX="9" refY="5" markerWidth="' + (mini ? 6 : 8) + '" markerHeight="' + (mini ? 6 : 8) + '" markerUnits="userSpaceOnUse" orient="auto"><path d="M0 0L10 5L0 10z" fill="currentColor"/></marker>' +
      '<pattern id="' + uid + 'h1" width="7" height="7" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="7" stroke="currentColor" stroke-width="1" opacity=".22"/></pattern>' +
      '<pattern id="' + uid + 'h2" width="4" height="4" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="4" stroke="currentColor" stroke-width="1.2" opacity=".32"/></pattern></defs>';
    // stage bands and headers
    var stageCols = {};
    MAP.states.forEach(function (st) { var c = LY.states[st.id][0]; var r = stageCols[st.step] = stageCols[st.step] || [99, -1]; r[0] = Math.min(r[0], c); r[1] = Math.max(r[1], c); });
    MAP.steps.forEach(function (stp, i) {
      var r = stageCols[stp.id]; if (!r) return;
      var xa = g.x0 + r[0] * g.cw - g.cw / 2 + 1, xb = g.x0 + r[1] * g.cw + g.cw / 2 - 1;

      if (i > 0) s += '<line class="g-sep" x1="' + xa + '" y1="4" x2="' + xa + '" y2="' + (H - 4) + '"/>';
      var lines = wrapWords(stp.label, Math.max(12, Math.floor((xb - xa) / (mini ? 5.6 : 6.1))));
      if (mini) lines = lines.slice(0, 2);
      lines.forEach(function (ln, k) { s += '<text class="g-step" text-anchor="middle" x="' + ((xa + xb) / 2) + '" y="' + (g.head[0] + k * (mini ? 10 : 12)) + '"' + (mini ? ' style="font-size:8.6px"' : '') + '>' + esc(ln) + '</text>'; });
      var dw = stageWords(stp.id);
      if (dw) s += '<text class="g-dstep" data-label="1" text-anchor="middle" x="' + ((xa + xb) / 2) + '" y="' + (g.head[2] + (lines.length > 2 ? 10 : 0)) + '"' + (mini ? ' style="font-size:8px"' : '') + '>' + esc(dw) + '</text>';
    });
    // edges
    var focus = !!(opts.on && opts.on.size);
    var edges = '', hits = '';
    MAP.branches.forEach(function (b) {
      var d = edgePath(g, b, mini);
      var f = opts.flow ? (opts.flow[b.id] || 0) : null;
      var zero = f !== null && f < 1e-12;
      var w = mini ? 1.1 : (f === null ? 1.3 : zero ? 1 : 0.7 + 10 * f);
      var on = focus && opts.on.has(b.id);
      if (on && mini) w = 3.2;
      if (on && !mini) w = Math.max(w, 2.4);
      var cls = 'g-edge' + (zero && !on ? ' zero' : '') + (on ? ' on' : '');
      if (opts.cur === b.id) edges += '<path class="g-halo" d="' + d + '" stroke-width="' + (w + 7) + '"/>';
      edges += '<path class="' + cls + '" d="' + d + '" stroke-width="' + w.toFixed(2) + '"' + (zero && !on ? '' : ' marker-end="url(#' + uid + 'a)"') + (opts.cur === b.id ? ' style="stroke-width:' + (w + 1.5).toFixed(2) + '"' : '') + '/>';
      var tip = '<b>' + esc(b.label) + '</b><br>Stage: ' + esc(STEP[STATE[b.from].step].label);
      if (opts.bv) {
        var v = opts.bv[b.id];
        tip += '<br>Chance at this branch: ' + num(v.chance, 'G', 'pct');
        if (f !== null) tip += '<br>Share of this piece taking it: ' + num(f, 'G', 'pct');
        if (b.tier) tip += '<br>Cost when taken: ' + num(v.cost, 'G', 'cost') + ' cheap-call equivalents, paid at ' + esc(TIER[b.tier].label);
      }
      if (opts.limTip && opts.limTip[b.id]) tip += '<br>Uses: ' + esc(opts.limTip[b.id].join(', '));
      hits += '<path class="g-hit" d="' + d + '" data-tip="' + esc(tip) + '"/>';
    });
    s += '<g>' + edges + '</g><g>' + hits + '</g>';
    // nodes
    MAP.states.forEach(function (st) {
      var p = pos(g, st.id), x = p.x - g.nw / 2, y = p.y - g.nh / 2;
      var cls = 'g-node' + (st.outcome ? ' outcome' : '') + (st.outcome && !OUT[st.id].right ? ' wrongout' : '') + (opts.start === st.id ? ' start' : '') + (opts.onStates && opts.onStates.has(st.id) ? ' on' : '');
      var tip = '<b>' + esc(st.label) + '</b><br>Stage: ' + esc(STEP[st.step].label) + (stageWords(st.step) ? ' (' + stageWords(st.step) + ')' : '');
      if (opts.stateFlow) tip += '<br>Share of this piece passing here: ' + num(opts.stateFlow[st.id] || 0, 'G', 'pct');
      if (opts.start === st.id) tip += '<br>This way of working starts here.';
      s += '<g class="' + cls + '" data-tip="' + esc(tip) + '"><rect x="' + x + '" y="' + y + '" width="' + g.nw + '" height="' + g.nh + '" rx="' + (mini ? 4 : 6) + '"/>';
      if (!mini) {
        var lines = wrapWords(st.label, st.outcome ? 12 : 14).slice(0, 3);
        var y0 = p.y - (lines.length - 1) * 5.3 + 3.2;
        lines.forEach(function (ln, k) { s += '<text text-anchor="middle" x="' + p.x + '" y="' + (y0 + k * 10.6).toFixed(1) + '">' + esc(ln) + '</text>'; });
      }
      s += '</g>';
    });
    s += '</svg>';
    el.innerHTML = s;
    el.classList.toggle('focus', focus);
  }

  var ROUTES = null, BV = null, EVC = null;
  function renderMap() {
    renderPickers();
    var cls = classFor(ST.policy);
    var stocks = ST.policy === 'library' ? stocksAt(ST.period) : {};
    EVC = E.evaluateClass(MAP, cur, ST.policy, cls, stocks);
    BV = C.branchValues(E, MAP, cur, ST.policy, cls, stocks);
    ROUTES = E.routes(MAP, cur, ST.policy, { classValues: cls, stocks: stocks, sortBy: ST.sortBy });
    if (ST.selAuto || !ROUTES.routes.some(function (r) { return r.id === ST.sel || r.id.indexOf(ST.sel + '>') === 0; })) {
      var top = ROUTES.routes.slice().sort(function (a, b) { return b.chance - a.chance; })[0];
      ST.sel = top ? top.id : null; ST.stepIdx = null; ST.selAuto = false; ST.userSel = false;
    }
    paintGraph();
    renderPiece(cls, stocks);
    renderTree();
    renderStepper();
  }
  function selBranches() {
    if (!ST.sel) return [];
    var list = ST.sel.split('>').map(function (i) { return MAP.branches[+i].id; });
    return ST.stepIdx === null ? list : list.slice(0, ST.stepIdx + 1);
  }
  function paintGraph() {
    var lit = ST.userSel || ST.stepIdx !== null;
    var bl = lit ? selBranches() : [];
    var on = new Set(bl), onS = new Set([POL[ST.policy].start]);
    bl.forEach(function (bid) { onS.add(BR[bid].from); onS.add(BR[bid].to); });
    drawGraph($('#graph'), { flow: EVC.branchFlow, stateFlow: EVC.stateFlow, bv: BV, on: on, onStates: onS, cur: ST.stepIdx !== null ? bl[bl.length - 1] : null, start: POL[ST.policy].start });
  }

  function buildTrie(routes) {
    var root = { kids: {}, order: [], chance: 0, carried: 0, key: '' };
    routes.forEach(function (r) {
      var node = root; node.chance += r.chance; node.carried += r.carried;
      r.branches.forEach(function (bid, i) {
        var bi = MAP.branches.indexOf(BR[bid]);
        var ch = node.kids[bi];
        if (!ch) { ch = node.kids[bi] = { bid: bid, kids: {}, order: [], chance: 0, carried: 0, key: (node.key ? node.key + '>' : '') + bi }; node.order.push(bi); }
        ch.chance += r.chance; ch.carried += r.carried; node = ch;
      });
      node.route = r;
    });
    return root;
  }
  function kidsOf(n) { var by = ST.sortBy; return n.order.map(function (k) { return n.kids[k]; }).sort(function (a, b) { return b[by] - a[by]; }); }
  function chainFrom(n) { var chain = [n]; while (!n.route && n.order.length === 1) { n = n.kids[n.order[0]]; chain.push(n); } return chain; }
  function renderTree() {
    var root = buildTrie(ROUTES.routes);
    var total = root.chance || 1;
    var head = '<div class="row small" data-tagged="G" style="justify-content:space-between;padding:2px 6px 6px"><span><b data-label="1">' + ROUTES.count + '</b> routes for this piece. Shares are of this piece; costs in cheap-call equivalents. ' + tagEl('G') + '</span><span class="muted">' + (ST.sortBy === 'chance' ? 'share' : 'share × cost') + '</span></div>';
    function row(n, depth) {
      var chain = chainFrom(n), end = chain[chain.length - 1];
      var kids = end.route ? [] : kidsOf(end);
      var open = ST.open[end.key] !== undefined ? ST.open[end.key] : end.chance / total >= 0.12 || depth === 0 && kids.length <= 4;
      var selected = ST.sel === end.key;
      var inSel = ST.sel && (ST.sel === end.key || ST.sel.indexOf(end.key + '>') === 0);
      var label = chain.map(function (c) { return esc(BR[c.bid].label); }).join(' <span class="muted">→</span> ');
      var val = ST.sortBy === 'chance' ? fmt(end.chance, 'pct') : fmt(end.carried, 'cost');
      var h = '<li><button type="button" class="trow' + (end.route ? ' leaf' : '') + '" data-key="' + end.key + '"' + (selected ? ' aria-current="true"' : '') + '>' +
        '<span class="tw">' + (end.route ? '•' : open ? '▾' : '▸') + '</span><span class="tl">' + label;
      if (end.route) h += '<br><b><i class="ocg ' + end.route.outcome + '"></i>' + esc(OUT[end.route.outcome].label) + '</b> <small>cost ' + esc(fmt(end.route.cost, 'cost')) + '</small>';
      else h += ' <small>(' + kids.length + ' ways on)</small>';
      h += '</span><span class="tv">' + esc(val) + '</span></button>';
      if (!end.route && (open || inSel)) {
        var big = kids.filter(function (k) { return k.chance / total >= 0.002; }), small = kids.filter(function (k) { return k.chance / total < 0.002; });
        h += '<ul>' + big.map(function (k) { return row(k, depth + 1); }).join('');
        if (small.length) {
          var sk = end.key + '#small', so = !!ST.open[sk], sum = small.reduce(function (a, k) { return a + k.chance; }, 0);
          h += '<li><button type="button" class="trow" data-small="' + sk + '"><span class="tw">' + (so ? '▾' : '▸') + '</span><span class="tl muted">' + small.length + ' rarer ways on, together</span><span class="tv">' + esc(fmt(sum, 'pct')) + '</span></button>' + (so ? '<ul>' + small.map(function (k) { return row(k, depth + 1); }).join('') + '</ul>' : '') + '</li>';
        }
        h += '</ul>';
      }
      return h + '</li>';
    }
    $('#tree').innerHTML = head + '<ul>' + kidsOf(root).map(function (k) { return row(k, 0); }).join('') + '</ul>';
    $('#treeTools').innerHTML = '<span class="small muted">Order the tree by</span>' + seg('Order', [['chance', 'Share of this piece'], ['carried', 'Cost it carries (share × cost)']], ST.sortBy);
  }

  function niceTicks(lo, hi, n) {
    var span = hi - lo || 1, step = Math.pow(10, Math.floor(Math.log10(span / n))), err = span / n / step;
    if (err >= 7.5) step *= 10; else if (err >= 3.5) step *= 5; else if (err >= 1.5) step *= 2;
    var t = [], v = Math.ceil(lo / step) * step;
    for (; v <= hi + step * 1e-9; v += step) t.push(+v.toFixed(10));
    return t;
  }

  function runEnsemble() {
    if (RUN) return;
    var n = +$('#ensN').value, seed = Math.round(+$('#ensSeed').value) || 0;
    RUN = C.ensembleRunner(E, MAP, TABLE, { n: n, seed: seed });
    $('#ensRun').disabled = true;
    // Yield through a message channel, not setTimeout: timers are clamped or throttled
    // in some panes and background tabs, which made the runs crawl. Each slice holds the
    // thread for at most about 40 ms, so the page keeps answering clicks and scrolls.
    var ch = typeof MessageChannel === 'function' ? new MessageChannel() : null;
    var later = ch ? function (f) { ch.port1.onmessage = function () { f(); }; ch.port2.postMessage(0); } : function (f) { setTimeout(f, 0); };
    (function tick() {
      var t0 = performance.now();
      while (!RUN.done() && performance.now() - t0 < 40) RUN.step(1);
      var k = RUN.count();
      $('#ensBar').style.width = (100 * k / n) + '%';
      $('#ensTxt').textContent = k + ' of ' + n + ' runs';
      if (!RUN.done()) { later(tick); return; }
      if (ch) { ch.port1.close(); ch.port2.close(); }
      ENS = RUN.result(); ENS.draws = RUN.draws(); RUN = null;
      $('#ensRun').disabled = false;
      renderEnsemble();
    })();
  }

  function scatter(title, kc, kw, w) {
    var W = 420, H = 300, L = 46, Rm = 12, T = 26, B = 40;
    var xs = ENS.draws.map(function (d) { return d[w].ratios[kc]; }), ys = ENS.draws.map(function (d) { return d[w].ratios[kw]; });
    var sx = xs.filter(isFinite).slice().sort(function (a, b) { return a - b; }), sy = ys.filter(isFinite).slice().sort(function (a, b) { return a - b; });
    var x0 = Math.min(E.percentile(sx, 0.5), 1), x1 = Math.max(E.percentile(sx, 99.5), 1), y0 = Math.min(E.percentile(sy, 0.5), 1), y1 = Math.max(E.percentile(sy, 99.5), 1);
    var px = (x1 - x0) * 0.06 || 0.02, py = (y1 - y0) * 0.06 || 0.02; x0 -= px; x1 += px; y0 -= py; y1 += py;
    var clipped = 0;
    var X = function (v) { return L + (v - x0) / (x1 - x0) * (W - L - Rm); }, Y = function (v) { return T + (y1 - v) / (y1 - y0) * (H - T - B); };
    var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="' + esc(title) + '"><text class="lbl-strong" x="' + L + '" y="14">' + esc(title) + '</text>' + svgTag(W - 22, 4, 'G');
    niceTicks(x0, x1, 5).forEach(function (t) { s += '<line x1="' + X(t) + '" x2="' + X(t) + '" y1="' + T + '" y2="' + (H - B) + '" stroke="var(--line)"/><text class="lbl" x="' + X(t) + '" y="' + (H - B + 14) + '" text-anchor="middle">' + fmt(t, 'ratio') + '</text>'; });
    niceTicks(y0, y1, 5).forEach(function (t) { s += '<line x1="' + L + '" x2="' + (W - Rm) + '" y1="' + Y(t) + '" y2="' + Y(t) + '" stroke="var(--line)"/><text class="lbl" x="' + (L - 5) + '" y="' + (Y(t) + 3.5) + '" text-anchor="end">' + fmt(t, 'ratio') + '</text>'; });
    s += '<line x1="' + X(1) + '" x2="' + X(1) + '" y1="' + T + '" y2="' + (H - B) + '" stroke="currentColor" stroke-dasharray="1 3"/><line x1="' + L + '" x2="' + (W - Rm) + '" y1="' + Y(1) + '" y2="' + Y(1) + '" stroke="currentColor" stroke-dasharray="1 3"/>';
    var dots = '';
    for (var i = 0; i < xs.length; i++) {
      var a = xs[i], b = ys[i]; if (!isFinite(a) || !isFinite(b)) { clipped++; continue; }
      if (a < x0 || a > x1 || b < y0 || b > y1) clipped++;
      dots += '<circle cx="' + X(Math.max(x0, Math.min(x1, a))).toFixed(1) + '" cy="' + Y(Math.max(y0, Math.min(y1, b))).toFixed(1) + '" r="1.7"/>';
    }
    s += '<g fill="currentColor" opacity=".35">' + dots + '</g>';
    s += '<text class="lbl" x="' + ((L + W - Rm) / 2) + '" y="' + (H - 8) + '" text-anchor="middle">cost ratio (left of the dotted line: cheaper)</text>';
    s += '<text class="lbl" transform="translate(12 ' + ((T + H - B) / 2) + ') rotate(-90)" text-anchor="middle">wrong-rate ratio (below: fewer wrong)</text>';
    return '<div class="chartbox" data-tagged="G">' + s + '</svg>' + (clipped ? '<p class="small muted">' + clipped + ' runs lie beyond the edges and are drawn on the edge.</p>' : '') + '</div>';
  }

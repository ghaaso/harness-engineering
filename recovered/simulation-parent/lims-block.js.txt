  // Two bars on one axis from WLO to WHI (every measured alone and last, with 0 and 1): alone hatched, last solid.
  function wbars(r) {
    var span = WHI - WLO || 1, z = 100 * (0 - WLO) / span;
    var bar = function (v, cls) { var x = 100 * (v - WLO) / span; return '<i class="' + cls + '" style="left:' + Math.min(z, x).toFixed(2) + '%;width:' + Math.abs(x - z).toFixed(2) + '%"></i>'; };
    return '<span class="wbars" aria-hidden="true">' + bar(r.alone, 'a p-hatch') + bar(r.last, 'l p-solid') + '<b style="left:' + z.toFixed(2) + '%"></b><b class="one" style="left:' + (100 * (1 - WLO) / span).toFixed(2) + '%"></b></span>';
  }
  // The alone and last cells of a row. A row the engine did not measure shows why, not a number.
  function wcells(r, attrs, bars) {
    var S = TX.sections.lims;
    if (!r) return '<span class="wv nm" style="grid-column:span 2"><small>…</small></span>';
    if (!r.measured) {
      var why = '<small>' + esc(S['nm_' + r.not_measured]) + (r.not_measured === 'gate_or_ladder' ? ' ↓' : '') + '</small>';
      return '<span class="wv nm"' + (attrs || '') + ' data-nm="' + r.not_measured + '" data-tip="' + esc(S['nm_why_' + r.not_measured]) + '"' + (r.not_measured === 'gate_or_ladder' ? ' data-gogate="1"' : '') + '>' + why + '</span>';
    }
    return '<span class="wv"' + (attrs || '') + ' data-tagged="G" data-alone="' + r.alone + '" data-last="' + r.last + '">' + (bars ? wbars(r) : '') + '<b class="wa">' + esc(fmt(r.alone, 'weight')) + '</b><b class="wl">' + esc(fmt(r.last, 'weight')) + '</b></span>';
  }

function makeResolver(all) {
  const set = new Set(all);
  return (tok) => {
    let t = tok.replace(/^\.?\//, '').replace(/[:#].*$/, '').replace(/\)$/, '');
    if (!/\.[a-z0-9]{1,5}$/i.test(t) && !t.endsWith('/')) return null; // not file-like (same rule as the earlier docs_extract.js)
    if (set.has(t)) return [t];
    const hits = all.filter((f) => f.endsWith('/' + t) || (t.endsWith('/') && f.startsWith(t)));
    if (t.endsWith('/')) return hits.length ? [t] : [];
    return hits;
  };
}
function cleanResolve(all, tok) {
  // cleaned rule: placeholders and globs are not citations; brace groups expand; a directory resolves if any file lies under it
  if (/[<*]/.test(tok)) return 'placeholder';
  let t = tok.replace(/^\.?\//, '').replace(/[:#].*$/, '').replace(/\)$/, '');
  const m = t.match(/^(.*)\{([^}]*)\}(.*)$/);
  const cands = m ? m[2].split(',').map((x) => m[1] + x + m[3]) : [t];
  let ok = 0, amb = 0;
  for (const c of cands) {
    if (c.endsWith('/')) { if (all.some((f) => f.startsWith(c) || f.includes('/' + c))) ok++; continue; }
    if (all.includes(c)) { ok++; continue; }
    const hits = all.filter((f) => f.endsWith('/' + c));
    if (hits.length === 1) ok++; else if (hits.length > 1) amb++;
  }
  if (ok === cands.length) return 'resolved';
  if (ok + amb === cands.length) return 'ambiguous';
  return 'none';
}
const resFrozen = makeResolver(frozenFiles), resMain = makeResolver(mainFiles), resLive = makeResolver(liveFiles);

function measure(name, docs, getText) {
  const R = { corpus: name, files: docs.length, bytes: 0, words: 0, est_tokens_bytes_div4: 0,
    file_tokens: 0, file_unique_frozen: 0, file_ambiguous_frozen: 0, file_none_frozen: 0, file_none_frozen_but_on_main_now: 0, file_none_frozen_but_in_live_tree: 0, file_absolute_or_outside: 0,
    clean_placeholder: 0, clean_resolved: 0, clean_ambiguous: 0, clean_none: 0, clean_none_examples: [], symbol_tokens: 0, symbol_unique_frozen: 0, symbol_ambiguous_frozen: 0, symbol_none_frozen: 0, examples_none: [] };
  for (const d of docs) {
    const txt = getText(d);
    R.bytes += Buffer.byteLength(txt);
    R.words += (txt.match(/\S+/g) || []).length;
    for (const m of txt.matchAll(/`([^`\s]{3,120})`/g)) {
      const tok = m[1];
      if (/^([A-Za-z]:[\\/]|\/c\/|~\/|\.\.\/|scratchpad|C:)/.test(tok)) { const r = resFrozen(tok); if (r) { R.file_tokens++; R.file_absolute_or_outside++; continue; } }
      const r = resFrozen(tok);
      if (r) {
        const cr = cleanResolve(frozenFiles, tok); R['clean_' + cr]++; if (cr === 'none' && R.clean_none_examples.length < 15 && Math.random() < 0.15) R.clean_none_examples.push(tok);
        R.file_tokens++;
        if (r.length === 1) R.file_unique_frozen++;
        else if (r.length > 1) R.file_ambiguous_frozen++;
        else { R.file_none_frozen++; if ((resMain(tok) || []).length) R.file_none_frozen_but_on_main_now++; else if ((resLive(tok) || []).length) R.file_none_frozen_but_in_live_tree++; if (R.examples_none.length < 12 && Math.random() < 0.2) R.examples_none.push(tok); }
        continue;
      }
      const nm = tok.replace(/\(.*$/, '').replace(/^.*[.:]/, '');
      if (/^[A-Za-z_][A-Za-z0-9_]{3,}$/.test(nm) && /[a-z][A-Z_]|_|[A-Z][a-z]+[A-Z]/.test(nm)) {
        R.symbol_tokens++;
        const ids = sym.get(nm);
        if (!ids) R.symbol_none_frozen++; else if (ids.size === 1) R.symbol_unique_frozen++; else R.symbol_ambiguous_frozen++;
      }
    }
  }
  R.est_tokens_bytes_div4 = Math.round(R.bytes / 4);
  return R;
}

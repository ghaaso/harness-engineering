const fs=require('fs'),rl=require('readline');
function shape(v,d){if(v===null)return 'null';if(Array.isArray(v))return v.length?['arr'+v.length, d<3?shape(v[0],d+1):'..']:'arr0';if(typeof v==='object'){const o={};for(const k of Object.keys(v))o[k]=d<3?shape(v[k],d+1):typeof v[k];return o;}if(typeof v==='string')return 'str'+v.length;return typeof v+':'+(typeof v==='number'||typeof v==='boolean'?v:'');}

// quote-aware split: mode 'seq' splits on ; && || newline; mode 'pipe' splits on a single |
function splitTop(str, mode) {
  const out = []; let cur = '', q = null; const s = String(str);
  for (let i = 0; i < s.length; i++) {
    const ch = s[i];
    if (q) { cur += ch; if (ch === '\\' && q === '"' && i + 1 < s.length) { cur += s[++i]; continue; } if (ch === q) q = null; continue; }
    if (ch === '"' || ch === "'") { q = ch; cur += ch; continue; }
    if (ch === '\\' && i + 1 < s.length) { cur += ch + s[++i]; continue; }
    if (mode === 'seq') {
      if (ch === ';' || ch === '\n') { out.push(cur); cur = ''; continue; }
      if ((ch === '&' && s[i + 1] === '&') || (ch === '|' && s[i + 1] === '|')) { out.push(cur); cur = ''; i++; continue; }
    } else if (ch === '|' && s[i + 1] !== '|' && s[i - 1] !== '|') { out.push(cur); cur = ''; continue; }
    cur += ch;
  }
  out.push(cur); return out.map((x) => x.trim()).filter(Boolean);
}

      const S = steps; if (!S.length) { wT.write(JSON.stringify({ file: rel, folder: short, sub, sid, ts0, ts1, branch: branch0, cwdRel, steps: 0, cls: 'empty' }) + '\n'); continue; }
      // windows (copied rule)
      const cps = new Set(compact); let win = 0; const winOf = [];
      for (let i = 0; i < S.length; i++) { if (cps.has(i)) win++; if (i > 0 && S[i - 1].ctx > 60000 && S[i].ctx < S[i - 1].ctx * 0.45) win++; winOf.push(win); }
      const laterInWin = []; for (let i = S.length - 1; i >= 0; i--) laterInWin[i] = i + 1 < S.length && winOf[i + 1] === winOf[i] ? laterInWin[i + 1] + 1 : 0;
      // transcript class (copied rule)
      const edits = {}; const lookTok = {}; let prodEdits = 0; let ctxSum = 0;
      for (const s of S) { ctxSum += s.ctx; for (const c of s.calls) { if (!c.cl) c.cl = classify(c); if (c.cl.k === 'edit') { edits[c.cl.z] = (edits[c.cl.z] || 0) + 1; if (['code', 'test', 'product-config'].includes(c.cl.z)) prodEdits++; } if (isLook(c.cl.k)) lookTok[c.cl.z] = (lookTok[c.cl.z] || 0) + c.rc; } }
      const prodLook = ['code', 'test', 'product-config', 'product-doc', 'claude-md'].reduce((a, z) => a + (lookTok[z] || 0), 0);
      const progLook = ['program-doc', 'program-evidence', 'memory', 'transcripts', 'testbed'].reduce((a, z) => a + (lookTok[z] || 0), 0);
      const allLook = Object.values(lookTok).reduce((a, b) => a + b, 0); const progEdits = (edits['program-doc'] || 0) + (edits['program-evidence'] || 0) + (edits.memory || 0);
      const tcls = prodEdits > 0 && prodEdits >= progEdits ? 'product-edit' : prodLook > progLook && prodLook > 20000 && prodLook >= 0.4 * allLook ? 'product-read' : 'program';
      const wcls = tcls === 'program' ? 'program' : 'product'; FALLBACK_CODE = wcls === 'product';
      wT.write(JSON.stringify({ file: rel, folder: short, sub, sid, ts0, ts1, branch: branch0, cwdRel, steps: S.length, ctxSum, cls: wcls, tcls, prodEdits }) + '\n');
      // later content (non-lookup) for citations and quotes; later edits
      const offs = []; let LC = ''; const EDITS = [];
      for (let i = 0; i < S.length; i++) {
        offs.push(LC.length); const s = S[i]; let piece = s.text || '';
        for (const c of s.calls) {
          if (c.ed) { piece += '\n' + c.ed.neu; EDITS.push({ i, f: normPath(c.ed.f, cwdRel), old: c.ed.old, write: c.name === 'Write' }); }
          else if (c.say) piece += '\n' + c.say;
          else if ((c.name === 'Bash' || c.name === 'PowerShell') && c.cl && (c.cl.k === 'edit-sh' || c.cl.k === 'act')) piece += '\n' + String((c.inpFull || {}).command || '');
        }
        LC += piece + '\n';
      }
      offs.push(LC.length);
      // quote index: each trimmed line of later content -> its last position (a quote is a whole read line reappearing as a line)
      const LCLINE = new Map(); { let pos = 0; for (const ln of LC.split('\n')) { const k = ln.trim().replace(/^[-*>+]\s+|^`|`$/g, '').trim(); if (k.length >= 40) LCLINE.set(k, pos); pos += ln.length + 1; } }
      const CITES = []; for (const m of LC.matchAll(CITE_G)) CITES.push({ pos: m.index, base: m[1].split('/').pop(), full: m[1], a: +m[2], b: m[3] ? +m[3] : +m[2] });
      // episodes (copied rule)
      const EP = []; let cur = null;
      const close = (endStep, endKind) => { if (cur && cur.calls.length) { cur.end = endKind; cur.endStep = endStep; EP.push(cur); } cur = null; };
      for (let i = 0; i < S.length; i++) {
        const s = S[i]; const looks = s.calls.filter((c) => isLook(c.cl.k)); const nonLook = s.calls.filter((c) => !isLook(c.cl.k) && c.cl.k !== 'neutral');
        if (!cur && looks.length) cur = { start: i, steps: 0, calls: [] };
        if (cur) {
          if (looks.length) cur.steps += nonLook.length ? 0 : 1;
          for (const c of looks) cur.calls.push({ i, c, n: c.name, k: c.cl.k, z: c.cl.z, t: c.cl.t, q: c.cl.q, g: c.cl.g, part: c.cl.part, rc: c.rc, later: laterInWin[i], paths: c.paths, head: c.head, err: c.err });
          if (nonLook.length) close(i, nonLook[0].cl.k + ':' + (nonLook[0].cl.z || ''));
          else if (!s.calls.length) close(i, 'answer');
        }
      }
      close(S.length - 1, 'transcript-end');
      // instances
      for (const e of EP) {
        const per = {}; let fetched = 0, carried = 0;
        for (const x of e.calls) { x.zone2 = callZone(x); x.form = form(x, x.zone2); x.qf = qf(x, wcls); x.cq = cqOf(x.qf); const tok = x.rc / CPT; fetched += tok; carried += tok * (x.later || 0); const p = (per[x.cq] = per[x.cq] || { fetched: 0, carried: 0, calls: 0 }); p.fetched += tok; p.carried += tok * (x.later || 0); p.calls++; }
        const dom = Object.entries(per).sort((a, b) => b[1].carried - a[1].carried || b[1].fetched - a[1].fetched)[0][0];
        // merge read spans over the episode
        const files = new Map();
        for (const x of e.calls) { if (!x.c.acc) continue; for (const [fp, en] of x.c.acc) { if (!fp) continue; let F = files.get(fp); if (!F) { F = { f: fp, lines: new Map(), ranges: [], whole: false, fileOnly: true, byCall: [] }; files.set(fp, F); } for (const [n, tx] of en.lines) F.lines.set(n, tx); F.ranges.push(...en.ranges); F.whole = F.whole || en.whole; if (en.rev) F.rev = en.rev; F.fileOnly = F.fileOnly && en.fileOnly && !en.lines.size && !en.ranges.length && !en.whole; F.byCall.push(e.calls.indexOf(x)); } }
        const endOff = offs[Math.min(e.endStep, S.length)] || 0; const endStep = e.endStep;
        const truth = [];
        for (const F of files.values()) {
          const spans = spansOf(F); const z = pzone(F.f);
          const text2n = new Map(); for (const [n, tx] of F.lines) { const k = tx.trim(); if (k.length >= 16 && /[A-Za-z]{3}/.test(k)) { if (!text2n.has(k)) text2n.set(k, []); text2n.get(k).push(n); } }
          const conf = { edit: new Set(), cite: new Set(), quote: new Set(), fileEdited: false, fileEditStep: null };
          for (const ed of EDITS) { if (ed.i < endStep || ed.f !== F.f) continue; conf.fileEdited = true; if (conf.fileEditStep == null) conf.fileEditStep = ed.i - endStep; for (const ol of String(ed.old).split('\n')) { const k = ol.trim(); if (k.length >= 16 && text2n.has(k)) for (const n of text2n.get(k)) conf.edit.add(n); } }
          const base = F.f.split('/').pop();
          const ambiguous = (BASECOUNT.get(base) || 0) > 1;
          for (const ct of CITES) { if (ct.pos < endOff || ct.base !== base) continue; if (ambiguous && !(ct.full.includes('/') && F.f.endsWith(ct.full.replace(/^\.\//, '')))) continue; for (const sp of spans) if (ct.a <= sp[1] + 2 && ct.b >= sp[0] - 2) { for (let n = Math.max(ct.a, sp[0]); n <= Math.min(ct.b, sp[1]); n++) conf.cite.add(n); if (ct.a > sp[1] || ct.b < sp[0]) conf.cite.add(Math.min(Math.max(ct.a, sp[0]), sp[1])); } }
          const cands = [...text2n.keys()].filter((k) => k.length >= 40 && k.length <= 400 && !/^(import |from \S+ import|#!|\/\/ |# ?-{3,}|\|[-: |]+\|$)/.test(k)).sort((a, b) => b.length - a.length).slice(0, 40);
          for (const k of cands) if ((LCLINE.get(k) || -1) >= endOff) for (const n of text2n.get(k)) conf.quote.add(n);
          let lineChars = 0; for (const tx of F.lines.values()) lineChars += tx.length + 1;
          const smp = []; const ns = [...F.lines.keys()].sort((a, b) => a - b); const step = Math.max(1, Math.floor(ns.length / 24)); for (let k = 0; k < ns.length; k += step) { const tx = F.lines.get(ns[k]); if (tx.trim().length >= 8) smp.push([ns[k], h8(tx)]); }
          const confLines = [...new Set([...conf.edit, ...conf.cite, ...conf.quote])].sort((a, b) => a - b);
          const confSpans = spans.filter((sp) => confLines.some((n) => n >= sp[0] && n <= sp[1]));
          truth.push({ f: F.f, zone: z, rev: F.rev || null, spans, whole: F.whole, fileOnly: F.fileOnly, nLines: F.lines.size, lineTok: Math.round(lineChars / CPT), samples: smp, conf: { edit: conf.edit.size, cite: conf.cite.size, quote: conf.quote.size, lines: confLines.slice(0, 200), spans: confSpans }, fileEdited: conf.fileEdited, fileEditAfterSteps: conf.fileEditStep, level: confLines.length ? 'confirmed' : 'read' });
        }
        const fc = e.calls[0];
        const inst = {
          id: sha(rel + '#' + e.start).slice(0, 12), file: rel, folder: short, sub, sid, cls: wcls, tcls,
          date: S[e.start].ts, branch: S[e.start].br || branch0, cwdRel: cwdRelOf(cwd), startStep: e.start, endStep, end: e.end, steps: e.steps, nCalls: e.calls.length,
          form: dom, formFirst: fc.cq, perForm: Object.fromEntries(Object.entries(per).map(([k, v]) => [k, { calls: v.calls, fetched: Math.round(v.fetched), carried: Math.round(v.carried) }])),
          fetched: Math.round(fetched), carried: Math.round(carried), laterFirst: fc.later,
          seed: seedOf(fc.c, cwdRel), firstCall: { tool: fc.n, kind: fc.k, qf: fc.qf, q: String(fc.n === 'Read' ? (fc.c.inpFull.file_path || '') : fc.n === 'Grep' ? fc.c.inpFull.pattern : fc.c.inpFull.command || fc.t || '').slice(0, 300) },
          calls: e.calls.map((x) => ({ tool: x.n, k: x.k, cq: x.cq, tok: Math.round(x.rc / CPT), later: x.later, q: String(x.n === 'Read' ? (x.c.inpFull.file_path || '') + (x.c.inpFull.offset != null ? ' @' + x.c.inpFull.offset + '+' + (x.c.inpFull.limit || '') : '') : x.n === 'Grep' ? x.c.inpFull.pattern + ' in ' + (x.c.inpFull.path || '.') + ' ' + (x.c.inpFull.output_mode || '') : x.c.inpFull.command || x.t || '').slice(0, 240) })),
          truth,
          confirmed: truth.some((t) => t.level === 'confirmed'), anyFileEdited: truth.some((t) => t.fileEdited),
        };
        wI.write(JSON.stringify(inst) + '\n'); nE++;
      }
      if (nT % 100 === 0) { fs.writeFileSync(W + '/out/extract-progress.json', JSON.stringify({ nT, nE, sec: (Date.now() - t0) / 1000 })); }
    }
  }
  wI.end(); wT.end();
  fs.writeFileSync(W + '/out/extract-progress.json', JSON.stringify({ done: true, nT, nE, sec: (Date.now() - t0) / 1000 }));
  console.log('transcripts', nT, 'instances', nE, 'sec', (Date.now() - t0) / 1000);
})();

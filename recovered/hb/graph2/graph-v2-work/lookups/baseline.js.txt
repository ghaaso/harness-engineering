function treeOf(commit) {
  const out = git(['ls-tree', '-r', '-l', commit]).toString().split('\n'); const T = new Map(); const need = [];
  for (const l of out) { const m = l.match(/^\d+ blob ([0-9a-f]{40})\s+(\d+|-)\t(.*)$/); if (!m) continue; const f = m[3]; if (!TEXT.test(f) || SKIP.test(f) || +m[2] > MAXB) continue; T.set(f, m[1]); if (!BLOB.has(m[1])) need.push([m[1], f, +m[2]]); }
  let k = 0;
  while (k < need.length) {
    const chunk = []; let sz = 0; while (k < need.length && (sz < 30e6 || !chunk.length)) { chunk.push(need[k]); sz += need[k][2]; k++; }
    const buf = git(['cat-file', '--batch'], chunk.map((c) => c[0]).join('\n') + '\n'); let p = 0;
    for (const [sha, f] of chunk) { const nl = buf.indexOf(10, p); const size = +buf.slice(p, nl).toString().split(' ')[2]; BLOB.set(sha, parse(f, buf.slice(nl + 1, nl + 1 + size).toString('utf8'))); p = nl + 1 + size + 1; }

}

function buildIndex(T) {
  const E = []; // entries: {p, a, b, kind, name, P}
  const files = new Map();
  const add = (f, P, cur) => { files.set(f, { P, cur }); E.push({ p: f, a: 1, b: P.n, kind: 'file', name: f.split('/').pop(), P, cur }); for (const s of P.syms) E.push({ p: f, a: s.a, b: s.b, kind: 'sym', name: s.name, P, cur }); for (const h of P.heads) E.push({ p: f, a: h.a, b: h.b, kind: 'head', name: h.t, P, cur }); for (const d of P.idDefs) E.push({ p: f, a: d.a, b: d.b, kind: 'iddef', name: d.id, P, cur }); };
  for (const [f, sha] of T) add(f, BLOB.get(sha), false);
  for (const [f, P] of UNTR) if (!files.has(f)) add(f, P, true);
  // BM25 (arm C) over entry text = name + path words (files: path words only)
  const inv = new Map(); const len = new Float32Array(E.length); let tot = 0;
  E.forEach((e, i) => { const ws = e.kind === 'file' ? words(e.p) : [...words(e.name), ...words(e.p.split('/').pop())]; len[i] = ws.length; tot += ws.length; const tf = new Map(); for (const w of ws) tf.set(w, (tf.get(w) || 0) + 1); for (const [w, c] of tf) { let l = inv.get(w); if (!l) inv.set(w, (l = [])); l.push(i, c); } });
  const byName = new Map(), byLower = new Map(), byPath = new Map(), byBase = new Map(), byId = new Map();
  E.forEach((e, i) => {
    if (e.kind === 'sym') { (byName.get(e.name) || byName.set(e.name, []).get(e.name)).push(i); const lo = e.name.toLowerCase(); (byLower.get(lo) || byLower.set(lo, []).get(lo)).push(i); }
    if (e.kind === 'file') { byPath.set(e.p, i); const b = e.name.toLowerCase(); (byBase.get(b) || byBase.set(b, []).get(b)).push(i); }
    if (e.kind === 'iddef') (byId.get(e.name) || byId.set(e.name, []).get(e.name)).push(i);
  });
  const lowerNames = [...byLower.keys()];
  const heads = []; E.forEach((e, i) => { if (e.kind === 'head') heads.push([e.name.toLowerCase(), i]); });
  return { E, files, inv, len, avg: tot / Math.max(1, E.length), byName, byLower, byPath, byBase, byId, lowerNames, heads };
}
// ---------- query from seed ----------
function queryOf(x) {
  const s = x.seed || {}; const q = { paths: [], terms: [], ids: [], phrases: [], scope: [] };
  if (s.kind === 'path') q.paths = [s.path, ...(s.paths || [])].filter(isRepoRel);
  if (s.kind === 'history') q.paths = (s.paths || []).filter(isRepoRel);
  if (s.kind === 'pattern' || s.kind === 'glob') { q.terms = s.terms || []; q.ids = s.ids || []; q.phrases = s.phrases || []; q.scope = [].concat(s.scope || []).filter((p) => isRepoRel(p) && p !== '.' && p !== ''); }
  q.paths = [...new Set(q.paths.map((p) => p.replace(/^\.\//, '')))];
  return q;
}
const inScope = (q, p) => !q.scope.length || q.scope.some((s) => p === s || p.startsWith(s.replace(/\/$/, '') + '/'));
function rank(cands, xid) { // cands: Map entryIdx -> [score, order]; ties broken by a seeded hash (order is used only for outline order)
  const seed = parseInt(String(xid).slice(0, 8), 16) | 0;
  const arr = [...cands].map(([i, [sc, ord]]) => [i, sc, ord, Math.imul((i ^ seed) + 0x9e3779b9, 2654435761) >>> 0]);
  arr.sort((a, b) => b[1] - a[1] || a[2] - b[2] || a[3] - b[3]);
  const tieAtCut = arr.length > K && arr[K - 1][1] === arr[K][1] && arr[K - 1][2] === arr[K][2];
  return { top: arr.slice(0, K).map((a) => a[0]), n: arr.length, tieAtCut };
}

function truthOf(x) {
  const conf = x.truth.some((t) => t.level === 'confirmed'); const prim = conf ? x.truth.filter((t) => t.level === 'confirmed') : x.truth;
  const m = new Map();

  return { conf, m };
}

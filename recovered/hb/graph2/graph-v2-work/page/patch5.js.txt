// one-off patch: repair backspace bytes left by a shell escape. Run from graph-v2:  node - < <this file>
const fs = require('fs');
const BS = String.fromCharCode(8);
let c = fs.readFileSync('check.js.txt', 'utf8');
c = c.split('.split(/' + BS + 'CQd+' + BS + '|' + BS + 'E1 to E3' + BS + '/)').join('.split(/\\bCQ\\d+\\b|\\bE1 to E3\\b/)');
fs.writeFileSync('check.js.txt', c);
let b = fs.readFileSync('build.js.txt', 'utf8');
b = b.split('replace(/' + BS + '([Tt])he the question/g').join('replace(/\\b([Tt])he the question/g');
fs.writeFileSync('build.js.txt', b);
console.log('left:', (c + b).split(BS).length - 1);

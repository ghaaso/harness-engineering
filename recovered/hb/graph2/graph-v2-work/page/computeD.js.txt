function bfs(adj, seed, maxHop) { const hop = new Map([[seed, 0]]); let fr = [seed]; for (let h = 1; h <= maxHop; h++) { const nx = []; for (const u of fr) for (const v of adj.get(u) || []) if (!hop.has(v)) { hop.set(v, h); nx.push(v); } fr = nx; } return hop; }
function ppr(adj, seed, alpha = 0.15, iters = 25) {
  let p = new Map([[seed, 1]]);
  for (let it = 0; it < iters; it++) { const q = new Map([[seed, alpha]]);
    for (const [u, m] of p) { const nb = adj.get(u); if (!nb || !nb.size) { q.set(seed, (q.get(seed) || 0) + (1 - alpha) * m); continue; } const sh = (1 - alpha) * m / nb.size; for (const v of nb) q.set(v, (q.get(v) || 0) + sh); }
    p = q; }
  return p;
}

function relLines(seedNodes, itemFile, max) {
  const out = [];

    const sF = col(e.s) === itemFile, dF = col(e.d) === itemFile;
    if (sIn && dF) out.push({ dir: '->', kind: e.kind, other: e.d, mine: e.s, evp: e.evp, evl: e.evl });
    else if (dIn && sF) out.push({ dir: '<-', kind: e.kind, other: e.s, mine: e.d, evp: e.evp, evl: e.evl });
  }
  out.sort((a, b) => (a.evp || '').localeCompare(b.evp || '') || (a.evl || 0) - (b.evl || 0));

function renderItem(seedNodes, item, seedLabel, seedFile) {
  const r = relLines(seedNodes, item.id, 3);
  const n = byId.get(item.id);
  let tree = n.path + '\n';
  for (const l of r.lines) tree += '  ' + (l.dir === '<-' ? '<-' : '') + l.kind + ' ' + label(l.other) + (l.evp ? ' @' + l.evp.split('/').pop() + ':' + l.evl : '') + '\n';
  if (r.all > r.lines.length) tree += '  +' + (r.all - r.lines.length) + ' more\n';
  if (item.co) tree += '  changed together in ' + item.co.support + ' commits\n';
  if (!r.all && !item.co && item.via) tree += '  via ' + byId.get(item.via).path + '\n';
  let tri = '';
  for (const l of r.lines) tri += (l.dir === '<-' ? l.other + ' ' + l.kind + ' ' + l.mine : l.mine + ' ' + l.kind + ' ' + l.other) + (l.evp ? ' @' + l.evp + ':' + l.evl : '') + '\n';
  if (item.co) tri += (seedFile || seedLabel) + ' co_changed ' + item.id + ' support=' + item.co.support + '\n';
  if (!r.all && !item.co && item.via) tri += seedLabel + ' reaches ' + item.via + '\n' + item.via + ' reaches ' + item.id + '\n';
  return { tree, tree_tok: countTokens(tree), triples: tri, triples_tok: countTokens(tri), n_rel: r.all };
}

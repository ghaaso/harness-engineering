function headlineOf(txt) {
  // priority: a paragraph that opens with a bold Answer; then the first prose paragraph under a heading named
  // what-this-changes, then verdict/answer/result/summary, then headline; else the first prose paragraph
  const paras = txt.split(/\r?\n\s*\r?\n/).map(p => p.trim()).filter(Boolean);
  const prose = p => !/^(#|\||```)/.test(p) && !/^Every number/i.test(p) && !/^Intervals:/i.test(p);
  const clean = p => p.replace(/\s*\r?\n\s*/g, ' ').replace(/^[-*]\s+/, '');
  let h = paras.find(p => /^\*\*Answer/i.test(p));
  if (!h) for (const re of [/^#+\s*(\d+\.\s*)?what this (changes|means)/i, /^#+\s*(\d+\.\s*)?(verdict|answer|result|summary)/i, /^#+\s*(\d+\.\s*)?headline/i]) {
    const i = paras.findIndex(p => re.test(p)); if (i < 0) continue;
    const body = paras[i].split(/\r?\n/).slice(1).join('\n').trim();
    const cand = [body, ...paras.slice(i + 1)].filter(Boolean).find(p => prose(p)); if (cand) { h = cand; break; } }
  if (!h) h = paras.find(prose) || '';
  h = clean(h).replace(/\*\*/g, '').replace(/`/g, '');
  return h.length > 900 ? h.slice(0, 900).replace(/\s\S*$/, '') + ' …' : h;
}

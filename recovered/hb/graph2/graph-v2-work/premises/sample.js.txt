function rng(seed) { return () => { seed |= 0; seed = (seed + 0x6d2b79f5) | 0; let t = Math.imul(seed ^ (seed >>> 15), 1 | seed); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
const r = rng(20261113);
const out = [];
for (const [id, kind, f] of FILES) {
  const L = fs.readFileSync(f, 'utf8').split(/\r?\n/);
  const units = []; let cur = null;
  const flush = () => { if (cur && cur.text.length >= 80) units.push(cur); cur = null; };
  L.forEach((line, i) => {
    const s = line.trim();
    if (!s) { flush(); return; }
    if (/^#/.test(s) || /^\|?\s*:?-{3,}/.test(s) || /^```/.test(s)) { flush(); return; }
    if (/^([-*]|\d+[.)])\s+/.test(s) || /^\|/.test(s)) { flush(); cur = { line: i + 1, text: s }; if (/^\|/.test(s)) flush(); return; }
    if (cur && /^\s+/.test(line)) { cur.text += ' ' + s; return; }
    if (cur && !/^\s/.test(line) && !/^([-*]|\d+[.)])/.test(s)) { cur.text += ' ' + s; return; }
    flush(); cur = { line: i + 1, text: s };
  });
  flush();
  // drop table header rows (the row right before a separator) by checking the next line
  const keep = units.filter((u) => !(/^\|/.test(u.text) && /^\|?\s*:?-{3,}/.test((L[u.line] || '').trim())));
  // seeded shuffle
  const idx = keep.map((_, k) => k); for (let k = idx.length - 1; k > 0; k--) { const j = Math.floor(r() * (k + 1)); [idx[k], idx[j]] = [idx[j], idx[k]]; }
  idx.slice(0, 9).forEach((k, n) => out.push({ file: id, kind, path: f, order: n + 1, line: keep[k].line, units: keep.length, text: keep[k].text.slice(0, 1100) }));
}

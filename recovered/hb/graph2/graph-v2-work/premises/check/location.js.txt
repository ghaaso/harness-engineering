// Location shares and writer shares of CQ1 reads, from rows.jsonl, with an own git first-added test of "tracked when read".
const fs=require('fs'),rl=require('readline');
const ROOT='c:/users/user/documents/projects/ems_workshop1/';
const added=new Map();let t=null;
for(const line of fs.readFileSync('my-git-added.txt','utf8').split('\n')){const s=line.trim();if(!s)continue;if(s.startsWith('@@')){t=Date.parse(s.split(' ')[1]);continue;}const k=s.toLowerCase();if(!added.has(k)||added.get(k)>t)added.set(k,t);}
const dirMin=new Map();for(const [p,tt] of added){const parts=p.split('/');for(let i=1;i<parts.length;i++){const d=parts.slice(0,i).join('/');if(!dirMin.has(d)||dirMin.get(d)>tt)dirMin.set(d,tt);}}
const grp=c=>c.startsWith('1 ')||c.startsWith('2 ')?'tracked':c.startsWith('3')?'folderUntracked':c.startsWith('4 ')?'scratch':c.startsWith('5 ')?'transcript':c.startsWith('7 ')?'otherOutside':'?';
(async()=>{const rows=[];const per={};
const r=rl.createInterface({input:fs.createReadStream('../rows.jsonl')});
for await(const l of r){const o=JSON.parse(l);const key=o.f+'|'+o.cid+'|'+o.i;per[key]=(per[key]||0)+1;rows.push({key,cls:o.cls,p:o.p,ts:o.ts,tok:o.tok,car:o.car,lvl:o.lvl,age:o.ageLastWriteH,vc:o.vc,kind:o.kind});}
const nCalls=Object.keys(per).length;const G={},W={};let T=0,C=0;const agree={yes:0,no:0,examples:{}};
for(const x of rows){const w=1/per[x.key];T+=x.tok;C+=x.car;const g=grp(x.cls);G[g]=G[g]||{calls:0,tok:0,car:0};G[g].calls+=w;G[g].tok+=x.tok;G[g].car+=x.car;
 // own tracked test
 const pl=x.p.toLowerCase();let mine='outside';
 if(pl.startsWith(ROOT)){const rel=pl.slice(ROOT.length).replace(/\/+$/,'');const ta=added.has(rel)?added.get(rel):(dirMin.has(rel)?dirMin.get(rel):null);mine=(ta!=null&&ta<=x.ts)?'tracked':'folderUntracked';}
 const theirs=g==='tracked'?'tracked':g==='folderUntracked'?'folderUntracked':'outside';
 if(mine===theirs)agree.yes++;else{agree.no++;const k=theirs+'->'+mine;agree.examples[k]=(agree.examples[k]||0)+1;}
 // writer level
 let wcls;if(x.lvl.startsWith('transcript'))wcls='a';else if(x.lvl.startsWith('session'))wcls=(x.age!=null&&x.age<24)?'b':'c';else if(x.lvl==='otherSession')wcls='d';else wcls='e';
 W[wcls]=W[wcls]||{calls:0,tok:0,car:0};W[wcls].calls+=w;W[wcls].tok+=x.tok;W[wcls].car+=x.car;}
const out={rowCalls:nCalls,rowTok:T,rowCar:C,location:{},writer:{},ownTrackedAgreement:agree};
const CLAIM_CALLS=6340,CLAIM_TOK=20704868,CLAIM_CAR=522160773;
for(const [g,v] of Object.entries(G))out.location[g]={calls:+v.calls.toFixed(1),callsShare:+(100*v.calls/CLAIM_CALLS).toFixed(1),tokShare:+(100*v.tok/CLAIM_TOK).toFixed(1),carShare:+(100*v.car/CLAIM_CAR).toFixed(1)};
out.location.noPathResidual={calls:CLAIM_CALLS-nCalls,callsShare:+(100*(CLAIM_CALLS-nCalls)/CLAIM_CALLS).toFixed(1)};
for(const [g,v] of Object.entries(W))out.writer[g]={calls:+v.calls.toFixed(1),callsShare:+(100*v.calls/nCalls).toFixed(1),tokShare:+(100*v.tok/T).toFixed(1),carShare:+(100*v.car/C).toFixed(1)};
const s=(ks,f)=>+ks.reduce((a,k)=>a+(W[k]?W[k][f]:0),0);
out.task24={calls:+(100*s(['a','b'],'calls')/nCalls).toFixed(1),tok:+(100*s(['a','b'],'tok')/T).toFixed(1),car:+(100*s(['a','b'],'car')/C).toFixed(1)};
out.loose={calls:+(100*s(['a','b','c'],'calls')/nCalls).toFixed(1),car:+(100*s(['a','b','c'],'car')/C).toFixed(1)};
fs.writeFileSync('location.json',JSON.stringify(out,null,1));console.log(JSON.stringify(out));})();

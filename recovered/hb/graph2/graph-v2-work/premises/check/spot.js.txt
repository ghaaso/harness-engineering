// Ground-truth spot checks: (1) 15 seeded repo paths vs git ls-files (read-only); (2) 5 seeded findings: quoted text present in cited file.
// Details go to spot.json in this folder only; stdout is aggregate.
const fs=require('fs'),rl=require('readline'),cp=require('child_process');const E=process.argv[2];
const REPO='C:/Users/User/Documents/projects/EMS_workshop1';const ROOT=REPO.toLowerCase()+'/';
let s=20261114;const rnd=()=>{s|=0;s=s+0x6D2B79F5|0;let t=Math.imul(s^s>>>15,1|s);t=t+Math.imul(t^t>>>7,61|t)^t;return((t^t>>>14)>>>0)/4294967296;};
const shuffle=a=>{for(let i=a.length-1;i>0;i--){const j=Math.floor(rnd()*(i+1));[a[i],a[j]]=[a[j],a[i]];}return a;};
const added=new Map();let t=null;for(const line of fs.readFileSync('my-git-added.txt','utf8').split('\n')){const x=line.trim();if(!x)continue;if(x.startsWith('@@')){t=Date.parse(x.split(' ')[1]);continue;}const k=x.toLowerCase();if(!added.has(k)||added.get(k)>t)added.set(k,t);}
(async()=>{const P=new Map();const r=rl.createInterface({input:fs.createReadStream('../rows.jsonl')});
for await(const l of r){const o=JSON.parse(l);const pl=o.p.toLowerCase();if(!pl.startsWith(ROOT)||pl.includes('/.claude/worktrees/')||o.kind.startsWith('directory'))continue;
 const lab=/^[12] /.test(o.cls)?'tracked':/^3/.test(o.cls)?'untracked':null;if(!lab)continue;const rel=o.p.slice(ROOT.length);
 if(!P.has(rel))P.set(rel,{lab,ts:o.ts});}
const keys=[...P.keys()].sort();const tr=shuffle(keys.filter(k=>P.get(k).lab==='tracked')).slice(0,8),un=shuffle(keys.filter(k=>P.get(k).lab==='untracked')).slice(0,7);
const det=[];let agree=0,explained=0;
for(const rel of [...tr,...un]){const x=P.get(rel);let now;try{cp.execFileSync('git',['-C',REPO,'-c','core.longpaths=true','ls-files','--error-unmatch','--',rel],{stdio:'pipe'});now='tracked';}catch(e){now='untracked';}
 const ta=added.get(rel.toLowerCase());const addedRel=ta==null?'never added':(ta<=x.ts?'added before read':'added after read');
 const ok=now===x.lab;if(ok)agree++;const why=ok?'':(x.lab==='untracked'&&addedRel==='added after read'?'committed after the read (label is as-of-read)':x.lab==='tracked'&&addedRel==='added before read'?'tracked on another ref or deleted since':'UNEXPLAINED');if(!ok&&why!=='UNEXPLAINED')explained++;
 det.push({rel,label:x.lab,lsFilesNow:now,gitAdd:addedRel,match:ok,why});}
// findings quote check
const F=JSON.parse(fs.readFileSync(E+'/findings-classified.json','utf8')).findings;const pick=shuffle(F.map(f=>f.id).sort()).slice(0,5);
const norm=x=>x.toLowerCase().replace(/[`*_>|#\[\]()"'\u2018\u2019\u201c\u201d]/g,' ').replace(/\s+/g,' ').trim();
const qd=[];for(const id of pick){const f=F.find(y=>y.id===id);let txt;try{txt=norm(fs.readFileSync(f.source_path,'utf8'));}catch(e){qd.push({id,file:'unreadable'});continue;}
 const q=norm(f.text.replace(/\.\.\.|\u2026/g,' '));const words=q.split(' ');const wins=[];for(let i=0;i+8<=words.length;i+=8)wins.push(words.slice(i,i+8).join(' '));
 const hit=wins.filter(w=>txt.includes(w)).length;let lineOk=null;
 try{const lines=fs.readFileSync(f.source_path,'utf8').split(/\r?\n/);const around=norm(lines.slice(Math.max(0,f.line-3),f.line+40).join(' '));lineOk=wins.length?around.includes(wins[0]):null;}catch(e){}
 qd.push({id,source_kind:f.source_kind,tracked:f.source_path.toLowerCase().startsWith(ROOT),windows:wins.length,windowsFound:hit,firstWindowNearCitedLine:lineOk});}
fs.writeFileSync('spot.json',JSON.stringify({paths:det,quotes:qd},null,1));
console.log(JSON.stringify({paths:{n:det.length,agree,explainedMismatches:explained,unexplained:det.filter(d=>d.why==='UNEXPLAINED').length,byLabel:det.map(d=>d.label[0]+(d.match?'+':'-')+(d.why?'('+d.why.slice(0,9)+')':'')).join(' ')},quotes:qd}));})();

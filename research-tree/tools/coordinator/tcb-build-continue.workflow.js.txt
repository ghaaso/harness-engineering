let dead = 0
const missing = [], notStarted = [], skipped = []
async function stage(label, phaseName, lines, opts) {
  if (doneSet.has(label)) { skipped.push(label); return { from_file: DONE + '/' + fname(label) + '.json' } }
  if (dead >= 3) { notStarted.push(label); return null }
  const prompt = lines.concat(['', CONTEXT, '', 'LAST STEP, always: write your whole return, as JSON, to ' + DONE + '/' + fname(label) + '.json (write a temporary file beside it, then rename it). This file is how the work is picked up again if the PC resets.']).join('\n')
  const r = await agent(prompt, Object.assign({ label: label, phase: phaseName }, opts))
  if (r == null) { dead++; missing.push(label); log('NULL ' + label + ' (' + dead + ' so far; at 3 no new agent starts)') }
  return r
}

async function reviewUnit(u, before) {
  await before
  const scripts = u.groups.flatMap(g => g.scripts)
  const rv = await stage('review:' + u.key, 'Review', [
    'Review one unit of the trusted scripts, refuting by default: unit ' + u.key + ' (' + u.name + '): ' + scripts.map(s => TCB + '/' + s).join(', ') + (u.key === 'U5' ? ', the briefs under ' + TCB + '/briefs/ and ' + TCB + '/thresholds.json' : '') + (u.key === 'U3' ? ', and the role files under ' + TCB + '/agents/, ' + TCB + '/profiles/ and ' + TCB + '/settings/' : '') + '. Judge them against their specs under ' + ROOT + '/spec/, ' + CONTRACT + ' and the rules of the design that the specs name.',
    'Read the code. Run the unit\'s tests. For each script make at least two mutations in a scratch copy under ' + ROOT + '/review-tmp/' + u.key + '/ that break the mechanism the script exists for (remove the epoch check, the lock, the hash comparison, the search for the quote, the three-valued result, the blinding) and see that a test fails: a mutation no test catches is a finding. Look for: a behaviour of the spec that the code does not do; an unsafe action (a kill of a process not proven to belong to the attempt, a write outside the run root, a network call outside snap.mjs, a path or a command built from untrusted text); a race between a check and the act that follows; a wall-clock read where the design counts; a "pass" where a check did not run; a statistic computed wrongly (check any formula by hand on a small case); a brief that hints at the hoped-for result or names a model tier. High and medium findings only, each with file and lines, the failing condition and the smallest fix. Do not edit the unit\'s files.',
  ], { schema: FINDINGS, model: 'opus', effort: 'high' })
  if (!rv) return { key: u.key, review: rv, fix: null }
  // a review finished before a restart does not prove that its fix was: the fix runs unless the journals hold its result too (2026-11-17)
  if (rv.from_file && doneSet.has('fix:' + u.key)) return { key: u.key, review: rv, fix: { from_file: DONE + '/fix-' + u.key + '.json' } }
  if (!rv.from_file && !(rv.findings.length || rv.mutations.some(m => !m.a_test_failed))) return { key: u.key, review: rv, fix: null }
  const fx = await stage('fix:' + u.key, 'Review', [
    'Apply the reviewer\'s findings to unit ' + u.key + ' of the trusted scripts (you are the single writer of this unit\'s scripts, tests, fixtures and other files for this step; never edit ' + TCB + '/lib.mjs: put what the library needs under lib_change_requests).     (rv.from_file ? RESUMED(u.key) : ''),
    ref(rv, 'The review'),
  ], { schema: FIXED, model: u.tier === 'opus' ? 'opus' : 'sonnet', effort: 'high' })
  return { key: u.key, review: rv, fix: fx }
}

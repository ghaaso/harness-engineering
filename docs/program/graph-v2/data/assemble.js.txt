function mdTable(text, start) {
  const L = text.split('\n'); let i = L.findIndex(l => start.test(l)); if (i < 0) throw new Error('table not found ' + start);
  while (i < L.length && !L[i].startsWith('|')) i++;

  return { head: rows[0], rows: rows.slice(1) };
}

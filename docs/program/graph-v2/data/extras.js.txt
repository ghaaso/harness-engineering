// bullets: lines starting with "- " (or "N. " when numbered) with indented continuation lines, until a blank line
function bulletsAfter(lines, start, numbered) {
  const out = []; let cur = null;
  const startRe = numbered ? /^\d+\.\s+/ : /^- /;
  for (let i = start; i < lines.length; i++) {
    const l = lines[i];
    if (startRe.test(l)) { if (cur !== null) out.push(cur); cur = l.replace(startRe, "").trim(); }
    else if (cur !== null && /^\s+\S/.test(l)) cur += " " + l.trim();
    else if (cur !== null && l.trim() === "" ) { if (numbered) continue; break; }
    else if (cur !== null) break;
  }
  if (cur !== null) out.push(cur);
  return need(out, "bullets after line " + start);
}

  need(m, "interval in '" + cell + "'");
  return [num(m[1]), num(m[2]), num(m[3])];
};

// ---- pending slots
function baseSlot(p) {
  const abs = path.join(REPO, p);
  if (!fs.existsSync(abs)) return { path: p, exists: false, label: "being measured in another session" };
  const lines = fs.readFileSync(abs, "utf8").replace(/\r/g, "").split("\n");
  const out = { path: p, exists: true };
  const dl = lines.find((l) => /\b20\d\d-\d\d-\d\d\b/.test(l));
  out.written = dl ? dl.trim() : null;
  let h = -1, lvl = 0;
  for (let i = 0; i < lines.length; i++) { const m = /^(#{1,6})\s+(.*)$/.exec(lines[i]); if (m && /headline|verdict|short answer|result/i.test(m[2])) { h = i; lvl = m[1].length; break; } }
  if (h < 0) { out.headline = null; out.numbers = []; return out; }
  const body = [];
  for (let i = h + 1; i < lines.length && body.length < 25; i++) { const m = /^(#{1,6})\s/.exec(lines[i]); if (m && m[1].length <= lvl) break; body.push(lines[i]); }
  out.headline = body.join("\n").trim();
  out.numbers = out.headline.match(/[+-]?\d[\d,]*(?:\.\d+)?%?/g) || [];
  return out;
}

  const chk = (c, what) => { if (!c) throw new Error("premise json disagrees with NOTES: " + what); };
  chk(reads_by_location[2].carried_pct === 67.8 && reads_by_location[1].carried_pct === 25.1 && reads_by_location[0].carried_pct === 4.1 && same_task.same_session_24h.carried_pct === 83.9, "location/same-session shares");
  chk(findings_rows[0].k === 12 && findings_rows[4].k === 32 && reuse.found === 43 && reuse.sampled === 56 && stale.stale === 2 && stale.drifted === 1 && stale.traced === 8, "finding counts");
  must(flat, "67.8 % carried", "67.8"); must(flat, "83.9 % carried", "83.9"); must(flat, "43 of 56 found", "reuse");
  return { result_path: dir + "result.json", notes_path: dir + "NOTES.md", date, what_it_measures, reads_by_location, total, same_task, findings_rows, premises_per_finding, reuse, stale, step4, could_not_measure, not_claimed };
}

// Renderers for the two finished measurements (lookup test set, premise measure). Loaded by build.js.txt with
// require(); every number comes from data/extras.json pending.<k>.structured, printed as given (no re-rounding),
// except axis ticks and positions, which carry class "cmp". Each renderer returns '' when its part is absent, so a
// missing field drops one block, never the slot.
'use strict';
module.exports = function (H) {
  const { esc, e, tag, fold, num, CQS } = H;
  const has = v => v !== undefined && v !== null && v !== '';
  const arr = v => Array.isArray(v) ? v : [];
  const pc = v => has(v) ? `${v} %` : 'n/a';
  const fmt = v => typeof v === 'number' ? num(v) : esc(v);
  const formName = r => esc(r.form_name || (r.form && CQS[r.form]) || r.form || '');
  const clsName = c => ({ product: 'product', program: 'program', all: 'all' }[c] || esc(c));
  const list = (xs, cls) => `<ul class="${cls || 'gfl'}">${arr(xs).map(t => `<li>${e(typeof t === 'string' ? t : JSON.stringify(t))}</li>`).join('')}</ul>`;
  // one 0-100 % bar with its value printed as given
  const pbar = (lab, v, cls, lw, extra) => !has(v) ? '' : `<div class="hbar" style="--lw:${lw || 200}px"><div class="lab">${lab}</div><div class="track"><div class="fill ${cls || 'g-docs'}" style="width:${Math.max(0.3, +v).toFixed(2)}%"></div>${extra || ''}<div class="val" style="left:${Math.min(+v, 74).toFixed(2)}%">${pc(v)}</div></div></div>`;
  const kv = o => Object.entries(o || {}).filter(([, v]) => typeof v !== 'object');
  const seeOf = p => `${tag('meas', esc(p.path))}`;
  const flat = p => String(p.text || '').replace(/\s+/g, ' ');
  const grab = (p, re) => { const x = flat(p).match(re); return x || null; };

  /* ---------- lookup test set: section B ---------- */
  function ltB(p) {
    const S = p.structured; if (!S) return '';
    const I = S.instances || {}, C = S.confirmed || {}, RP = S.replayable || {}, R1 = S.r1_falsifier || {};
    const LAB = { all: 'lookups in the test set', transcripts: 'transcripts they come from (session logs, subagents included)', product: 'lookups in product work', program: 'lookups in program and document work', product_carried_M: 'million carried tokens, product', program_carried_M: 'million carried tokens, program',
      all_pct: 'all lookups', edit_or_cite_pct: 'all lookups, by an edit or citation only (no quote)', product_pct: 'product lookups', program_pct: 'program lookups', product_carried_weighted_pct: 'product, weighted by carried tokens', program_carried_weighted_pct: 'program, weighted by carried tokens',
      product_carried_pct: 'product, by carried tokens', program_carried_pct: 'program, by carried tokens' };
    const lab = k => esc(LAB[k] || k.replace(/_pct$/, '').replace(/_/g, ' '));
    const cnt = kv(I).map(([k, v]) => `<div class="kfig"><b>${fmt(v)}</b><span>${lab(k)}</span></div>`).join('');
    const confBars = kv(C).filter(([k]) => /pct/.test(k)).map(([k, v]) => pbar(lab(k), v, 'g-docs', 330)).join('');
    const repBars = kv(RP).filter(([k]) => /pct/.test(k)).map(([k, v]) => pbar(lab(k), v, 'g-index', 330)).join('');
    const repN = has(RP.all_n) ? `${fmt(RP.all_n)} lookups in all replay with no leak (the count behind the first bar).` : '';
    const r1rows = arr(R1.rows).map(r => `<div class="r1row"><div class="lab"><b>${formName(r)}</b><small>${clsName(r.cls)} · ${fmt(r.instances)} lookups</small></div><div class="track"><i class="half cmp" style="left:50%"></i><div class="fill ${r.at_least_half ? 't-ok' : 't-warn'}" style="width:${Math.max(0.3, +r.confirmed_any_pct).toFixed(2)}%"></div><div class="val" style="left:${Math.min(+r.confirmed_any_pct, 74).toFixed(2)}%">${pc(r.confirmed_any_pct)} confirmed</div></div><div class="r1v">${r.at_least_half ? 'at least half: index or graph replay can score it' : 'under half: must be scored by an agent run'}</div></div>`).join('');
    return `<div class="slot filled meas-block"><div class="slh">${seeOf(p)} <b>Measured.</b> The test set of real lookups, taken from recorded sessions.</div>
${p.head ? `<p>${esc(p.head)}</p>` : ''}
${cnt ? `<h4>Size of the test set (lookup instances)</h4><div class="kfigs">${cnt}</div>` : ''}
${confBars ? `<h4>Confirmed truth: lookups whose read span a later edit, citation or quote used (0 to 100 %)</h4>${confBars}` : ''}
${repBars ? `<h4>Replayable with no leak: the needed files unchanged at the frozen commit (0 to 100 %)</h4>${repBars}${repN ? `<p class="small">${repN}</p>` : ''}` : ''}
${r1rows ? `<h4>The falsifier of recommendation 1, per question form</h4><p class="small">${e(String(R1.rule || '').replace(/([^.])$/, '$1.'))} The dashed line marks half; green bars reach it, amber bars do not.</p><div class="r1">${r1rows}</div>` : ''}
${S.design_change && !p.skipDesign ? `<p class="note"><b>What this changes in the design.</b> ${e(S.design_change)}</p>` : ''}
${fold('What this measurement could not measure (' + arr(S.could_not_measure).length + ')', list(S.could_not_measure))}</div>`;
  }

  /* ---------- lookup test set: section I (the arm to beat) ---------- */
  function ltI(p) {
    const S = p.structured; if (!S) return '';
    const arms = Object.fromEntries(arr(S.arms).map(a => [a.key, a.name]));
    const armN = k => esc(arms[k] ? `${k}, ${arms[k]}` : k);
    const armS = k => esc(arms[k] ? `${k} (${arms[k].replace(/ over .*/, '').split(/[,(]/)[0].trim()})` : k);
    const armLegend = arr(S.arms).length ? `<p class="small">Arms: ${arr(S.arms).map(a => `<b>${esc(a.key)}</b> ${esc(a.name)}`).join(' · ')}. Token figures are medians over the lookups an arm hit: the ten result lines plus the text of the first hit.</p>` : '';
    const BL = arr(S.baselines);
    const blMax = 100;
    const grp = []; for (const b of BL) { const k = b.cls + '|' + b.seeds; let g = grp.find(x => x.k === k); if (!g) grp.push(g = { k, cls: b.cls, seeds: b.seeds, rows: [] }); g.rows.push(b); }
    const blRows = grp.map(g => { const b0 = g.rows[0]; return `<div class="blg"><h5>${clsName(g.cls)} lookups, ${g.seeds === 'all' ? 'every seed' : 'seeds that a search found'}: ${fmt(b0.n_scorable)} scorable of ${fmt(b0.n_dev)} development lookups${has(b0.any_arm_pct) ? ` · any arm hits ${pc(b0.any_arm_pct)}` : ''}</h5>` + g.rows.map(b => `<div class="hbar" style="--lw:330px"><div class="lab"><b>${armN(b.arm)}</b><br><small>${has(b.tokens_first) ? fmt(b.tokens_first) + ' tokens to first hit' : ''}${has(b.tokens_cheapest) ? ' · ' + fmt(b.tokens_cheapest) + ' cheapest' : ''}${has(b.share_of_all_dev) ? ' · ' + pc(b.share_of_all_dev) + ' of all development lookups' : ''}</small></div><div class="track"><div class="fill ${b.arm === 'C' ? 'hl' : 'g-index'}" style="width:${Math.max(0.3, +b.hit10_pct).toFixed(2)}%"></div><div class="val" style="left:${Math.min(+b.hit10_pct, 62).toFixed(2)}%">${pc(b.hit10_pct)} in top 10${has(b.carried_weighted_pct) ? ' · ' + pc(b.carried_weighted_pct) + ' by carried tokens' : ''}</div></div></div>`).join('') + '</div>'; }).join('');
    const rn = grab(p, /Reading notes: (.*?\(`tie_at_cut_pct`\)\.)/), ep = grab(p, /(These shares differ from use-cases\.md table 3\.1 .*?"what a test instance asks"\.)/);
    const readNotes = (rn || ep) ? `<p class="note"><b>Reading notes (from the NOTES file).</b> ${rn ? e(rn[1].replace(/\s*\(`?tie_at_cut_pct`?\)/, '').replace(/^path/, 'Path')) : ''} A hit is an overlap with the span the agent read, not the whole needed span.${ep ? ' <b>Why the shares per form differ from section B:</b> ' + e(ep[1]) : ''}</p>` : '';
    const TB = arr(S.to_beat);
    const toks = TB.flatMap(r => [r.tokens_first, r.tokens_cheapest, r.truth_tokens, r.agent_fetched]).filter(v => has(v) && +v > 0).map(Number);
    let tbChart = '';
    if (TB.length) {
      const lo = toks.length ? Math.floor(Math.log10(Math.min(...toks))) : 1, hi = toks.length ? Math.ceil(Math.log10(Math.max(...toks))) : 6;
      const lx = v => (100 * (Math.log10(v) - lo) / Math.max(1, hi - lo)).toFixed(2);
      const ax = [...Array(hi - lo + 1)].map((_, i) => `<span class="cmp" style="left:${(100 * i / Math.max(1, hi - lo)).toFixed(2)}%">${num(Math.pow(10, lo + i))}</span>`).join('');
      const MK = [['tokens_first', 'tk-first', 'best arm: tokens to its first hit'], ['tokens_cheapest', 'tk-cheap', 'best arm: tokens to its cheapest hit'], ['truth_tokens', 'tk-truth', 'the span the agent read (the used span where confirmed)'], ['agent_fetched', 'tk-agent', 'what the agent itself fetched']];
      const mark = (r, [k, c]) => has(r[k]) && +r[k] > 0 ? `<i class="tk ${c}" style="left:${lx(+r[k])}%" title="${esc(k)}"></i>` : '';
      const row = r => `<div class="tbrow"><div class="lab"><b>${formName(r)}</b><small>${pc(r.share_pct)} of ${clsName(r.cls)} carried tokens · ${fmt(r.n)} scorable · best arm ${armS(r.best_arm)}</small></div>
<div class="track"><div class="fill ${r.best_arm === 'C' ? 'hl' : 'g-index'}" style="width:${Math.max(0.3, +r.hit_pct || 0).toFixed(2)}%"></div><div class="val" style="left:${Math.min(+r.hit_pct || 0, 66).toFixed(2)}%">${pc(r.hit_pct)}</div></div>
<div class="tktrack">${MK.map(mm => mark(r, mm)).join('')}</div>
<div class="tkv">${MK.map(([k, c]) => has(r[k]) ? `<span><i class="tk ${c}"></i>${fmt(r[k])}</span>` : '').join('')}</div></div>`;
      const legend = `<div class="tklegend">${MK.map(([, c, l]) => `<span><i class="tk ${c}"></i>${l}</span>`).join('')} <span class="small">(token axis logarithmic; medians)</span></div>`;
      const classes = [...new Set(TB.map(r => r.cls))];
      tbChart = classes.map(cl => `<div class="tb tbcls"><h5 class="tbsub">${clsName(cl)} lookups (${TB.filter(r => r.cls === cl).length} question forms)</h5>${legend}<div class="tbrow head"><div class="lab">Question form</div><div>Hit rate of the best index-only arm (0 to 100 %)</div><div class="tkaxis">${ax}</div><div>median tokens, as marked</div></div>${TB.filter(r => r.cls === cl).map(row).join('')}</div>`).join('');
    }
    return `<div class="card meas-block" style="margin-top:14px"><h3>The arm to beat: an index with no graph</h3><div class="tagline">${seeOf(p)} Step 1 of ruling R-7 already run for its baselines, on the development part of the test set.</div>
${blRows ? `<h4>Index-only arms, top-10 hit rate on scorable lookups</h4>${armLegend}${blRows}` : ''}${readNotes}
${tbChart ? `<h4>Per question form: the hit rate a graph must beat, and the tokens it must beat it in</h4>${tbChart}` : ''}
${S.to_beat_caveat ? `<p class="note"><b>Caveat.</b> ${e(S.to_beat_caveat)}</p>` : ''}
${fold('What this measurement could not measure (' + arr(S.could_not_measure).length + ')', list(S.could_not_measure))}</div>`;
  }

  /* ---------- premise measure: section E ---------- */
  function pmE(p) {
    const S = p.structured; if (!S) return '';
    const fx = grab(p, /Rounding only: transcripts ([\d.]+) % \(not ([\d.]+) %\), other outside sources ([\d.]+) % \(not ([\d.]+) %\)/);
    const RL = arr(S.reads_by_location).map(r => { const o = Object.assign({}, r);
      if (fx && /^transcripts/.test(r.where) && String(r.calls_pct) === fx[2]) { o.calls_pct = fx[1]; o.fixed = fx[2]; }
      if (fx && /^other outside/.test(r.where) && String(r.calls_pct) === fx[4]) { o.calls_pct = fx[3]; o.fixed = fx[4]; }
      return o; });
    const LC = ['pm0', 'pm1', 'pm2', 'pm3', 'pm4', 'pm5', 'pm6'];
    const stack = RL.length ? `<div class="stack pmstack">${RL.map((r, i) => `<div class="${LC[i % LC.length]}" style="width:${Math.max(0, +r.carried_pct).toFixed(2)}%" title="${esc(r.where)}: ${pc(r.carried_pct)}">${+r.carried_pct >= 8 ? pc(r.carried_pct) : ''}</div>`).join('')}</div>
<div class="glegend pmlist">${RL.map((r, i) => `<span><i class="sw ${LC[i % LC.length]}"></i>${e(r.where)}: <b>${pc(r.carried_pct)}</b> carried · ${pc(r.fetched_pct)} fetched · ${pc(r.calls_pct)} of calls${r.fixed ? ` (corrected in the NOTES recomputation, not ${r.fixed} %)` : ''}</span>`).join('')}</div>` : '';
    const ST = S.same_task || {};
    const stRows = Object.entries(ST).map(([k, v]) => { const o = typeof v === 'object' && v ? v : { carried_pct: v };
      const val = o.carried_pct ?? o.pct ?? o.share_pct; const SLAB = { strict_same_transcript: 'the same agent transcript wrote the file', same_session_24h: 'the same session (orchestrator and its subagents) wrote it under 24 hours before', loose_any_age: 'the same session wrote it, at any age (upper bound)' };
      return pbar(esc(o.label || SLAB[k] || k.replace(/_/g, ' ')) + (has(o.calls_pct) ? `<br><small>${pc(o.calls_pct)} of calls</small>` : ''), val, 'g-docs', 420); }).join('');
    const u1 = grab(p, /counting recorded writes alone, the share is (\d+) % of calls and (\d+) % of carried tokens/), u2 = grab(p, /One session that ran (\d+) hours holds (\d+) % of the calls/);
    const upNote = (u1 || u2) ? `<p class="note"><b>Read the middle bar as an upper figure.</b>${u1 ? ` Counting recorded writes alone, it is ${u1[1]} % of calls and ${u1[2]} % of carried tokens.` : ''}${u2 ? ` One session that ran ${u2[1]} hours holds ${u2[2]} % of the calls, so the session-level figures lean on it.` : ''}</p>` : '';
    const FR = arr(S.findings_rows);
    const xp = v => Math.max(0, Math.min(100, +v)).toFixed(2);
    const dots = FR.length ? `<div class="dotc pmdots"><div class="dotaxis">${[0, 20, 40, 60, 80, 100].map(t => `<span class="cmp" style="left:${t}%">${t} %</span>`).join('')}</div>${FR.map(r => { const w = arr(r.wilson), b = arr(r.bootstrap);
      return `<div class="drow"><div class="dlab">${e(r.label)}<small><b>${fmt(r.k)} of ${fmt(r.n)} = ${pc(r.pct)}</b>${w.length ? ` · Wilson [${w[0]}, ${w[1]}]` : ''}${b.length ? ` · bootstrap by file [${b[0]}, ${b[1]}]` : ' · no bootstrap'}</small></div><div class="dtrack">${w.length ? `<i class="ci" style="left:${xp(w[0])}%;width:${(xp(w[1]) - xp(w[0])).toFixed(2)}%"></i>` : ''}${b.length ? `<i class="ci2" style="left:${xp(b[0])}%;width:${(xp(b[1]) - xp(b[0])).toFixed(2)}%"></i>` : ''}<i class="pt" style="left:${xp(r.pct)}%"></i></div></div>`; }).join('')}<div class="dfoot"><span>0 %</span><span>share of the sampled findings; thick bar Wilson 95 %, thin line bootstrap over files 95 %</span><span>100 %</span></div></div>` : '';
    const PF = S.premises_per_finding;
    const pf = !PF ? '' : typeof PF === 'object' ? `<p class="small"><b>Premises per finding:</b> ${kv(PF).map(([k, v]) => `${esc({ q1: 'lower quartile', q3: 'upper quartile' }[k] || k.replace(/_/g, ' '))} ${fmt(v)}`).join(' · ')}</p>` : `<p class="small"><b>Premises per finding:</b> ${fmt(PF)}</p>`;
    const RU = S.reuse || {}, SL = S.stale || {};
    const known = (o, ks) => ks.every(k => has(o[k]));
    const ruTxt = known(RU, ['found', 'sampled']) ? `${fmt(RU.found)} of ${fmt(RU.sampled)} sampled findings that appear in the transcripts were shown again in another session${has(RU.two_or_more) ? `, ${fmt(RU.two_or_more)} in two or more` : ''}${has(RU.median_gap_h) ? `; median gap ${fmt(RU.median_gap_h)} hours` : ''}.${known(RU, ['files_two_plus', 'files_total']) ? ` ${fmt(RU.files_two_plus)} of ${fmt(RU.files_total)} files were read in two or more sessions${has(RU.files_carried_pct) ? `, carrying ${pc(RU.files_carried_pct)} of the carried tokens` : ''}.` : ''} An exposure shows the text was shown, not that it was used.` : kv(RU).map(([k, v]) => `${esc(k.replace(/_/g, ' '))} ${fmt(v)}`).join(' · ');
    const slTxt = known(SL, ['stale', 'traced']) ? `Of ${fmt(SL.traced)} findings traced through the git history, ${fmt(SL.stale)} had gone stale${has(SL.drifted) ? ` and ${fmt(SL.drifted)} drifted` : ''}.${has(SL.exposures) ? ` The stale ones were shown again ${fmt(SL.exposures)} times after the change${has(SL.pairs) ? `, in ${fmt(SL.pairs)} session-finding pairs` : ''}.` : ''}${has(SL.wrote_after) ? ` Times the stale sentence was written into a file afterwards: ${fmt(SL.wrote_after)}. Whether a decision rested on it was not established.` : ''}` : kv(SL).map(([k, v]) => `${esc(k.replace(/_/g, ' '))} ${fmt(v)}`).join(' · ');
    const loopM = (ruTxt || slTxt) ? `<div class="loopm"><div class="lmc"><b>On the loop, step 1 → reuse: are recorded findings read again?</b><span>${ruTxt}</span></div><div class="lmc"><b>On the loop, steps 2 to 4 → stale: did a changed premise matter?</b><span>${slTxt}</span></div></div>` : '';
    return `<div class="slot filled meas-block"><div class="slh">${seeOf(p)} <b>Measured.</b> ${e(S.what_it_measures || 'Share of earlier conclusions whose premises are inside the repository.')}</div>
${p.head ? `<p>${esc(p.head)}</p>` : ''}
${stack ? `<h4>Where the reads of earlier conclusions come from, by carried tokens</h4>${stack}` : ''}
${stRows ? `<h4>How much of that is the same task re-reading its own notes (by carried tokens, 0 to 100 %)</h4>${stRows}${upNote}` : ''}
${dots ? `<h4>What the sampled findings rest on</h4>${dots}` : ''}${pf}
${loopM}
${arr(S.step4).length ? `<h4>What this changes in the design</h4>${list(S.step4)}` : ''}
${S.not_claimed ? `<p class="note"><b>Not claimed.</b> ${e(Array.isArray(S.not_claimed) ? S.not_claimed.join(' ') : S.not_claimed)}</p>` : ''}
${fold('What this measurement could not measure (' + arr(S.could_not_measure).length + ')', list(S.could_not_measure))}</div>`;
  }
  /* ---------- embeddings trial: section G ---------- */
  // values printed at 4 decimals as the coordinator's report prints them; axis ticks carry class "cmp"
  const r4 = x => { if (typeof x !== 'number' || !isFinite(x)) throw new Error('embeddings: not a number ' + x); return (Math.round(Math.abs(x) * 1e4 + 1e-9) / 1e4).toFixed(4); };
  const s4 = x => (+r4(x) === 0 ? '' : x > 0 ? '+' : '−') + r4(x);
  const iv = o => `${s4(o.diff)} [${s4(o.lo)}, ${s4(o.hi)}]`;
  const WORD = w => w === 'cannot tell (lower)' ? 'lower' : w;
  const wtag = w => `<span class="ew w-${esc(WORD(w).replace(/[^a-z]+/g, '-'))}">${esc(WORD(w))}</span>`;
  function embNames(S) { const N = Object.fromEntries(arr(S.arms).map(a => [a.key, a.name])); return t => String(t)
    .replace(/\bE1 to E4\b/g, `${N.E1}, ${N.E2}, ${N.E3} and ${N.E4}`).replace(/^E5\.$/, `The ${N.E5}.`).replace(/\bE5's\b/g, `the ${N.E5}'`)
    .replace(/(^|[.!?] )E5\b/g, (m, a) => a + 'The ' + N.E5).replace(/\bE5\b/g, 'the ' + N.E5).replace(/\bE([1-4])\b/g, (m, k) => N['E' + k])
    .replace(/\bPPR\b/g, 'rooted PageRank').replace(/\bCO\b/g, 'co-change confidence').replace(/\bRRF\b/g, 'the simple ranker').replace(/P@10/g, 'top-ten precision').replace(/(^|[.!?] )(rooted PageRank|co-change confidence|the simple ranker)/g, (m, a, b) => a + b[0].toUpperCase() + b.slice(1)); }
  const bnd = v => (v < 0 ? '−' : '+') + Math.abs(v);
  const DESC = { E1: 'knowledge-graph embedding', E2: 'knowledge-graph embedding', E3: 'knowledge-graph embedding', E4: 'graph-neural encoder', E5: 'RotatE specialists, one per area, a router picks two', PLACEBO: 'a random score fused the same way; checks the harness' };
  function embR(p) {
    const S = p.structured; if (!S) return '';
    const nm = embNames(S), te = t => e(nm(t));
    const A = arr(S.arms), real = A.filter(a => a.key !== 'PLACEBO');
    const V = [['fused_pool', 'Fused, pool (the primary)', 'Fused with the simple ranker, inside the candidate pool: the primary contrast'], ['alone_pool', 'Alone, pool', 'The arm alone, inside the candidate pool'], ['fused_unrestricted', 'Fused, all code files', 'Fused with the simple ranker, ranked over all code files'], ['alone_unrestricted', 'Alone, all code files', 'The arm alone, ranked over all code files']];
    // one x range for all four views: the real arms' intervals and the adds bar, in steps of 0.01 (the placebo may be clipped)
    const STEP = 0.01;
    const lo = Math.floor(Math.min(...real.flatMap(a => V.map(([k]) => a[k].lo))) / STEP - 1e-9) * STEP;
    const hi = (Math.ceil(Math.max(S.adds_bar, ...real.flatMap(a => V.map(([k]) => a[k].hi))) / STEP - 1e-9) + 1) * STEP;
    const xp = v => (100 * (Math.max(lo, Math.min(hi, v)) - lo) / (hi - lo)).toFixed(2);
    const B0 = S.match_band[0], B1 = S.match_band[1];
    const ticks = []; for (let t = Math.round(lo / STEP); t <= Math.round(hi / STEP); t++) ticks.push(t);
    const bg = (h) => `<rect x="${xp(B0)}%" y="0" width="${(xp(B1) - xp(B0)).toFixed(2)}%" height="${h}" class="efband"/>${ticks.map(t => `<line x1="${xp(t * STEP)}%" x2="${xp(t * STEP)}%" y1="0" y2="${h}" class="efgrid"/>`).join('')}<line x1="${xp(0)}%" x2="${xp(0)}%" y1="0" y2="${h}" class="efzero"/><line x1="${xp(S.adds_bar)}%" x2="${xp(S.adds_bar)}%" y1="0" y2="${h}" class="efbar"/>`;
    const axis = `<svg class="efsvg" width="100%" height="64" aria-hidden="true">${bg(64)}<text x="${((+xp(B0) + +xp(B1)) / 2).toFixed(2)}%" y="17" text-anchor="middle" class="eflab">matches band</text><text x="${xp(S.adds_bar)}%" dx="-5" y="37" text-anchor="end" class="eflab efbarl">adds bar</text>${ticks.filter(t => t % 2 === 0).map(t => `<text x="${xp(t * STEP)}%" y="58" text-anchor="middle" class="cmp eftick">${t === 0 ? '0' : (t > 0 ? '+' : '−') + (Math.abs(t) * STEP).toFixed(2)}</text>`).join('')}</svg>`;
    const row = (a, k) => { const o = a[k]; const cl = o.lo < lo, ch = o.hi > hi, dOff = o.diff < lo || o.diff > hi;
      const cls = WORD(o.word) === 'lower' ? 'eflow' : WORD(o.word) === 'adds' ? 'efadd' : 'efmat';
      const svg = `<svg class="efsvg" width="100%" height="100%" role="img" aria-label="${esc(a.name)}: ${esc(iv(o))}, ${esc(WORD(o.word))}">${bg('100%')}
<line x1="${xp(o.lo)}%" x2="${xp(o.hi)}%" y1="50%" y2="50%" class="efci ${cls}"/>${cl ? `<text x="0%" y="50%" dy="5" text-anchor="start" class="efarr ${cls}f">◀</text>` : ''}${ch ? `<text x="100%" y="50%" dy="5" text-anchor="end" class="efarr ${cls}f">▶</text>` : ''}${dOff ? `<text x="${o.diff < lo ? '2.2%' : '97.8%'}" y="50%" dy="-8" text-anchor="${o.diff < lo ? 'start' : 'end'}" class="efoff">point ${s4(o.diff)}, off the scale</text>` : `<circle cx="${xp(o.diff)}%" cy="50%" r="6.5" class="efpt ${cls}f"/>`}</svg>`;
      return `<div class="efrow${a.key === 'PLACEBO' ? ' efplac' : ''}"><div class="eflabc"><b>${esc(a.name)}</b><small>${esc(DESC[a.key] || '')}</small></div><div class="efplot">${svg}</div><div class="efval"><b>${iv(o)}</b> ${wtag(o.word)}<small>scores ${r4(a.level[k])}</small></div></div>`; };
    const views = V.map(([k, , title], vi) => `<div class="efview" data-v="${k}"${vi ? ' hidden' : ''}><h5>${esc(title)}</h5><div class="efrow efaxis"><div class="eflabc"></div><div class="efplot">${axis}</div><div class="efval small">difference [95 % interval], word</div></div>${A.map(a => row(a, k)).join('')}<div class="efrow effoot"><div></div><div class="efplot"><span>← lower than the simple ranker</span><span>difference in top-ten precision</span><span>higher →</span></div><div></div></div></div>`).join('');
    const C = S.comparators, cp = C.pool, cu = C.unrestricted;
    const levels = `<p class="small">The simple ranker (rooted PageRank fused with co-change confidence) scores <b>${r4(cp.RRF)}</b> in the pool and <b>${r4(cu.RRF)}</b> over all code files. In the pool, rooted PageRank alone scores ${r4(cp.PPR)}, co-change confidence alone ${r4(cp.CO)}, a same-folder rule ${r4(cp.DIR)} and a random order ${r4(cp.RANDOM)}.</p>`;
    const E1 = A.find(a => a.key === 'E1');
    const pil = `<p class="small"><b>${esc(E1.name)} on the ${num(S.pilot_n)} commits the pilot's set also holds:</b> fused, pool ${iv(S.pilot_shared)}${S.pilot_shared.lo > 0 ? ', wholly above zero' : ''}${S.pilot_shared.diff < S.adds_bar ? `, still under the +${S.adds_bar} bar` : ''}.</p>`;
    const fa = real.map(a => a.answer_level.fused), allIn = real.every(a => a.answer_level.fused_vs.lo < 0 && a.answer_level.fused_vs.hi > 0);
    const ans = `<p class="small"><b>Answer level</b> (how much of the reference answer fits in ${num(S.answer_budget)} tokens; the protocol gives no word there): the simple ranker ${r4(S.answer_level_rrf)}, the fused arms ${r4(Math.min(...fa))} to ${r4(Math.max(...fa))}${allIn ? ', each difference with an interval that includes zero' : ''}; ${esc(E1.name)} fused ${iv(E1.answer_level.fused_vs)}.</p>`;
    const seg = `<div class="seg" role="group" aria-label="Contrast" data-ctl="embv">${V.map(([k, l], i) => `<button type="button" data-v="${k}" aria-pressed="${i === 0}">${esc(l)}</button>`).join('')}</div>`;
    const how = `<p class="efhow"><b>How to read it.</b> Each row is one arm: the line is its 95 % interval, the dot its difference to the simple ranker in top-ten precision; the shaded band is ${bnd(B0)} to ${bnd(B1)}, the dashed line the bar of ${bnd(S.adds_bar)}. An arm "adds" when its interval lies above zero and its gain reaches the bar; it "matches" when the whole interval lies inside the band, which means within ${S.adds_bar}, not "no difference"; "lower" when the interval lies wholly below zero.</p>`;
    const why = `<ol class="efwhy">${arr(S.why).map(w => `<li><b>${te(w.lead)}</b> ${te(w.text)}</li>`).join('')}</ol>`;
    const H = S.heldout, HV = H.vs || {};
    const hcell = (r, k) => `${r4(r[k])}${HV[r.key] ? `<br><small>${iv(HV[r.key][k])}</small>` : ''}`;
    const htab = `<div class="tblwrap"><table class="eftab"><thead><tr><th>Ranker</th><th>Pool, top-ten precision</th><th>All code files, top-ten precision</th><th>Answer level</th></tr></thead><tbody>${arr(H.rows).map(r => `<tr${r.key === 'RRF' ? ' class="efref"' : ''}><td><b>${esc(r.name)}</b></td><td>${hcell(r, 'pool')}</td><td>${hcell(r, 'unrestricted')}</td><td>${hcell(r, 'answer_level')}</td></tr>`).join('')}</tbody></table></div>`;
    const Fz = H.frozen;
    const lim = `<ol class="eflim">${arr(S.limits).map(t => `<li>${te(t)}</li>`).join('')}</ol>`;
    return `<div class="slot filled"><div class="meas-block efblock"><div class="slh">${tag('meas', esc(S.report_path))} <b>Measured, judged and audited</b> (report of ${esc(S.date)}; figures read from the lane's tuning report and the grader's returns).</div>
<h3 class="efq">Do learned graph embeddings add to the simple ranker?</h3>
<p class="efans"><b>${te(S.answer_bold)}</b></p>${arr(S.answer_paragraphs).map(t => `<p>${te(t)}</p>`).join('')}
<h4>Difference to the simple ranker, per arm (the later ${num(S.late_n)} tuning commits)</h4>${seg}<div class="efchart">${views}</div>
${how}${levels}${pil}${ans}
<h4>Why: six reasons from the audit</h4><p class="small">${te(S.why_sources)}</p>${why}<p class="note"><b>The sentence the evidence supports:</b> ${te(S.why_summary)}</p>
<h4>Held-out: ${esc(H.heading.replace(/; one run of the frozen arm; described, not judged/, ''))}</h4>
<p class="small">One run of the frozen arm, described and not judged. Frozen before any held-out unit ran: ${esc(nm(Fz.arm))}, setting ${esc(Fz.config_id)} (${num(Fz.dim)} dimensions), weight ${num(Fz.weight)}, ${num(Fz.epochs)} epochs. Under the two RotatE rows: the difference to the simple ranker [95 % interval].</p>${htab}<p class="small">${te(H.note)}</p>
<h4>What the lane's result file lacks or gets wrong (${arr(S.limits).length})</h4><p class="small">${te(S.limits_intro)} Titles only; the full text is in <code>${esc(S.report_path)}</code>, section 6.</p>${lim}
<h4>What it does not say</h4>${list(arr(S.does_not_say).map(nm))}</div></div>`;
  }
  return { ltB, ltI, pmE, embR };

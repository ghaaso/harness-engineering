#!/bin/bash
# Usage: yt-transcript.sh <videoId> <outfile>
# Fetches the YouTube auto/manual transcript as plain text via an ephemeral uvx run (nothing installed).
set -u
ID="$1"; OUT="$2"
uvx --from youtube-transcript-api youtube_transcript_api.exe "$ID" --format text > "$OUT" 2>"$OUT.err"
RC=$?
if [ $RC -ne 0 ] || [ ! -s "$OUT" ]; then echo "FAILED rc=$RC"; cat "$OUT.err" | tail -5; exit 1; fi
echo "OK $(wc -w < "$OUT") words -> $OUT"

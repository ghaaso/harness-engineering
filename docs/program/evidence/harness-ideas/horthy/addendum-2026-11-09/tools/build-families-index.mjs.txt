// Writes a compact index of the 838 families (fixed; never re-clustered) for the addendum's mapping stage.
// Usage: node build-families-index.mjs <review/families.json> <out.md>
import fs from 'node:fs';
const [inp, out] = process.argv.slice(2);
const fams = JSON.parse(fs.readFileSync(inp, 'utf8'));
const G = { G1: 'W1 + L2 — CLAUDE.md length, context budget, compaction', G2: 'W2 + W3 — routine compliance, hooks, cold resume', G3: 'W4 + W5 — agents, skills, workflow library', G4: 'L1 + L3 — effort routing, verifiers, backpressure, benchmarks', G5: 'none / cross-cutting — software-factory method, product, loops, process' };
const lines = ['# The 838 claim families of the first pass (fixed ids; do not merge, split or rename them)', '', 'Format: `id | final disposition | bears on | cards | claim`', ''];
for (const g of Object.keys(G)) {
  const l = fams.filter(f => f.group === g);
  lines.push('## ' + g + ' — ' + G[g] + ' (' + l.length + ' families)', '');
  for (const f of l) lines.push(f.id + ' | ' + f.final_disposition + ' | ' + ((f.bears_on || []).join(',') || 'none') + ' | ' + f.members.length + ' | ' + String(f.label).replace(/\s+/g, ' ').trim());
  lines.push('');
}
fs.writeFileSync(out, lines.join('\n'));
console.log('families', fams.length, 'bytes', lines.join('\n').length);

// Renders the timeline JSON into reader files: one text file per date window with every Dex-authored post
// (text, quoted post, reply target, article body). Reposts of other people are listed by id only.
// Usage: node x-render-timeline.mjs <timeline.json> <outdir> <screen_name>
import fs from 'node:fs';
import path from 'node:path';
const [inp, outDir, user] = process.argv.slice(2);
const all = JSON.parse(fs.readFileSync(inp, 'utf8'));
fs.mkdirSync(outDir, { recursive: true });
const ts = s => (s.created_timestamp ? s.created_timestamp * 1000 : new Date(s.created_at).getTime());
const day = s => new Date(ts(s)).toISOString().slice(0, 10);
const own = all.filter(s => ((s.author && s.author.screen_name) || '').toLowerCase() === user.toLowerCase()).sort((a, b) => ts(a) - ts(b));
const others = all.filter(s => ((s.author && s.author.screen_name) || '').toLowerCase() !== user.toLowerCase());
function articleText(a) {
  const blocks = (a.content && a.content.blocks) || [];
  const out = ['=== ARTICLE: ' + a.title + ' (created ' + a.created_at + ', modified ' + a.modified_at + ') ==='];
  for (const b of blocks) { const ty = b.type || ''; let p = ''; if (ty.startsWith('header')) p = '## '; else if (ty.includes('unordered')) p = '- '; else if (ty.includes('ordered')) p = '1. '; else if (ty === 'blockquote') p = '> '; out.push(p + (b.text || '')); }
  return out.join('\n');
}
function render(s) {
  const L = [];
  L.push('----- POST ' + s.id + ' | ' + day(s) + ' | https://x.com/' + user + '/status/' + s.id + ' | replies ' + s.replies + ' likes ' + s.likes + ' views ' + s.views);
  if (s.replying_to) L.push('(reply to @' + s.replying_to + (s.replying_to_status ? ' status ' + s.replying_to_status : '') + ')');
  L.push(s.text || (s.raw_text && s.raw_text.text) || '');
  if (s.quote) L.push('QUOTED @' + ((s.quote.author && s.quote.author.screen_name) || '?') + ' (' + s.quote.id + '): ' + (s.quote.text || ''));
  if (s.article) L.push(articleText(s.article));
  if (s.media && s.media.all && s.media.all.length) L.push('[media: ' + s.media.all.map(m => m.type + (m.altText ? ' alt="' + m.altText + '"' : '')).join('; ') + ']');
  return L.join('\n');
}
const windows = [['2026-09-01', '2026-09-30'], ['2026-10-01', '2026-10-14'], ['2026-10-15', '2026-10-31'], ['2026-11-01', '2026-11-30']];
const index = [];
for (const [a, b] of windows) {
  const list = own.filter(s => day(s) >= a && day(s) <= b);
  if (!list.length) continue;
  const file = path.join(outDir, 'x-timeline-' + a + '_' + b + '.txt');
  const head = 'SOURCE: @' + user + ' timeline via https://api.fxtwitter.com/2/profile/' + user + '/statuses (head posts and self-authored replies the endpoint returns; reply threads under each post are NOT included; ' + list.length + ' posts, ' + a + ' to ' + b + ')\n\n';
  fs.writeFileSync(file, head + list.map(render).join('\n\n') + '\n');
  const words = list.map(render).join(' ').split(/\s+/).length;
  index.push({ file: file.replace(/\\/g, '/'), from: a, to: b, posts: list.length, words, ids: list.map(s => s.id) });
  console.log(path.basename(file), 'posts', list.length, 'words', words);
}
fs.writeFileSync(path.join(outDir, 'x-timeline-index.json'), JSON.stringify({ own: own.length, reposts_of_others: others.map(s => ({ id: s.id, author: s.author && s.author.screen_name, date: day(s) })), windows: index }, null, 1));
const arts = own.filter(s => s.article).map(s => day(s) + ' ' + s.id + ' ' + s.article.title);
console.log('articles:', arts.join(' | '));

// after.js.txt — do sessions make more mistakes after a compaction than before one (FixFlow and the clinic testbed).
// Record: docs/program/AFTER-COMPACTION-2026-11.md (part 1 is the scoring this script implements).
// Makes no model calls. Reads main-session transcripts of the two project groups only.
//
// Run from this folder (scripts are saved as .js.txt, so they run on stdin):
//   node - survey                      < after.js.txt   what ran before the scoring was fixed: nothing by position except calls between boundaries
//   node - index  <cacheDir>           < after.js.txt   needs <cacheDir>/index.json from the summary check's reader (see README); writes <cacheDir>/calls.json
//   node - sample <cacheDir>           < after.js.txt   picks the hand-read sample by hash; writes <cacheDir>/hand-read.txt (blind) and audit-skeleton.json
//   node - count  <cacheDir> [outDir]  < after.js.txt   windows, signals, placebo, resampling -> after-table.txt, after.json (reads <outDir>/audit.json)
//
// <cacheDir> holds commands and tool results (calls.json, hand-read.txt). Keep it OUTSIDE the repository and delete it
// when done. Nothing written to [outDir] holds a sentence from a message, a summary or a tool result, or a command.
//
// The reader is the summary check's (docs/program/evidence/summary-check/check.js.txt): its boundaries, input ends,
// call classes and fingerprints are used as they are; this script adds the call stream, the windows and the signals.

}

// ---------- the kinds of error result (scoring 1.3): the first that fits, in this order ----------
const P = {
  ended: /^\s*\[Tool call interrupted: the session ended/,
  ownerRefused: /The user doesn't want to (?:proceed|take this action)|\[Request interrupted by user/,
  harnessRefused: /^(?:<tool_use_error>)?Blocked: |is blocked\. This path is protected|denied by a built-in|Permission to use .{1,80} has been denied|Start sessions only on|needs the user's own menu|hook error|outside the directories this session has acce|requires approval|was blocked by/m,
  environment: /Command timed out|timed out after|Connection timed out|ETIMEDOUT|EAI_AGAIN|ECONNREFUSED|ECONNRESET|Could not resolve host|is not installed\b|ENAMETOOLONG|did not finish rendering/,
  missingPath: /No such file or directory|cannot access '|Cannot find path|cannot find the (?:file|path) specified|Could not find a part of the path|Could not find file|ENOENT|File does not exist|Directory does not exist|Path does not exist/,
  missingCommand: /command not found|is not recognized as (?:an internal|the name of a cmdlet|a name of a cmdlet)|: not found\s*$|Python was not found|No such tool available/m,
  missingRef: /unknown revision or path not in the working tree|bad revision|[Nn]ot a valid object name|pathspec '.*' did not match|did not match any file|couldn't find remote ref|not a git repository|Could not resolve to a (?:PullRequest|Repository|node)|no pull requests found|invalid reference|bad object|does not point to a valid object|exists on disk, but not in|is not a commit|no such ref|not something we can merge|no such remote|unknown ref|Needed a single revision|Could not bind/,
  shellSyntax: /syntax error near unexpected token|syntax error: unexpected|unexpected EOF while looking for matching|unterminated quoted string|ParserError|ParseException|The string is missing the terminator|Missing closing '[)}]'|Unexpected token '.{1,40}' in expression or statement|Missing expression after/,
  badArgs: /unknown option|invalid option|unrecognized (?:option|argument)|unrecognized arguments|unknown (?:flag|switch|command|subcommand)|illegal option|is not a git command|unknown shorthand flag|A parameter cannot be found that matches parameter name|no such option|InputValidationError|^Pass \{|items also needs|Invalid workflow script|^usage: /im,
  scriptCrash: /Traceback \(most recent call last\)|^Node\.js v\d+|^\s*(?:Syntax|Reference|Type|Range)Error\b|^\w*Error: |Uncaught|Unhandled(?:PromiseRejection| exception)|^\s+at .+\(node:|KeyError|AssertionError|Exception in thread|panicked at/m,
};
const EK = ['owner refused', 'harness refused', 'environment', 'missing path', 'missing command', 'missing reference', 'shell syntax', 'bad arguments', 'file tool refused', 'script crashed', 'check failed', 'other'];
function errorKind(c, t) {
  if (P.ownerRefused.test(t)) return 'owner refused';
  if (P.harnessRefused.test(t)) return 'harness refused';
  if (P.environment.test(t)) return 'environment';
  if (P.missingPath.test(t)) return 'missing path';
  if (P.missingCommand.test(t)) return 'missing command';
  if (P.missingRef.test(t)) return 'missing reference';
  if (c.shell && P.shellSyntax.test(t)) return 'shell syntax';
  if (P.badArgs.test(t)) return 'bad arguments';
  if (/^(Read|Edit|Write|NotebookEdit|MultiEdit)$/.test(c.name)) return 'file tool refused';
  if (c.k !== 'test or check run' && P.scriptCrash.test(t)) return 'script crashed';
  if (c.k === 'test or check run') return 'check failed';
  return 'other';
}
// the same diagnostics on a line of a result that is not flagged as an error (scoring 1.3)
const NF = {
  'missing path': /^\S+: .*(?:No such file or directory|cannot access)|^[\w\-]+ : Cannot find path/m,
  'missing command': /^\S+: .*command not found|^[\w\-]+ : The term '|^Python was not found/m,
  'missing reference': /^fatal: .*(?:unknown revision|[Nn]ot a valid object name|bad revision|not a git repository|Needed a single revision)|^error: pathspec '/m,
};
const notFlagged = (t) => { for (const k of Object.keys(NF)) if (NF[k].test(t)) return k; return null; };
// the subject of an error, for "the same error met again" (A1)
function subjectOf(kind, c, t) {
  const first = (...res) => { for (const re of res) { const m = re.exec(t); if (m) return m[1]; } return null; };
  if (kind === 'missing path') {
    if (!c.shell && c.fpath) return R.normPath(c.fpath);
    const s = first(/cannot access '([^'\n]+)'/, /can't read ([^:\n]+): No such file/, /cannot open '([^'\n]+)'/, /cannot change to '([^'\n]+)'/, /Cannot find path '([^'\n]+)'/, /\[Errno 2\] No such file or directory: '([^'\n]+)'/, /no such file or directory,? (?:open|stat|scandir|lstat|access|unlink|rmdir|mkdir)? ?'([^'\n]+)'/i, /^\S+: (?:line \d+: )?(?:cd: )?([^:\n]+): No such file or directory/m);
    return s ? R.normPath(s.trim()) : '?';
  }
  if (kind === 'missing command') { if (/Python was not found/.test(t)) return 'python'; const s = first(/^(?:\S+: )?(?:line \d+: )?([\w.\-]+): command not found/m, /'([^'\n]+)' is not recognized/, /The term '([^'\n]+)'/, /No such tool available: (\S+)/); return s ? s.toLowerCase() : '?'; }
  if (kind === 'missing reference') { if (/not a git repository/.test(t)) return 'not a repository'; const s = first(/ambiguous argument '([^'\n]+)'/, /pathspec '([^'\n]+)'/, /bad revision '([^'\n]+)'/, /[Nn]ot a valid object name:? '?([^\s'\n]+)/, /couldn't find remote ref (\S+)/, /PullRequest with the number of (\d+)/); return s || '?'; }
  if (kind === 'harness refused') return /Blocked: (?:sleep|Start-Sleep)/.test(t) ? 'wait guard' : /path is protected/.test(t) ? 'protected path' : /built-in/.test(t) ? 'safety check' : /Start sessions only on/.test(t) ? 'session rule' : /hook error|outside the directories/.test(t) ? 'hook' : /own menu/.test(t) ? 'owner menu' : 'other guard';
  if (kind === 'file tool refused') return (/String to replace not found/.test(t) ? 'text not found' : /Found \d+ matches/.test(t) ? 'not unique' : /modified since read/.test(t) ? 'changed since read' : /has not been read yet/.test(t) ? 'not read yet' : /exceeds maximum allowed/.test(t) ? 'too large' : /EISDIR|is a directory/i.test(t) ? 'is a directory' : 'other') + '|' + R.normPath(c.fpath || '');
  return null;
}
// a finer label for the kinds that count in M, for the breakdown added after the first count (table section J)
function subKind(c, t) {
  const k = c.ekind;
  if (k === 'missing command') return /Python was not found/.test(t) || /^(python3?|py)$/.test(c.subj || '') ? 'python, which is not on this PC\'s PATH' : 'another program or tool';
  if (k === 'shell syntax') return /unexpected EOF while looking for matching/.test(t) ? 'a command cut short or a quote left open' : 'another parse error';
  if (k === 'script crashed') return /AssertionError|assert [^\n]{0,80}count\(|JS anchor|anchor: |Error\(["'`]missing|expected \d+ of/.test(t) ? 'the text it was to replace was not in the file' : /SyntaxError|Invalid regular expression|bad escape|Unterminated|Unexpected token/.test(t) ? 'the script itself did not parse' : /ModuleNotFoundError|MODULE_NOT_FOUND|Cannot find module/.test(t) ? 'a module it could not load' : 'another crash';
  if (k === 'file tool refused') return String(c.subj || '').split('|')[0] || 'other';
  if (k === 'harness refused') return c.subj || 'other guard';
  if (k === 'bad arguments') return c.shell ? 'a program\'s options' : /update_plan/.test(c.name) ? 'the plan tool' : c.name === 'Monitor' ? 'the Monitor tool' : 'another tool';
  return '';
}
const A1_FAMILY = new Set(['missing path', 'missing command', 'missing reference', 'file tool refused', 'harness refused']);
function isNoop(k, t) {
  return (/git commit/.test(k) && /nothing to commit|nothing added to commit|no changes added to commit/.test(t)) || (/git push/.test(k) && /Everything up-to-date/.test(t)) || (/git merge/.test(k) && /Already up to date/.test(t))
    || (/gh pr create/.test(k) && /already exists/.test(t)) || (/gh pr merge/.test(k) && /already merged/.test(t)) || (/git cherry-pick/.test(k) && /previous cherry-pick is now empty|nothing to commit/.test(t));
}
// options before the verb are dropped first, as the reader does in classifyCall
const flatCmd = (c) => (c.length > 20000 ? c.slice(0, 20000) : c).replace(/\s-[cC]\s+(?:"[^"\n]{0,300}"|'[^'\n]{0,300}'|[^\s"'=]{1,200}(?:=(?:"[^"\n]{0,300}"|'[^'\n]{0,300}'|[^\s"']{0,300}))?)/g, ' ');
const DISCARD = /\bgit\s+(?:checkout|restore)\b[^;&|\n]*\s--\s+\S|\bgit\s+restore\s+(?!--staged\b)[^;&|\n-]*\S|\bgit\s+reset\s+--hard\b|\bgit\s+revert\b|\bgit\s+stash\b(?!\s+(?:list|show|pop|apply|drop|branch))|\bgit\s+clean\s+-[a-zA-Z]*f/;
const checkKey = (cmd) => sha1(cmd.replace(/\s+/g, ' ').replace(/\s*\|\s*(tail|head)\b.*$/, '').trim().toLowerCase()).slice(0, 12);   // the reader's normalisation of a check command
// the files a call reads, each with the lines it reads (scoring 1.3 A3 and addendum 1.11 A3s):
// 'whole', 'tail', 'unknown', or a list of [first line, last line]
function readItems(name, inp, cmd) {
  if (name === 'Read') { if (!inp.file_path) return []; const off = +inp.offset || 0, lim = +inp.limit || 0; return [{ p: R.normPath(String(inp.file_path)), r: (off || lim) ? [[off || 1, lim ? (off || 1) + lim - 1 : 1e9]] : 'whole' }]; }
  if (name !== 'Bash' && name !== 'PowerShell') return [];
  const out = [], head = cmd.split('<<')[0];                           // nothing after a here-document marker is a command
  for (const seg of head.split(/&&|\|\||[;|\n]/)) {
    const s = seg.trim(), m = /^(cat|head|tail|nl|less|more|bat|type|Get-Content|gc)\b([\s\S]*)$/.exec(s), ms = m ? null : /^sed\s+(?:-[A-Za-z]*n[A-Za-z]*|--quiet|--silent)\b([\s\S]*)$/.exec(s);
    if (!m && !ms) continue;
    const args = m ? m[2] : ms[1], verb = m ? m[1] : 'sed', paths = []; let skip = false, r = verb === 'sed' ? 'unknown' : verb === 'tail' ? 'tail' : verb === 'head' ? [[1, 10]] : 'whole';
    if (verb === 'head') { const n = /(?:^|\s)-n\s*(\d+)|(?:^|\s)-(\d+)(?=\s|$)/.exec(args); if (n) r = [[1, +(n[1] || n[2])]]; }
    if (verb !== 'sed' && verb !== 'head' && verb !== 'tail') { const n = /-(?:TotalCount|First|Head)\s+(\d+)/i.exec(args); if (n) r = [[1, +n[1]]]; else if (/-(?:Tail|Last)\s+\d+/i.test(args)) r = 'tail'; }
    for (let tok of args.split(/\s+/)) {
      if (skip) { skip = false; continue; }
      if (/^\d?>>?$/.test(tok)) { skip = true; continue; }             // the word after a redirect is written, not read
      if (/^\d?>/.test(tok)) continue;
      tok = tok.replace(/^["']|["']$/g, '');
      if (verb === 'sed' && /^\d+(?:,\d+)?p(?:;\d+(?:,\d+)?p)*$/.test(tok)) { r = tok.split(';').map(x => { const q = /^(\d+)(?:,(\d+))?p$/.exec(x); return [+q[1], +(q[2] || q[1])]; }); continue; }
      if (!tok || tok.startsWith('-') || /[<>*?$`{}]/.test(tok)) continue;
      if (R.EXT_RE.test(tok)) paths.push(R.normPath(tok));
    }
    for (const p of paths) out.push({ p, r });
  }
  return out;
}

// ---------- stage: index (the call stream of every main transcript in scope) ----------
async function readCalls(f, sid) {
  const calls = [], byId = new Map(); let ln = 0, lastOwn = null, lastModel = '';
  const rl = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
  for await (const line of rl) {
    ln++;
    if (line.startsWith('{"type":"api-request') || line.startsWith('{"type":"file-history')) continue;
    let o; try { o = JSON.parse(line); } catch { continue; }
    if (o.isSidechain) continue;
    if (o.timestamp && o.timestamp > CALL_CUT) continue;
    const own = !(o.sessionId && o.sessionId !== sid);
    if (o.type === 'assistant' && o.message) {
      const m = o.message; if (m.model && m.model !== '<synthetic>' && m.usage) lastModel = m.model;
      if (Array.isArray(m.content)) for (const b of m.content) if (b.type === 'tool_use') {
        const inp = b.input || {}, cl = R.classifyCall(b.name, inp), shell = b.name === 'Bash' || b.name === 'PowerShell', cmd = shell ? String(inp.command || '') : '';
        const c = { ln, own, name: b.name, k: cl.k, state: cl.state, shell, model: lastModel, fp: R.fingerprint(b.name, inp), d: R.describeCall(b.name, inp), cmd: cmd.slice(0, 900) };
        const fpath = inp.file_path || inp.notebook_path || ''; if (fpath) c.fpath = String(fpath);
        c.rr = readItems(b.name, inp, cmd); c.rp = [...new Set(c.rr.map(x => x.p))];
        if (b.name === 'Edit') c.ed = { p: R.normPath(String(inp.file_path || '')), oh: sha1(String(inp.old_string || '')), nh: sha1(String(inp.new_string || '')) };
        if (b.name === 'Write') c.wr = { p: R.normPath(String(inp.file_path || '')), ch: sha1(String(inp.content || '')) };
        if (shell && DISCARD.test(flatCmd(cmd))) c.r2 = true;
        if (cl.k === 'test or check run') c.ck = checkKey(cmd);
        calls.push(c); byId.set(b.id, c); if (own) lastOwn = c;
      }
      continue;
    }
    if (o.type === 'user' && o.message) {
      const ct = o.message.content, intr = (t) => { if (own && lastOwn && /^\s*\[Request interrupted/.test(t || '')) lastOwn.interrupt = true; };
      if (typeof ct === 'string') { if (!o.isCompactSummary) intr(ct); continue; }
      if (Array.isArray(ct)) for (const b of ct) {
        if (!b) continue;
        if (b.type === 'text') intr(b.text);
        else if (b.type === 'tool_result') {
          const c = byId.get(b.tool_use_id); if (!c || c.res) continue; c.res = true;
          const t = typeof b.content === 'string' ? b.content : R.collectStrings(b.content, []).join('\n');
          c.rs = t.trim().slice(-200);
          if (P.ended.test(t)) { c.ended = true; continue; }
          c.err = !!b.is_error;
          const note = /\[This command modified \d+ files? you've previously read: ([^\]]+?)\. Call Read before editing\.\]/.exec(t);   // the harness's own note of a file the command changed
          if (note) c.mod = note[1].split(/,\s*/).map(x => x.trim().replace(/\\/g, '/').toLowerCase()).map(x => x.slice(x.lastIndexOf('/') + 1));
          if (c.err) { c.ekind = errorKind(c, t); const s = subjectOf(c.ekind, c, t); if (s) c.subj = s; const sk = subKind(c, t); if (sk) c.sub = sk; c.rh = t.slice(0, 400); c.rt = t.slice(-700); if (c.name === 'Read') { c.rp = []; c.rr = []; } }
          else if (c.shell) { const nf = notFlagged(t); if (nf) { c.nf = nf; c.subj = subjectOf(nf, c, t); const m = NF[nf].exec(t); c.rh = t.slice(Math.max(0, m.index - 200), m.index + 400); c.rt = t.slice(-300); } }
          if (c.shell && c.state && isNoop(c.k, t)) { c.noop = true; if (!c.rt) c.rt = t.slice(-500); }
          if (c.ck) c.chk = (c.err || R.testCounts(t).some(x => /^failed\|(?!0+$)/.test(x))) ? 'fail' : 'pass';
        }
      }
    }
  }
  // signals that look back over the session (scoring 1.3): A1, A2, R1, R3
  const seenErr = new Map(), lastOk = new Map(), edits = new Map(), lastRun = new Map();
  calls.forEach((c, i) => {
    const fam = c.ekind && A1_FAMILY.has(c.ekind) ? c.ekind : c.nf || null;
    if (fam && c.subj && c.subj !== '?') { const key = fam + '|' + c.subj; if (seenErr.has(key)) c.a1 = seenErr.get(key); else seenErr.set(key, c.ln); }
    const ok = c.res && !c.err && !c.ended;
    if (c.state) { const j = lastOk.get(c.fp); if (j !== undefined && i - j <= LOOKBACK) c.a2 = calls[j].ln; if (ok) lastOk.set(c.fp, i); }
    if (ok && (c.ed || c.wr)) {
      const p = (c.ed || c.wr).p, list = edits.get(p) || edits.set(p, []).get(p);
      if (c.ed) { for (let x = list.length - 1; x >= 0; x--) { const e = list[x]; if (i - e.i > LOOKBACK) break; if (e.ed && e.ed.oh === c.ed.nh && e.ed.nh === c.ed.oh && c.ed.oh !== c.ed.nh) { c.r1 = calls[e.i].ln; break; } } }
      else { let between = false; for (let x = list.length - 1; x >= 0; x--) { const e = list[x]; if (i - e.i > LOOKBACK) break; if (e.wr && e.wr.ch === c.wr.ch) { if (between) c.r1 = calls[e.i].ln; break; } between = true; } }
      list.push({ i, ed: c.ed, wr: c.wr });
    }
    if (c.ck && c.chk) { const prev = lastRun.get(c.ck); if (c.chk === 'fail' && prev && prev.chk === 'pass' && i - prev.i <= LOOKBACK) c.r3 = calls[prev.i].ln; lastRun.set(c.ck, { i, chk: c.chk }); }
    delete c.ed; delete c.wr; delete c.fp;
  });
  return calls;
}
async function stageIndex(cacheDir) {
  const { files, inScope, leftOut } = scopeFiles(), out = { callCut: CALL_CUT, inScope, leftOut, files: [] }, t0 = Date.now();
  for (const it of files) { const calls = await readCalls(it.f, it.sid); out.files.push({ g: it.g, sid: it.sid, calls }); }
  fs.writeFileSync(path.join(cacheDir, 'calls.json'), JSON.stringify(out));
  const n = out.files.reduce((a, f) => a + f.calls.filter(c => c.own).length, 0);
  console.log(`INDEX_DONE transcripts=${files.length} own-calls=${n} copied-calls=${out.files.reduce((a, f) => a + f.calls.filter(c => !c.own).length, 0)} seconds=${((Date.now() - t0) / 1000).toFixed(0)} cut=${CALL_CUT}`);
}

// ---------- windows (scoring 1.2) and placebo points (1.4) ----------
function load(cacheDir) {
  const idx = JSON.parse(fs.readFileSync(path.join(cacheDir, 'index.json'), 'utf8')), cj = JSON.parse(fs.readFileSync(path.join(cacheDir, 'calls.json'), 'utf8'));
  if (idx.startedAt !== CALL_CUT) throw new Error(`the reader's index.json was made with cut ${idx.startedAt}; run it with SUMMARY_CHECK_CUT=${CALL_CUT}`);
  const byId = new Map(cj.files.map(f => [f.sid, f])), scTail = JSON.parse(fs.readFileSync(path.join(SC_DIR, 'audit.json'), 'utf8')).tail, scRows = new Map(JSON.parse(fs.readFileSync(path.join(SC_DIR, 'check.json'), 'utf8')).rows.map(r => [r.id, r]));
  const sessions = []; let laterAll = 0;
  for (const f of cj.files) { f.calls.forEach((c, i) => { c.pos = i; }); f.own = f.calls.filter(c => c.own); f.own.forEach((c, i) => { c.oi = i; }); f.sid8 = f.sid.slice(0, 8); }
  for (const s of idx.sessions) {
    const f = byId.get(s.sid); if (!f) throw new Error('no call stream for ' + s.sid);
    f.comps = s.comps.map(c => { const id = `${s.g}-${s.sid8}-${String(c.n).padStart(2, '0')}`, sc = scRows.get(id), counted = c.ts <= BOUNDARY_CUT; if (!counted) laterAll++;
      if (counted && (!sc || sc.pre !== c.pre || sc.prepared !== !!c.precomputed)) throw new Error('boundary ' + id + ' does not match the summary check');
      return { id, n: c.n, line: c.line, inputEnd: c.inputEnd, ts: c.ts, pre: c.pre, prepared: !!c.precomputed, model: R.MODEL_NAME[c.model] || c.model || '?', counted, band: c.pre < 300000 ? 0 : c.pre < 400000 ? 1 : c.pre < 900000 ? 2 : 3, overtaken: counted ? (scTail[id] || ['?'])[0] === 'overtaken' : null }; }).sort((x, y) => x.line - y.line);
    if (f.comps.some(c => c.counted)) sessions.push(f);
  }
  return { sessions, files: cj.files, inScope: cj.inScope, leftOut: cj.leftOut, laterAll };
}
const rankBefore = (own, line) => { let lo = 0, hi = own.length; while (lo < hi) { const m = (lo + hi) >> 1; if (own[m].ln < line) lo = m + 1; else hi = m; } return lo; };
// windows of W calls a side around every boundary on record; a stretch under 2W is cut in the middle; no call in two windows
function boundaryWindows(f, W) {
  const own = f.own, cs = f.comps, eb = cs.map(c => rankBefore(own, c.inputEnd)), sa = cs.map(c => rankBefore(own, c.line)), w = cs.map(() => ({}));
  cs.forEach((c, k) => {
    if (k === 0) w[k].before = [Math.max(0, eb[0] - W), eb[0]];
    const end = k + 1 < cs.length ? eb[k + 1] : own.length, G = Math.max(0, end - sa[k]);
    if (k + 1 < cs.length) { if (G >= 2 * W) { w[k].after = [sa[k], sa[k] + W]; w[k + 1].before = [end - W, end]; } else { const h = Math.floor(G / 2); w[k].after = [sa[k], sa[k] + h]; w[k + 1].before = [sa[k] + h, end]; } }
    else w[k].after = [sa[k], Math.min(sa[k] + W, own.length)];
    w[k].tail = [eb[k], sa[k]]; w[k].stretchAfter = G;
  });
  return { eb, sa, w };
}
function placeboPoints(f, bw) {
  const own = f.own, cs = f.comps, pts = [], add = (p, where, from) => pts.push({ p, where, dist: p - (from || 0), before: [p - N, p], after: [p, p + N] });
  if (bw.eb[0] >= 4 * N) add(Math.floor(bw.eb[0] / 2), 'before the first boundary');
  for (let k = 0; k + 1 < cs.length; k++) { const G = bw.eb[k + 1] - bw.sa[k]; if (G >= 4 * N) add(bw.sa[k] + Math.floor(G / 2), 'between two boundaries', bw.sa[k]); }
  const last = cs.length - 1, Gl = own.length - bw.sa[last]; if (Gl >= 3 * N) add(bw.sa[last] + Math.floor((Gl + N) / 2), 'after the last boundary', bw.sa[last]);
  return pts;
}
const slice = (f, r) => f.own.slice(r[0], r[1]);
const noMaps = () => ({ before: new Map(), after: new Map() });
// A3: a read in the window that names a path a read in the reference calls also named
const samePath = (p, q) => p === q || p.endsWith('/' + q) || q.endsWith('/' + p);
function reReads(win, ref) {
  const refs = []; for (const c of ref) if (c.rp && c.rp.length) for (const p of c.rp) refs.push([p, c.ln]);
  const out = new Map(); if (!refs.length) return out;
  for (const c of win) if (c.rp && c.rp.length) { const hit = refs.find(([q]) => c.rp.some(p => samePath(p, q))); if (hit) out.set(c, hit[1]); }
  return out;
}
const refBefore = (f, r) => { const first = f.own[r[0]]; return first ? f.calls.slice(Math.max(0, first.pos - N), first.pos) : []; };
// A3s (addendum 1.11): the same lines of the same file, with no write to it between the two reads that the calls show
const overlap = (a, b) => { if (a === 'unknown' || b === 'unknown') return false; if (a === 'whole' || b === 'whole') return true; if (a === 'tail' || b === 'tail') return a === b; return a.some(([s1, e1]) => b.some(([s2, e2]) => s1 <= e2 && s2 <= e1)); };
const WRITE_RE = /\bsed\s+(?:-[A-Za-z]*i|--in-place)|\bperl\s+-[A-Za-z0-9]*i|\bgit\s+(?:checkout|restore|apply|stash|reset|revert|merge|rebase|cherry-pick|pull)\b|\b(?:mv|cp|patch|tee)\b|write_text|write_bytes|writeFileSync|\.write\(|--write\b|ruff\s+format|--fix\b/;
function wroteTo(c, p) {
  const base = p.slice(p.lastIndexOf('/') + 1);
  if ((c.name === 'Edit' || c.name === 'Write' || c.name === 'NotebookEdit') && c.fpath) return samePath(R.normPath(c.fpath), p);
  if (!c.shell) return false;
  if (c.mod && c.mod.includes(base)) return true;
  if (!c.cmd.toLowerCase().includes(base)) return false;
  return WRITE_RE.test(c.cmd) || new RegExp('>>?\\s*["\']?[^\\s"\'|;&]*' + base.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'i').test(c.cmd);
}
function reReadsStrict(f, win, ref) {
  const refs = []; for (const c of ref) for (const x of c.rr || []) refs.push([x, c]);
  const out = new Map(); if (!refs.length) return out;
  for (const c of win) for (const x of c.rr || []) {
    if (out.has(c)) break;
    for (const [y, rc] of refs) {
      if (!samePath(x.p, y.p) || !overlap(x.r, y.r)) continue;
      if (wroteTo(c, x.p) || f.calls.slice(rc.pos + 1, c.pos).some(k => wroteTo(k, x.p))) continue;
      out.set(c, rc.ln); break;
    }
  }
  return out;
}
// every window of the count: the pairs at boundaries, the pairs at placebo points, the fresh starts
function buildAll(D) {
  const pairs = [], plc = [], leftOut = [], laterBoundaries = [];
  for (const f of D.sessions) {
    const bw = boundaryWindows(f, N); f.bw = bw;
    f.comps.forEach((c, k) => {
      if (!c.counted) { laterBoundaries.push(c.id); return; }
      const B = slice(f, bw.w[k].before), A = slice(f, bw.w[k].after), T = slice(f, bw.w[k].tail);
      const rb = refBefore(f, bw.w[k].before), ra = B.concat(T);
      const pr = { kind: 'boundary', f, c, before: B, after: A, tail: T, a3: { before: reReads(B, rb), after: reReads(A, ra) }, a3s: { before: reReadsStrict(f, B, rb), after: reReadsStrict(f, A, ra) } };
      if (B.length >= MIN_WINDOW && A.length >= MIN_WINDOW) pairs.push(pr); else leftOut.push({ id: c.id, before: B.length, after: A.length });
    });
    for (const pt of placeboPoints(f, bw)) { const B = slice(f, pt.before), A = slice(f, pt.after); const rb = refBefore(f, pt.before); plc.push({ kind: 'placebo', f, where: pt.where, dist: pt.dist, before: B, after: A, a3: { before: reReads(B, rb), after: reReads(A, B) }, a3s: { before: reReadsStrict(f, B, rb), after: reReadsStrict(f, A, B) } }); }
  }
  // no call in two windows
  const seen = new Set(); for (const x of pairs.concat(plc)) for (const c of x.before.concat(x.after)) { const key = x.f.sid + ':' + c.pos; if (seen.has(key)) throw new Error('a call sits in two windows: ' + key); seen.add(key); }
  const fresh = []; for (const f of D.files) { if (!f.calls.length || !f.calls[0].own || f.own.length < N) continue; fresh.push({ kind: 'fresh', f, before: [], after: f.own.slice(0, N), a3: noMaps(), a3s: noMaps() }); }
  return { pairs, plc, leftOut, laterBoundaries, fresh };
}

      const obs = (arr) => { let a = 0, d = 0, a2 = 0, d2 = 0; BIN.forEach((b, bi) => { if (bi < 2) { a += tot(arr(G[g][bi])); d += tot(G[g][bi].n); } else { a2 += tot(arr(G[g][bi])); d2 += tot(G[g][bi].n); } }); return { early: 100 * a / d, late: 100 * a2 / d2, x1: a, n1: d, x2: a2, n2: d2 }; };

      if (g === 'comp') { const base = cmpB.get('M').rb; let ex = 0; J.where.comp.slice(0, 5).forEach(b => { ex += (b.mRate - base) * 30 / 100; }); out(`      [E] M above the rate of the 30 calls before a compaction (${f2(base)}), summed over calls 1 to 150 as if every stretch ran that long: ${f2(ex)} mistakes for each compaction`); const first = (J.where.comp[0].mRate - base) * 30 / 100; out(`      [E] of it in calls 1 to 30: ${f2(first)}; at the boundaries' own figures (${f2(cmpB.get('M').ra)} after, ${f2(base)} before): ${f2(cmpB.get('M').d * N / 100)} in the ${N} calls after a compaction`); J.where.excess = { base, perCompaction: ex, first30: first, pairFirst30: cmpB.get('M').d * N / 100 }; }

  }

  for (const id of ['M', 'E', 'B']) { const sg = sigById.get(id), S0 = sums(sg, W.pairs), T0 = { xb: tot(S0.xb), nb: tot(S0.nb), xa: tot(S0.xa), na: tot(S0.na) };
    const ds = D.sessions.map((f, i) => ({ sid8: f.sid8, pairs: W.pairs.filter(x => x.f === f).length, d: 100 * (T0.xa - S0.xa[i]) / (T0.na - S0.na[i]) - 100 * (T0.xb - S0.xb[i]) / (T0.nb - S0.nb[i]), xa: S0.xa[i], xb: S0.xb[i] })).sort((a, b) => a.d - b.d);

// ---------- stage: survey (as run before the scoring was fixed; no signal by position) ----------
async function stageSurvey() {
  const { files, inScope, leftOut } = scopeFiles(), q = (a, p) => { if (!a.length) return 0; const s = [...a].sort((x, y) => x - y); return s[Math.min(s.length - 1, Math.floor(p * s.length))]; };
  const toolN = {}, errTool = {}, combo = {}; let calls = 0, errs = 0, copied = 0, nb = 0, nbOld = 0, t2 = 0, checks = 0, checkFail = 0, stateCalls = 0, interrupts = 0;
  const gFirst = [], gMid = [], gLast = []; let sessWith = 0;
  for (const it of files) {
    const comps = (await R.findSummaries(it.f, it.sid, CALL_CUT)).filter(c => !c.copied && c.sumLine), lines = [], tools = new Map(); let ln = 0;
    const rl = readline.createInterface({ input: fs.createReadStream(it.f), crlfDelay: Infinity });
    for await (const line of rl) {
      ln++; if (line.startsWith('{"type":"api-request') || line.startsWith('{"type":"file-history')) continue;
      let o; try { o = JSON.parse(line); } catch { continue; }
      if (o.isSidechain || (o.timestamp && o.timestamp > CALL_CUT)) continue;
      const own = !(o.sessionId && o.sessionId !== it.sid);
      if (o.type === 'assistant' && o.message && Array.isArray(o.message.content)) { for (const b of o.message.content) if (b.type === 'tool_use') { if (!own) { copied++; continue; } calls++; lines.push(ln); toolN[b.name] = (toolN[b.name] || 0) + 1; const cl = R.classifyCall(b.name, b.input || {}); tools.set(b.id, { name: b.name, k: cl.k, state: cl.state }); if (cl.state) stateCalls++; } continue; }
      if (o.type !== 'user' || !o.message || !own) continue;
      const ct = o.message.content; if (typeof ct === 'string') { if (/^\[Request interrupted/.test(ct)) interrupts++; continue; }
      if (Array.isArray(ct)) for (const b of ct) { if (!b) continue; if (b.type === 'text' && /^\[Request interrupted/.test(b.text || '')) interrupts++; if (b.type !== 'tool_result') continue; const t = tools.get(b.tool_use_id); if (!t) continue;
        const text = typeof b.content === 'string' ? b.content : R.collectStrings(b.content, []).join('\n'), shell = t.name === 'Bash' || t.name === 'PowerShell';
        if (t.k === 'test or check run') { checks++; if (b.is_error || R.testCounts(text).some(x => /^failed\|(?!0$)/.test(x))) checkFail++; }
        if (!b.is_error) { if (shell && notFlagged(text)) t2++; continue; }
        errs++; errTool[shell ? t.name : /^mcp__/.test(t.name) ? 'mcp__*' : t.name] = (errTool[shell ? t.name : /^mcp__/.test(t.name) ? 'mcp__*' : t.name] || 0) + 1;
        const k = P.ended.test(text) ? 'no result: the session ended' : errorKind({ shell, name: t.name, k: t.k }, text); combo[k] = (combo[k] || 0) + 1; }
    }
    if (!comps.length) continue; sessWith++; nb += comps.length; nbOld += comps.filter(c => c.ts <= BOUNDARY_CUT).length;
    const between = (a, b) => lines.filter(l => l > a && l < b).length;
    comps.forEach((c, i) => (i ? gMid : gFirst).push(between(i ? comps[i - 1].line : 0, c.line))); gLast.push(between(comps[comps.length - 1].line, Infinity));
  }
  const desc = (name, a) => console.log(`${name}: n=${a.length} min=${Math.min(...a)} p10=${q(a, .1)} median=${q(a, .5)} p90=${q(a, .9)} max=${Math.max(...a)}`);
  console.log(`folders in scope ${inScope}, left out ${leftOut}; main transcripts ${files.length}; calls read up to ${CALL_CUT}`);
  console.log(`tool calls by the main sessions ${calls}; copied calls ${copied}; boundaries ${nb} in ${sessWith} sessions, of them at the summary check's cut ${nbOld}`);
  desc('calls from a session\'s start to its first boundary', gFirst); desc('calls between two boundaries', gMid); desc('calls after the last boundary', gLast);
  for (const n of [30, 50]) console.log(`  ${n} a side: stretches between boundaries with a middle of ${2 * n} or more calls: ${gMid.filter(g => g >= 4 * n).length} of ${gMid.length}`);
  console.log(`error results ${errs} (${(100 * errs / calls).toFixed(1)} for each 100 calls); by tool ${JSON.stringify(errTool)}`);
  console.log('error results by kind, whole record:'); for (const [k, v] of Object.entries(combo).sort((a, b) => b[1] - a[1])) console.log('  ' + String(v).padStart(5) + '  ' + k);
  console.log(`shell results not flagged that hold a diagnostic line ${t2}; check or test runs ${checks}, failing ${checkFail}; state-changing calls ${stateCalls}; interrupts ${interrupts}`);
  console.log('tool names: ' + Object.entries(toolN).filter(([k]) => !/^mcp__/.test(k)).sort((a, b) => b[1] - a[1]).map(([k, v]) => k + ' ' + v).join(', ') + ', mcp tools ' + Object.entries(toolN).filter(([k]) => /^mcp__/.test(k)).reduce((a, [, v]) => a + v, 0));
}

(async () => {
  if (stage === 'survey') return stageSurvey();
  if (stage === 'index') return stageIndex(a1);
  if (stage === 'sample') return stageSample(a1);
  if (stage === 'count') return stageCount(a1, a2);
  console.log('stages: survey | index <cacheDir> | sample <cacheDir> | count <cacheDir> [outDir]');
})();

function readerPrompt(r) {
  return [
    'You are one of three readers. A design for running research with a tree of language-model sessions (one coordinating session, one child session per research field, short-lived helper agents under each child) is about to be judged against published evidence. You are not shown the design, on purpose: your job is to bring back what is known, not an opinion of it.',
    '',
    'Your questions:',
    r.qs.map(q => QUESTIONS[q]).join('\n'),
    '',
    'For each question find the strongest published evidence, for and against. One finding is one claim from one source. Aim for 10 to 14 findings in all, the most decision-relevant first; a few good ones beat many weak ones. For every finding give: the claim in one or two plain sentences; the mechanism (why it happens) if the source gives one; the source (title, authors or organisation, year, an address that opens); where in the source the claim is (section, table or figure); the number with its unit and what it was compared with; the conditions; the kind of evidence; which questions it bears on (their D-numbers); and in "for_design" one sentence on what a designer of such a tree should do differently because of it.',
    '',
    'Possible starting points. They are from memory and unverified: treat each as a lead to open, not as a fact. If one does not exist or does not say what its title suggests, say so in "notes". Search beyond them.',
    r.seeds,
    '',
    'Ids: ' + r.key + '-01, ' + r.key + '-02, and so on.',
    '',
    WEB_RULES,
  ].join('\n')
}

function recheckPrompt(res, r) {
  return [
    'A reader brought back the findings below from published work. Open each source again yourself and say whether it supports the claim, the number and the conditions as written. Refute by default: a claim you cannot find in the source is "not_supported"; a source you cannot open is "could_not_open" (try the address, then one search for the title). If the claim is right but the number, the comparison or the conditions are off, the verdict is "confirmed_with_correction" and you give the corrected claim and number. In "conditions_that_limit_it" say what a careful reader must know before applying the claim elsewhere: the models tested, the tasks, the size of the study, who ran it and whether anyone independent repeated it. Check every finding. If there are more than 14, check those that carry a number first and mark the ones you did not get to as "not_reached". Use the same ids.',
    '',
    'Findings (JSON):',
    JSON.stringify(res.findings),
    '',
    WEB_RULES,
  ].join('\n')
}

function attackPrompt(r) {
  return [ATTACK_HEAD, '', r.lens, '', 'Ids: ' + r.key + '-01, ' + r.key + '-02, and so on.'].join('\n')
}

// Readers go straight into their re-check (no barrier); refuters run beside them and never see the literature.
const litP = pipeline(
  READERS,
  r => agent(readerPrompt(r), { label: 'read:' + r.key, phase: 'Read', schema: LIT, model: 'sonnet', effort: 'high' }),
  (res, r) => {
    if (!res || !res.findings || !res.findings.length) return null
    return agent(recheckPrompt(res, r), { label: 'recheck:' + r.key, phase: 'Recheck', schema: RECHECK, model: 'opus', effort: 'medium' })
      .then(chk => ({ key: r.key, findings: res.findings, searched_not_found: res.searched_not_found || [], notes: res.notes || '', checks: chk ? chk.checks : null }))
  }
)
const attP = parallel(REFUTERS.map(r => () =>
  agent(attackPrompt(r), { label: 'attack:' + r.key, phase: 'Attack', schema: ATTACK, model: 'opus', effort: 'high' })
    .then(a => (a ? { key: r.key, findings: a.findings, rows_that_hold: a.rows_that_hold, overall: a.overall } : null))
))
const both = await Promise.all([litP, attP])
const lit = both[0].filter(Boolean)
const att = both[1].filter(Boolean)
log('readers returned: ' + lit.length + ' of ' + READERS.length + '; refuters returned: ' + att.length + ' of ' + REFUTERS.length)

const kept = []
const dropped = []
for (const r of lit) {
  const byId = {}
  for (const c of (r.checks || [])) byId[c.id] = c
  if (!r.checks) log('re-check of ' + r.key + ' returned nothing: all its ' + r.findings.length + ' findings are dropped as unchecked')
  for (const f of r.findings) {
    const c = byId[f.id]
    if (c && (c.verdict === 'confirmed' || c.verdict === 'confirmed_with_correction')) {
      kept.push({
        id: f.id, claim: c.corrected_claim || f.claim, mechanism: f.mechanism || '', number: c.corrected_number || f.number,
        conditions: f.conditions, limits: c.conditions_that_limit_it || '', source_title: f.source_title, authors_or_org: f.authors_or_org || '',
        year: f.year, url: f.url, where_in_source: f.where_in_source || '', evidence_kind: f.evidence_kind, bears_on: f.bears_on,
        for_design: f.for_design, verdict: c.verdict, recheck_note: c.note,
      })
    } else {
      dropped.push({ id: f.id, source_title: f.source_title, verdict: c ? c.verdict : 'not_checked', note: c ? c.note : 'the re-check returned nothing for this id' })
    }
  }
}
log('literature findings kept after the re-check: ' + kept.length + '; dropped: ' + dropped.length)

const attackFlat = att.flatMap(a => a.findings.map(f => Object.assign({ lens: a.key }, f)))
const holds = att.flatMap(a => a.rows_that_hold.map(h => Object.assign({ lens: a.key }, h)))

const map = await agent(mapPrompt, { label: 'map', phase: 'Map', schema: MAP, model: 'opus', effort: 'high' })

return {
  counts: { readers: lit.length, refuters: att.length, literature_kept: kept.length, literature_dropped: dropped.length, attack_findings: attackFlat.length },
  sources: kept.map(f => ({ id: f.id, title: f.source_title, who: f.authors_or_org, year: f.year, url: f.url, kind: f.evidence_kind, number: f.number, verdict: f.verdict })),
  dropped: dropped,
  not_found: lit.map(r => ({ reader: r.key, searched_not_found: r.searched_not_found })),
  attack: attackFlat.map(f => ({ id: f.id, title: f.title, target: f.target, kind: f.kind, severity: f.severity })),
  map: map,
}

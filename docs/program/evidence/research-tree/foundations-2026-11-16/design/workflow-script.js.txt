    'How to write it. Resolve conflicts between layers and say how in your return. One name for one thing throughout (one task table, one status file, one commit record). Do not keep a rule that has no runner. Where two layers propose two mechanisms for the same job, keep the simpler. Keep the pilot set small: a script belongs to the pilot only if a problem rated high would otherwise be unguarded when the first field runs. Plain words, short sentences, no filler, tokens never money, at most about 520 lines of at most 105 characters (tables may run longer). Do not invent measured numbers: a formula with named inputs is right; a made-up figure is wrong.',
    '',
    'THE SIX LAYERS (JSON):',
    JSON.stringify(layers),
  ].join('\n')
}

const FINDINGS = {
  type: 'object',
  properties: {
    findings: { type: 'array', items: { type: 'object', properties: {
      id: { type: 'string' }, title: { type: 'string' }, target: { type: 'string' },
      kind: { type: 'string', enum: ['classical_assumption_does_not_hold', 'guarantee_overstated', 'no_runner_or_cannot_run_here', 'too_much_machinery', 'missing', 'contradiction', 'does_not_deliver_the_owners_requirement', 'analogy_is_decoration', 'other'] },
      argument: { type: 'string' }, lines: { type: 'string' }, severity: { type: 'string', enum: ['high', 'medium', 'low'] }, smallest_fix: { type: 'string' },
    }, required: ['id', 'title', 'target', 'kind', 'argument', 'lines', 'severity', 'smallest_fix'] } },
    holds: { type: 'array', items: { type: 'string' } },
    overall: { type: 'string' },
  },
  required: ['findings', 'holds', 'overall'],
}

const ATTACKERS = [
  { key: 'A1', lens: 'Classical assumptions. For each borrowed design in the document, name the assumptions under which the classical result holds (deterministic re-execution, fail-stop components, independent errors, a trusted computing base that the subjects cannot reach, a known key distribution, stationary behaviour, honest sensors) and say whether each holds for language-model sessions on this harness. Find every place where the document claims a guarantee that the classical result does not give under these faults, and every place where a famous name is used as decoration: the rule would read the same without it. Check the mapping files for what the experts themselves said fails.' },
  { key: 'A2', lens: 'Can it run here? Take the FACTS ABOUT THE HARNESS as the only things that exist. For each rule, find the runner and ask whether that runner can do it: is the script given what it needs (what can a script not read?), does the harness really offer the enforcement point, who starts the script and what happens between a coordinator\'s turns, what happens after a reboot, and which steps silently need the owner. Find bootstrap problems (a check that needs the thing it checks), ordering problems between scripts, and anything "believed, not verified" that the design leans on without a fallback. Use the model-check result given below.' },
  { key: 'A3', lens: 'Simplicity and cost. This protocol guards a literature research programme of seven fields. Estimate the machinery\'s own weight: how many scripts, how many lines, how many tokens and hours the checks themselves use per field, against what they protect. Which mechanisms buy most of the protection for least effort? Which belong to "later" or to "never" for a programme of this size? Propose the smallest set that keeps the problems rated high guarded, and an order to build it in. Also say where the design would cost more tokens than the fault it prevents. Use the cost computation given below; do not invent figures.' },
  { key: 'A4', lens: 'The owner\'s requirement. He asked for (a) experts first, reviews second: a context filled precisely with checked knowledge, then a review that is not a raw, biased opinion; (b) a tree built from solved designs, his own B+ tree and distributed-systems ideas among them, used properly and not as decoration; (c) everything judged by the seven measures; (d) a structure so careful that a mistake cannot spread. Check the document against each. Where does it drop, weaken or merely name one of his ideas? (Version 1 withdrew his "split on overflow"; does version 2 treat overflow as a real B+ tree would?) Where does it decide something that amendment 1 of version 1 records as his own word (who reviews, how a second expert works) without leaving it to him? Is there a path from this document to the owner being able to see the seven measures for the tree?' },
]

function attackPrompt(a, extra) {
  return [
    'You are attacking a design, not improving its prose. Read all of ' + DOC + ' (version 2 of a research tree protocol). Refute by default. Your lens:',
    a.lens,
    '',
    CONTEXT,
    '',
    'Rules for findings. Cite the document by line number. Give the smallest change that fixes each problem; prefer removing or merging to adding. Also list what holds, so that the result is not one-sided. Leave out what you would rate low. At most 12 findings, the most damaging first. Ids ' + a.key + '-01, ' + a.key + '-02, and so on.',
    '',
    'RESULTS YOU MAY USE',
    extra,
  ].join('\n')
}

const RULING = {
  type: 'object',
  properties: {
    edits: { type: 'array', items: { type: 'object', properties: {
      rests_on: { type: 'array', items: { type: 'string' } }, section: { type: 'string' }, edit: { type: 'string' }, why: { type: 'string' },
    }, required: ['rests_on', 'section', 'edit', 'why'] } },
    rejected: { type: 'array', items: { type: 'object', properties: { id: { type: 'string' }, why: { type: 'string' } }, required: ['id', 'why'] } },
    pilot_set_final: { type: 'array', items: { type: 'string' } },
    moved_to_later: { type: 'array', items: { type: 'string' } },
    for_the_judge: { type: 'array', items: { type: 'string' } },
    not_established: { type: 'array', items: { type: 'string' } },
    verdict: { type: 'string' },
  },
  required: ['edits', 'rejected', 'pilot_set_final', 'moved_to_later', 'for_the_judge', 'not_established', 'verdict'],
}

const REVISED = {
  type: 'object',
  properties: {
    path: { type: 'string' }, lines: { type: 'integer' },
    change_log: { type: 'array', items: { type: 'string' } },
    not_applied: { type: 'array', items: { type: 'object', properties: { edit: { type: 'string' }, why: { type: 'string' } }, required: ['edit', 'why'] } },
  },
  required: ['path', 'lines', 'change_log', 'not_applied'],
}

// Stage 1: the six layers in parallel. The execution layer goes straight on to its model check and the fidelity review.
const layerP = LAYERS.map(l => agent(layerPrompt(l), { label: 'design:' + l.key, phase: 'Design', schema: LAYER, model: 'opus', effort: 'high' }))
const modelP = layerP[0].then(spec => {
  if (!spec) return null
  return agent(modelPrompt(spec), { label: 'model:check', phase: 'Model', schema: MODEL, model: 'opus', effort: 'high' })
    .then(m => {
      if (!m) return { model: null, fidelity: null }
      return agent(fidelityPrompt(spec, m), { label: 'model:fidelity', phase: 'Model', schema: FIDELITY, model: 'opus', effort: 'high' })
        .then(f => ({ model: m, fidelity: f }))
    })
})

const layersRaw = await Promise.all(layerP)
const layers = {}
LAYERS.forEach((l, i) => { if (layersRaw[i]) layers[l.key + ' ' + l.title] = layersRaw[i]; else log('layer ' + l.key + ' returned nothing: version 2 lacks ' + l.title) })
log('layers returned: ' + Object.keys(layers).length + ' of ' + LAYERS.length)

// Stage 2: one document. Runs while the model check is still going.
const integrated = await agent(integratePrompt(layers), { label: 'integrate', phase: 'Integrate', schema: INTEGRATED, model: 'opus', effort: 'high' })
const mc = await modelP
const modelBrief = JSON.stringify(mc && mc.model ? {
  properties_checked: mc.model.properties_checked, ablations: mc.model.ablations, flaws_in_the_spec: mc.model.flaws_in_the_spec, limits: mc.model.limits,
  fidelity: mc.fidelity ? { reproduces: mc.fidelity.reproduces, rules_missing_from_the_model: mc.fidelity.rules_missing_from_the_model, vacuous_or_weak_properties: mc.fidelity.vacuous_or_weak_properties, model_errors: mc.fidelity.model_errors, extra_flaws_in_the_spec: mc.fidelity.extra_flaws_in_the_spec, verdict: mc.fidelity.verdict } : 'the fidelity review returned nothing',
} : 'the model check returned nothing')
const costLayer = layersRaw[4]
const costBrief = costLayer ? String(costLayer.computed || 'the cost designer returned no computation') : 'the cost layer returned nothing'
const complLayer = layersRaw[5]
const extra = [
  'The model check of the execution layer (properties, ablations, flaws found, and the fidelity review of the model):', modelBrief, '',
  'The cost designer\'s computation (real output of its script):', costBrief, '',
  'The completeness analysis (what nothing covers, and the estimate of unfound failure modes):', complLayer ? String(complLayer.computed || '') + '\n' + JSON.stringify(complLayer.problems_covered) : 'the completeness layer returned nothing',
].join('\n')

// Stage 3: four attackers on the integrated document.
const attacks = (await parallel(ATTACKERS.map(a => () =>
  agent(attackPrompt(a, extra), { label: 'attack:' + a.key, phase: 'Attack', schema: FINDINGS, model: 'opus', effort: 'high' })
    .then(r => (r ? { key: a.key, findings: r.findings, holds: r.holds, overall: r.overall } : null))
))).filter(Boolean)
log('attackers returned: ' + attacks.length + ' of ' + ATTACKERS.length)
const allFindings = attacks.flatMap(a => a.findings.map(f => Object.assign({ lens: a.key }, f)))

// Stage 4: rule on the findings, then apply the accepted edits.
const ruling = await agent([
  'Four refuters attacked version 2 of a research tree protocol (' + DOC + '), each with a different lens; a model check of its execution layer and a fidelity review of that model also reported flaws. Rule on all of it. Read the document yourself.',
  '1. Check each finding\'s line citation and argument. Reject a finding that misreads the document, or that asks for more machinery than its risk is worth, with the reason. Be as hard on the refuters as they were on the design.',
  '2. Every flaw that the model check found with all mechanisms present must be either fixed by an edit or rejected with a reason: none may be dropped silently.',
  '3. Turn what you accept into edits: each names the section and says exactly what to change, add or delete. Merge findings that ask for the same edit. Prefer deleting or moving to "later" over adding.',
  '4. Settle the pilot set: the scripts and files that must exist before the first field runs, kept as small as the problems rated high allow; and what moves to "later".',
  '5. Never decide what the document records as the owner\'s decision; an edit may only make the options and the default clearer.',
  '6. "for_the_judge": where the refuters disagree with each other or with the designers and you are unsure. "not_established": what nothing settles.',
  'No web. Do not create, edit or delete any file. This task is not about the product whose folder you start in.',
  '',
  CONTEXT,
  '',
  'THE FINDINGS (JSON):', JSON.stringify(allFindings), '',
  'WHAT THE REFUTERS SAY HOLDS:', JSON.stringify(attacks.map(a => ({ lens: a.key, holds: a.holds, overall: a.overall }))), '',
  'THE MODEL CHECK AND ITS FIDELITY REVIEW:', modelBrief,
].join('\n'), { label: 'settle:rule', phase: 'Settle', schema: RULING, model: 'opus', effort: 'high' })

let revised = null
if (ruling && ruling.edits && ruling.edits.length) {
  revised = await agent([
    'Apply the edits below to ' + DOC + ' and save it in place. Apply every edit exactly; if one cannot be applied as written, apply its intent and list it under not_applied with the reason. Renumber rules only if an edit deletes or adds one, and then fix every reference to a renumbered rule (the table of problems, the pilot list). Replace the pilot section with the final pilot set and what moved to later. Keep the document\'s voice: plain words, short sentences, tokens never money, no invented figures. Do not add anything the edits do not ask for. Afterwards read the whole document once more for a rule that now has no runner, a reference to something deleted, or two names for one thing, and fix those. Write only this file. This task is not about the product whose folder you start in.',
    '',
    'EDITS (JSON):', JSON.stringify(ruling.edits), '',
    'FINAL PILOT SET:', JSON.stringify(ruling.pilot_set_final), '',
    'MOVED TO LATER:', JSON.stringify(ruling.moved_to_later),
  ].join('\n'), { label: 'settle:revise', phase: 'Settle', schema: REVISED, model: 'opus', effort: 'high' })
} else {
  log('no edits were ruled: the document stands as integrated')
}

return {
  doc: DOC,
  layers: Object.keys(layers).map(k => ({ layer: k, scripts: layers[k].scripts.map(s => s.name + ' (' + s.stage + ')'), pilot_minimum: layers[k].pilot_minimum, contradicts_the_frame: layers[k].contradicts_the_frame, for_the_owner: layers[k].for_the_owner, left_out: layers[k].left_out.map(x => x.what) })),
  cost_computed: costBrief,
  completeness_computed: complLayer ? String(complLayer.computed || '') : '',
  model: mc && mc.model ? { files: mc.model.files_written, how_to_run: mc.model.how_to_run, properties: mc.model.properties_checked.map(p => ({ name: p.name, kind: p.kind, holds: p.holds, states: p.states_explored })), ablations: mc.model.ablations.map(a => ({ removed: a.removed, breaks: a.property_that_breaks })), flaws: mc.model.flaws_in_the_spec.map(f => ({ title: f.title, fix: f.fix })), limits: mc.model.limits } : null,
  fidelity: mc && mc.fidelity ? { reproduces: mc.fidelity.reproduces, missing: mc.fidelity.rules_missing_from_the_model, vacuous: mc.fidelity.vacuous_or_weak_properties.map(v => v.property + ': ' + v.why), model_errors: mc.fidelity.model_errors, extra_flaws: mc.fidelity.extra_flaws_in_the_spec.map(f => f.title), verdict: mc.fidelity.verdict } : null,
  integrated: integrated ? { lines: integrated.lines, conflicts_resolved: integrated.conflicts_resolved, for_the_owner: integrated.for_the_owner, doubts: integrated.doubts } : null,
  attack: allFindings.map(f => ({ id: f.id, title: f.title, kind: f.kind, severity: f.severity })),
  ruling: ruling ? { edits: ruling.edits.length, edit_list: ruling.edits.map(e => e.section + ': ' + e.edit), rejected: ruling.rejected, pilot_set_final: ruling.pilot_set_final, moved_to_later: ruling.moved_to_later, for_the_judge: ruling.for_the_judge, not_established: ruling.not_established, verdict: ruling.verdict } : null,
  revised: revised,
}

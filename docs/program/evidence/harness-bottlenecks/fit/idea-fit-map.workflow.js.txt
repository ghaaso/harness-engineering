export const meta = {
  name: 'idea-fit-map',
  description: 'Map the surviving Horthy and BAML ideas onto the 21 bottlenecks and re-screen the dropped ones under the bottleneck lens',
  phases: [
    { title: 'Screen', detail: 'five Sonnet readers over the 783 dropped Horthy families', model: 'sonnet' },
    { title: 'Map', detail: 'three Opus mappers over the 78 Horthy survivors, one over the BAML list', model: 'opus' },
  ],
}

const IN = args.inDir
const COMPILE = args.compile
const BN = args.bottlenecks

const PRE = `You are one lane of a read-only mapping job. Read these three files first, in full, with the Read tool:
1. ${IN}\\brief.md  (the lens, the fit scale and the hard rules; it governs everything you return)
2. ${COMPILE}  (the bottleneck compile: corrected diagnoses in section 1, what the testbed built in section 2, class / first step / proof for each of B01 to B21 in section 3, cross-cutting mechanisms in section 7)
3. ${BN}  (the owner's 21 bottlenecks, verbatim)
Then read your lane file in full. It is a markdown table with long rows; if a Read is truncated, continue with offset until you have read every row.`

const BIDS = { type: 'array', items: { type: 'string', pattern: '^B(0[1-9]|1[0-9]|2[01])$' } }

const SCREEN_SCHEMA = {
  type: 'object',
  properties: {
    lane_file: { type: 'string' },
    rows_read: { type: 'integer' },
    candidates: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          id: { type: 'string' },
          disposition_in_pr: { type: 'string', enum: ['skip', 'already-done'] },
          bottlenecks: BIDS,
          claim_short: { type: 'string', description: 'The claim in at most 25 words, faithful to the row.' },
          why_it_fell: { type: 'string', description: 'The recorded reason in at most 30 words, faithful to the row.' },
          fell_reason_class: { type: 'string', enum: ['weak-evidence', 'already-in-fixflow-on-paper', 'no-fixflow-weakness-matched', 'reopens-an-owner-skip', 'cost-or-risk', 'not-transferable', 'other'] },
          reason_still_holds_against_bottleneck: { type: 'string', enum: ['yes', 'partly', 'no', 'unknown'] },
          what_it_would_add: { type: 'string', description: 'At most 40 words: the concrete thing in the testbed record it would have changed, and how.' },
          strength: { type: 'string', enum: ['strong', 'moderate', 'weak'] },
        },
        required: ['id', 'disposition_in_pr', 'bottlenecks', 'claim_short', 'why_it_fell', 'fell_reason_class', 'reason_still_holds_against_bottleneck', 'what_it_would_add', 'strength'],
      },
    },
    notes: { type: 'string', description: 'At most 80 words: patterns you saw, and anything you could not read.' },
  },
  required: ['lane_file', 'rows_read', 'candidates', 'notes'],
}

const MAPPING = {
  type: 'object',
  properties: {
    id: { type: 'string' },
    source: { type: 'string', enum: ['horthy', 'baml'] },
    disposition_in_pr: { type: 'string', enum: ['adopt', 'trial', 'borrow'] },
    idea: { type: 'string', description: 'The narrowed version in at most 45 words. Not the original claim.' },
    bottlenecks: BIDS,
    fit: { type: 'string', enum: ['direct', 'supporting', 'fixflow-only', 'conflicts', 'none'] },
    how_it_fits: { type: 'string', description: 'At most 60 words. For direct or supporting: the artefact in harness v2 (a rule, a script, a brief line, a workflow step, a CI check) and the thing in the testbed record it would have changed. For fixflow-only: which weakness. For conflicts: the measured fact.' },
    relation_to_compile: { type: 'string', enum: ['already-cited-in-compile', 'strengthens-first-step', 'adds-a-step', 'replaces-first-step', 'no-relation'] },
    attaches_to: { type: 'string', description: 'The compile section and bottleneck row it attaches to, or empty.' },
    caveat: { type: 'string', description: 'At most 40 words: what limits it here, or empty.' },
    added_cost: { type: 'string', enum: ['none', 'script', 'brief-or-config-line', 'extra-agents-per-phase', 'extra-agents-per-finding', 'standing-process'] },
    needs_owner_decision: { type: 'boolean' },
    proof_on_record: { type: 'string', description: 'At most 40 words: the replay or measurement on the testbed record that would test it, or empty.' },
    reopened_part: { type: 'string', description: 'Empty unless you propose reviving a part the PR refuter cut; then the part and why the cut does not hold against a bottleneck.' },
  },
  required: ['id', 'source', 'disposition_in_pr', 'idea', 'bottlenecks', 'fit', 'how_it_fits', 'relation_to_compile', 'attaches_to', 'caveat', 'added_cost', 'needs_owner_decision', 'proof_on_record', 'reopened_part'],
}

const MAP_SCHEMA = {
  type: 'object',
  properties: {
    lane_file: { type: 'string' },
    rows_read: { type: 'integer' },
    mappings: { type: 'array', items: MAPPING },
    notes: { type: 'string', description: 'At most 120 words: bottlenecks no idea in this lane reaches, and ideas that pull against each other.' },
  },
  required: ['lane_file', 'rows_read', 'mappings', 'notes'],
}

const BAML_SCHEMA = {
  type: 'object',
  properties: {
    lane_file: { type: 'string' },
    cards_read: { type: 'integer' },
    mappings: { type: 'array', items: MAPPING },
    fallen_candidates: SCREEN_SCHEMA.properties.candidates,
    notes: { type: 'string' },
  },
  required: ['lane_file', 'cards_read', 'mappings', 'fallen_candidates', 'notes'],
}

const screenPrompt = (file) => `${PRE}

Your lane file: ${IN}\\${file}
Each row is one claim family from Dex Horthy's published work that PR #186 dropped: either "skip" (usually weak evidence, cost, or no FixFlow weakness matched) or "already-done" (FixFlow's ship-plan harness has the mechanism on paper). The last column says why.

Your job is a recall screen under a lens PR #186 never used. For every row ask: read on its own, does this claim name a mechanism that bears on one of the 21 bottlenecks B01 to B21? Return a candidate when it does, even though the PR dropped it. Do not return a row whose only link is topic overlap.

For "already-done" rows, return one only when the mechanism it names is what a bottleneck's first step calls for (compile section 3) or is something the testbed record shows was missing or failed; class it already-in-fixflow-on-paper.

For each candidate, classify why it fell, and say whether that reason still holds when the question is the bottleneck and not FixFlow's W1 to W5. "Weak evidence" still holds unless the clinic record itself supplies evidence; "no FixFlow weakness matched" usually does not hold; "reopens an owner skip" (hooks, the context system, a measurement pipeline, a pre-commit gate) still needs the owner.

Expect roughly 5 to 20 percent of rows to be candidates. Err toward including a row when unsure; a later Opus pass prunes. Report rows_read as the number of table rows you actually read. Return only the structured result.`

const mapPrompt = (file) => `${PRE}

Your lane file: ${IN}\\${file}
Each row is one idea from Dex Horthy's published work that survived PR #186's review as adopt or trial. The column "What survived the refuter" is the idea; the Addendum column, where filled, replaces or adds to it.

Return one mapping for every row in the file, none skipped. For each: which of B01 to B21 it bears on (an empty list is a normal answer), the fit on the brief's scale, how it fits, its relation to what the compile already proposes for that bottleneck, what limits it, what it costs, whether it needs an owner decision, and what replay or measurement on the testbed record would test it.

Be strict. These ideas were narrowed to fit FixFlow's ship plan; many will be fixflow-only and that is the honest answer. Where the narrowed version names a FixFlow file or phase (the ship plan's section 0, P1, the PR body), ask whether the same mechanism has a counterpart in the testbed record (a slice gate, a claims line, the board, a resume); if it does, map it and say what the counterpart is; if it does not, it is fixflow-only.

In notes, name the bottlenecks nothing in your lane reaches and any two ideas that pull against each other. Return only the structured result.`

const bamlPrompt = `${PRE}

Your lane file: ${IN}\\baml-post-review.md
It holds four extracts of the BAML ideas document (PR #184): the owner summary with the four trials (I-13, D-13, T-08, P-10); the post-review list of all 89 cards with dispositions and reasons; the rulings; and the structural comparison, whose section 4.6 lists five structural borrows S-1 to S-5.

Two jobs.
1. mappings: one mapping for each of the four trials and each of the five borrows (nine in all; use disposition_in_pr "trial" for the trials and "borrow" for S-1 to S-5, source "baml"). Same rules as the brief: bottlenecks, fit, how it fits, relation to the compile, caveat, cost, owner decision, proof on the record.
2. fallen_candidates: screen the 85 cards that ended already-done or skip under the bottleneck lens, which PR #184 never used. Return a candidate when the card names a mechanism that bears on one of B01 to B21 (for example typed claims for B12, output contracts for B16, checks that must be able to fail for B17). Classify why it fell and whether that reason still holds against the bottleneck. The tool itself is ruled out (no install, a second billing path, a second orchestrator); do not propose it.

Report cards_read. Return only the structured result.`

const out = { screen: [], map: [], baml: null }

phase('Screen')
const screenFiles = ['horthy-fallen-1.md', 'horthy-fallen-2.md', 'horthy-fallen-3.md', 'horthy-fallen-4.md', 'horthy-fallen-5.md']
const screened = await parallel(screenFiles.map((f, i) => () =>
  agent(screenPrompt(f), { label: `screen:fallen-${i + 1}`, phase: 'Screen', schema: SCREEN_SCHEMA, model: 'sonnet', effort: 'medium', agentType: 'general-purpose' })))
out.screen = screened.map((r, i) => r || { lane_file: screenFiles[i], rows_read: 0, candidates: [], notes: 'AGENT FAILED' })
log(`Screen: ${out.screen.reduce((n, r) => n + r.candidates.length, 0)} candidates from ${out.screen.reduce((n, r) => n + r.rows_read, 0)} rows; failed lanes: ${screened.filter((r) => !r).length}`)

phase('Map')
const mapFiles = ['horthy-survivors-A.md', 'horthy-survivors-B.md', 'horthy-survivors-C.md']
const mapped = await parallel([
  ...mapFiles.map((f, i) => () =>
    agent(mapPrompt(f), { label: `map:survivors-${'ABC'[i]}`, phase: 'Map', schema: MAP_SCHEMA, model: 'opus', effort: 'high', agentType: 'general-purpose' })),
  () => agent(bamlPrompt, { label: 'map:baml', phase: 'Map', schema: BAML_SCHEMA, model: 'opus', effort: 'high', agentType: 'general-purpose' }),
])
out.map = mapped.slice(0, 3).map((r, i) => r || { lane_file: mapFiles[i], rows_read: 0, mappings: [], notes: 'AGENT FAILED' })
out.baml = mapped[3] || { lane_file: 'baml-post-review.md', cards_read: 0, mappings: [], fallen_candidates: [], notes: 'AGENT FAILED' }
log(`Map: ${out.map.reduce((n, r) => n + r.mappings.length, 0)} Horthy mappings, ${out.baml.mappings.length} BAML mappings, ${out.baml.fallen_candidates.length} BAML fallen candidates; failed lanes: ${mapped.filter((r) => !r).length}`)

return out

const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

// ================================================================ run
phase('Lock')
const lockRes = await parallel([
  () => agent(pL1, { label: 'fix:lock', phase: 'Lock', schema: BUILD_SCHEMA, ...OPUS }),
  () => agent(pB1, { label: 'bench:write', phase: 'Lock', schema: BUILD_SCHEMA, ...OPUS }),
])
const lock = lockRes[0]
const benchWrite = lockRes[1]
if (!lock) return { failed: 'lock', benchWrite }

phase('Freeze')
let freeze = await agent(pG2(''), { label: 'grade:refreeze', phase: 'Freeze', schema: FREEZE_SCHEMA, ...OPUS })
if (!freeze) return { failed: 'freeze', lock, benchWrite }
if (!freeze.matches_addendum_4) return { stopped: 'freeze-mismatch', lock, benchWrite, freeze }

phase('Preflight')
let pre = await agent(pP1(''), { label: 'audit:preflight-1', phase: 'Preflight', schema: AUDIT_SCHEMA, ...OPUS })
const fixes = []
let round = 0
while (pre && (!pre.ready_for_heldout || !pre.bench_ready) && round < 2) {
  round++
  log('Pre-flight round ' + round + ': ' + pre.defects.length + ' defects, ready=' + pre.ready_for_heldout + ', bench=' + pre.bench_ready)
  const fix = await agent(pFix(JSON.stringify(pre.defects, null, 1), round), { label: 'fix:preflight-' + round, phase: 'Preflight', schema: BUILD_SCHEMA, ...OPUS })
  const refreeze = await agent(pG2('This is a re-freeze after pre-flight fix round ' + round + ': code hashes may have changed. Re-freeze every declared record and the question code with the current code hash, re-run every lock-check, and rewrite PLAN.json only if it must change. Step a is already done; confirm selection_v2.json is unchanged.'), { label: 'grade:refreeze-' + (round + 1), phase: 'Preflight', schema: FREEZE_SCHEMA, ...OPUS })
  fixes.push({ round, fix, refreeze })
  if (refreeze) freeze = refreeze
  pre = await agent(pP1('This is re-audit number ' + (round + 1) + '. A fix lane and a re-freeze ran after the previous audit (its notes are in audit/preflight/). Re-check everything, not only the earlier defects.'), { label: 'audit:preflight-' + (round + 1), phase: 'Preflight', schema: AUDIT_SCHEMA, ...OPUS })
}
if (!pre || !pre.ready_for_heldout) return { stopped: 'preflight', lock, benchWrite, freeze, fixes, pre }

phase('Timing')
let bench = null
if (pre.bench_ready) {
  bench = await agent(pB2, { label: 'bench:run', phase: 'Timing', schema: BENCH_SCHEMA, ...OPUS })
} else {
  log('Engine timing skipped: the pre-flight audit says the benches are not ready.')
}

phase('Held-out')
const run = await agent(pR1, { label: 'run:heldout', phase: 'Held-out', schema: RUN_SCHEMA, ...OPUS })
if (!run || !run.complete) return { stopped: 'runner', lock, benchWrite, freeze, fixes, pre, bench, run }

phase('Score')
const grade = await agent(pG3, { label: 'grade:heldout', phase: 'Score', schema: HELDOUT_SCHEMA, ...OPUS })
if (!grade) return { failed: 'grade', lock, benchWrite, freeze, fixes, pre, bench, run }

phase('Refute')
const ref = await parallel([
  () => agent(pV1, { label: 'refute:rescore', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS }),
  () => agent(pV2, { label: 'refute:discipline', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS }),
  () => agent(pV3, { label: 'refute:claims', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS }),
])

return { lock, benchWrite, freeze, fixes, pre, bench, run, grade, refuters: { rescore: ref[0], discipline: ref[1], claims: ref[2] } }

// ================================================================ run
phase('Research')
const lanesRaw = await parallel([
  () => agent(pR1, { label: 'research:graphify', phase: 'Research', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR2, { label: 'research:temporal', phase: 'Research', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR3, { label: 'research:graphrag', phase: 'Research', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR4, { label: 'research:agents', phase: 'Research', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR5, { label: 'research:evaluation', phase: 'Research', schema: LANE_SCHEMA, ...SONNET }),
])
const names = ['graphify', 'temporal', 'graphrag', 'agents', 'evaluation']
const lanes = {}
lanesRaw.forEach((r, i) => { if (r) lanes[names[i]] = { summary: r.summary, findings: r.findings, what_it_means_for_the_test: r.what_it_means_for_the_test, could_not_verify: r.could_not_verify } })
const missing = names.filter(n => !lanes[n])
if (missing.length) log('Research lanes with no result: ' + missing.join(', '))
if (!lanes.graphify) return { failed: 'research:graphify', lanes }

phase('Critique')
const cr = await parallel([
  () => agent(pC1(lanes), { label: 'critique:assumptions', phase: 'Critique', schema: CRIT_SCHEMA, ...OPUS }),
  () => agent(pC2(lanes), { label: 'critique:architecture', phase: 'Critique', schema: ARCH_SCHEMA, ...OPUS }),
])
const audit = cr[0]
const arch = cr[1]
if (!audit || !arch) return { failed: 'critique', lanes, audit, arch }
const refuter = await agent(pC3(lanes, audit, arch), { label: 'critique:refuter', phase: 'Critique', schema: REFUTE_SCHEMA, ...OPUS })

phase('Design')
const design = await agent(pD1(lanes, audit, arch, refuter), { label: 'design:protocol', phase: 'Design', schema: DESIGN_SCHEMA, ...OPUS })
let designCheck = null
if (design) designCheck = await agent(pD2(design), { label: 'design:check', phase: 'Design', schema: REFUTE_SCHEMA, ...OPUS })

return { lanes, audit, arch, refuter, design, designCheck }

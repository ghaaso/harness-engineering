  return EN.cmp.filter(function(x){ return x.g===g&&x.c===c&&x.k===k; }).map(function(x){
    return {t:ENGS[x.lo==='SQLite'?'relational':x.lo==='LadybugDB'?'ladybug':'duckpgq']+' faster than '+ENGS[x.hi==='SQLite'?'relational':x.hi==='LadybugDB'?'ladybug':'duckpgq']+': '+(x.holds?'the difference holds':'inside the noise, not stated')+(x.q?' (only without the facts load)':''), tip:(x.holds?'Difference that holds':'Inside the noise')+', '+fms(Math.abs(x.d))+' ms wall'+(x.w&&x.w.length?'. '+x.w.map(function(i){return EN.wq[i];}).join(' '):'')};
  });
}

  var p=[], y=8, cls=['primary','S1','base','S2','tq'];

      var right=X(rg[1]);

      else p.push(mark(E[2],X(M.m),cy,E[3],false,6));
      if(lab){ var room=x1-right-10; var lt=room>lab.length*6.4+4 ? txt(right+9,cy+4,lab,{size:12,fill:'var(--ink2)'}) : txt(X(rg[0])-9,cy+4,fitText(lab,Math.max(40,X(rg[0])-x0-12),12),{size:12,fill:'var(--ink2)',anchor:'end'}); if(narrow&&room<=lab.length*6.4+4&&X(rg[0])-x0<lab.length*6.4){ lt=txt(14,cy+20,lab,{size:12,fill:'var(--ink2)'}); rh+=12; } p.push(lt); }
      var tip=E[1]+', '+CLSL[c]+', '+(kind==='warm'?'warm median ':'cold median ')+fms(M.m)+' ms; spread between passes '+fms(M.sp)+' ms; passes '+M.pp.map(fms).join(', ')+' ms'+(kind==='cold'&&r.cwf?'.       p.push('<rect x="0" y="'+y+'" width="'+w+'" height="'+rh+'" fill="transparent" data-tip="'+esc(tip)+'"/>');
      y+=rh;
    });

}

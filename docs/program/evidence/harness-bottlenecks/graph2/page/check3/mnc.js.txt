function find(o, p, out) {
  if (Array.isArray(o)) { if (/must_not|claims|discipline/i.test(p)) out.push([p, o]); o.forEach((x, i) => find(x, p + '[' + i + ']', out)); }
  else if (o && typeof o === 'object') for (const k of Object.keys(o)) find(o[k], p + '.' + k, out);
  return out;
}

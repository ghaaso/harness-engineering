const fs = require('fs');
const P = 'C:/Users/User/Documents/projects/EMS_workshop1/docs/program/evidence/harness-bottlenecks/graph2/page/';
const before = fs.readFileSync(P + 'work3/alltext-before.txt', 'utf8').split(/\r?\n/).map(s => s.trim()).filter(Boolean);
const afterTxt = fs.readFileSync(P + 'work/text/alltext.txt', 'utf8');
const after = new Set(afterTxt.split(/\r?\n/).map(s => s.trim()).filter(Boolean));
const flatAfter = afterTxt.replace(/\s+/g, ' ');
const seen = new Set();
const gone = [];
for (const l of before) {
  if (after.has(l) || seen.has(l)) continue;
  seen.add(l);
  if (flatAfter.includes(l.replace(/\s+/g, ' '))) continue;
  gone.push(l);
}
console.log('lines before', before.length, 'distinct gone', gone.length);
for (const g of gone) console.log('- ' + g.slice(0, 700));
// section headings
const html0 = fs.readFileSync(P + 'work3/index.before-l.html', 'utf8');
const html1 = fs.readFileSync('C:/Users/User/Documents/projects/EMS_workshop1/docs/program/graph-test/index.html', 'utf8');
const ids = h => [...h.matchAll(/<section[^>]*id="([^"]+)"/g)].map(m => m[1]);
console.log('sections before', ids(html0).join(','));
console.log('sections after ', ids(html1).join(','));
console.log('sizes', html0.length, html1.length, Buffer.byteLength(html1));

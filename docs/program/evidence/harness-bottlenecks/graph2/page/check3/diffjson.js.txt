const fs = require('fs');
const base = 'C:/Users/User/Documents/projects/EMS_workshop1/docs/program/evidence/harness-bottlenecks/graph2/';
const a = JSON.parse(fs.readFileSync(base + 'page/work3/k0-result.before.json', 'utf8'));
const b = JSON.parse(fs.readFileSync(base + 'widen/k0/result.json', 'utf8'));
function flat(o, p, out) {
  if (o && typeof o === 'object') { for (const k of Object.keys(o)) flat(o[k], p ? p + '.' + k : k, out); }
  else out[p] = o;
  return out;
}
const fa = flat(a, '', {}), fb = flat(b, '', {});
let changed = [], removed = [], added = [];
for (const k of Object.keys(fa)) {
  if (!(k in fb)) removed.push(k + ' = ' + JSON.stringify(fa[k]).slice(0, 200));
  else if (JSON.stringify(fa[k]) !== JSON.stringify(fb[k])) changed.push(k + ' : ' + JSON.stringify(fa[k]).slice(0, 300) + '  =>  ' + JSON.stringify(fb[k]).slice(0, 300));
}
for (const k of Object.keys(fb)) if (!(k in fa)) added.push(k + ' = ' + JSON.stringify(fb[k]).slice(0, 300));
console.log('top-level keys before', Object.keys(a));
console.log('top-level keys after', Object.keys(b));
console.log('CHANGED', changed.length); changed.forEach(x => console.log('  ' + x));
console.log('REMOVED', removed.length); removed.slice(0, 80).forEach(x => console.log('  ' + x));
console.log('ADDED', added.length); added.slice(0, 200).forEach(x => console.log('  ' + x));

// Checks the built page against data/data.json.

// Reads index.html, the record, and (if present) the all-states text dump made by
// ../evidence/harness-bottlenecks/graph2/page/job-text.json with the screenshot tool.

const html = fs.readFileSync(path.join(HERE, 'index.html'), 'utf8');
const md = fs.readFileSync(path.join(HERE, '..', 'GRAPH-TEST-RESULT-2026-11.md'), 'utf8');
const dumpPath = path.join(HERE, '..', 'evidence', 'harness-bottlenecks', 'graph2', 'page', 'work', 'text', 'alltext.txt');
const problems = [], notes = [];

/* 1. size and sections */
const size = Buffer.byteLength(html);
if (size >= 500000) problems.push('index.html is ' + size + ' bytes (limit 500,000)');
notes.push('index.html ' + size + ' bytes; record ' + md.split('\n').length + ' lines');
if (md.split('\n').length >= 150) problems.push('record has 150 lines or more');
for (const l of 'abcdefghijk') { if (!new RegExp('<section class="sec" id="' + l + '"><h2>').test(html)) problems.push('section ' + l.toUpperCase() + ' missing'); }
/* 1b. scope block first (update 2026-11-13): above the seven cards and section A, linking to K */
{ const iS = html.indexOf('<section class="scope" id="scope"'), iA = html.indexOf('<section class="sec" id="a"'), iV = html.indexOf('class="verdicts');
  if (iS < 0) problems.push('scope block missing');
  else { if (!(iS < iA && (iV < 0 || iS < iV))) problems.push('scope block is not above the seven cards and section A');
    const scope = html.slice(iS, html.indexOf('</section>', iS));
    if (!/href="#k/.test(scope)) problems.push('scope block does not link to section K');
    if ((scope.match(/class="lyr no/g) || []).length < 4) problems.push('scope layer picture shows fewer than four untested layers');
    if ((scope.match(/class="lyr yes/g) || []).length < 1) problems.push('scope layer picture shows no tested layer'); } }
/* 1c. J points to K instead of repeating it; card 7 stays; card 6 reports */
{ const iJ = html.indexOf('<section class="sec" id="j"'), J = html.slice(iJ, html.indexOf('<section class="sec" id="k"'));
  if (!/<a class="pointk" href="#k-wider"/.test(J)) problems.push('section J has no pointer to the wider test in K');
  if (/class="mx"|class="phases"/.test(J)) problems.push('section J repeats the wider-test picture from K');
  if ((html.match(/<div class="vcard/g) || []).length < 7) problems.push('fewer than seven verdict cards');
  if (/measuring now/i.test(html)) problems.push('engines still shown as "measuring now"'); }
/* 1d. section K: every outside source item carries a who tag and a strength tag; the wider test is marked as a design */
{ const iK = html.indexOf('<section class="sec" id="k"'), K = html.slice(iK, html.indexOf('</main>', iK) > 0 ? html.indexOf('</main>', iK) : undefined);
  const nWho = (K.match(/class="tag who/g) || []).length, nStr = (K.match(/class="tag str/g) || []).length;
  if (nWho < 4 || nWho !== nStr) problems.push('section K: who tags (' + nWho + ') and strength tags (' + nStr + ') missing or unpaired');
  else notes.push('section K: ' + nWho + ' outside items, each with a who tag and a strength tag');
  if (!/id="k-wider"/.test(K) || !/DESIGN, NOT A RESULT/.test(K)) problems.push('section K: wider test not marked as a design');
  if ((K.match(/<div class="krow(?! khead)/g) || []).length !== d.k.rows.length) problems.push('section K: owner table rows differ from data (' + d.k.rows.length + ')');
  // the owner's words in the table are verbatim pieces of his three messages (a "..." marks a cut); the one row the design check added says so
  const Q = d.k.owner_quotes.join(' | ').toLowerCase(), H = html.replace(/&#39;/g, "'").replace(/&quot;/g, '"').replace(/&amp;/g, '&');
  let nv = 0;
  d.k.rows.forEach((r, i) => { if (r.words.startsWith('(not in your words')) return;
    const pieces = r.words.split(/\s*\.\.\.\s*/).filter(Boolean);
    if (!pieces.every(p => Q.includes(p.toLowerCase()))) problems.push('section K row ' + (i + 1) + ': words are not verbatim from the owner: ' + r.words.slice(0, 80));
    else if (!pieces.every(p => H.includes(p))) problems.push('section K row ' + (i + 1) + ': owner words not shown on the page');
    else nv++; });
  notes.push('section K: ' + nv + ' rows of the owner\'s words checked verbatim against his messages'); }
/* network: the only addresses allowed are outside-source links (<a href>) to the listed hosts; nothing is loaded */
{ const LINK_HOSTS = ['github.com', 'www.kubeblogs.com', 'rentierdigital.xyz'];
  const stripped = html.replace(/http:\/\/www\.w3\.org\/2000\/svg/g, '').replace(/<a [^>]*href="https:\/\/([^/"]+)[^"]*"/g, (m, h) => LINK_HOSTS.includes(h) ? '<a' : m);
  if (/https?:\/\//.test(stripped)) problems.push('page contains a network address outside the allowed source links');
  const links = (html.match(/<a [^>]*href="https:\/\/[^"]+"/g) || []);
  notes.push(links.length + ' outside-source links (allowed hosts: ' + LINK_HOSTS.join(', ') + ')'); }
if (/<script[^>]+src=|<link[^>]+href=|<img[^>]+src="https?:|@import|url\(\s*["']?https?:/.test(html)) problems.push('page loads an external file');

/* 2. every number in the data */
const vals = [];
(function walk(o) {
  if (o == null) return;
  if (typeof o === 'number') { vals.push(o); return; }
  if (typeof o === 'string') { (o.replace(/,(?=\d{3})/g, '').match(/\d+(?:\.\d+)?/g) || []).forEach(x => vals.push(+x)); return; }
  if (Array.isArray(o)) return o.forEach(walk);
  if (typeof o === 'object') Object.keys(o).forEach(k => { walk(o[k]); (k.match(/\d+(?:\.\d+)?/g) || []).forEach(x => vals.push(+x)); });
})(d);
// derived, documented on the page: 1.96 (standard errors for a 95 % interval), 95 (per cent), counts of shown items
const ALLOW = new Set([1.96, 95]);
const absVals = vals.map(Math.abs);
function inData(tok, isPct) {
  const x = Math.abs(+tok.replace(/,/g, ''));
  if (ALLOW.has(x)) return true;
  const dec = (tok.split('.')[1] || '').length;
  const m = Math.pow(10, dec);
  const eq = v => Math.abs(Math.round(v * m + 1e-7) / m - x) < 1e-9 || Math.abs(Math.round(v * m - 1e-7) / m - x) < 1e-9;
  if (absVals.some(eq)) return true;
  if (isPct && absVals.some(v => Math.abs(Math.round(v * 100) - x) < 1e-9)) return true;
  if (dec === 0 && absVals.some(v => v === x)) return true;
  return false;
}
function pageText() {
  let t = html.replace(/<script[\s\S]*?<\/script>/g, ' ').replace(/<style[\s\S]*?<\/style>/g, ' ').replace(/<[^>]+>/g, '\n').replace(/&amp;/g, '&').replace(/&quot;/g, '"').replace(/&lt;/g, '<').replace(/&gt;/g, '>');
  if (fs.existsSync(dumpPath)) { t += '\n' + fs.readFileSync(dumpPath, 'utf8'); notes.push('all-states dump included (' + fs.statSync(dumpPath).size + ' bytes)'); }
  else problems.push('no all-states dump: run job-text.json first');
  return t;
}
const text = pageText().replace(/−/g, '-');
const lines = text.split('\n');
const bad = new Map();
let checked = 0;
for (const line of lines) {
  // skip file paths and configuration ids (shown as code), dates and times
  const clean = line.replace(/[\w./-]+\.(tsx?|jsx?|py|mjs|json)\b/g, ' ').replace(/\b\d{4}-\d{2}-\d{2}(T[\d:.]+Z)?\b/g, ' ').replace(/\b\d{2}:\d{2}\b/g, ' ').replace(/\b[a-z_]+=[\w.>+-]+/gi, ' ').replace(/\b(T1x?|T[2-5]|H[1-5]b?|S[12]|k[46]|B[12])\b/g, ' ').replace(/#\d+/g, ' ');
  const re = /(\d[\d,]*(?:\.\d+)?)(\s*%)?/g; let mm;
  while ((mm = re.exec(clean))) {
    const tok = mm[1].replace(/,$/, ''); checked++;
    if (!inData(tok, !!mm[2])) { const k = tok + (mm[2] ? ' %' : ''); if (!bad.has(k)) bad.set(k, line.trim().slice(0, 140)); }
  }
}
notes.push(checked + ' number tokens checked');
/* 2b. no broken values in any state: NaN, undefined, null, or a missing-value dash where a number belongs */
['NaN', 'undefined', 'null', '[object Object]'].forEach(w => { if (new RegExp('\\b' + w.replace(/[[\]]/g, '\\$&') + '\\b').test(text.replace(/\b\w+=null\b/g, ' '))) problems.push('page text contains "' + w + '"'); }); // hops=null is a recorded configuration code (unbounded hops), shown in mono beside its plain name
if (/recall –|F1 –|: –|\(– /.test(text)) problems.push('a missing value (–) is shown where a number belongs');
/* 2c. counts in prose are read from data: the sources hold no hand-typed sample sizes */
{ const src = fs.readFileSync(path.join(HERE, 'build.js.txt'), 'utf8') + fs.readFileSync(path.join(HERE, 'src', 'client.js.txt'), 'utf8');
  const typed = src.match(/\b(75|65|418|770|49|48|31|30|23|22) (commits|tasks|questions|times|untouched)\b/g) || [];
  if (typed.length) problems.push('hand-typed counts in the sources: ' + [...new Set(typed)].join(', ')); else notes.push('no hand-typed sample sizes in build.js.txt or client.js.txt'); }
for (const [k, l] of bad) problems.push('number not in data: ' + k + '   in: ' + l);

];

// the generated record (GRAPH-TEST-RESULT-2026-11.md) is held to the same must-not-claim lists as the page

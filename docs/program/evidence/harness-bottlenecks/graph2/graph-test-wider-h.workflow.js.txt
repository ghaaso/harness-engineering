const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

// ================================================================ run
const out = await parallel([
  async () => {
    const page = await agent(pPage, { label: 'page:update', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
    if (!page) return { failed: 'page' }
    const check = await agent(pPageCheck, { label: 'page:check', phase: 'Page', schema: REFUTE_SCHEMA, ...OPUS })
    let fix = null
    if (check && (!check.ok || check.defects.length > 0 || check.mismatches.length > 0)) {
      fix = await agent(pPageFix(JSON.stringify({ ok: check.ok, defects: check.defects, mismatches: check.mismatches, claims: check.claims.filter(c => c.result !== 'survives') }, null, 1)), { label: 'page:apply-check', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
    }
    return { page, check, fix }
  },
  () => agent(pDesign, { label: 'design:v2', phase: 'Design', schema: BUILD_SCHEMA, ...OPUS }),
  async () => {
    const k0 = await agent(pK0, { label: 'k0:replay', phase: 'Cache replay', schema: K0_SCHEMA, ...OPUS })
    if (!k0) return { failed: 'k0' }
    const audit = await agent(pK0Audit, { label: 'k0:audit', phase: 'Cache replay', schema: REFUTE_SCHEMA, ...OPUS })
    return { k0, audit }
  },

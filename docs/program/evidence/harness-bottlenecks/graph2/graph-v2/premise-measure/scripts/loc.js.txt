// ---- shell parsing: cd, VAR=, $VAR expansion, candidate path tokens
function shellParse(cmd, cwd0) {
  const vars = {}; let cwd = cwd0; const reads = [], writes = [];
  const segs = String(cmd || '').split(/&&|\|\||;|\n|\|/);
  for (let seg of segs) {
    seg = seg.trim(); if (!seg) continue;
    seg = seg.replace(/\$\{?(\w+)\}?/g, (m, v) => (vars[v] !== undefined ? vars[v] : m));
    let m;
    while ((m = seg.match(/^(?:export\s+)?([A-Za-z_]\w*)=("([^"]*)"|'([^']*)'|(\S*))\s*/))) { vars[m[1]] = m[3] ?? m[4] ?? m[5] ?? ''; seg = seg.slice(m[0].length); }
    if ((m = seg.match(/^cd\s+("([^"]+)"|'([^']+)'|(\S+))/))) { const d = norm(m[2] || m[3] || m[4]); cwd = isAbs(d) ? d : norm(cwd + '/' + d); continue; }
    // writes
    for (const r of seg.matchAll(/(?:^|[^0-9&2])>>?\s*("([^"]+)"|'([^']+)'|([^\s;|&)]+))/g)) { const w = r[2] || r[3] || r[4]; if (w && !/^\/dev\/null$|^&/.test(w)) writes.push(w); }
    for (const r of seg.matchAll(/\b(?:tee(?:\s+-a)?|Out-File(?:\s+-FilePath)?|Set-Content(?:\s+-Path)?|mkdir\s+-p)\s+("([^"]+)"|'([^']+)'|(\S+))/g)) writes.push(r[2] || r[3] || r[4]);
    const toks = [...seg.matchAll(/"([^"]*)"|'([^']*)'|(\S+)/g)].map((x) => x[1] ?? x[2] ?? x[3]);
    const head = toks[0] || '';
    if (/^(cp|mv)$/.test(head) && toks.length >= 3) writes.push(toks[toks.length - 1]);
    for (let k = 1; k < toks.length; k++) {
      const t = toks[k]; if (!t || t.startsWith('-') || /[|^*?()\[\]{}$<>]|\\\||\\[bdswn.(]/.test(t)) continue;
      if (!(/[\/\\]/.test(t) || /\.[A-Za-z]{1,5}$/.test(t))) continue;
      if (/^https?:/.test(t)) continue;
      const n = norm(t); let abs = isAbs(n) ? n : norm(cwd + '/' + n);
      if (!isAbs(n) && /^(docs|backend|src|technician-app|ops|scripts)\//.test(n) && !fs.existsSync(abs) && fs.existsSync(REPO + '/' + n)) abs = REPO + '/' + n;
      reads.push(fixDup(abs));
    }
  }
  const wabs = writes.map((w) => { const n = norm(w); return isAbs(n) ? n : norm(cwd + '/' + n); });
  return { reads, writes: wabs };
}

// ---- classify one absolute path at a time
function classify(abs, tsRead) {
  const s = abs; const low = s.toLowerCase();
  const rel = relOf(s);
  if (rel !== null) {
    const r = rel.toLowerCase();
    const isDir = !/\.[a-z0-9]{1,5}$/.test(r);
    const at = isDir ? dirAdded.get(r.replace(/\/$/, '')) : added.get(r);
    const program = r.startsWith('docs/program') || r === 'docs' ;
    let vc; if (at === undefined) vc = 'never'; else if (at <= tsRead) vc = 'tracked'; else vc = 'later';
    if (vc === 'tracked') return { cls: program ? '2 tracked program document' : '1 product code or tracked product document', vc, rel: r, isDir };
    return { cls: program || !/^(backend|src|technician-app|ops|scripts|public|\.github)\//.test(r) ? '3 program file not in version control when read' : '3b product-area file not in version control when read', vc, rel: r, isDir };
  }
  if (/\/\.claude\/projects\/[^/]+\/memory\//i.test(s)) return { cls: '7 other outside the repository (memory notes)' };
  if (/\/\.claude\/projects\//i.test(s)) return { cls: '5 session transcript' };
  if (/bundled-skills|\/\.claude\/(skills|plugins)\//i.test(s)) return { cls: '7 other outside the repository (skill and tool files)' };
  if (/\/Temp\/|scratchpad|\/\.fixflow-cache\//i.test(s)) {
    // a scratch clone or copy of the repository: the tail is a repository path known to git
    const m = s.match(/\/((?:docs|backend|src|technician-app|ops|scripts)\/.+|CLAUDE\.md|ARCHITECTURE\.md)$/);
    if (m) { const r = m[1].toLowerCase(); const at = added.get(r); if (at !== undefined && at <= tsRead) return { cls: r.startsWith('docs/program') ? '2 tracked program document' : '1 product code or tracked product document', vc: 'tracked', rel: r, clone: true }; }
    return { cls: '4 scratch folder outside the repository' };
  }
  if (/Clinic|24_Clinic/i.test(s)) return { cls: '7 other outside the repository (the other codebase)' };
  return { cls: '7 other outside the repository (elsewhere)' };
}

  ].join('\n')
}

const opts = (label, ph, schema, effort) => ({ label, phase: ph, schema, model: 'opus', effort: effort || 'high', agentType: 'general-purpose' })
const brief = o => o ? JSON.stringify(o).slice(0, 6000) : 'NO RESULT (the agent failed)'

phase('Fix')
const a1 = await agent(pA1, opts('fix:harness+rerun', 'Fix', BUILD_SCHEMA))
if (!a1) return { failed: 'harness fix returned nothing' }
const [a2, a3] = await parallel([
  () => agent(pA2L, opts('fix:ladybug', 'Fix', BUILD_SCHEMA)),
  () => agent(pA2D, opts('fix:duckpgq', 'Fix', BUILD_SCHEMA)),
])
log('harness ' + a1.status.slice(0, 120) + ' | ladybug ' + (a2 ? a2.status.slice(0, 120) : 'FAILED') + ' | duckpgq ' + (a3 ? a3.status.slice(0, 120) : 'FAILED'))
if (!a2 || !a3) return { harnessFix: a1, ladybugFix: a2, duckpgqFix: a3, failed: 'a graph arm fix returned nothing' }

phase('Score 1')
const g1 = await agent(pG1 + '\n\nFIXERS SELF-REPORTS (verify what you rely on)\nharness: ' + brief(a1) + '\nladybug: ' + brief(a2) + '\nduckpgq: ' + brief(a3), opts('grade:stage1', 'Score 1', GRADE1_SCHEMA))
if (!g1 || !g1.gate_ok) {
  log('stage-1 gate not passed: ' + (g1 ? g1.gate_detail.slice(0, 300) : 'grader failed'))
  return { harnessFix: a1, ladybugFix: a2, duckpgqFix: a3, grade1: g1, stopped: 'gate' }
}
log('stage 1 scored; gate ok')

phase('Tune')
const tuners = await parallel([
  () => agent(pTR, opts('tune:relational', 'Tune', BUILD_SCHEMA)),
  () => agent(pTS, opts('tune:search', 'Tune', BUILD_SCHEMA)),
  () => agent(pTL, opts('tune:ladybug', 'Tune', BUILD_SCHEMA)),
  () => agent(pTD, opts('tune:duckpgq', 'Tune', BUILD_SCHEMA)),
])
const names = ['relational', 'search', 'ladybug', 'duckpgq']
const tunersBrief = tuners.map((t, i) => names[i] + ': ' + (t ? JSON.stringify({ status: t.status, key_numbers: t.key_numbers, problems: t.problems }).slice(0, 2500) : 'NO RESULT')).join('\n')
log('tuners done: ' + tuners.map((t, i) => names[i] + (t ? ' ok' : ' FAILED')).join(', '))

phase('Select')
const g2 = await agent(pG2(JSON.stringify({ leaderboard: g1.leaderboard, tq: g1.tq, observations: g1.observations, problems: g1.problems }).slice(0, 7000), tunersBrief), opts('grade:select', 'Select', SELECT_SCHEMA))

phase('Audit')
const v = g2 ? await agent(pV(JSON.stringify(g2).slice(0, 9000)), opts('audit:selection', 'Audit', AUDIT_SCHEMA)) : null
log('selection ' + (g2 ? 'done' : 'FAILED') + ' | audit ' + (v ? ('ready_for_heldout=' + v.ready_for_heldout + ', defects ' + v.defects.length) : 'FAILED'))
return { harnessFix: a1, ladybugFix: a2, duckpgqFix: a3, grade1: g1, tuners: { relational: tuners[0], search: tuners[1], ladybug: tuners[2], duckpgq: tuners[3] }, selection: g2, audit: v }

const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

// ================================================================ run
const branches = await parallel([
  async () => {
    phase('Corrections')
    const c1 = await agent(pC1, { label: 'grade:corrections', phase: 'Corrections', schema: BUILD_SCHEMA, ...OPUS })
    if (!c1) return { failed: 'corrections' }
    const c2 = await agent(pC2, { label: 'verify:corrections', phase: 'Corrections', schema: REFUTE_SCHEMA, ...OPUS })
    let c3 = null
    if (c2 && (c2.mismatches.length > 0 || c2.defects.length > 0)) {
      c3 = await agent(pC3(JSON.stringify({ mismatches: c2.mismatches, defects: c2.defects, claims: c2.claims.filter(c => c.result !== 'survives') }, null, 1)), { label: 'grade:apply-check', phase: 'Corrections', schema: BUILD_SCHEMA, ...OPUS })
    }
    const page = await agent(pP1, { label: 'page:build', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
    return { corrections: c1, check: c2, applied: c3, page }
  },
  async () => {
    const bench = await agent(pB4, { label: 'bench:run', phase: 'Timing', schema: BENCH_SCHEMA, ...OPUS })
    let benchAudit = null
    if (bench) benchAudit = await agent(pB5, { label: 'bench:audit', phase: 'Timing', schema: REFUTE_SCHEMA, ...OPUS })
    return { bench, benchAudit }
  },
])

return { report: branches[0], timing: branches[1] }

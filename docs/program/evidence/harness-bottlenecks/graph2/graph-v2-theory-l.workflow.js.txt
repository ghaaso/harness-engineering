  'THE GROUND (what agents really ask, and what v1 is):', JSON.stringify(ground, null, 1), '',
  withLanes(lanes),
  'THE VERIFIERS (corrections; they override the lanes):', JSON.stringify(verifs, null, 1), '',
  'WRITE the design in this order, each part short and exact:',
  '1. THE COMPETENCY QUESTIONS the graph must answer, taken from the recorded use (ranked by what they cost today), not from what a graph can do. Mark the ones that are not graph questions and are left to reading.',
  '2. THE MODEL: node kinds, relation kinds, properties, where an n-ary or hyper relation is used, identity over time, confidence and evidence; each derived from a competency question, with the modelling theory it follows.',
  '3. CONSTRUCTION: how each relation is extracted, with which method and tool, and the accuracy to expect; what replaces name matching.',
  '4. DERIVED FACTS, TIME AND INVALIDATION: base facts, rules, materialised answers, provenance, incremental maintenance; where a conclusion made by a model sits and how it is kept honest.',
  '5. RANKING: the principled ranker, its features, its training and validation without leaking the future, and how it is compared with v1 and with the plain changed-together list.',
  '6. SELECTION: how the smallest sufficient answer for a token budget is chosen.',
  '7. THE AGENT TOOL: operations, arguments, output format and caps, the description text budget; each tied to its published source.',
  '8. STORAGE: the deliberate choices, and what is not worth doing at this size.',
  'For EVERY component give: the theory and source it comes from; what it replaces in v1; the alternatives rejected and why; the expected effect and why; the test on TUNING data that decides whether it stays (an ablation with a stated threshold: a component that does not beat the simpler form on tuning data is cut); the build cost; what must be installed (name, source, size, offline or not); and a tier: tier 0 = plain code, no install, can start now; tier 1 = needs an install the owner must approve; tier 2 = only pays off at a larger scale, not built now.',
  'ALSO WRITE: the list "not worth it at this size" with the reason and the scale at which it would be; the validation plan (only tuning data and rolling-origin splits: the 75 held-out commits of the finished test are spent and may be used only as a final descriptive check, declared as such; fresh commits after the frozen commit are few); the build order; the decisions that are the owner (installs, each with size and source; any step that sends content to a model; the order against other work); and a table of the owner own words against the component that answers each, with no gap left unnamed.',
  'Be honest about the ceiling: by the owner own records a knowledge layer acts on about a quarter of the tokens (fetched content) and on the number of lookup steps; say what share of that the design could plausibly move and that only an agent test can show it.',
  'Keep DESIGN.md under 350 lines, plain language, every term defined once. No component without a source; no source without a link or a full citation.',
].join('\n')

// ---------------------------------------------------------------- Refute
const pRefute = (design, ground) => common(
  'REFUTER. Attack the graph v2 design before anything is built.',
  OUT + '/REFUTE.md'
) + [
  'THE DESIGN (JSON; the full text is ' + OUT + '/DESIGN.md, read it, and the lane and verifier notes in the same folder):', JSON.stringify(design, null, 1), '',
  'THE GROUND:', JSON.stringify(ground, null, 1), '',
  'ATTACK: (1) Is any component still improvised: a choice with no source, or a source that does not say what the design claims? (2) Over-engineering: which components answer a question agents rarely ask (check against the recorded use), or give a benefit that a cheaper thing already gives (the changed-together list, a grep, a file the agent already has in its prefix)? (3) Scale: which components import a technique whose published benefit exists only at a scale far above this graph? (4) Learning on little data: about 65 tuning commits and 140 in all: can the ranker be trained and validated honestly, or will it fit noise; is the rolling-origin plan sound; is there any path by which the spent held-out commits or the future leak in (history edges, embeddings, document text written after the fact)? (5) Is each tuning test able to fail: is the threshold stated, is the simpler form really the strongest simple form? (6) Does the design serve the measured use, or the questions a graph is good at? (7) What is missing that the theory clearly calls for? (8) Are the owner words covered honestly: graph theory, knowledge graph, temporal knowledge graph, spatial and temporal topology, caching, invalidation, semantic relations, small and fast? (9) Does anything in it promise a token saving that only an agent test can show?',
  'Return each claim of the design with result "survives", "needs qualification" (give it) or "refuted"; the components to cut; the components missing; and the sentences nobody may say to the owner about this design.',
].join('\n')

const SONNET = { model: 'sonnet', effort: 'high', agentType: 'general-purpose' }
const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

// ================================================================ run
const first = await parallel([
  () => agent(pUse, { label: 'ground:use-cases', phase: 'Ground', schema: USE_SCHEMA, ...OPUS }),
  () => agent(pAudit, { label: 'ground:v1-audit', phase: 'Ground', schema: AUDIT_SCHEMA, ...OPUS }),
  () => agent(pR1, { label: 'theory:construction', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR2, { label: 'theory:modelling', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR3, { label: 'theory:ranking', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR4, { label: 'theory:selection-interface', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR5, { label: 'theory:time-incremental', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
  () => agent(pR6, { label: 'theory:storage', phase: 'Theory', schema: LANE_SCHEMA, ...SONNET }),
])
const ground = { use_cases: first[0], v1_audit: first[1] }
const names = ['construction', 'modelling', 'ranking', 'selection_interface', 'time_incremental', 'storage']
const lanes = {}
names.forEach((n, i) => { const r = first[i + 2]; if (r) lanes[n] = { summary: r.summary, findings: r.findings, recommendations: r.recommendations, earlier_research_missed: r.earlier_research_missed, could_not_verify: r.could_not_verify } })
const missing = names.filter(n => !lanes[n])
if (missing.length) log('Theory lanes with no result: ' + missing.join(', '))
if (!ground.use_cases) log('The use-case lane returned nothing: the design will say so.')
if (Object.keys(lanes).length < 4) return { failed: 'theory', ground, lanes }

phase('Verify')
const pick = (keys) => { const o = {}; keys.forEach(k => { if (lanes[k]) o[k] = lanes[k] }); return o }
const ver = await parallel([
  () => agent(pVerify('a', pick(['construction', 'modelling', 'storage'])), { label: 'verify:construction-modelling-storage', phase: 'Verify', schema: VERIFY_SCHEMA, ...OPUS }),
  () => agent(pVerify('b', pick(['ranking', 'selection_interface', 'time_incremental'])), { label: 'verify:ranking-selection-time', phase: 'Verify', schema: VERIFY_SCHEMA, ...OPUS }),
])
const verifs = { a: ver[0], b: ver[1] }

phase('Design')
const design = await agent(pDesign(ground, lanes, verifs), { label: 'design:graph-v2', phase: 'Design', schema: DESIGN_SCHEMA, ...OPUS })
if (!design) return { failed: 'design', ground, lanes, verifs }

phase('Refute')
const refute = await agent(pRefute(design, ground), { label: 'refute:graph-v2', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS })

return { ground, lanes, verifs, design, refute }

const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

// ================================================================ run
phase('Install')
const install = await agent(pInstall, { label: 'install:graphify', phase: 'Install', schema: BUILD_SCHEMA, ...OPUS })
if (!install || !install.ok) return { stopped: 'install', install }

phase('Fit')
const fit = await agent(pFit, { label: 'fit:tuning', phase: 'Fit', schema: BUILD_SCHEMA, ...OPUS })
if (!fit || !fit.ok) return { stopped: 'fit', install, fit }

phase('Select')
const select = await agent(pSelect, { label: 'grade:select', phase: 'Select', schema: SELECT_SCHEMA, ...OPUS })
if (!select || !select.ok) return { stopped: 'select', install, fit, select }

phase('Held-out')
const run = await agent(pRun, { label: 'run:heldout', phase: 'Held-out', schema: BUILD_SCHEMA, ...OPUS })
if (!run || !run.ok) return { stopped: 'runner', install, fit, select, run }

phase('Score')
const score = await agent(pScore, { label: 'grade:heldout', phase: 'Score', schema: SCORE_SCHEMA, ...OPUS })
if (!score) return { failed: 'score', install, fit, select, run }

phase('Refute')
const ref = await parallel([
  () => agent(pV1, { label: 'refute:run', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS }),
  () => agent(pV2, { label: 'refute:claims', phase: 'Refute', schema: REFUTE_SCHEMA, ...OPUS }),
])
const v1 = ref[0]
const v2 = ref[1]

phase('Page')
let page = null
let pageCheck = null
let pageFix = null
if (v1 && v1.ok) {
  page = await agent(pPage(score, v1, v2), { label: 'page:graphify', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
  if (page) {
    pageCheck = await agent(pPageCheck, { label: 'page:check4', phase: 'Page', schema: REFUTE_SCHEMA, ...OPUS })
    if (pageCheck && (!pageCheck.ok || pageCheck.defects.length > 0 || pageCheck.mismatches.length > 0)) {
      pageFix = await agent(pPageFix(JSON.stringify({ ok: pageCheck.ok, defects: pageCheck.defects, mismatches: pageCheck.mismatches, claims: pageCheck.claims.filter(c => c.result !== 'survives') }, null, 1)), { label: 'page:apply-check4', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
    }
  }
} else {
  log('Page not updated: the run audit did not pass, so the result is not shown until it is judged.')
}

return { install, fit, select, run, score, refuters: { run: v1, claims: v2 }, page, pageCheck, pageFix }

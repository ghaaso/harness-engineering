// rooted PageRank by power iteration over an index; dangling mass returns to the root
function rootedPR(root, alpha, iters) {

    p = q; }
  const out = new Map(); ids.forEach((x, i) => { if (p[i]) out.set(x, p[i]); }); return out;
}

const OPUS = { model: 'opus', effort: 'high', agentType: 'general-purpose' }

const out = await parallel([
  async () => {
    const page = await agent(pPage, { label: 'page:graph-v2', phase: 'Page', schema: BUILD_SCHEMA, ...OPUS })
    if (!page) return { failed: 'page' }
    const check = await agent(pCheck, { label: 'page:check', phase: 'Check', schema: CHECK_SCHEMA, ...OPUS })
    let fix = null
    if (check && (!check.ok || check.defects.length > 0 || check.mismatches.length > 0)) {
      fix = await agent(pFix(JSON.stringify(check, null, 1)), { label: 'page:apply-check', phase: 'Check', schema: BUILD_SCHEMA, ...OPUS })
    }
    return { page, check, fix }
  },
  () => agent(pM1, { label: 'measure:lookup-testset', phase: 'Measure', schema: MEASURE_SCHEMA, ...OPUS }),
  () => agent(pM2, { label: 'measure:premises', phase: 'Measure', schema: MEASURE_SCHEMA, ...OPUS }),
])

return { page: out[0], lookups: out[1], premises: out[2] }

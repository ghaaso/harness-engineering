const SECTION_RE = /^\s*(?:#{1,4}\s*)?(?:\*\*)?(\d)\.\s+(?:Primary Request|Key Technical|Files and Code|Errors and [Ff]ixes|Problem Solving|All [Uu]ser [Mm]essages|Pending Tasks|Current Work|Optional Next Step)/;
const isRefNotPath = (r) => !EXT_RE.test(r) && !/^\d+(\.\d+)*$/.test(r);
function pathFromTick(c) {
  let s = c.trim().replace(/(?::\d+(?:-\d+)?|#L\d+(?:-L?\d+)?)$/, '');
  if (!/[\\/]/.test(s) || /[<>*?{}|$"'`…,;=]|\.\.\.|^https?:|^\w+:\/\/|^-/.test(s) || !EXT_RE.test(s)) return null;
  const abs = /^[A-Za-z]:[\\/]/.test(s);
  if (/\s/.test(s) && !abs) return null;
  const segs = s.split(/[\\/]+/).filter((x, i) => x !== '' || i > 0);
  for (const g of segs) if (g !== '' && !/^[\w.@~+\- ()\[\]:]+$/.test(g)) return null;
  const last = segs[segs.length - 1];
  if (!/^[\w.@+\-]+\.[A-Za-z0-9]+$/.test(last)) return null;
  // repair R3 after the first count: shorthand for several files is not one path
  if (/[^\\/.]\.\.[^\\/.]/.test(s)) return null;                                                  // a range, "006..010"
  if (/XX|NN|(?<![A-Za-z])N(?![A-Za-z])|[a-z]N\.[A-Za-z0-9]+$/.test(s)) return null;               // a placeholder name
  for (let i = 0; i < segs.length - 1; i++) if (EXT_RE.test(segs[i])) return null;                // "a.md/b.md": two names
  for (let i = 1; i < segs.length - 1; i++) if (segs[i].length === 1) return null;                // "name-a/b/c.md"
  const stem = last.replace(/\.[^.]+$/, ''), prev = segs.length > 1 ? segs[segs.length - 2] : '';
  if ((/^\d+$/.test(stem) || stem.length === 1) && /(\d|-[A-Za-z])$/.test(prev)) return null;      // "ref-1/2.md"
  return s;
}
// One claim = kind + subject + value, once per summary.
function extractClaims(text, sid8s) {
  const out = new Map(), lines = text.split('\n'); let sec = 0;
  const add = (kind, subj, val, li, col, len, extra) => {
    const key = kind + '|' + subj + '|' + val; if (out.has(key)) return;
    const L = lines[li], a = Math.max(0, col - 230), b = Math.min(L.length, col + len + 230);
    out.set(key, Object.assign({ kind, subj, val, sec, li, ctx: (a > 0 ? '…' : '') + L.slice(a, b) + (b < L.length ? '…' : '') }, extra || {}));
  };
  for (let li = 0; li < lines.length; li++) {
    const raw = lines[li]; const hm = SECTION_RE.exec(raw); if (hm) sec = +hm[1];
    if (/read the full transcript at:|\.claude[\\/]projects[\\/]/.test(raw)) continue;          // the harness's own footer
    // ---- file paths: backticked tokens first, then bare tokens outside backticks
    let bare = raw.replace(/`([^`\n]+)`/g, (m, c, off) => { const p = pathFromTick(c); if (p) add('path', normPath(p), '', li, off, m.length, { shown: p }); return ' '.repeat(m.length); });
    for (const m of bare.matchAll(RX.barePath)) { const p = pathFromTick(m[1]); if (p && !/\s/.test(p)) add('path', normPath(p), '', li, m.index, m[0].length, { shown: p }); }
    // ---- everything else works on the line without backticks and bold marks
    const pl = raw.replace(/[`*]/g, '');
    for (const m of pl.matchAll(RX.hex)) {
      const h = m[1]; if (!/\d/.test(h) || !/[a-f]/.test(h)) continue;
      if (!(h.length <= 12 || h.length === 40)) continue;
      if (sid8s.has(h.slice(0, 8)) && h.length <= 8) continue;                                     // a session id's first group
      const before = pl.slice(Math.max(0, m.index - 16), m.index);
      if (/(sha-?256|sha-?1|md5|digest|checksum)\W{0,4}$/i.test(before) || /(session|sessions|agent|task|local_)\W{0,2}$/i.test(before)) continue;
      add('commit', h, '', li, m.index, h.length);
    }
    if (sec === 6) continue;                                // "All user messages" quotes what was said then; the kinds below state where things stand (repair R5)
    const narr = (idx) => RX.narr.test(pl.slice(Math.max(0, idx - 50), idx));
    const later = (end) => /^\W{0,3}(or later|or newer)\b/.test(pl.slice(end, end + 16));
    for (const m of pl.matchAll(RX.headA)) if (isRefNotPath(m[1]) && !narr(m.index) && !later(m.index + m[0].length)) add('head', m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headB)) if (isRefNotPath(m[1]) && !narr(m.index)) add('head', m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headC)) if (!narr(m.index) && !/review|PASS|FAIL|verdict|round|second-pass/i.test(m[0])) add('head', '#' + m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headD)) {
      if (narr(m.index)) continue;
      const pre = pl.slice(Math.max(0, m.index - 120), m.index); let subj = null, at = -1;
      for (const r of pre.matchAll(/#(\d{1,4})\b/g)) if (r.index > at) { at = r.index; subj = '#' + r[1]; }
      for (const r of pre.matchAll(new RegExp('(?<![\\w\\/.\\-])' + REF + '(?![\\w\\/])', 'g'))) if (r.index > at && isRefNotPath(r[1])) { at = r.index; subj = r[1]; }
      if (subj && /[0-9a-f]{7,40}/.test(pre.slice(at + subj.length))) continue;                       // another id sits between the ref and this one
      add('head', subj || '@own', m[1], li, m.index, m[0].length);                                    // no branch named: the session's own branch (addition A1)
    }
    const prGuarded = (idx) => RX.prGuard.test(pl.slice(Math.max(0, idx - 40), idx));
    for (const m of pl.matchAll(RX.prA)) { if (prGuarded(m.index)) continue; const st = m[2].toLowerCase(); add('pr', '#' + m[1], st === 'draft' ? 'open' : st, li, m.index, m[0].length); }
    for (const m of pl.matchAll(RX.prB)) { if (prGuarded(m.index)) continue; const st = m[1].toLowerCase(); add('pr', '#' + m[2], st === 'draft' ? 'open' : st, li, m.index, m[0].length); }
    for (const m of pl.matchAll(RX.mergedAs)) if (!prGuarded(m.index)) add('merge', '#' + m[1], m[2], li, m.index, m[0].length);   // addition A2
    const tCtx = RX.tCtx.test(pl) || /\d\s+passed\b/.test(pl), instr = (idx) => RX.instr.test(pl.slice(Math.max(0, idx - 60), idx));
    if (tCtx) {
      for (const m of pl.matchAll(RX.t1)) if (!instr(m.index)) add('test', UNIT[m[2].toLowerCase()], numNorm(m[1]), li, m.index, m[0].length);
      for (const m of pl.matchAll(RX.t2)) if (!instr(m.index)) add('test', 'passed', numNorm(m[1]), li, m.index, m[0].length);
      for (const m of pl.matchAll(RX.t3)) if (!instr(m.index)) add('test', 'passed', numNorm(m[1]), li, m.index, m[0].length);
    }
    for (const m of pl.matchAll(RX.cnt)) {
      const n = numNorm(m[1]), plural = /s$/i.test(m[2]); if ((n === '1') === plural) continue;        // "USER-15 row", "Item 3 file": a label, not a count
      if (RX.cntLabel.test(pl.slice(Math.max(0, m.index - 12), m.index))) continue;
      add('count', m[2].toLowerCase().replace(/s$/, '') + 's', n, li, m.index, m[0].length);
    }
  }
  return [...out.values()];
}

// ---------- what a tool result or message shows about heads, pull requests and test runs ----------
function headObsFromOutput(cmd, text, push) {
  const c = cmd || '';
  for (const m of text.matchAll(/^\[([^\s\]]+)(?: \(root-commit\))? ([0-9a-f]{7,40})\]/gm)) if (m[1] !== 'detached') push(m[1], m[2], 'commit');
  for (const m of text.matchAll(/^\s*[ +*!=\-]?\s*([0-9a-f]{7,40})\.\.\.?([0-9a-f]{7,40})\s+(\S+)\s+->\s+(\S+)/gm)) {
    if (/^origin\//.test(m[4])) push(m[4], m[2], 'fetch'); else { push('origin/' + m[4], m[2], 'push'); if (m[3] !== 'HEAD') push(m[3], m[2], 'push'); }
  }
  if (/\bgit\b[^|;&]*\bbranch\b/.test(c)) for (const m of text.matchAll(/^[*+ ]\s+(\S+)\s+([0-9a-f]{7,40})\s/gm)) if (!/^\(/.test(m[1])) push(m[1].replace(/^remotes\//, ''), m[2], 'branch');
  if (/worktree\s+list/.test(c)) for (const m of text.matchAll(/^\S.*?\s+([0-9a-f]{7,40})\s+\[([^\]]+)\]/gm)) push(m[2], m[1], 'worktree');
  for (const m of text.matchAll(/^\*?\s*(?:commit\s+)?([0-9a-f]{7,40})\s+\(([^)\n]+)\)/gm)) for (let r of m[2].split(',')) { r = r.trim().replace(/^HEAD -> /, ''); if (!r || r === 'HEAD' || /^tag:/.test(r)) continue; push(r, m[1], 'log'); }
  for (const m of text.matchAll(/^([0-9a-f]{40})\s+refs\/(heads|remotes)\/(\S+)/gm)) push(m[2] === 'remotes' ? m[3] : (/ls-remote/.test(c) ? 'origin/' + m[3] : m[3]), m[1], 'refs');
  const rp = /\bgit\b(?:\s+-C\s+\S+)?\s+rev-parse\s+(?:--short(?:=\d+)?\s+|--verify\s+)*([A-Za-z0-9._\/\-]+)\s*$/.exec(c.trim());
  if (rp && rp[1] !== 'HEAD' && !/^-/.test(rp[1])) { const t = text.trim(); if (/^[0-9a-f]{7,40}$/.test(t)) push(rp[1], t, 'rev-parse'); }
}
function ghObs(text, pushHead, pushPr) {
  if (!/"(?:headRefOid|state|number)"/.test(text)) return;
  // walk JSON-ish objects loosely: split on object boundaries and read the fields that sit together
  for (const chunk of text.split(/\}\s*,\s*\{|\n\s*\n/)) {
    const num = /"number"\s*:\s*(\d+)/.exec(chunk), st = /"state"\s*:\s*"(OPEN|MERGED|CLOSED)"/.exec(chunk), oid = /"headRefOid"\s*:\s*"([0-9a-f]{40})"/.exec(chunk), ref = /"headRefName"\s*:\s*"([^"]+)"/.exec(chunk), mg = /"merged(?:At)?"\s*:\s*(true|"[0-9T:\-Z.]+")/.exec(chunk);
    if (num && (st || mg)) pushPr(+num[1], st ? st[1].toLowerCase() : 'merged', 'gh-json');
    if (oid && num) pushHead('#' + num[1], oid[1], 'gh-json');
    if (oid && ref) pushHead('origin/' + ref[1], oid[1], 'gh-json');
  }
}
function prObsFromOutput(cmd, text, pushPr) {
  for (const m of text.matchAll(/(?:Merged|Squashed and merged|Rebased and merged) pull request [^\n#]*#(\d+)/g)) pushPr(+m[1], 'merged', 'gh-merge');
  for (const m of text.matchAll(/Pull request [^\n#]*#(\d+)[^\n]*was already merged/g)) pushPr(+m[1], 'merged', 'gh-merge');
  for (const m of text.matchAll(/Closed pull request [^\n#]*#(\d+)/g)) pushPr(+m[1], 'closed', 'gh-close');
  if (/\bgh\b[^|;&]*\bpr\s+create\b/.test(cmd || '')) for (const m of text.matchAll(/github\.com\/[^\s\/]+\/[^\s\/]+\/pull\/(\d+)/g)) pushPr(+m[1], 'open', 'gh-create');
  if (/\bgh\b[^|;&]*\bpr\s+list\b/.test(cmd || '')) for (const m of text.matchAll(/^(\d+)\t[^\n]*\t(OPEN|MERGED|CLOSED|DRAFT)\b/gm)) pushPr(+m[1], m[2] === 'DRAFT' ? 'open' : m[2].toLowerCase(), 'gh-list');
  if (/\bgh\b[^|;&]*\bpr\s+view\b/.test(cmd || '')) { const n = /^number:\t(\d+)/m.exec(text), s = /^state:\t(OPEN|MERGED|CLOSED|DRAFT)/m.exec(text); if (n && s) pushPr(+n[1], s[1] === 'DRAFT' ? 'open' : s[1].toLowerCase(), 'gh-view'); }
}
function testCounts(text) {
  const out = [];
  for (const m of text.matchAll(RX.t1)) out.push(UNIT[m[2].toLowerCase()] + '|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.t2)) out.push('passed|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.t3)) out.push('passed|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.tObsA)) out.push(UNIT[m[1].toLowerCase()] + '|' + m[2]);
  for (const m of text.matchAll(RX.tObsB)) out.push(UNIT[m[1].toLowerCase()] + '|' + m[2]);
  return out;
}

// ---------- stage: index ----------
async function findSummaries(f, sid, cutoff) {
  const comps = []; let ln = 0;
  const rl = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
  for await (const line of rl) {
    ln++;
    if (line.startsWith('{"type":"api-request-blob"') || line.startsWith('{"type":"file-history')) continue;
    const isC = line.includes('"compact_boundary"'), isS = line.includes('"isCompactSummary":true');
    if (!isC && !isS) continue;
    let o; try { o = JSON.parse(line); } catch { continue; }
    if (o.isSidechain) continue;
    if (cutoff && o.timestamp && o.timestamp > cutoff) continue;
    if (o.type === 'system' && o.subtype === 'compact_boundary') {
      const m = o.compactMetadata || {}, ps = m.preservedSegment || {};
      comps.push({ line: ln, ts: o.timestamp, pre: m.preTokens || 0, post: m.postTokens || 0, ms: m.durationMs || 0, trigger: m.trigger || '?', precomputed: !!m.precomputed, headUuid: ps.headUuid || null, tailUuid: ps.tailUuid || null, branch: o.gitBranch || '', copied: !!(o.sessionId && o.sessionId !== sid), sumLine: 0, sumTs: null, text: null });
      continue;
    }
    if (o.type === 'user' && o.isCompactSummary && o.message && comps.length && !comps[comps.length - 1].sumLine) {
      const c = comps[comps.length - 1], ct = o.message.content;
      c.sumLine = ln; c.sumTs = o.timestamp; c.text = typeof ct === 'string' ? ct : Array.isArray(ct) ? ct.filter(b => b && b.type === 'text').map(b => b.text || '').join('\n') : '';
    }
  }
  return comps;
}
async function scanTranscript(f, comps, wanted, cutoff) {
  // one pass over the whole file: what the session saw (tool results, user-role messages, attachments) and what it said
  const occ = new Map();                                             // token key -> { seen: [lines], said: [lines] }
  const hit = (key, cls, ln) => { let o = occ.get(key); if (!o) { o = { seen: [], said: [] }; occ.set(key, o); } const a = o[cls]; if (a[a.length - 1] !== ln) a.push(ln); };
  const headObs = [], prObs = [], runs = [], opened = new Map(), uuidLine = new Map(), tsLines = [], models = [];
  const tools = new Map();                                           // tool_use id -> { name, cmd, fp }
  const recent = [];                                                 // the latest calls and messages, for the kept tail (addition A3)
  const post = [];                                                   // per boundary: the next window's first six calls and model messages (scoring 1.10)
  const act = (a) => { recent.push(a); if (recent.length > 800) recent.splice(0, 200); if (a.k !== 'message arrived') for (const p of post) if (p.items.length < 6 && a.line > p.c.line) p.items.push(a); return a; };
  const hex7 = new Map(); for (const h of wanted.hex) { const k = h.slice(0, 7); (hex7.get(k) || hex7.set(k, []).get(k)).push(h); }
  // tokens that can only have come from one place: task ids, hexadecimal ids, long numbers. Their first line in the
  // transcript tells whether a summary that holds one saw the kept tail (added with scoring 1.9).
  const firstTok = new Map();
  const toks = (text) => { const out = new Set(); if (!text) return out; for (const m of text.matchAll(/(?<![A-Za-z0-9_\-\/\\.])(?:[a-z][a-z0-9]{8}|[0-9a-f]{7,40}|\d{5,})(?![A-Za-z0-9_\-])/g)) { const t = m[0]; if (/^\d+$/.test(t) || (/\d.*\d/.test(t) && /[a-z]/.test(t))) { out.add(t); if (out.size >= 400) break; } } return out; };
  const noteToks = (text, ln) => { const s = toks(text); for (const t of s) if (!firstTok.has(t)) firstTok.set(t, ln); return s; };
  const scan = (text, cls, ln, opt) => {
    if (!text) return;
    // repair R1: an id that starts a longer hexadecimal string (a 64-character digest, an upper-case hash) counts as seen
    if (hex7.size) for (const m of text.matchAll(/(?<![0-9a-zA-Z])[0-9a-fA-F]{7,}(?![0-9a-zA-Z])/g)) { const t = m[0].toLowerCase(), c = hex7.get(t.slice(0, 7)); if (c) for (const w of c) if (t === w || t.startsWith(w) || w.startsWith(t)) hit('h|' + w, cls, ln); }
    if (wanted.base.size) {
      const t2 = text.replace(/\\+/g, '/').replace(/\/{2,}/g, '/').toLowerCase().replace(/(?<![a-z0-9])([a-z]):\//g, '/$1/');
      for (const m of t2.matchAll(/[a-z0-9_@+\-][a-z0-9_.@+\-]*\.[a-z0-9]{1,8}(?![a-z0-9_])/g)) {
        const ps = wanted.base.get(m[0]); if (!ps) continue; const end = m.index + m[0].length;
        hit('b|' + m[0], cls, ln);                                   // repair R2: the file name alone is on record
        for (const p of ps) { const st = end - p.length; if (st < 0 || !t2.startsWith(p, st)) continue; const pc = st > 0 ? t2[st - 1] : ''; if (pc && /[a-z0-9_.@+\-]/.test(pc)) continue; hit('p|' + p, cls, ln); }
      }
    }
    const t3 = opt && opt.stripLineNumbers ? text.replace(/^\s*\d+[\t→]/gm, '') : text;
    for (const k of testCounts(t3)) hit('t|' + k, cls, ln);
    for (const m of t3.matchAll(RX.cnt)) hit('c|' + m[2].toLowerCase().replace(/s$/, '') + 's|' + numNorm(m[1]), cls, ln);
    if (wanted.nums.size) for (const m of t3.matchAll(/(?<![\d.,])\d[\d,]*(?![\d]|[.,]\d)/g)) { const n = numNorm(m[0]); if (wanted.nums.has(n)) hit('n|' + n, cls, ln); }
  };
  const proseObs = (text, ln) => {                                   // a message that states a head or a pull request state in the summary's own forms
    for (const c of extractClaims(text, new Set())) { if (c.kind === 'head') headObs.push({ ref: c.subj, sha: c.val, line: ln, form: 'prose' }); else if (c.kind === 'pr') prObs.push({ n: +c.subj.slice(1), state: c.val, line: ln, form: 'prose' }); }
  };
  // where the summary's input ended (scoring 1.2), and what lies between that point and the boundary
  const closeTail = (c) => {
    const headLine = c.headUuid && uuidLine.has(c.headUuid) ? uuidLine.get(c.headUuid) : c.line;
    let tsLine = c.line; if (c.precomputed && c.sumTs) for (const [l, ts] of tsLines) { if (l >= c.line) break; if (ts > c.sumTs) { tsLine = l; break; } }
    c.headLine = headLine; c.inputEnd = Math.min(headLine, tsLine); c.inputEndTs = c.ts;
    for (const [l, ts] of tsLines) if (l >= c.inputEnd) { c.inputEndTs = ts; break; }
    const tail = recent.filter(a => a.line >= c.inputEnd && a.line < c.line);
    // what first appears inside the kept tail, and how much of it the summary holds
    const only = { said: new Set(), seen: new Set() };
    for (const a of tail) { for (const t of a.toks || []) if (firstTok.get(t) >= c.inputEnd) only.said.add(t); for (const t of a.rtoks || []) if (firstTok.get(t) >= c.inputEnd) only.seen.add(t); }
    for (const t of only.seen) only.said.delete(t);
    const held = (set) => [...set].filter(t => c.text.includes(t));
    c.tailOnly = { fromResults: only.seen.size, fromResultsInSummary: held(only.seen).length, fromCalls: only.said.size, fromCallsInSummary: held(only.said).length, ex: held(only.seen).slice(0, 4) };
    // echoes: a run of four words from a tail result that the summary repeats. Pass two drops the ones found earlier in the transcript.
    const sumFlat = c.text.replace(/\s+/g, ' '), cand = new Set();
    for (const a of tail) for (const ln2 of (a.rfull || '').split('\n')) {
      const w = ln2.trim().split(/\s+/).filter(Boolean);
      for (let i = 0; i + 4 <= w.length && cand.size < 12; i++) { const g = w.slice(i, i + 4).join(' '); if (g.length >= 28 && /\d/.test(g) && /^[A-Za-z0-9 =_.,:\/()#%+\-]+$/.test(g) && sumFlat.includes(g)) cand.add(g); }
    }
    c.echoCand = [...cand];
    c.tailKeep = tail.map(a => ({ k: a.k, state: a.state, fp: a.fp, full: a.full }));
    post.push({ c, items: [] });
    c.tail = tail.map(a => { const b = Object.assign({}, a); delete b.toks; delete b.rtoks; delete b.rfull; delete b.full; delete b.fp; return b; });
  };
  let ln = 0, lastModel = '', ci = 0;
  const rl = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
  for await (const line of rl) {
    ln++;
    while (ci < comps.length && ln >= comps[ci].line) { comps[ci].model = lastModel; closeTail(comps[ci]); ci++; }
    if (ci >= comps.length && (post[post.length - 1].items.length >= 6 || ln > comps[comps.length - 1].line + 800)) break;   // six steps after the last boundary, then stop
    if (line.startsWith('{"type":"api-request') || line.startsWith('{"type":"file-history')) continue;
    let o; try { o = JSON.parse(line); } catch { continue; }
    if (o.isSidechain) continue;
    if (cutoff && o.timestamp && o.timestamp > cutoff) continue;
    if (o.uuid) uuidLine.set(o.uuid, ln);
    if (o.timestamp) tsLines.push([ln, o.timestamp]);
    if (o.type === 'assistant' && o.message) {
      const m = o.message; if (m.model && m.model !== '<synthetic>' && m.usage) lastModel = m.model;
      if (Array.isArray(m.content)) for (const b of m.content) {
        if (b.type === 'text') { scan(b.text, 'said', ln); const tk = noteToks(b.text, ln); if ((b.text || '').trim()) act({ line: ln, k: 'model text', state: false, d: b.text.trim().slice(0, 260), toks: tk, full: b.text.slice(0, 6000) }); }
        else if (b.type === 'thinking') { scan(b.thinking, 'said', ln); noteToks(b.thinking, ln); }
        else if (b.type === 'tool_use') {
          const inp = b.input || {}, cl = classifyCall(b.name, inp), all = collectStrings(inp, []).join('\n');
          const a = act({ line: ln, k: cl.k, state: cl.state, tool: b.name, d: describeCall(b.name, inp), toks: noteToks(all, ln), fp: fingerprint(b.name, inp) });
          tools.set(b.id, { name: b.name, cmd: typeof inp.command === 'string' ? inp.command : '', fp: inp.file_path || inp.notebook_path || (b.name === 'Read' || b.name === 'Glob' || b.name === 'Grep' ? inp.path : '') || '', a });
          scan(collectStrings(inp, []).join('\n'), 'said', ln);
        }
      }
      continue;
    }
    if (o.type === 'user' && o.message) {
      const ct = o.message.content;
      if (o.isCompactSummary) { const st = typeof ct === 'string' ? ct : collectStrings(ct, []).join('\n'); scan(st, 'said', ln); noteToks(st, ln); continue; }   // an earlier summary is the model's own words
      if (typeof ct === 'string') { scan(ct, 'seen', ln); proseObs(ct, ln); const tk = noteToks(ct, ln); if (ct.trim()) act({ line: ln, k: 'message arrived', state: false, d: arrived(ct), rtoks: tk }); continue; }
      if (Array.isArray(ct)) for (const b of ct) {
        if (!b) continue;
        if (b.type === 'text') { scan(b.text, 'seen', ln); proseObs(b.text || '', ln); const tk = noteToks(b.text, ln); if ((b.text || '').trim()) act({ line: ln, k: 'message arrived', state: false, d: arrived(b.text), rtoks: tk }); }
        else if (b.type === 'tool_result') {
          const t = tools.get(b.tool_use_id) || { name: '?', cmd: '', fp: '' }, text = typeof b.content === 'string' ? b.content : collectStrings(b.content, []).join('\n');
          const echo = t.name === 'Write' || t.name === 'Edit' || t.name === 'NotebookEdit' || t.name === 'MultiEdit';   // the result repeats what the model wrote
          const rt = noteToks(text, ln);
          if (t.a) { t.a.rLine = ln; t.a.err = !!b.is_error; t.a.r = text.trim().slice(-240); t.a.counts = testCounts(text).slice(-4); t.a.rtoks = rt; t.a.rfull = echo ? '' : text.slice(0, 8000); }
          if (t.fp && !b.is_error) { const p = normPath(String(t.fp)); if (!opened.has(p)) opened.set(p, ln); }
          scan(text, echo ? 'said' : 'seen', ln, { stripLineNumbers: t.name === 'Read' });
          if (echo) continue;
          if (t.name === 'Bash' || t.name === 'PowerShell') {
            headObsFromOutput(t.cmd, text, (ref, sha, form) => headObs.push({ ref, sha, line: ln, form }));
            prObsFromOutput(t.cmd, text, (n, state, form) => prObs.push({ n, state, line: ln, form }));
            const tc = testCounts(text); if (tc.length) runs.push({ line: ln, cmd: sha1(t.cmd.replace(/\s+/g, ' ').replace(/\s*\|\s*(tail|head)\b.*$/, '').trim().toLowerCase()).slice(0, 12), counts: [...new Set(tc)] });
          } else if (t.name === 'Agent' || t.name === 'Task' || /SendMessage|get_status|list_sessions/.test(t.name)) proseObs(text, ln);
          ghObs(text, (ref, sha, form) => headObs.push({ ref, sha, line: ln, form }), (n, state, form) => prObs.push({ n, state, line: ln, form }));
        }
      }
      continue;
    }
    if (o.type === 'attachment' && o.attachment) {
      const t = collectStrings(o.attachment, []).join('\n'); scan(t, 'seen', ln); const tk = noteToks(t, ln);
      if (o.attachment.type === 'queued_command' || o.attachment.type === 'task_status' || o.attachment.type === 'hook_additional_context') { proseObs(t, ln); if (o.attachment.type !== 'hook_additional_context') act({ line: ln, k: 'message arrived', state: false, d: arrived(t), rtoks: tk }); }
      continue;
    }
    if (o.type === 'queue-operation' && typeof o.content === 'string') { scan(o.content, 'seen', ln); proseObs(o.content, ln); continue; }
    if (o.type === 'system' && typeof o.content === 'string') { scan(o.content, 'seen', ln); noteToks(o.content, ln); continue; }
  }
  while (ci < comps.length) { comps[ci].model = lastModel; closeTail(comps[ci]); ci++; }
  // scoring 1.10: did the next window repeat something its kept tail had already done
  const runs3 = (t) => { const w = (t || '').toLowerCase().split(/[^a-z0-9]+/).filter(Boolean), out = new Set(); for (let i = 0; i + 3 <= w.length; i++) out.add(w[i] + ' ' + w[i + 1] + ' ' + w[i + 2]); return out; };
  for (const p of post) {
    const c = p.c, tk = c.tailKeep || [], r = { state: 0, said: 0, shell: 0, sameFile: 0, mostShared: 0, anyCmd: 0, anyCmdClasses: [] }; c.next = [];
    for (const a of p.items) {
      let what = '';
      if (a.k === 'model text') { const A = runs3(a.full); for (const t of tk) { if (t.k !== 'model text') continue; const B = runs3(t.full); if (A.size < 8 || B.size < 8) continue; let inter = 0; for (const x of A) if (B.has(x)) inter++; r.mostShared = Math.max(r.mostShared, Math.round(100 * inter / A.size)); if (inter / A.size > 0.5) { what = 'said'; break; } } }
      else if (a.fp && tk.some(t => t.fp === a.fp && t.k === a.k)) what = a.state ? 'state' : a.k === 'shell' ? 'shell' : '';
      else if (a.fp && a.fp.startsWith('file:') && tk.some(t => t.fp && t.fp.startsWith('file:') && t.fp.slice(0, t.fp.lastIndexOf(':')) === a.fp.slice(0, a.fp.lastIndexOf(':')))) r.sameFile++;   // the same file again with other text: more work on it, not a repeat
      if (a.fp && a.fp.startsWith('cmd:') && tk.some(t => t.fp === a.fp)) { r.anyCmd++; r.anyCmdClasses.push(a.k); }   // any shell or node command run again with the same text, whatever its class (review T-1)
      if (what) r[what]++;
      c.next.push({ k: a.k, state: !!a.state, repeat: what });
    }
    c.repeats = r; delete c.tailKeep;
  }
  const withCand = comps.filter(c => c.echoCand && c.echoCand.length);
  if (withCand.length) {
    const lim = Math.max(...withCand.map(c => c.inputEnd)); let l2 = 0;
    const live = withCand.map(c => ({ c, left: new Set(c.echoCand) }));
    const rl2 = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
    for await (const line of rl2) { l2++; if (l2 >= lim) break; if (line.startsWith('{"type":"api-request') || line.startsWith('{"type":"file-history')) continue;
      for (const x of live) { if (l2 >= x.c.inputEnd || !x.left.size) continue; for (const g of x.left) if (line.includes(g)) x.left.delete(g); } }
    for (const x of live) x.c.echo = x.left.size;
  }
  for (const c of comps) { if (c.echo === undefined) c.echo = 0; delete c.echoCand; }
  return { occ, headObs, prObs, runs, opened, uuidLine, tsLines };
}
// the class of one tool call, for the kept-tail listing (scoring 1.9): which calls change state
function classifyCall(name, inp) {
  if (name === 'Bash' || name === 'PowerShell') {
    const c = String(inp.command || '');
    // options before the verb are dropped first, so no pattern has to walk through quoted values (a nested one hung the run)
    const c2 = c.length > 20000 ? c.slice(0, 20000) : c;
    const flat = c2.replace(/\s-[cC]\s+(?:"[^"\n]{0,300}"|'[^'\n]{0,300}'|[^\s"'=]{1,200}(?:=(?:"[^"\n]{0,300}"|'[^'\n]{0,300}'|[^\s"']{0,300}))?)/g, ' ');
    const gm = [...flat.matchAll(/\bgit\s+(commit|push|merge|rebase|reset|cherry-pick|tag|pull|revert|am|checkout|switch|branch|worktree|add|stash|fetch|status|log|diff|show|rev-parse|ls-files|ls-remote)\b/g)].map(m => m[1]);
    const ghm = [...c.matchAll(/\bgh\b\s+pr\s+(create|merge|close|edit|ready|comment|view|list|checks|status)\b/g)].map(m => m[1]);
    const st = gm.filter(v => /^(commit|push|merge|rebase|reset|cherry-pick|tag|pull|revert|am)$/.test(v)).map(v => 'git ' + v).concat(ghm.filter(v => /^(create|merge|close|edit|ready)$/.test(v)).map(v => 'gh pr ' + v));
    if (st.length) return { k: [...new Set(st)].join(' + '), state: true };
    if (/\b(pytest|jest|vitest|npm (run )?test|check\.py|ff-gates|typecheck|ruff|mypy|lint|merge-ready)\b/.test(c)) return { k: 'test or check run', state: false };
    if (gm.length || ghm.length) return { k: 'git or gh status', state: false };
    return { k: 'shell', state: false };
  }
  if (/^(Write|Edit|NotebookEdit|MultiEdit)$/.test(name)) return { k: 'file write', state: true };
  if (/^(Read|Grep|Glob|ToolSearch)$/.test(name)) return { k: 'read', state: false };
  if (/SendMessage|send_message/.test(name)) return { k: 'message sent', state: true };
  if (/^(Agent|Task|Workflow)$/.test(name) || /start_session|hand_off/.test(name)) return { k: 'agent or session started', state: true };
  if (/update_plan|wrap_up|mark_chapter|TodoWrite/.test(name)) return { k: 'plan update', state: false };
  return { k: 'other tool', state: false };
}
function arrived(t) {                                                // scratch only: what a message that came in says, shortened
  const st = /<status>([^<]*)<\/status>/.exec(t), su = /<summary>([\s\S]*?)<\/summary>/.exec(t);
  if (/<task-notification>/.test(t) && (st || su)) return ('task notification: ' + (st ? st[1] : '') + ' | ' + (su ? su[1] : '')).replace(/\s+/g, ' ').slice(0, 260);
  return t.trim().replace(/\s+/g, ' ').slice(0, 260);
}
function fingerprint(name, inp) {                                    // what makes two calls "the same" (scoring 1.10)
  if (name === 'Bash' || name === 'PowerShell') return 'cmd:' + sha1(String(inp.command || '').replace(/\s+/g, ' ').trim());
  if (/^(Write|Edit|NotebookEdit|MultiEdit)$/.test(name)) return 'file:' + normPath(String(inp.file_path || inp.notebook_path || '')) + ':' + sha1(String(inp.content || '') + '|' + String(inp.new_string || '') + '|' + String(inp.new_source || ''));
  return name + ':' + sha1(collectStrings(inp, []).join('|'));
}
function describeCall(name, inp) {                                   // scratch only: for the hand read
  if (name === 'Bash' || name === 'PowerShell') return String(inp.command || '').replace(/\s+/g, ' ').slice(0, 230);
  if (inp.file_path || inp.notebook_path) return String(inp.file_path || inp.notebook_path);
  return (name + ' ' + collectStrings(inp, []).join(' | ').replace(/\s+/g, ' ')).slice(0, 230);
}
function lastSections(text) {
  const out = { 7: [], 8: [], 9: [] }; let sec = 0;
  for (const l of text.split('\n')) { const hm = SECTION_RE.exec(l); if (hm) sec = +hm[1]; if (/^If you need specific details from before compaction/.test(l)) break; if (out[sec]) out[sec].push(l); }
  const body = text.split(/\nIf you need specific details from before compaction/)[0];
  return { s7: out[7].join('\n'), s8: out[8].join('\n'), s9: out[9].join('\n'), end: body.slice(-1100) };
}
async function stageIndex(cacheDir) {
  fs.mkdirSync(cacheDir, { recursive: true });
  const t0 = Date.now(), startedAt = process.env.SUMMARY_CHECK_CUT || new Date(t0).toISOString();   // pass the first count's cut to keep the same summaries

  const bornBy = (s) => { try { return fs.statSync(path.join(ROOT, s)).birthtime.toISOString() <= startedAt; } catch (e) { return true; } };

      for (const cl of c.claims) {

    const sess = { g: it.g, sid: it.sid, sid8: it.sid.slice(0, 8), headObs: ev.headObs, prObs: ev.prObs, runs: ev.runs, comps: [] };
    comps.forEach((c, i) => {
      const before = (arr, lim) => { let n = 0, first = 0, last = 0; for (const l of arr) { if (l >= lim) break; n++; if (!first) first = l; last = l; } return [n, first, last]; };

        else if (cl.kind === 'path') { cl.ev = stat('p|' + cl.subj); let op = 0; for (const [p, l] of ev.opened) { if (l >= c.line) continue; if (p === cl.subj || p.endsWith('/' + cl.subj) || cl.subj.endsWith('/' + p)) { op = op ?         footer: /written without seeing them/.test(c.text), tail: c.tail, tailOnly: c.tailOnly, echo: c.echo, next: c.next || [], repeats: c.repeats || { state: 0, said: 0, shell: 0, sameFile: 0, mostShared: 0, anyCmd: 0, anyCmdClasses: [] }, claims: c.claims }, lastSections(c.text)));
    });
    out.sessions.push(sess);
    process.stderr.write(`index ${it.g} ${sess.sid8} summaries=${comps.length} claims=${comps.reduce((a, c) => a + c.claims.length, 0)} (${((Date.now() - t0) / 1000).toFixed(0)}s)\n`);
  }
  fs.writeFileSync(path.join(cacheDir, 'index.json'), JSON.stringify(out));
  const ns = out.sessions.reduce((a, s) => a + s.comps.length, 0), nc = out.sessions.reduce((a, s) => a + s.comps.reduce((x, c) => x + c.claims.length, 0), 0);
  console.log(`INDEX_DONE transcripts=${files.length} sessions-with-summaries=${out.sessions.length} summaries=${ns} claims=${nc} copied-markers=${out.copiedMarkers} seconds=${((Date.now() - t0) / 1000).toFixed(0)} cut=${startedAt}`);
}

function git(repo, args, input) { const r = spawnSync('git', ['-C', repo, ...args], { input, encoding: 'utf8', maxBuffer: 1 << 29 }); return { status: r.status, out: r.stdout || '' }; }
const sameSha = (a, b) => !!a && !!b && (a.startsWith(b) || b.startsWith(a));
function loadRepo(g, tokens) {
  const repo = REPOS[g]; const R = { g, repo, ok: fs.existsSync(repo), full: new Map(), ct: new Map(), pathCt: new Map(), baseIdx: new Map(), refLog: new Map(), anc: new Map(), worktrees: [] };
  if (!R.ok) return R;
  const list = [...tokens];
  const bc = git(repo, ['cat-file', '--batch-check'], list.join('\n') + '\n').out.split('\n');
  list.forEach((t, i) => { const m = /^([0-9a-f]{40}) (\w+) /.exec(bc[i] || ''); if (m) R.full.set(t, { sha: m[1], type: m[2] }); });
  for (const l of git(repo, ['log', '--all', '--format=%H %ct']).out.split('\n')) { const [h, c] = l.split(' '); if (h) R.ct.set(h, +c); }
  let ct = 0;
  for (const l of git(repo, ['log', '--all', '--name-only', '--format=@%ct']).out.split('\n')) {
    if (!l) continue; if (l[0] === '@') { ct = +l.slice(1); continue; }
    const p = normPath(l), prev = R.pathCt.get(p); if (prev === undefined || ct < prev) R.pathCt.set(p, ct);
  }
  for (const p of R.pathCt.keys()) { const b = baseKey(p); (R.baseIdx.get(b) || R.baseIdx.set(b, []).get(b)).push(p); }
  R.worktrees = git(repo, ['worktree', 'list', '--porcelain']).out.split('\n').filter(l => l.startsWith('worktree ')).map(l => l.slice(9));
  R.refs = new Set(git(repo, ['for-each-ref', '--format=%(refname:short)']).out.split('\n').filter(Boolean));
  R.isAnc = (x, y) => { const k = x + ' ' + y; if (!R.anc.has(k)) R.anc.set(k, git(repo, ['merge-base', '--is-ancestor', x, y]).status === 0); return R.anc.get(k); };
  R.headAt = (ref, T) => { if (!R.refs.has(ref)) return null; if (!R.refLog.has(ref)) R.refLog.set(ref, git(repo, ['log', '--first-parent', '--format=%H %ct', ref]).out.split('\n').filter(Boolean).map(l => { const [h, c] = l.split(' '); return [h, +c]; })); for (const [h, c] of R.refLog.get(ref)) if (c <= T) return h; return null; };
  R.pathTime = (p) => { const c = R.baseIdx.get(baseKey(p)); if (!c) return undefined; let best; for (const r of c) if (r === p || r.endsWith('/' + p) || p.endsWith('/' + r)) { const t = R.pathCt.get(r); if (best === undefined || t < best) best = t; } return best; };
  // repair R2: a path is on disk when it exists as written, or when a file under the project ends with it
  R.onDisk = (p) => { try { if (/^\/[a-z]\//.test(p)) return fs.existsSync(toFs(p)); for (const root of [repo, ...R.worktrees]) if (fs.existsSync(path.join(root, p))) return true; } catch {} const c = DISK.get(baseKey(p)); if (c) for (const r of c) if (r.endsWith('/' + p)) return true; return false; };
  R.nameKnown = (p) => R.baseIdx.has(baseKey(p)) || DISK.has(baseKey(p));
  return R;
}
const DISK = new Map();                                              // file name -> paths on disk under the two projects and the lab cache
function walkDisk(roots) {
  const skip = new Set(['.git', 'node_modules', '.venv', 'venv', '__pycache__', '.gradle', '.expo', '.pytest_cache', '.mypy_cache', '.ruff_cache']); let n = 0;
  const rec = (dir, depth) => { if (n > 600000 || depth > 14) return; let es; try { es = fs.readdirSync(dir, { withFileTypes: true }); } catch { return; }
    for (const e of es) { if (e.isDirectory()) { if (!skip.has(e.name)) rec(path.join(dir, e.name), depth + 1); } else { n++; const p = normPath(path.join(dir, e.name)), b = baseKey(p); (DISK.get(b) || DISK.set(b, []).get(b)).push(p); } } };
  for (const r of roots) if (fs.existsSync(r)) rec(r, 0);
  return n;
}
function loadGitHub(cacheDir) {

  const r = spawnSync('gh', ['pr', 'list', '--state', 'all', '--limit', '1000', '--json', 'number,state,createdAt,mergedAt,closedAt,isDraft,headRefName,headRefOid,mergeCommit'], { cwd: REPOS.fixflow, encoding: 'utf8', maxBuffer: 1 << 27 });

}
function prCommits(G, cacheDir, n) {

}

  fs.writeFileSync(path.join(outDir, 'check-table.txt'), L.join('\n') + '\n');
  fs.writeFileSync(path.join(outDir, 'check.json'), JSON.stringify(J));
  console.log(`TABLE_DONE summaries=${N} claims=${F2} wrong(hand)=${WH} overtaken=${ov.length} repeats=${rp.length} lines=${L.length}`);
}

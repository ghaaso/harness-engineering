const isRefNotPath = (r) => !EXT_RE.test(r) && !/^\d+(\.\d+)*$/.test(r);
function pathFromTick(c) {
  let s = c.trim().replace(/(?::\d+(?:-\d+)?|#L\d+(?:-L?\d+)?)$/, '');
  if (!/[\\/]/.test(s) || /[<>*?{}|$"'`…,;=]|\.\.\.|^https?:|^\w+:\/\/|^-/.test(s) || !EXT_RE.test(s)) return null;
  const abs = /^[A-Za-z]:[\\/]/.test(s);
  if (/\s/.test(s) && !abs) return null;
  const segs = s.split(/[\\/]+/).filter((x, i) => x !== '' || i > 0);
  for (const g of segs) if (g !== '' && !/^[\w.@~+\- ()\[\]:]+$/.test(g)) return null;
  if (!/^[\w.@+\-]+\.[A-Za-z0-9]+$/.test(segs[segs.length - 1])) return null;
  return s;
}
// One claim = kind + subject + value, once per summary.
function extractClaims(text, sid8s) {
  const out = new Map(), lines = text.split('\n'); let sec = 0;
  const add = (kind, subj, val, li, col, len, extra) => {
    const key = kind + '|' + subj + '|' + val; if (out.has(key)) return;
    const L = lines[li], a = Math.max(0, col - 230), b = Math.min(L.length, col + len + 230);
    out.set(key, Object.assign({ kind, subj, val, sec, li, ctx: (a > 0 ? '…' : '') + L.slice(a, b) + (b < L.length ? '…' : '') }, extra || {}));
  };
  for (let li = 0; li < lines.length; li++) {
    const raw = lines[li]; const hm = SECTION_RE.exec(raw); if (hm) sec = +hm[1];
    if (/read the full transcript at:|\.claude[\\/]projects[\\/]/.test(raw)) continue;          // the harness's own footer
    // ---- file paths: backticked tokens first, then bare tokens outside backticks
    let bare = raw.replace(/`([^`\n]+)`/g, (m, c, off) => { const p = pathFromTick(c); if (p) add('path', normPath(p), '', li, off, m.length, { shown: p }); return ' '.repeat(m.length); });
    for (const m of bare.matchAll(RX.barePath)) { const p = pathFromTick(m[1]); if (p && !/\s/.test(p)) add('path', normPath(p), '', li, m.index, m[0].length, { shown: p }); }
    // ---- everything else works on the line without backticks and bold marks
    const pl = raw.replace(/[`*]/g, '');
    for (const m of pl.matchAll(RX.hex)) {
      const h = m[1]; if (!/\d/.test(h) || !/[a-f]/.test(h)) continue;
      if (!(h.length <= 12 || h.length === 40)) continue;
      if (sid8s.has(h.slice(0, 8)) && h.length <= 8) continue;                                     // a session id's first group
      const before = pl.slice(Math.max(0, m.index - 16), m.index);
      if (/(sha-?256|sha-?1|md5|digest|checksum)\W{0,4}$/i.test(before) || /(session|sessions|agent|task|local_)\W{0,2}$/i.test(before)) continue;
      add('commit', h, '', li, m.index, h.length);
    }
    const narr = (idx) => RX.narr.test(pl.slice(Math.max(0, idx - 50), idx));
    for (const m of pl.matchAll(RX.headA)) if (isRefNotPath(m[1]) && !narr(m.index)) add('head', m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headB)) if (isRefNotPath(m[1]) && !narr(m.index)) add('head', m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headC)) if (!narr(m.index)) add('head', '#' + m[1], m[2], li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.headD)) {
      if (narr(m.index)) continue;
      const pre = pl.slice(Math.max(0, m.index - 120), m.index); let subj = null, at = -1;
      for (const r of pre.matchAll(/#(\d{1,4})\b/g)) if (r.index > at) { at = r.index; subj = '#' + r[1]; }
      for (const r of pre.matchAll(new RegExp('(?<![\\w\\/.\\-])' + REF + '(?![\\w\\/])', 'g'))) if (r.index > at && isRefNotPath(r[1])) { at = r.index; subj = r[1]; }
      if (!subj || /[0-9a-f]{7,40}/.test(pre.slice(at + subj.length))) continue;                      // another id sits between the ref and this one
      add('head', subj, m[1], li, m.index, m[0].length);
    }
    const prs = [...pl.matchAll(RX.prNum)];
    for (let k = 0; k < prs.length; k++) {
      const m = prs[k], end = m.index + m[0].length;
      let after = pl.slice(end, Math.min(pl.length, end + 45)); const ca = after.search(/[.;]|#\d/); if (ca >= 0) after = after.slice(0, ca);
      let before = pl.slice(Math.max(0, m.index - 35), m.index); const cb = Math.max(before.lastIndexOf('.'), before.lastIndexOf(';'), before.search(/#\d[^#]*$/) >= 0 ? before.search(/#\d[^#]*$/) + 2 : -1); if (cb >= 0) before = before.slice(cb + 1);
      const aw = [...after.matchAll(RX.prState)], bw = [...before.matchAll(RX.prState)];
      const a = aw.length ? aw[0][1].toLowerCase() : null, b = bw.length ? bw[bw.length - 1][1].toLowerCase() : null;
      let st = null, scope = '';
      if (a && b && a !== b) continue;
      if (a) { st = a; scope = before + ' ' + after.slice(0, aw[0].index); } else if (b) { st = b; scope = before.slice(Math.max(0, bw[bw.length - 1].index - 25)) + ' ' + after.slice(0, 12); }
      if (!st || RX.prGuard.test(scope)) continue;
      add('pr', '#' + m[1], st === 'draft' ? 'open' : st, li, m.index, m[0].length, st === 'draft' ? { draft: 1 } : null);
    }
    for (const m of pl.matchAll(RX.t1)) add('test', UNIT[m[2].toLowerCase()], numNorm(m[1]), li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.t2)) add('test', 'passed', numNorm(m[1]), li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.t3)) add('test', 'passed', numNorm(m[1]), li, m.index, m[0].length);
    for (const m of pl.matchAll(RX.cnt)) add('count', m[2].toLowerCase().replace(/s$/, '') + 's', numNorm(m[1]), li, m.index, m[0].length);
  }
  return [...out.values()];
}

// ---------- what a tool result or message shows about heads, pull requests and test runs ----------
function headObsFromOutput(cmd, text, push) {
  const c = cmd || '';
  for (const m of text.matchAll(/^\[([^\s\]]+)(?: \(root-commit\))? ([0-9a-f]{7,40})\]/gm)) if (m[1] !== 'detached') push(m[1], m[2], 'commit');
  for (const m of text.matchAll(/^\s*[ +*!=\-]?\s*([0-9a-f]{7,40})\.\.\.?([0-9a-f]{7,40})\s+(\S+)\s+->\s+(\S+)/gm)) {
    if (/^origin\//.test(m[4])) push(m[4], m[2], 'fetch'); else { push('origin/' + m[4], m[2], 'push'); if (m[3] !== 'HEAD') push(m[3], m[2], 'push'); }
  }
  if (/\bgit\b[^|;&]*\bbranch\b/.test(c)) for (const m of text.matchAll(/^[*+ ]\s+(\S+)\s+([0-9a-f]{7,40})\s/gm)) if (!/^\(/.test(m[1])) push(m[1].replace(/^remotes\//, ''), m[2], 'branch');
  if (/worktree\s+list/.test(c)) for (const m of text.matchAll(/^\S.*?\s+([0-9a-f]{7,40})\s+\[([^\]]+)\]/gm)) push(m[2], m[1], 'worktree');
  for (const m of text.matchAll(/^\*?\s*(?:commit\s+)?([0-9a-f]{7,40})\s+\(([^)\n]+)\)/gm)) for (let r of m[2].split(',')) { r = r.trim().replace(/^HEAD -> /, ''); if (!r || r === 'HEAD' || /^tag:/.test(r)) continue; push(r, m[1], 'log'); }
  for (const m of text.matchAll(/^([0-9a-f]{40})\s+refs\/(heads|remotes)\/(\S+)/gm)) push(m[2] === 'remotes' ? m[3] : (/ls-remote/.test(c) ? 'origin/' + m[3] : m[3]), m[1], 'refs');
  const rp = /\bgit\b(?:\s+-C\s+\S+)?\s+rev-parse\s+(?:--short(?:=\d+)?\s+|--verify\s+)*([A-Za-z0-9._\/\-]+)\s*$/.exec(c.trim());
  if (rp && rp[1] !== 'HEAD' && !/^-/.test(rp[1])) { const t = text.trim(); if (/^[0-9a-f]{7,40}$/.test(t)) push(rp[1], t, 'rev-parse'); }
}
function ghObs(text, pushHead, pushPr) {
  if (!/"(?:headRefOid|state|number)"/.test(text)) return;
  // walk JSON-ish objects loosely: split on object boundaries and read the fields that sit together
  for (const chunk of text.split(/\}\s*,\s*\{|\n\s*\n/)) {
    const num = /"number"\s*:\s*(\d+)/.exec(chunk), st = /"state"\s*:\s*"(OPEN|MERGED|CLOSED)"/.exec(chunk), oid = /"headRefOid"\s*:\s*"([0-9a-f]{40})"/.exec(chunk), ref = /"headRefName"\s*:\s*"([^"]+)"/.exec(chunk), mg = /"merged(?:At)?"\s*:\s*(true|"[0-9T:\-Z.]+")/.exec(chunk);
    if (num && (st || mg)) pushPr(+num[1], st ? st[1].toLowerCase() : 'merged', 'gh-json');
    if (oid && num) pushHead('#' + num[1], oid[1], 'gh-json');
    if (oid && ref) pushHead('origin/' + ref[1], oid[1], 'gh-json');
  }
}
function prObsFromOutput(cmd, text, pushPr) {
  for (const m of text.matchAll(/(?:Merged|Squashed and merged|Rebased and merged) pull request [^\n#]*#(\d+)/g)) pushPr(+m[1], 'merged', 'gh-merge');
  for (const m of text.matchAll(/Pull request [^\n#]*#(\d+)[^\n]*was already merged/g)) pushPr(+m[1], 'merged', 'gh-merge');
  for (const m of text.matchAll(/Closed pull request [^\n#]*#(\d+)/g)) pushPr(+m[1], 'closed', 'gh-close');
  if (/\bgh\b[^|;&]*\bpr\s+create\b/.test(cmd || '')) for (const m of text.matchAll(/github\.com\/[^\s\/]+\/[^\s\/]+\/pull\/(\d+)/g)) pushPr(+m[1], 'open', 'gh-create');
  if (/\bgh\b[^|;&]*\bpr\s+list\b/.test(cmd || '')) for (const m of text.matchAll(/^(\d+)\t[^\n]*\t(OPEN|MERGED|CLOSED|DRAFT)\b/gm)) pushPr(+m[1], m[2] === 'DRAFT' ? 'open' : m[2].toLowerCase(), 'gh-list');
  if (/\bgh\b[^|;&]*\bpr\s+view\b/.test(cmd || '')) { const n = /^number:\t(\d+)/m.exec(text), s = /^state:\t(OPEN|MERGED|CLOSED|DRAFT)/m.exec(text); if (n && s) pushPr(+n[1], s[1] === 'DRAFT' ? 'open' : s[1].toLowerCase(), 'gh-view'); }
}
function testCounts(text) {
  const out = [];
  for (const m of text.matchAll(RX.t1)) out.push(UNIT[m[2].toLowerCase()] + '|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.t2)) out.push('passed|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.t3)) out.push('passed|' + numNorm(m[1]));
  for (const m of text.matchAll(RX.tObsA)) out.push(UNIT[m[1].toLowerCase()] + '|' + m[2]);
  for (const m of text.matchAll(RX.tObsB)) out.push(UNIT[m[1].toLowerCase()] + '|' + m[2]);
  return out;
}

// ---------- stage: index ----------
async function findSummaries(f, sid, cutoff) {
  const comps = []; let ln = 0;
  const rl = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
  for await (const line of rl) {
    ln++;
    if (line.startsWith('{"type":"api-request-blob"') || line.startsWith('{"type":"file-history')) continue;
    const isC = line.includes('"compact_boundary"'), isS = line.includes('"isCompactSummary":true');
    if (!isC && !isS) continue;
    let o; try { o = JSON.parse(line); } catch { continue; }
    if (o.isSidechain) continue;
    if (cutoff && o.timestamp && o.timestamp > cutoff) continue;
    if (o.type === 'system' && o.subtype === 'compact_boundary') {
      const m = o.compactMetadata || {}, ps = m.preservedSegment || {};
      comps.push({ line: ln, ts: o.timestamp, pre: m.preTokens || 0, post: m.postTokens || 0, ms: m.durationMs || 0, trigger: m.trigger || '?', precomputed: !!m.precomputed, headUuid: ps.headUuid || null, tailUuid: ps.tailUuid || null, copied: !!(o.sessionId && o.sessionId !== sid), sumLine: 0, sumTs: null, text: null });
      continue;
    }
    if (o.type === 'user' && o.isCompactSummary && o.message && comps.length && !comps[comps.length - 1].sumLine) {
      const c = comps[comps.length - 1], ct = o.message.content;
      c.sumLine = ln; c.sumTs = o.timestamp; c.text = typeof ct === 'string' ? ct : Array.isArray(ct) ? ct.filter(b => b && b.type === 'text').map(b => b.text || '').join('\n') : '';
    }
  }
  return comps;
}
async function scanTranscript(f, comps, wanted) {
  // one pass over the whole file: what the session saw (tool results, user-role messages, attachments) and what it said
  const occ = new Map();                                             // token key -> { seen: [lines], said: [lines] }
  const hit = (key, cls, ln) => { let o = occ.get(key); if (!o) { o = { seen: [], said: [] }; occ.set(key, o); } const a = o[cls]; if (a[a.length - 1] !== ln) a.push(ln); };
  const headObs = [], prObs = [], runs = [], opened = new Map(), uuidLine = new Map(), tsLines = [], models = [];
  const tools = new Map();                                           // tool_use id -> { name, cmd, fp }
  const hex7 = new Map(); for (const h of wanted.hex) { const k = h.slice(0, 7); (hex7.get(k) || hex7.set(k, []).get(k)).push(h); }
  const scan = (text, cls, ln, opt) => {
    if (!text) return;
    if (hex7.size) for (const m of text.matchAll(/(?<![0-9a-zA-Z])[0-9a-f]{7,40}(?![0-9a-zA-Z])/g)) { const c = hex7.get(m[0].slice(0, 7)); if (c) for (const w of c) if (m[0] === w || m[0].startsWith(w) || w.startsWith(m[0])) hit('h|' + w, cls, ln); }
    if (wanted.base.size) {
      const t2 = text.replace(/\\+/g, '/').replace(/\/{2,}/g, '/').toLowerCase();
      for (const m of t2.matchAll(/[a-z0-9_@+\-][a-z0-9_.@+\-]*\.[a-z0-9]{1,8}(?![a-z0-9_])/g)) {
        const ps = wanted.base.get(m[0]); if (!ps) continue; const end = m.index + m[0].length;
        for (const p of ps) { const st = end - p.length; if (st < 0 || !t2.startsWith(p, st)) continue; const pc = st > 0 ? t2[st - 1] : ''; if (pc && /[a-z0-9_.@+\-]/.test(pc)) continue; hit('p|' + p, cls, ln); }
      }
    }
    const t3 = opt && opt.stripLineNumbers ? text.replace(/^\s*\d+[\t→]/gm, '') : text;
    for (const k of testCounts(t3)) hit('t|' + k, cls, ln);
    for (const m of t3.matchAll(RX.cnt)) hit('c|' + m[2].toLowerCase().replace(/s$/, '') + 's|' + numNorm(m[1]), cls, ln);
    if (wanted.nums.size) for (const m of t3.matchAll(/(?<![\d.,])\d[\d,]*(?![\d]|[.,]\d)/g)) { const n = numNorm(m[0]); if (wanted.nums.has(n)) hit('n|' + n, cls, ln); }
  };
  const proseObs = (text, ln) => {                                   // a message that states a head or a pull request state in the summary's own forms
    for (const c of extractClaims(text, new Set())) { if (c.kind === 'head') headObs.push({ ref: c.subj, sha: c.val, line: ln, form: 'prose' }); else if (c.kind === 'pr') prObs.push({ n: +c.subj.slice(1), state: c.val, line: ln, form: 'prose' }); }
  };
  let ln = 0, lastModel = '', ci = 0;
  const rl = readline.createInterface({ input: fs.createReadStream(f), crlfDelay: Infinity });
  for await (const line of rl) {
    ln++;
    while (ci < comps.length && ln >= comps[ci].line) { comps[ci].model = lastModel; ci++; }
    if (ci >= comps.length && ln > comps[comps.length - 1].line) break;   // nothing after the last boundary is needed
    if (line.startsWith('{"type":"api-request') || line.startsWith('{"type":"file-history')) continue;
    let o; try { o = JSON.parse(line); } catch { continue; }
    if (o.isSidechain) continue;
    if (o.uuid) uuidLine.set(o.uuid, ln);
    if (o.timestamp) tsLines.push([ln, o.timestamp]);
    if (o.type === 'assistant' && o.message) {
      const m = o.message; if (m.model && m.model !== '<synthetic>' && m.usage) lastModel = m.model;
      if (Array.isArray(m.content)) for (const b of m.content) {
        if (b.type === 'text') scan(b.text, 'said', ln); else if (b.type === 'thinking') scan(b.thinking, 'said', ln);
        else if (b.type === 'tool_use') { const inp = b.input || {}; tools.set(b.id, { name: b.name, cmd: typeof inp.command === 'string' ? inp.command : '', fp: inp.file_path || inp.notebook_path || (b.name === 'Read' || b.name === 'Glob' || b.name === 'Grep' ? inp.path : '') || '' }); scan(collectStrings(inp, []).join('\n'), 'said', ln); }
      }
      continue;
    }
    if (o.type === 'user' && o.message) {
      const ct = o.message.content;
      if (o.isCompactSummary) { scan(typeof ct === 'string' ? ct : collectStrings(ct, []).join('\n'), 'said', ln); continue; }   // an earlier summary is the model's own words
      if (typeof ct === 'string') { scan(ct, 'seen', ln); proseObs(ct, ln); continue; }
      if (Array.isArray(ct)) for (const b of ct) {
        if (!b) continue;
        if (b.type === 'text') { scan(b.text, 'seen', ln); proseObs(b.text || '', ln); }
        else if (b.type === 'tool_result') {
          const t = tools.get(b.tool_use_id) || { name: '?', cmd: '', fp: '' }, text = typeof b.content === 'string' ? b.content : collectStrings(b.content, []).join('\n');
          const echo = t.name === 'Write' || t.name === 'Edit' || t.name === 'NotebookEdit' || t.name === 'MultiEdit';   // the result repeats what the model wrote
          if (t.fp && !b.is_error) { const p = normPath(String(t.fp)); if (!opened.has(p)) opened.set(p, ln); }
          scan(text, echo ? 'said' : 'seen', ln, { stripLineNumbers: t.name === 'Read' });
          if (echo) continue;
          if (t.name === 'Bash' || t.name === 'PowerShell') {
            headObsFromOutput(t.cmd, text, (ref, sha, form) => headObs.push({ ref, sha, line: ln, form }));
            prObsFromOutput(t.cmd, text, (n, state, form) => prObs.push({ n, state, line: ln, form }));
            const tc = testCounts(text); if (tc.length) runs.push({ line: ln, cmd: sha1(t.cmd.replace(/\s+/g, ' ').replace(/\s*\|\s*(tail|head)\b.*$/, '').trim().toLowerCase()).slice(0, 12), counts: [...new Set(tc)] });
          } else if (t.name === 'Agent' || t.name === 'Task' || /SendMessage|get_status|list_sessions/.test(t.name)) proseObs(text, ln);
          ghObs(text, (ref, sha, form) => headObs.push({ ref, sha, line: ln, form }), (n, state, form) => prObs.push({ n, state, line: ln, form }));
        }
      }
      continue;
    }
    if (o.type === 'attachment' && o.attachment) { const t = collectStrings(o.attachment, []).join('\n'); scan(t, 'seen', ln); if (o.attachment.type === 'queued_command' || o.attachment.type === 'task_status' || o.attachment.type === 'hook_additional_context') proseObs(t, ln); continue; }
    if (o.type === 'queue-operation' && typeof o.content === 'string') { scan(o.content, 'seen', ln); proseObs(o.content, ln); continue; }
    if (o.type === 'system' && typeof o.content === 'string') { scan(o.content, 'seen', ln); continue; }
  }
  while (ci < comps.length) { comps[ci].model = lastModel; ci++; }
  return { occ, headObs, prObs, runs, opened, uuidLine, tsLines };
}
async function stageIndex(cacheDir) {
  fs.mkdirSync(cacheDir, { recursive: true });
  const startedAt = new Date().toISOString(), t0 = Date.now();
  const stores = fs.readdirSync(ROOT, { withFileTypes: true }).filter(e => e.isDirectory()).map(e => e.name).sort();

  const sid8s = new Set(files.map(x => x.sid.slice(0, 8)));
  const out = { startedAt, replayCut: REPLAY_CUT, foldersInScope: scope.length, foldersLeftOut: stores.length - scope.length, mainTranscripts: files.length, copiedMarkers: 0, markersWithoutSummary: 0, sessions: [] };
  for (const it of files) {

    out.copiedMarkers += all.filter(c => c.copied).length; out.markersWithoutSummary += all.filter(c => !c.copied && !c.sumLine).length;
    const comps = all.filter(c => !c.copied && c.sumLine); if (!comps.length) continue;
    const wanted = { hex: new Set(), base: new Map(), nums: new Set() };
    for (const c of comps) {
      c.claims = extractClaims(c.text, sid8s); c.chars = c.text.length;
      for (const cl of c.claims) {
        if (cl.kind === 'commit') wanted.hex.add(cl.subj); else if (cl.kind === 'head') wanted.hex.add(cl.val);
        else if (cl.kind === 'path') { const b = baseKey(cl.subj); (wanted.base.get(b) || wanted.base.set(b, []).get(b)).push(cl.subj); }
        else if (cl.kind === 'test' || cl.kind === 'count') wanted.nums.add(cl.val);
      }
    }
    for (const [b, ps] of wanted.base) wanted.base.set(b, [...new Set(ps)]);
    const ev = await scanTranscript(it.f, comps, wanted);
    const sess = { g: it.g, sid: it.sid, sid8: it.sid.slice(0, 8), headObs: ev.headObs, prObs: ev.prObs, runs: ev.runs, comps: [] };
    comps.forEach((c, i) => {
      const headLine = c.headUuid && ev.uuidLine.has(c.headUuid) ? ev.uuidLine.get(c.headUuid) : c.line;
      let tsLine = c.line; if (c.precomputed && c.sumTs) { for (const [l, ts] of ev.tsLines) { if (l >= c.line) break; if (ts > c.sumTs) { tsLine = l; break; } } }
      const inputEnd = Math.min(headLine, tsLine);
      const before = (arr, lim) => { let n = 0, first = 0, last = 0; for (const l of arr) { if (l >= lim) break; n++; if (!first) first = l; last = l; } return [n, first, last]; };
      const stat = (key) => { const o = ev.occ.get(key); if (!o) return { seen: 0, said: 0 }; const [ns, fs1, ls] = before(o.seen, c.line), [nd, fd] = before(o.said, c.line); return { seen: ns, firstSeen: fs1, lastSeen: ls, said: nd, firstSaid: fd }; };
      for (const cl of c.claims) {
        if (cl.kind === 'commit') cl.ev = stat('h|' + cl.subj); else if (cl.kind === 'head') cl.ev = stat('h|' + cl.val);
        else if (cl.kind === 'path') { cl.ev = stat('p|' + cl.subj); let op = 0; for (const [p, l] of ev.opened) { if (l >= c.line) continue; if (p === cl.subj || p.endsWith('/' + cl.subj) || cl.subj.endsWith('/' + p)) { op = op ? Math.min(op, l) : l; } } cl.ev.opened = op; }
        else if (cl.kind === 'test') { cl.ev = stat('t|' + cl.subj + '|' + cl.val); cl.ev.num = stat('n|' + cl.val); }
        else if (cl.kind === 'count') { cl.ev = stat('c|' + cl.subj + '|' + cl.val); cl.ev.num = stat('n|' + cl.val); }
      }
      sess.comps.push({ n: i + 1, line: c.line, sumLine: c.sumLine, ts: c.ts, sumTs: c.sumTs, pre: c.pre, post: c.post, ms: c.ms, precomputed: c.precomputed, model: c.model, chars: c.chars, headLine, inputEnd, afterReplayCut: c.ts > REPLAY_CUT, claims: c.claims });
    });
    out.sessions.push(sess);
    process.stderr.write(`index ${it.g} ${sess.sid8} summaries=${comps.length} claims=${comps.reduce((a, c) => a + c.claims.length, 0)} (${((Date.now() - t0) / 1000).toFixed(0)}s)\n`);
  }
  fs.writeFileSync(path.join(cacheDir, 'index.json'), JSON.stringify(out));
  const ns = out.sessions.reduce((a, s) => a + s.comps.length, 0), nc = out.sessions.reduce((a, s) => a + s.comps.reduce((x, c) => x + c.claims.length, 0), 0);
  console.log(`INDEX_DONE transcripts=${files.length} sessions-with-summaries=${out.sessions.length} summaries=${ns} claims=${nc} copied-markers=${out.copiedMarkers} seconds=${((Date.now() - t0) / 1000).toFixed(0)} cut=${startedAt}`);
}

function git(repo, args, input) { const r = spawnSync('git', ['-C', repo, ...args], { input, encoding: 'utf8', maxBuffer: 1 << 29 }); return { status: r.status, out: r.stdout || '' }; }
const sameSha = (a, b) => !!a && !!b && (a.startsWith(b) || b.startsWith(a));
function loadRepo(g, tokens) {
  const repo = REPOS[g]; const R = { g, repo, ok: fs.existsSync(repo), full: new Map(), ct: new Map(), pathCt: new Map(), baseIdx: new Map(), refLog: new Map(), anc: new Map(), worktrees: [] };
  if (!R.ok) return R;
  const list = [...tokens];
  const bc = git(repo, ['cat-file', '--batch-check'], list.join('\n') + '\n').out.split('\n');
  list.forEach((t, i) => { const m = /^([0-9a-f]{40}) (\w+) /.exec(bc[i] || ''); if (m) R.full.set(t, { sha: m[1], type: m[2] }); });
  for (const l of git(repo, ['log', '--all', '--format=%H %ct']).out.split('\n')) { const [h, c] = l.split(' '); if (h) R.ct.set(h, +c); }
  let ct = 0;
  for (const l of git(repo, ['log', '--all', '--name-only', '--format=@%ct']).out.split('\n')) {
    if (!l) continue; if (l[0] === '@') { ct = +l.slice(1); continue; }
    const p = normPath(l), prev = R.pathCt.get(p); if (prev === undefined || ct < prev) R.pathCt.set(p, ct);
  }
  for (const p of R.pathCt.keys()) { const b = baseKey(p); (R.baseIdx.get(b) || R.baseIdx.set(b, []).get(b)).push(p); }
  R.worktrees = git(repo, ['worktree', 'list', '--porcelain']).out.split('\n').filter(l => l.startsWith('worktree ')).map(l => l.slice(9));
  R.refs = new Set(git(repo, ['for-each-ref', '--format=%(refname:short)']).out.split('\n').filter(Boolean));
  R.isAnc = (x, y) => { const k = x + ' ' + y; if (!R.anc.has(k)) R.anc.set(k, git(repo, ['merge-base', '--is-ancestor', x, y]).status === 0); return R.anc.get(k); };
  R.headAt = (ref, T) => { if (!R.refs.has(ref)) return null; if (!R.refLog.has(ref)) R.refLog.set(ref, git(repo, ['log', '--first-parent', '--format=%H %ct', ref]).out.split('\n').filter(Boolean).map(l => { const [h, c] = l.split(' '); return [h, +c]; })); for (const [h, c] of R.refLog.get(ref)) if (c <= T) return h; return null; };
  R.pathTime = (p) => { const c = R.baseIdx.get(baseKey(p)); if (!c) return undefined; let best; for (const r of c) if (r === p || r.endsWith('/' + p) || p.endsWith('/' + r)) { const t = R.pathCt.get(r); if (best === undefined || t < best) best = t; } return best; };
  R.onDisk = (p) => { try { if (/^[a-z]:\//.test(p)) return fs.existsSync(p); for (const root of [repo, ...R.worktrees]) if (fs.existsSync(path.join(root, p))) return true; } catch {} return false; };
  return R;
}
function loadGitHub(cacheDir) {
  const f = path.join(cacheDir, 'github.json');
  if (fs.existsSync(f)) return JSON.parse(fs.readFileSync(f, 'utf8'));
  const r = spawnSync('gh', ['pr', 'list', '--state', 'all', '--limit', '1000', '--json', 'number,state,createdAt,mergedAt,closedAt,isDraft,headRefName,headRefOid'], { cwd: REPOS.fixflow, encoding: 'utf8', maxBuffer: 1 << 27 });
  const G = { ok: r.status === 0, at: new Date().toISOString(), prs: {}, commits: {} };
  if (G.ok) for (const p of JSON.parse(r.stdout)) G.prs[p.number] = p;
  fs.writeFileSync(f, JSON.stringify(G)); return G;
}
function prCommits(G, cacheDir, n) {

  const r = spawnSync('gh', ['pr', 'view', String(n), '--json', 'commits'], { cwd: REPOS.fixflow, encoding: 'utf8', maxBuffer: 1 << 26 });

}

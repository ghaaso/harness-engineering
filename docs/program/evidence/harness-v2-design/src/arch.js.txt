  function libFactor(p, K) { return p.lib ? 1 - p.e * p.hitMax * K / (K + p.khalf) + p.look : 1; }

  /* steady state: what each limit allows in slices per hour, and the knee */
  function capacity(p) {
    var f = libFactor(p, p.S * p.perSlice / 2);
    var cycle = p.r * (p.k * (p.tw + p.tc) + p.tv) + p.tm + p.mc * p.tc;
    var perSliceBurn = p.r * (p.k * p.tw + p.tv) * p.mult * f;
    var lim = [{ key: 'check', v: p.slots * 60 / (p.tc * (p.r * p.k + p.mc)) }];
    if (p.shared) lim.push({ key: 'root', v: 60 / (p.r * p.tv + p.tm + p.mc * p.tc) });
    else { lim.push({ key: 'review', v: 60 / (p.r * p.tv) }); lim.push({ key: 'merge', v: 60 / (p.tm + p.mc * p.tc) }); }
    lim.push({ key: 'tokens', v: (p.budget / p.win) / perSliceBurn * 60 });
    var low = lim.slice().sort(function (a, b) { return a.v - b.v; })[0], perLane = 60 / cycle;
    return { lim: lim, cap: low.v, which: low.key, perLane: perLane, knee: low.v / perLane, planned: Math.max(1, low.key === "tokens" ? Math.ceil(low.v / perLane) : Math.round(p.margin * low.v / perLane)), at: function (n) { return Math.min(n * perLane, low.v); } };
  }

  /* stock-and-flow run, one-minute steps */
  function simulate(p, H) {
    var Nrun = p.planner ? Math.min(p.N, capacity(p).planned) : p.N;
    var s = { B: p.S, W: 0, Q: 0, C: 0, V: 0, G: 0, Mg: 0, M: 0, T: 0, K: 0, stopped: false };
    var out = { t: [], B: [], W: [], Q: [], C: [], V: [], G: [], M: [], T: [], K: [], stop: [] };
    var acc = { work: 0, check: 0, gate: 0, held: 0, stop: 0 }, done = null, stops = 0, burnt = 0;
    for (var t = 0; t <= H; t++) {
      if (t % p.win === 0) { s.T = 0; s.stopped = false; }
      var f = libFactor(p, s.K), active = s.W + s.Q + s.C + s.V + s.G + s.Mg;
      if (t % 5 === 0) { out.t.push(t / 60); ['B', 'W', 'Q', 'C', 'V', 'M', 'K'].forEach(function (k) { out[k].push(s[k]); }); out.G.push(s.G + s.Mg); out.T.push(Math.min(100, 100 * s.T / p.budget)); out.stop.push(s.stopped ? 1 : 0); }
      if (s.M >= p.S - 0.5 && done === null) done = t;
      if (s.stopped) { acc.stop += active; continue; }
      var rvCap = p.shared ? Math.max(0, 1 - s.Mg) : 1, reviewing = Math.min(s.V, rvCap), waitRev = s.V - reviewing;
      /* the governor: slow the lanes so that the window lasts */
      var left = p.win - (t % p.win), would = ((s.W + reviewing) + (s.Q + s.G + waitRev) * p.wq) * p.mult * f * left, room = 0.98 * p.budget - s.T;
      var g = p.governor && would > room ? Math.max(0, room / would) : 1;
      var st = Math.max(0, Math.min(s.B, Math.min(Nrun, p.width) - active)); if (p.stagger) st = Math.min(st, 0.2); if (g < 1) st = 0;
      s.B -= st; s.W += st;
      var wOut = g * s.W / p.tw; s.W -= wOut; s.Q += wOut;
      var mergeShare = p.mc * p.tc / (p.tm + p.mc * p.tc), slotsEff = Math.max(0, p.slots - Math.min(1, s.Mg) * mergeShare);
      var cIn = Math.max(0, Math.min(s.Q, slotsEff - s.C)); s.Q -= cIn; s.C += cIn;
      var cOut = s.C / p.tc, more = cOut * (1 - 1 / p.k); s.C -= cOut; s.W += more; s.V += cOut - more;
      var vOut = g * reviewing / p.tv, back = vOut * (1 - 1 / p.r); s.V -= vOut; s.W += back; s.G += vOut - back;
      var gIn = Math.max(0, Math.min(s.G, 1 - s.Mg)); s.G -= gIn; s.Mg += gIn;
      var mOut = s.Mg / (p.tm + p.mc * p.tc); s.Mg -= mOut; s.M += mOut;
      if (p.lib) { s.K += mOut * p.perSlice; s.K -= s.K * p.staleShare * mOut; }
      var bn = (g * (s.W + reviewing) + (s.Q + s.G + waitRev) * p.wq) * p.mult * f; s.T += bn; burnt += bn;
      acc.work += g * (s.W + reviewing); acc.held += (1 - g) * (s.W + reviewing); acc.check += s.Q + s.C; acc.gate += waitRev + s.G + s.Mg;
      if (s.T >= p.budget) { s.stopped = true; stops++; }
    }
    out.done = done; out.acc = acc; out.Nrun = Nrun; out.stops = stops; out.burnt = burnt / 60; return out;
  }

  function at(run, key) { var i = Math.min(run.t.length - 1, Math.round(ST.t * 12)); return run[key][i]; }
  function sdDiagram(run) {

  function wavesOf(slices, check) {
    var left = slices.slice(), done = {}, waves = [], guard = 0;
    var clash = function (a, b) { return a.writes.some(function (f) { return b.writes.indexOf(f) >= 0; }); };
    while (left.length && guard++ < 20) {
      var ready = left.filter(function (s) { return s.needs.every(function (d) { return done[d]; }); }), wave = [];
      ready.forEach(function (s) { if (!check || wave.every(function (w) { return !clash(w, s); })) wave.push(s); });
      if (!wave.length) break;
      waves.push(wave); wave.forEach(function (s) { done[s.id] = 1; left.splice(left.indexOf(s), 1); });
    }
    var hidden = []; waves.forEach(function (w) { for (var i = 0; i < w.length; i++) for (var j = i + 1; j < w.length; j++) if (clash(w[i], w[j])) hidden.push([w[i], w[j]]); });
    return { waves: waves, hidden: hidden };
  }

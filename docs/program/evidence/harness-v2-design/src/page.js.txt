    ['parked', /\bparked starts?\b/i], ['refuter', /\brefuters?\b/], ['verdict', /\btyped verdict\b/i], ['highrisk', /\bhigh-risk\b/i], ['head', /\bfinal head\b/i], ['compile', /\bthe compile\b/i], ['heartbeat', /\bheartbeats?\b/i]];
  function autolink(t) {
    var parts = String(t).split(/(\{\{[^}]+\}\})/), used = {};
    parts.forEach(function (p) { var m = /^\{\{t:([a-z0-9_]+)\|/.exec(p); if (m) used[m[1]] = 1; });
    AUTO.forEach(function (a) {
      if (used[a[0]]) return;
      for (var i = 0; i < parts.length; i++) {
        if (parts[i].indexOf('{{') === 0) continue;
        var m = a[1].exec(parts[i]); if (!m) continue;
        used[a[0]] = 1;
        parts.splice(i, 1, parts[i].slice(0, m.index), '{{t:' + a[0] + '|' + m[0] + '}}', parts[i].slice(m.index + m[0].length));
        return;
      }
    });
    return parts.join('');
  }

  }

  }

  }

  function simulateQueue(st) {
    var du = D.sims.queue.durations, alone = st.gate === 'win' ? du.win1 : du.con1, pair = st.gate === 'win' ? du.win2 : du.con2;
    var S = [], i; for (i = 0; i < st.n; i++) S.push({ id: i + 1, rel: null, start: null, end: null });
    var t = 0, running = [], waiting = [], next = 0, guard = 0;
    function release(k) { while (k-- > 0 && next < st.n) { S[next].rel = t; waiting.push(S[next]); next++; } }
    if (st.release === 'all') release(st.n); else release(2);
    while (guard++ < 200) {
      var free = st.slots - running.length, k = Math.min(free, waiting.length);
      if (k > 0) { var conc = running.length + k; for (i = 0; i < k; i++) { var s = waiting.shift(); s.start = t; s.end = t + (conc >= 2 ? pair : alone); running.push(s); } }
      if (!running.length) break;
      var tn = Math.min.apply(null, running.map(function (r) { return r.end; }));
      t = tn; running = running.filter(function (r) { return r.end > t; });
      if (st.release === 'two' && running.length + waiting.length === 0) release(2);
      if (!running.length && !waiting.length && next >= st.n) break;
    }
    return S;
  }

  function mergeOutcome(st) {
    var named = st.reader !== 'nobody', indep = st.reader === 'reviewer';

    var why = [];
    if (st.reader === 'reviewer') why.push('An independent reviewer read the final commit. The verdict names the commit that gets merged.');

    else why.push('Only the branch’s own session reviewed the final commit. ' + (st.risk ? 'On a high-risk branch that does not count as a read.' : 'On a low-risk branch that is allowed, and recorded.'));
    if (st.mode === 'today') return { cls: indep ? 'ok' : 'bad', head: indep ? 'Merged' : 'Merged, unread', why: why.concat(indep ? [] : ['Nothing checks who read the final commit, and nothing is written down.']), merged: true, v2: false };
    if (!problem) return { cls: 'ok', head: 'Merged', why: why, merged: true, v2: true };
    if (st.mode === 'record') return { cls: 'flag', head: 'Merged, and flagged as unread', why: why.concat(['Record-only: the queue blocks nothing. It writes the flag into the merge record.']), merged: true, v2: true };

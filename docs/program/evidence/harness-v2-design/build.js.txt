/* ---------- 3. number lint: digits that are not inside a tagged number ---------- */
const ALLOW = [/\bB\d\d\b/g, /\bD\d\d\b/g, /\bG\d-F\d+\b/g, /\b[A-Z]{1,2}-\d+\b/g, /PR #\d+/g, /\b(phase|Phase|mechanism|Mechanism|mechanisms|Mechanisms|section|sections|step|Step)s? \d+(, \d+)*( (and|to) \d+)?/g,
  /\b\d{1,2} (January|February|March|April|May|June|July|August|September|October|November|December|Oct|Nov)\b/g, /\b(October|November) \d{1,2}\b/g, /\b\d\d:\d\d(:\d\d)?\b/g, /\b20\d\d\b/g, /§\d[\d.]*/g,
  /\bthe 21\b/g, /\b21 (problems|unsolved|problem)\b/g, /\bof the 21\b/g, /\bP[0-3]s?\b/g, /\bP0a\b/g, /\bLever \d\b/g, /the 10 and 11 November/g, /of 10 and 11 November/g, /\bv2\b/g, /\bWSL2\b/g, /\bWindows 11\b/g, /\b24 (event|hours|quiet hours)\b/g, /\b2\d cards\b/g, /\b400k\b/g, /\b15-minute\b/g, /'20 PASS' stood against 19 lines/g, /nine checks while its record said 55/g];
const lint = [];
(function walk(v, where) {
  if (typeof v === 'string') {
    let s = v.replace(/\{\{[^}]+\}\}/g, ' ');
    ALLOW.forEach(re => { s = s.replace(re, ' '); });
    const hits = s.match(/\d[\d,.]*\d|\d/g);
    if (hits) lint.push(where + ': ' + hits.join(' | ') + '   <<  ' + v.slice(0, 110));
  } else if (Array.isArray(v)) v.forEach((x, i) => walk(x, where + '[' + i + ']'));
  else if (v && typeof v === 'object') Object.keys(v).forEach(k => {
    if (where === '' && (k === 'numbers' || k === 'shutdowns' || k === 'glossary')) return;
    if (where === '.meta' && (k === 'sources' || k === 'limits' || k === 'date')) return;
    if (/^(id|key|d|start|end|buildFrom|ideas|mechs|durations|measuredWaits|rows|n|phase|from|to|share|bands|gates|perGate|limit|after|sessionsDefault|sessionsMin|sessionsMax|made|ref)$/.test(k) && typeof v[k] !== 'string') return;
    if (k === 'ref' || k === 'id' || k === 'key' || k === 'd' || k === 'said' || k === 'round' || k === 'chips') return;
    walk(v[k], where + '.' + k);
  });
})(D, '');

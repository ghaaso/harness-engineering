// Refuter's checks on the library pathway map. Run from this folder:
//   node - < refute-map.js.txt
// Uses process.cwd(), never __dirname. Reads the map files only; writes
// refute-map-console.txt beside itself (and prints the same lines).
// Part A re-runs the engine and compares with the record. Part B decomposes the
// best-value result. Part C re-runs the model with questionable pieces changed
// (tilt tests). Variants change a deep copy of map.json in memory, never the file.

// ------------------------------------------------------------------ B. decomposition
say('');
say('== B. Where the library\'s extra cost over the ladder comes from (best values, last period)');
function flowCost(policy, stocks) {
  const ev = E.evaluatePolicy(MAP, BEST, policy, stocks);
  const cm = E.compile(MAP);
  // per-branch expected cost = sum over classes of class chance * branch flow * branch cost
  const base = {};
  const pol = cm.policies[cm.policyIndex[policy]];
  const groups = {};
  pol.combos.forEach(combo => {
    const cls = E.evaluateClass(MAP, BEST, policy, combo.values, stocks);
    if (!cls.chance) return;
    // recompute branch costs in this class scope
    const scope = Object.assign({}, E.prepareParams(MAP, BEST).values);
    cm.stockIds.forEach(id => { scope[id] = stocks[id] || 0; });
    cm.stockValues.forEach(v => { scope[v.id] = v.fn(scope); });
    Object.assign(scope, pol.flags, combo.flags);
    cm.derived.forEach(d => { scope[d.id] = d.fn(scope); });
    cm.branches.forEach(b => {
      const fl = cls.branchFlow[b.id];
      if (!fl || b.tier < 0) return;
      const k = b.cost.fn(scope);
      const g = b.id.replace(/^(l\d+)_.*$/, '$1');
      const key = g === 'l5' ? (b.id.indexOf('none') > 0 || b.id.indexOf('wrong_') > 0 ? 'l5 cheap attempt (no or wrong entry)' : 'l5 cheap attempt (good entry, shortened)') : g;
      groups[key] = (groups[key] || 0) + cls.chance * fl * k;
    });
  });
  return { total: ev.overall.cost, groups };
}
const decL = flowCost('library', last.stocks), decC = flowCost('cheap_ladder', {});
const keys = Array.from(new Set(Object.keys(decL.groups).concat(Object.keys(decC.groups)))).sort();
keys.forEach(k => say('  ', k.padEnd(44), 'library', f2(decL.groups[k] || 0), 'ladder', f2(decC.groups[k] || 0), 'difference', f2((decL.groups[k] || 0) - (decC.groups[k] || 0))));
say('   totals: library', f2(decL.total), 'ladder', f2(decC.total));

});

say('== C. Tilt tests. Each row: best values (last period) and a 1000-run ensemble, seed 20261112.');
function runVariant(name, map, table) {
  const best = E.paramsFromTable(table, 'best');
  const t = E.timePath(map, best);
  const L = t.periods[t.periods.length - 1];
  const en = E.ensemble(map, table, { n: 1000, seed: 20261112 });
  const s = en.spread.last;
  say(name.padEnd(46),
    '| best: cL', f2(L.ratios.cost_vs_ladder), 'wL', f2(L.ratios.wrong_vs_ladder), 'cT', f2(L.ratios.cost_vs_top), 'wT', f2(L.ratios.wrong_vs_top),
    'ws', f2(L.stockValues.wrong_share),
    '| ens p50: cL', f2(s.cost_vs_ladder.p50), 'wL', f2(s.wrong_vs_ladder.p50), 'cT', f2(s.cost_vs_top.p50), 'wT', f2(s.wrong_vs_top.p50),
    '| p5 cL', f2(s.cost_vs_ladder.p5), 'wL', f2(s.wrong_vs_ladder.p5),
    '| better vs ladder', pc(en.shareBetterVs.cheap_ladder.last), 'vs top', pc(en.shareBetterVs.top_first.last));
  return { t, en };
}
const setRow = (table, id, lo, best, hi) => { const r = table.parameters.find(x => x.id === id); r.low = lo; r.best = best; r.high = hi; };

// C2: no "stealing". Coverage counts every entry; a held problem's entry is bad
// in proportion to the bad entries that cover problems the good ones do not.
function noSteal(map) {
  const m = clone(map);
  m.stocks.values.find(v => v.id === 'wrong_share').expr = '1 - safediv(1 - exp(-E_good / k_fill), 1 - exp(-(E_good + E_bad) / k_fill))';
  m.stocks.values.find(v => v.id === 'fill').expr = '1 - exp(-(E_good + E_bad) / k_fill)';
  return m;
}
runVariant('C2 bad entries cover, not steal', noSteal(MAP), TABLE);

// C3: bad entries fade from retrieval at the stale rate (a stale entry about
// code that is gone stops matching). Tests the absorbing bad stock.
function retire(map) {
  const m = clone(map);
  m.stocks.update.find(u => u.id === 'E_bad').expr = 'E_bad - delta_stale * E_bad + delta_stale * E_good + episodes_per_period * (W_bad_right + W_bad_escape - W_repair)';
  return m;
}
runVariant('C3 bad entries fade at the stale rate', retire(MAP), TABLE);

// C4: a good entry in hand also goes up the ladder with the work.
function upLadder(map, rightExpr, wrongExpr, costExpr, callsExpr) {
  const m = clone(map);
  const outcome = new Set(m.states.filter(s => s.outcome).map(s => s.id));
  const set = new Set(); const stack = ['L6_wrong'];
  while (stack.length) { const s = stack.pop(); if (set.has(s) || outcome.has(s)) continue; set.add(s); m.branches.filter(b => b.from === s).forEach(b => stack.push(b.to)); }
  const newStates = m.states.filter(s => set.has(s.id)).map(s => Object.assign({}, s, { id: s.id + '_E' }));
  const newBranches = m.branches.filter(b => set.has(b.from)).map(b => Object.assign({}, b, { id: b.id + '_E', from: b.from + '_E', to: set.has(b.to) ? b.to + '_E' : b.to }));
  newBranches.forEach(b => {
    if (b.id === 'l8_right_E') { b.chance = rightExpr; b.cost = costExpr; b.calls = callsExpr; }
    if (b.id === 'l8_wrong_E') { b.chance = wrongExpr; b.cost = costExpr; b.calls = callsExpr; }
  });
  m.states = m.states.concat(newStates);
  m.branches = m.branches.concat(newBranches);
  ['l5_exact_wrong', 'l5_similar_wrong', 'l5_full_wrong', 'l5_part_wrong'].forEach(id => { m.branches.find(b => b.id === id).to = 'L6_wrong_E'; });
  m.measures.forEach(me => {
    if (me.flow_into) me.flow_into = me.flow_into.concat(me.flow_into.filter(s => set.has(s)).map(s => s + '_E'));
    if (me.flow_on) me.flow_on = me.flow_on.concat(me.flow_on.filter(b => newBranches.some(nb => nb.id === b + '_E')).map(b => b + '_E'));
  });
  return m;
}

// ------------------------------------------------------------------ D. corner search
say('');
say('== D. Can any setting inside the ranges make the library cheaper than the ladder? (coordinate search)');
function search(map, table, objective, label) {
  const rows = table.parameters.filter(r => !r.run_setting && r.low !== r.high);
  let p = E.paramsFromTable(table, 'best');
  const score = q => { const t = E.timePath(map, q, { lean: true }); return objective(t); };
  let cur = score(p);
  for (let sweep = 0; sweep < 4; sweep++) {
    rows.forEach(r => {
      for (let i = 0; i <= 8; i++) {
        const v = r.low + (r.high - r.low) * i / 8;
        const q = Object.assign({}, p, { [r.id]: v });
        let s; try { s = score(q); } catch (e) { continue; }
        if (s < cur - 1e-12) { cur = s; p = q; }
      }
    });
  }
  say('  ' + label + ': minimum found', f2(cur));
  const moved = table.parameters.filter(r => Math.abs(p[r.id] - r.best) > 1e-9).map(r => r.id + '=' + (+p[r.id].toFixed(3)));
  say('    settings moved from best:', moved.join(', '));
  return p;
}

//   node - < patch-map-rev4.js.txt
// Uses process.cwd(), never __dirname. Safe to run more than once: a rerun makes no edit.
// Edits the text in place, block by block, so the file keeps its hand layout.
// What it does:
//   R18   declares the four groups of library-side unknowns (lib_groups) with plain labels.
//   R18a  replaces the three reference settings with middle, the eight corners, all_good and
//         worst_corner (plus all_good_costly, which the page brief asks for). "unfavourable" is retired.
//   R18b  rewrites the tie note: four shared levels per run, one per group.
//   R18c  declares the square.
//   R19   rewrites the weight block: two weights, alone and last, as shares of the full fall.
//   R20   declares the shared groups gate and ladder, with directions, and the shared table.
//   R21   adds p_promote, k_promo_exact and k_promo_similar (drawn from an added stream, so every
//         revision-3 draw stays the same), the derived values promo and g_lib, the gate branches on
//         g_lib, the promoted-check link with the loops it closes, and its measures.
// It changes no state, no other branch, no stock, no attribute and no policy. It checks that.

const NL = String.fromCharCode(10);
function J(v) {
  if (Array.isArray(v)) return '[' + v.map(J).join(', ') + ']';
  if (v && typeof v === 'object') { const k = Object.keys(v); return k.length ? '{ ' + k.map(x => JSON.stringify(x) + ': ' + J(v[x])).join(', ') + ' }' : '{}'; }
  return JSON.stringify(v);
}
let edits = 0;

// ---------------------------------------------------------------- block helpers
function lines() { return raw.split(NL); }
function blockAt(key) {
  const L = lines();
  const start = L.findIndex(l => l.startsWith('  "' + key + '":'));
  if (start < 0) return null;
  let end = start + 1;
  while (end < L.length && !L[end].startsWith('  "') && L[end] !== '}') end++;
  let last = end - 1;
  while (last > start && L[last].trim() === '') last--;
  return { start, last };
}
function setBlock(key, text) {
  const b = blockAt(key);
  if (!b) throw new Error('No block ' + key);
  const L = lines();
  const now = L.slice(b.start, b.last + 1).join(NL);
  const trail = now.endsWith(',') ? ',' : '';
  const want = text + trail;
  if (now === want) return;
  raw = L.slice(0, b.start).concat(want.split(NL), L.slice(b.last + 1)).join(NL);
  edits++;
}
function insertAfter(afterKey, key, text, gap) {
  if (blockAt(key)) { setBlock(key, text); return; }
  const b = blockAt(afterKey);
  if (!b) throw new Error('No block ' + afterKey);
  const L = lines();
  const lastLine = L[b.last].endsWith(',') ? L[b.last] : L[b.last] + ',';
  const add = (gap ? [''] : []).concat((text + ',').split(NL));
  raw = L.slice(0, b.last).concat([lastLine], add, L.slice(b.last + 1)).join(NL);
  edits++;
}
function swap(a, b) { if (raw.includes(b)) return; if (!raw.includes(a)) throw new Error('Text not found: ' + a.slice(0, 90)); raw = raw.replace(a, b); edits++; }
function arrayBlock(key, items) { return '  "' + key + '": [' + NL + items.map(x => '    ' + J(x)).join(',' + NL) + NL + '  ]'; }
function lineKey(key, value) { return '  "' + key + '": ' + J(value); }

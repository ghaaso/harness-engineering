// drawn as the share each step keeps. Run from this folder: node - < patch-page-ux1.js.txt   (a rerun makes 0 edits)
'use strict';
const fs = require('fs');
const path = require('path');
const SRC = path.join(process.cwd(), 'page-src');
let edits = 0;
function patch(file, pairs) {
  const p = path.join(SRC, file);
  let s = fs.readFileSync(p, 'utf8');
  pairs.forEach(([from, to, name]) => {
    if (s.includes(to)) return;                      // already applied
    if (!s.includes(from)) throw new Error(file + ': anchor not found for "' + name + '"');
    s = s.replace(from, () => to); edits++;
  });
  fs.writeFileSync(p, s);
}

// text.json: wording and the map's row height (room for the count in each box)
(function () {
  const p = path.join(SRC, 'text.json');
  const raw = fs.readFileSync(p, 'utf8');
  const j = JSON.parse(raw);
  const set = (o, k, v) => { if (o[k] !== v) { o[k] = v; edits++; } };
  set(j.sections.map, 'lede', 'The map starts with all work. Pick a way of working, or one kind of work. Tap a route in the tree to light it up, then step through it.');
  set(j.sections.map, 'graph_note', 'Boxes are states and lines are branches. The bold number in a box counts the pieces of work, of a hundred, that pass through it. A thicker line carries more work. Faded boxes and dotted lines carry none under this choice.');
  set(j.sections.map, 'scroll_note', 'On a narrow screen the map scrolls sideways.');
  set(j.sections.funnel, 'lede', 'Start with a hundred pieces of work the cheap model would get wrong alone. The bold number is how many are left after each step. The height of a bar is the share that step keeps.');
  set(j.sections.funnel, 'marks', 'Faint mark: the share the step keeps at the all-good setting.');
  set(j.layout, 'row_h', 66);
  set(j.layout, 'node_h', 56);
  const out = JSON.stringify(j, null, raw.includes('\n  "') ? 2 : 1) + (raw.endsWith('\n') ? '\n' : '');
  if (out !== raw) fs.writeFileSync(p, out);
})();

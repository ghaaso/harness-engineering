  // ============================================================ mod: map (section 1)
  // A short version of the hundred pieces above the detailed graph, three ways of working in aligned rows,
  // plus counts on the graph's lines, coloured endings and larger text. Redeclares the section-1 functions.
  var map_ORDER = ['top_first', 'cheap_ladder', 'library'];
  var map_EV = {};      // per way of working: { sf, bf, outs } at the current settings and scope
  var map_GOOD = ['L5_exact', 'L5_similar', 'L5_full', 'L5_part'];

  function geom(mini) {
    return mini ? { cw: LY.col_w, rh: 34, x0: LY.x0, y0: 96, nw: LY.node_w - 20, nh: 20, head: [16, 30, 44], lh: 13 }
      : { cw: 108, rh: 86, x0: 50, y0: 172, nw: 86, nh: 76, head: [20, 36, 52], lh: 16 };
  }

  function map_evalAll(stocks) {
    var all = ST.scope === 'all';
    map_ORDER.forEach(function (p) {
      var st = p === 'library' ? stocks : {};
      if (all) { var ev = E.evaluatePolicy(MAP, cur, p, st); map_EV[p] = { sf: ev.stateFlow, bf: ev.branchFlow, outs: ev.overall.outcomes, wrong: ev.overall.wrong }; }
      else { var ec = E.evaluateClass(MAP, cur, p, classFor(p), st); map_EV[p] = { sf: ec.stateFlow, bf: ec.branchFlow, outs: ec.outcomes, wrong: ec.wrong }; }
    });
  }
  function map_sum(o, ids) { return ids.reduce(function (a, k) { return a + (o[k] || 0); }, 0); }

  // The moments of the short version. Each returns its parts: [share of a hundred, words, tone].
  var map_MOMENTS = [
    { q: 'Is the library looked up?', parts: function (p, e) {
      if (p !== 'library') return [[1, 'no library here', 'none']];
      var l = e.sf.L2_lookup || 0; return [[l, 'looked up', 'mid'], [Math.max(0, 1 - l), 'not looked up', 'light']];
    } },
    { q: 'What does the cheap model get?', parts: function (p, e) {
      return [[map_sum(e.sf, map_GOOD), 'a good entry', 'good'], [e.sf.L5_wrong || 0, 'a wrong entry', 'bad'], [e.sf.L5_none || 0, 'no entry', 'light']];
    } },
    { q: 'Is the cheap first try right?', parts: function (p, e) {
      return [[e.sf.L6_right || 0, 'right', 'good'], [e.sf.L6_wrong || 0, 'wrong', 'bad']];
    } },
    { q: 'Does the gate catch a wrong first try?', parts: function (p, e) {
      return [[e.bf.l6_caught || 0, 'caught, retried', 'good'], [e.bf.l6_escaped || 0, 'slips past', 'bad']];
    } },
    { q: 'Is the cheap retry right?', parts: function (p, e) {
      return [[e.sf.L7_right || 0, 'right', 'good'], [e.sf.L7_wrong || 0, 'wrong', 'bad']];
    } },
    { q: 'Is Opus right?', parts: function (p, e) {
      return [[e.sf.L8_right || 0, 'right', 'good'], [e.sf.L8_wrong || 0, 'wrong', 'bad']];
    } },
    { q: 'Is Fable right?', parts: function (p, e) {
      return [[e.sf.L9_right || 0, 'right', 'good'], [e.sf.L9_wrong || 0, 'wrong', 'bad']];
    } },
    { q: 'How does it end?', parts: function (p, e) {
      return [[e.outs.EQUAL || 0, 'accepted and right', 'good'], [e.outs.ESCAPED_FOUND || 0, 'wrong, found later', 'bad'], [e.outs.ESCAPED_SILENT || 0, 'wrong, never found', 'bad'], [e.outs.UNRESOLVED || 0, 'caught, never solved', 'bad']];
    } }
  ];
  function map_c(v) { return fmt(100 * v, 'count'); }

  function map_shortNow() {
    var p = ST.policy, e = map_EV[p], kind = ST.scope === 'all' ? '' : ' of this kind';
    var b = function (v) { return '<b>' + esc(map_c(v)) + '</b>'; };
    var out = ['Of 100 pieces' + kind + ' under ' + esc(PNAME[p].toLowerCase()) + ':'];
    if (p === 'library') out.push(b(e.sf.L2_lookup || 0) + ' are looked up, and ' + b(map_sum(e.sf, map_GOOD)) + ' reach the cheap model with a good entry.');
    var cheap = map_sum(e.sf, ['L6_right', 'L6_wrong']);
    if (cheap > 1e-9) {
      out.push('The cheap model gets ' + b(e.sf.L6_right || 0) + ' right first time and ' + b(e.sf.L6_wrong || 0) + ' wrong.');
      out.push('The gate catches ' + b(e.bf.l6_caught || 0) + ' of those wrong ones.');
    } else out.push('No piece goes to the cheap model.');
    out.push(b(e.sf.L8_opus || 0) + ' reach Opus and ' + b(e.sf.L9_fable || 0) + ' reach Fable.');
    out.push(b(e.outs.EQUAL || 0) + ' end accepted and right; ' + b(e.wrong) + ' end wrong.');
    return out.join(' ');
  }
  function map_graphNow() {
    var all = ST.scope === 'all', sf = all ? EVP.stateFlow : EVC.stateFlow, bf = all ? EVP.branchFlow : EVC.branchFlow;
    var nS = MAP.states.filter(function (s) { return (sf[s.id] || 0) >= 1e-12; }).length;
    var nB = MAP.branches.filter(function (b) { return (bf[b.id] || 0) >= 1e-12; }).length;
    var nL = MAP.branches.filter(function (b) { return (bf[b.id] || 0) >= 0.01; }).length;
    var top = MAP.branches.slice().sort(function (a, c) { return (bf[c.id] || 0) - (bf[a.id] || 0); })[0];
    return '<b>' + nS + '</b> of the ' + MAP.states.length + ' boxes and <b>' + nB + '</b> of the ' + MAP.branches.length + ' lines carry work under ' + esc(PNAME[ST.policy].toLowerCase()) + '. ' +
      '<b>' + nL + '</b> lines carry at least one piece and show their count. ' +
      'The busiest line: ' + esc(top.label.toLowerCase()) + ', <b>' + esc(map_c(bf[top.id] || 0)) + '</b> pieces.';
  }

  function renderMap() {
    var cls = classFor(ST.policy);
    var stocks = ST.policy === 'library' ? stocksAt(ST.period) : {};
    var libStocks = stocksAt(ST.period);
    EVC = E.evaluateClass(MAP, cur, ST.policy, cls, stocks);
    EVP = E.evaluatePolicy(MAP, cur, ST.policy, stocks);
    BV = C.branchValues(E, MAP, cur, ST.policy, cls, stocks);
    ROUTES = E.routes(MAP, cur, ST.policy, { classValues: cls, stocks: stocks, sortBy: ST.sortBy });
    map_evalAll(libStocks);
    if (ST.selAuto || !ROUTES.routes.some(function (r) { return r.id === ST.sel || r.id.indexOf(ST.sel + '>') === 0; })) {
      var top = ROUTES.routes.slice().sort(function (a, b) { return b.chance - a.chance; })[0];
      ST.sel = top ? top.id : null; ST.stepIdx = null; ST.selAuto = false; ST.userSel = false;
    }
    var lede = $('#mapLede');
    if (lede) lede.innerHTML = 'Start with the short version: 100 pieces of work, three ways of working, one row each. Below it, the full map shows every box and line. Tap a route in the tree to light it up.';
    renderPickers();
    paintGraph();
    renderPiece(cls, stocks);
    renderTree();
    renderStepper();
  }

        else { var lane = LY.lanes[b.id] !== undefined ? LY.lanes[b.id] : z.r; var x1 = a.x + g.nw / 2, x2 = z.x - g.nw / 2; lx = (x1 + x2) / 2; ly = g.y0 + lane * g.rh + o; }
        var t = fmt(100 * f, 'count'), tw = 8 * t.length + 6, tries = adj ? [0, 16, -16, 32, -32] : [0, 40, -40, 80, -80], ok = false;
        for (var i = 0; i < tries.length && !ok; i++) { var X = adj ? lx : lx + tries[i], Y = adj ? ly + tries[i] : ly; if (map_free(X, Y, tw)) { lx = X; ly = Y; ok = true; } }
        placed.push([lx, ly, tw]);
        labs += '<text class="ux-map-elab' + (on ? ' on' : '') + '" text-anchor="middle" x="' + lx.toFixed(1) + '" y="' + (ly + 4.5).toFixed(1) + '" style="font-size:13px">' + esc(t) + '</text>';
      }
    });
    s += '<g>' + edges + '</g><g>' + hits + '</g>';
    MAP.states.forEach(function (st) {
      var p = pos(g, st.id), x = p.x - g.nw / 2, y = p.y - g.nh / 2;
      var sf = opts.stateFlow ? (opts.stateFlow[st.id] || 0) : null;
      var right = st.outcome && OUT[st.id].right;
      var cls = 'g-node' + (st.outcome ? ' outcome' : '') + (st.outcome && !OUT[st.id].right ? ' wrongout' : '') + (right ? ' ux-map-rightout' : '') + (opts.start === st.id ? ' start' : '') + (opts.onStates && opts.onStates.has(st.id) ? ' on' : '') + (sf !== null && sf < 1e-12 && !mini ? ' idle' : '');
      var tip = '<b>' + esc(st.label) + '</b><br>Stage: ' + esc(STEP[st.step].label) + (stageWords(st.step) ? ' (' + stageWords(st.step) + ')' : '');
      if (opts.stateFlow) tip += '<br>Share of ' + (opts.all ? 'all work' : 'this piece') + ' passing here: ' + num(sf, 'G', 'pct') + (sf < 1e-12 ? '<br>No work reaches this box under this choice.' : '');
      if (opts.start === st.id) tip += '<br>This way of working starts here.';
      s += '<g class="' + cls + '" data-tip="' + esc(tip) + '">';
      if (!mini && st.outcome) s += '<text class="ux-map-endw ' + (right ? 'good' : 'bad') + '" text-anchor="middle" x="' + p.x + '" y="' + (y - 7) + '" style="font-size:13px">' + (right ? 'RIGHT ENDING' : 'WRONG ENDING') + '</text>';
      s += '<rect x="' + x + '" y="' + y + '" width="' + g.nw + '" height="' + g.nh + '" rx="' + (mini ? 4 : 6) + '"/>';
      if (!mini) {
        var lines = wrapWords(st.label, st.outcome ? 11 : 12).slice(0, 4);
        var lsp = 14.5, blockH = lines.length * lsp + (sf !== null ? 17 : 0);
        var y0 = p.y - blockH / 2 + 11;
        lines.forEach(function (ln, j) { s += '<text class="' + (st.outcome ? 'g-out' : 'g-lab') + '" text-anchor="middle" x="' + p.x + '" y="' + (y0 + j * lsp).toFixed(1) + '" style="font-size:13px">' + esc(ln) + '</text>'; });
        if (sf !== null) s += '<text class="g-cnt" data-tagged="G" text-anchor="middle" x="' + p.x + '" y="' + (y0 + lines.length * lsp + 3).toFixed(1) + '" style="font-size:14px">' + esc(fmt(100 * sf, 'count')) + (st.outcome ? ' of 100' : '') + '</text>';
      }
      s += '</g>';
    });
    if (!mini) s += '<g class="ux-map-labs" data-tagged="G">' + labs + '</g>';
    s += '</svg>';
    el.innerHTML = s;
    el.classList.toggle('focus', focus);
  }

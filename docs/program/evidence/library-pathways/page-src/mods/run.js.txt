  function run_mid() { if (!RUN_MID) RUN_MID = E.timePath(MAP, BEST); return RUN_MID; }
  function run_head(q, how, now) {
    return '<div class="ux-run-head"><h3>' + esc(q) + '</h3><p class="ux-run-how"><b>How to read it.</b> ' + how + '</p><p class="ux-run-now"><b>What it shows now.</b> ' + now + '</p></div>';

  function run_x(v) { return fmt(v, 'ratio'); }
  function run_word(v) { return Math.abs(v - 1) < 0.005 ? 'same' : v < 1 ? 'lower' : 'higher'; }
  function run_nudge(items, gap) {
    items.sort(function (a, b) { return a.y - b.y; });
    for (var i = 1; i < items.length; i++) if (items[i].y - items[i - 1].y < gap) items[i].y = items[i - 1].y + gap;
    return items;
  }

  // ---------- picture 1: the three ways
  function run_ways(top, lad, last, first) {
    var rows = [['top_first', top, null], ['cheap_ladder', lad, null], ['library', last.library, first.library]];
    var maxC = Math.max(top.cost, lad.cost, last.library.cost, first.library.cost) || 1;
    var maxW = Math.max(top.wrong, lad.wrong, last.library.wrong, first.library.wrong) || 1;
    var W = 600, bx = 64, bw = 300, y = 6, s = '<svg viewBox="0 0 ' + W + ' 450" role="img" aria-label="Cost per piece and wrong rate of the three ways of working">';
    rows.forEach(function (r) {
      var lib = r[0] === 'library', v = r[1], f = r[2], bh = 44 + 2 * 44;
      if (lib) s += '<rect x="1" y="' + (y - 4) + '" width="' + (W - 2) + '" height="' + (bh + 4) + '" rx="10" fill="none" stroke="currentColor" stroke-width="2.5"/>';
      s += '<text class="ux-run-sn' + (lib ? ' lib' : '') + '" x="' + (lib ? 12 : 4) + '" y="' + (y + 22) + '">' + esc(PNAME[r[0]]) + (lib ? ' (last period)' : '') + '</text>';
      [['Cost', v.cost, maxC, f && f.cost, 'cost', ' per piece'], ['Wrong', v.wrong, maxW, f && f.wrong, 'pct', ' in a hundred pieces']].forEach(function (k, i) {
        var ry = y + 34 + i * 46, len = Math.max(2, k[1] / k[2] * bw), isP = k[4] === 'pct';
        var val = isP ? fmt(k[1] * 100, 'funnel') : fmt(k[1], 'cost'), fv = k[3] === null || k[3] === undefined ? null : (isP ? fmt(k[3] * 100, 'funnel') : fmt(k[3], 'cost'));
        s += '<text class="ux-run-rl" x="' + (lib ? 12 : 4) + '" y="' + (ry + 18) + '">' + k[0] + '</text>' +
          '<rect x="' + bx + '" y="' + ry + '" width="' + len.toFixed(1) + '" height="26" rx="3" fill="' + (lib ? 'var(--ink)' : 'var(--mid)') + '"' + (lib ? '' : ' opacity=".55"') + '/>' +
          '<text class="ux-run-nv" x="' + (bx + len + 8).toFixed(1) + '" y="' + (ry + 14) + '">' + esc(val) + '</text>' +
          '<text class="ux-run-nu" x="' + (bx + len + 8).toFixed(1) + '" y="' + (ry + 30) + '">' + esc((isP ? 'in a hundred pieces' : 'per piece') + (fv !== null ? ', first period ' + fv : '')) + '</text>';
      });
      y += bh + 14;
    });
    s += svgTag(W - 16, 434, 'G') + '</svg>';
    return '<div class="ux-run-svg" data-tagged="G">' + s + '</div>';
  }

  // ---------- picture 2: the two comparisons, bars from the line at one
  function run_cmp(R, F) {
    var RS = TX.sections.run, W = 600, pl = 92, pr = 408, y0 = 0;
    var blocks = [
      ['Against the cheap model with the ladder', 'the cheap model with the ladder', 'cost_vs_ladder', 'wrong_vs_ladder'],
      ['Against the strong model directly', 'the strong model directly', 'cost_vs_top', 'wrong_vs_top']];
    var s = '<svg viewBox="0 0 ' + W + ' 450" role="img" aria-label="The library as a multiple of each other way, last period">';
    blocks.forEach(function (b) {
      var vals = [R[b[2]], R[b[3]], F[b[2]], F[b[3]]].filter(isFinite).concat([1]);
      var lo = Math.min.apply(null, vals), hi = Math.max.apply(null, vals), ext = Math.max(hi - lo, 0.4), c = (hi + lo) / 2;
      lo = c - ext / 2 * 1.2; hi = c + ext / 2 * 1.2;
      var X = function (v) { return pl + (Math.max(lo, Math.min(hi, v)) - lo) / (hi - lo) * (pr - pl); }, x1 = X(1);
      s += '<text class="ux-run-sn" x="4" y="' + (y0 + 18) + '">' + esc(b[0]) + '</text>';
      var anchorEnd = x1 > (pl + pr) / 2;
      s += '<text class="ux-run-nu" x="' + (anchorEnd ? x1 - 6 : x1 + 6) + '" y="' + (y0 + 40) + '" text-anchor="' + (anchorEnd ? 'end' : 'start') + '">line at one: same as ' + esc(b[1]) + '</text>';
      s += '<text class="ux-run-nu" x="' + (x1 - 8) + '" y="' + (y0 + 62) + '" text-anchor="end">← lower</text><text class="ux-run-nu" x="' + (x1 + 8) + '" y="' + (y0 + 62) + '">higher →</text>';
      [['Cost', b[2]], ['Wrong rate', b[3]]].forEach(function (k, i) {
        var v = R[k[1]], f = F[k[1]], cy = y0 + 72 + i * 44, good = v < 1, same = Math.abs(v - 1) < 0.005;
        s += '<text class="ux-run-rl" x="4" y="' + (cy + 19) + '">' + k[0] + '</text>' +
          '<rect x="' + Math.min(x1, X(v)).toFixed(1) + '" y="' + cy + '" width="' + Math.max(2.5, Math.abs(X(v) - x1)).toFixed(1) + '" height="28" rx="2" fill="' + (same ? 'var(--mid)' : good ? 'var(--good)' : 'var(--bad)') + '"/>' +
          '<text class="ux-run-nv" x="' + (pr + 16) + '" y="' + (cy + 14) + '">' + esc(run_x(v)) + ' times, ' + run_word(v) + '</text>' +
          '<text class="ux-run-nu" x="' + (pr + 16) + '" y="' + (cy + 30) + '">first period ' + esc(run_x(f)) + '</text>';
      });
      var ay = y0 + 164;
      s += '<line x1="' + x1 + '" x2="' + x1 + '" y1="' + (y0 + 68) + '" y2="' + ay + '" stroke="currentColor" stroke-width="2.4"/>' +
        '<line x1="' + pl + '" x2="' + pr + '" y1="' + ay + '" y2="' + ay + '" stroke="currentColor" stroke-width="1"/>';
      niceTicks(lo, hi, 5).forEach(function (t) { s += '<line x1="' + X(t) + '" x2="' + X(t) + '" y1="' + ay + '" y2="' + (ay + 5) + '" stroke="currentColor"/><text class="ux-run-tk" x="' + X(t) + '" y="' + (ay + 20) + '" text-anchor="middle">' + fmt(t, 'ratio') + '</text>'; });
      s += '<text class="ux-run-nu" x="' + ((pl + pr) / 2) + '" y="' + (ay + 40) + '" text-anchor="middle">library as a multiple of this way (one = the same)</text>';
      y0 += 222;
    });
    s += svgTag(W - 16, 434, 'G') + '</svg>';
    return '<div class="ux-run-svg" data-tagged="G">' + s + '</div>';
  }

  function run_ratioChart(title, ref, kc, kw) {
    var W = 600, H = 350, L = 54, Rm = 150, T = 34, B = 56;
    var xs = PATH.periods.map(function (p) { return p.period + 1; });
    var cs = PATH.periods.map(function (p) { return p.ratios[kc]; }), ws = PATH.periods.map(function (p) { return p.ratios[kw]; });

    var lo = Math.min.apply(null, all), hi = Math.max.apply(null, all), ext = Math.max(hi - lo, 0.06), pad = ext * 0.2, c = (hi + lo) / 2;
    lo = c - ext / 2 - pad; hi = c + ext / 2 + pad;
    var pw = W - L - Rm;
    var X = function (v) { return L + (v - 1) / Math.max(1, NPER - 1) * pw; }, Y = function (v) { return T + (hi - Math.min(hi, Math.max(lo, v))) / (hi - lo) * (H - T - B); };

    s += '<text class="ux-run-sn" x="4" y="20">' + esc(title) + '</text>';

    niceTicks(lo, hi, 5).forEach(function (t) { s += '<line x1="' + L + '" x2="' + (L + pw) + '" y1="' + Y(t).toFixed(1) + '" y2="' + Y(t).toFixed(1) + '" stroke="var(--line-2)" stroke-width=".7"/><text class="ux-run-tk" x="' + (L - 6) + '" y="' + (Y(t) + 4.5).toFixed(1) + '" text-anchor="end">' + fmt(t, 'ratio') + '</text>'; });

  function updateSliderLabels() {
    var changed = [], perG = {};
    MAP.parameters.forEach(function (p) {
      var row = ROWS[p.id], ch = Math.abs(cur[p.id] - row.best) > 1e-12;
      if (ch) { changed.push(p); perG[PSTAGE[p.id]] = (perG[PSTAGE[p.id]] || 0) + 1; }
      var el = document.getElementById('sv-' + p.id); if (!el) return;
      el.innerHTML = num(cur[p.id], ch ? 'G' : row.tag, p.integer ? 'int' : 'auto');
      el.closest('.slider').classList.toggle('changed', ch);
    });
    $$('.ux-run-gc').forEach(function (e) { var n = perG[e.getAttribute('data-gc')]; e.innerHTML = n ? ' · <b>' + n + ' changed</b>' : ''; });
    var list = '';
    if (changed.length) {
      list = '<div class="ux-run-chg"><b data-label="1">Changed from the middle values</b> <span class="small muted">(' + changed.length + '). A moved value is your guess, so it shows G.</span><ul>' + changed.map(function (p) {
        var row = ROWS[p.id], up = cur[p.id] > row.best, k = p.integer ? 'int' : 'auto';
        return '<li><span class="ux-run-cn">' + esc(pname(p.id)) + '</span><span class="ux-run-cv">' + (up ? '▲ higher' : '▼ lower') + ': ' + num(cur[p.id], 'G', k) + ' <span class="muted">was ' + esc(fmt(row.best, k)) + ' (middle)</span></span></li>';
      }).join('') + '</ul></div>';
    }
    $('#changedNote').innerHTML = changed.length ? list : 'All settings at their middle values.';
  }

  function fmt(v, kind) {

  function roundPath(pts, rad) {
    var clean = [pts[0]];
    for (var i = 1; i < pts.length; i++) { var q = clean[clean.length - 1]; if (Math.abs(pts[i][0] - q[0]) > 0.01 || Math.abs(pts[i][1] - q[1]) > 0.01) clean.push(pts[i]); }

    for (var k = 1; k < clean.length - 1; k++) {
      var p0 = clean[k - 1], p1 = clean[k], p2 = clean[k + 1];
      var d1x = p1[0] - p0[0], d1y = p1[1] - p0[1], d2x = p2[0] - p1[0], d2y = p2[1] - p1[1];
      var l1 = Math.hypot(d1x, d1y), l2 = Math.hypot(d2x, d2y);
      var r = Math.min(rad, l1 / 2, l2 / 2);
      var a = [p1[0] - d1x / l1 * r, p1[1] - d1y / l1 * r], b = [p1[0] + d2x / l2 * r, p1[1] + d2y / l2 * r];
      d += 'L' + a[0].toFixed(1) + ' ' + a[1].toFixed(1) + 'Q' + p1[0].toFixed(1) + ' ' + p1[1].toFixed(1) + ' ' + b[0].toFixed(1) + ' ' + b[1].toFixed(1);
    }
    var z = clean[clean.length - 1];
    return d + 'L' + z[0].toFixed(1) + ' ' + z[1].toFixed(1);

      if (opts.cur === b.id) edges += '<path class="g-halo" d="' + d + '" stroke-width="' + (w + 7) + '"/>';
      edges += '<path class="' + cls + '" d="' + d + '" stroke-width="' + w.toFixed(2) + '"' + (zero && !on ? '' : ' marker-end="url(#' + uid + 'a)"') + (opts.cur === b.id ? ' style="stroke-width:' + (w + 1.5).toFixed(2) + '"' : '') + '/>';
      var tip = '<b>' + esc(b.label) + '</b><br>Stage: ' + esc(STEP[STATE[b.from].step].label);

  function buildTrie(routes) {
    var root = { kids: {}, order: [], chance: 0, carried: 0, key: '' };
    routes.forEach(function (r) {
      var node = root; node.chance += r.chance; node.carried += r.carried;
      r.branches.forEach(function (bid, i) {
        var bi = MAP.branches.indexOf(BR[bid]);
        var ch = node.kids[bi];
        if (!ch) { ch = node.kids[bi] = { bid: bid, kids: {}, order: [], chance: 0, carried: 0, key: (node.key ? node.key + '>' : '') + bi }; node.order.push(bi); }
        ch.chance += r.chance; ch.carried += r.carried; node = ch;
      });
      node.route = r;
    });
    return root;
  }
  function kidsOf(n) { var by = ST.sortBy; return n.order.map(function (k) { return n.kids[k]; }).sort(function (a, b) { return b[by] - a[by]; }); }
  function chainFrom(n) { var chain = [n]; while (!n.route && n.order.length === 1) { n = n.kids[n.order[0]]; chain.push(n); } return chain; }

  // The square as SVG for one sheet and one ratio. Shade: solid below one, hatched above one, darker
  // further from one. Every cell prints its value. The thick line runs between cells on either side of one.
  function squareSvg(sheetId, ratio) {
    var q = sheetOf(sheetId), cells = q.sheet.cells, n = q.levels.length, c = 40, L = 66, T = 8, W = L + n * c + 12, H = T + n * c + 52;
    var dev = 1e-9;
    cells.forEach(function (row) { row.forEach(function (x) { var v = x.ratios[ratio]; if (isFinite(v)) dev = Math.max(dev, Math.abs(v - 1)); }); });
    var X = function (j) { return L + j * c; }, Y = function (i) { return T + (n - 1 - i) * c; };
    var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" data-sheet="' + sheetId + '" data-ratio="' + ratio + '" role="img" aria-label="' + esc(CT.square_title) + '">' +
      '<defs><pattern id="sqh" width="6" height="6" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="6" stroke="currentColor" stroke-width="2.2"/></pattern></defs>';
    cells.forEach(function (row, i) {
      row.forEach(function (x, j) {
        var v = x.ratios[ratio], k = C.sideOfOne(v, 0), a = 0.08 + 0.5 * Math.min(1, Math.abs(v - 1) / dev);
        var sel = ST.sqCell && ST.sqCell[0] === i && ST.sqCell[1] === j;
        s += '<g class="sqc" role="button" tabindex="0" data-i="' + i + '" data-j="' + j + '" data-v="' + v + '">' +
          '<rect x="' + X(j) + '" y="' + Y(i) + '" width="' + c + '" height="' + c + '" fill="' + (k === 'above' ? 'url(#sqh)' : 'currentColor') + '" opacity="' + a.toFixed(3) + '"/>' +
          '<rect x="' + X(j) + '" y="' + Y(i) + '" width="' + c + '" height="' + c + '" class="sqf' + (sel ? ' sel' : '') + '"/>' +
          '<text class="sqv' + (k !== 'above' && a > 0.4 ? ' inv' : '') + '" x="' + (X(j) + c / 2) + '" y="' + (Y(i) + c / 2 + 4) + '" text-anchor="middle">' + esc(fmt(v, 'ratio')) + '</text></g>';
      });
    });
    // the line where the ratio is one: every cell edge with one side below one and the other not
    var d = '';
    for (var i = 0; i < n; i++) for (var j = 0; j < n; j++) {
      var lo = cells[i][j].ratios[ratio] < 1;
      if (j + 1 < n && lo !== (cells[i][j + 1].ratios[ratio] < 1)) d += 'M' + X(j + 1) + ' ' + Y(i) + 'v' + c;
      if (i + 1 < n && lo !== (cells[i + 1][j].ratios[ratio] < 1)) d += 'M' + X(j) + ' ' + Y(i) + 'h' + c;
    }
    if (d) s += '<path class="sqone" d="' + d + '"/>';
    var gx = LIBGI[q.x], gy = LIBGI[q.y], by = T + n * c;
    s += '<text class="sqa" x="' + L + '" y="' + (by + 16) + '">' + esc(gx.poor) + '</text><text class="sqa" x="' + (L + n * c) + '" y="' + (by + 16) + '" text-anchor="end">' + esc(gx.good) + '</text>' +
      '<text class="sqt" x="' + (L + n * c / 2) + '" y="' + (by + 40) + '" text-anchor="middle">' + esc(gx.label) + ' →</text>' +
      '<text class="sqa" transform="translate(' + (L - 8) + ' ' + by + ') rotate(-90)">' + esc(gy.poor) + '</text><text class="sqa" transform="translate(' + (L - 8) + ' ' + T + ') rotate(-90)" text-anchor="end">' + esc(gy.good) + '</text>' +
      '<text class="sqt" transform="translate(' + (L - 30) + ' ' + (T + n * c / 2) + ') rotate(-90)" text-anchor="middle">' + esc(gy.short.replace(/^./, function (ch) { return ch.toUpperCase(); })) + ' →</text>';
    return s + '</svg>';
  }

  function niceTicks(lo, hi, n) {
    var span = hi - lo || 1, step = Math.pow(10, Math.floor(Math.log10(span / n))), err = span / n / step;
    if (err >= 7.5) step *= 10; else if (err >= 3.5) step *= 5; else if (err >= 1.5) step *= 2;
    var t = [], v = Math.ceil(lo / step) * step;
    for (; v <= hi + step * 1e-9; v += step) t.push(+v.toFixed(10));
    return t;
  }

  function runEnsemble() {
    if (RUNS) return;

    RUNS = TIES.map(function (tie) { return C.ensembleRunner(E, MAP, TABLE, { n: n, seed: seed, tie: tie }); });
    $('#ensRun').disabled = true;
    // Yield through a message channel, not setTimeout: timers are clamped or throttled in some panes
    // and background tabs. Each slice holds the thread for at most about 40 ms.
    var ch = typeof MessageChannel === 'function' ? new MessageChannel() : null;
    var later = ch ? function (f) { ch.port1.onmessage = function () { f(); }; ch.port2.postMessage(0); } : function (f) { setTimeout(f, 0); };
    var total = n * TIES.length;
    (function tick() {
      var t0 = performance.now();
      var r = RUNS.filter(function (x) { return !x.done(); })[0];
      while (r && performance.now() - t0 < 40) { r.step(1); if (r.done()) r = RUNS.filter(function (x) { return !x.done(); })[0]; }
      var k = RUNS.reduce(function (a, x) { return a + x.count(); }, 0);
      $('#ensBar').style.width = (100 * k / total) + '%';
      $('#ensTxt').textContent = k + ' of ' + total + ' runs';
      if (r) { later(tick); return; }
      if (ch) { ch.port1.close(); ch.port2.close(); }
      ENSR = {};
      RUNS.forEach(function (x) { var res = x.result(); res.draws = x.draws(); ENSR[x.tie] = res; });
      RUNS = null;
      $('#ensRun').disabled = false;
      renderEnsemble();
    })();
  }
  function spreadChart(title, ids, S) {

  function scatter(title, kc, kw, w) {
    var W = 420, H = 300, L = 46, Rm = 12, T = 26, B = 40;
    var xs = ENS.draws.map(function (d) { return d[w].ratios[kc]; }), ys = ENS.draws.map(function (d) { return d[w].ratios[kw]; });
    var sx = xs.filter(isFinite).slice().sort(function (a, b) { return a - b; }), sy = ys.filter(isFinite).slice().sort(function (a, b) { return a - b; });
    var x0 = Math.min(E.percentile(sx, 0.5), 1), x1 = Math.max(E.percentile(sx, 99.5), 1), y0 = Math.min(E.percentile(sy, 0.5), 1), y1 = Math.max(E.percentile(sy, 99.5), 1);
    var px = (x1 - x0) * 0.06 || 0.02, py = (y1 - y0) * 0.06 || 0.02; x0 -= px; x1 += px; y0 -= py; y1 += py;
    var clipped = 0;
    var X = function (v) { return L + (v - x0) / (x1 - x0) * (W - L - Rm); }, Y = function (v) { return T + (y1 - v) / (y1 - y0) * (H - T - B); };
    var s = '<svg viewBox="0 0 ' + W + ' ' + H + '" role="img" aria-label="' + esc(title) + '"><text class="lbl-strong" x="' + L + '" y="14">' + esc(title) + '</text>' + svgTag(W - 22, 4, 'G');
    niceTicks(x0, x1, 5).forEach(function (t) { s += '<line x1="' + X(t) + '" x2="' + X(t) + '" y1="' + T + '" y2="' + (H - B) + '" stroke="var(--line)"/><text class="lbl" x="' + X(t) + '" y="' + (H - B + 14) + '" text-anchor="middle">' + fmt(t, 'ratio') + '</text>'; });
    niceTicks(y0, y1, 5).forEach(function (t) { s += '<line x1="' + L + '" x2="' + (W - Rm) + '" y1="' + Y(t) + '" y2="' + Y(t) + '" stroke="var(--line)"/><text class="lbl" x="' + (L - 5) + '" y="' + (Y(t) + 3.5) + '" text-anchor="end">' + fmt(t, 'ratio') + '</text>'; });
    s += '<line x1="' + X(1) + '" x2="' + X(1) + '" y1="' + T + '" y2="' + (H - B) + '" stroke="currentColor" stroke-dasharray="1 3"/><line x1="' + L + '" x2="' + (W - Rm) + '" y1="' + Y(1) + '" y2="' + Y(1) + '" stroke="currentColor" stroke-dasharray="1 3"/>';
    var dots = '';
    for (var i = 0; i < xs.length; i++) {
      var a = xs[i], b = ys[i]; if (!isFinite(a) || !isFinite(b)) { clipped++; continue; }
      if (a < x0 || a > x1 || b < y0 || b > y1) clipped++;
      dots += '<circle cx="' + X(Math.max(x0, Math.min(x1, a))).toFixed(1) + '" cy="' + Y(Math.max(y0, Math.min(y1, b))).toFixed(1) + '" r="1.7"/>';
    }
    s += '<g fill="currentColor" opacity=".35">' + dots + '</g>';
    s += '<text class="lbl" x="' + ((L + W - Rm) / 2) + '" y="' + (H - 8) + '" text-anchor="middle">cost ratio (left of the dotted line: cheaper)</text>';
    s += '<text class="lbl" transform="translate(12 ' + ((T + H - B) / 2) + ') rotate(-90)" text-anchor="middle">wrong-rate ratio (below: fewer wrong)</text>';
    return '<div class="chartbox" data-tagged="G">' + s + '</svg>' + (clipped ? '<p class="small muted">' + clipped + ' runs lie beyond the edges and sit on the edge.</p>' : '') + '</div>';
  }

    if (CM.promoMeasure) rows.push(['promo', 'More caught at the first gate by promoted checks, of the same hundred', 'funnel']);

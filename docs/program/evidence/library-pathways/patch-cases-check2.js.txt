let applied = 0;
EDITS.forEach(([cid, slot, oldT, newT]) => {
  const c = d.cases.find(x => x.id === cid);
  if (!c) { problems.push('no case ' + cid); return; }
  const o = typeof slot === 'number' ? c.steps[slot - 1] : c;
  const k = typeof slot === 'number' ? 'note' : slot;
  if (o[k].includes(oldT)) { o[k] = o[k].split(oldT).join(newT); applied++; }
  else if (!o[k].includes(newT)) problems.push('text not found in ' + cid + ' ' + slot + ': ' + oldT.slice(0, 60));
});

// The walks wrote three dates as "2026-11-0x". An earlier pass changed them to plain words in
// cases.json by hand; this keeps that change reproducible from the walk files.
d.cases.forEach(c => { if (/2026-11-0x/.test(c.when || '')) { c.when = c.when.replace(/2026-11-0x/g, 'early November 2026'); applied++; } });

// every sentence in the case text at most 20 words (the build's own splitter and count)
const sentenceOf = s => s.replace(/<[^>]*>/g, ' ').replace(/\{\{[^}]+\}\}/g, 'X').replace(/&[a-z]+;/g, ' ');
const sentences = s => sentenceOf(s).split(/(?<=[.!?])\s+(?=[A-Z'‘“(0-9X])/).map(x => x.trim()).filter(Boolean);
const wc = s => s.split(/\s+/).filter(t => /[A-Za-z0-9]/.test(t)).length;
d.cases.forEach(c => {
  const texts = ['title', 'story', 'entry', 'worked'].map(k => [k, c[k] || '']).concat(c.steps.map(s => ['step ' + s.step, s.note]));
  texts.forEach(([k, t]) => sentences(t).forEach(x => { if (wc(x) > 20) problems.push(c.id + ' ' + k + ' (' + wc(x) + '): ' + x.slice(0, 80)); }));
});
if (JSON.stringify(d.cases.map(c => c.steps.map(s => [s.status, !!s.ks]))) !== marksBefore) problems.push('a status or a ks mark changed');
if (problems.length) { console.error(problems.join('\n')); process.exit(1); }
fs.writeFileSync(FILE, JSON.stringify(d, null, 1));
console.log('cases.json: ' + applied + ' sentences rewritten this run, ' + EDITS.length + ' in all; no sentence over 20 words; statuses and ks marks unchanged');

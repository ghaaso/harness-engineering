// ------------------------------------------------------------------ the page's own number formats (page.js fmt)
function fmt(v, kind) {
  if (typeof v === 'string') return v;
  if (v === null || v === undefined || !isFinite(v)) return 'none';
  const a = Math.abs(v);
  if (kind === 'pct') { const x = v * 100; return (x !== 0 && Math.abs(x) < 0.1 ? x.toFixed(2) : x.toFixed(1)) + '%'; }
  if (kind === 'ratio') return a >= 10 ? v.toFixed(1) : v.toFixed(2);
  if (kind === 'cost') return a >= 100 ? v.toFixed(0) : v.toFixed(1);
  if (kind === 'int') return Math.round(v).toLocaleString('en');
  if (kind === 'rho') return (v > 0 ? '+' : v < 0 ? '−' : '') + Math.abs(v).toFixed(2);
  if (kind === 'funnel') return v === 0 ? '0' : a >= 1 ? v.toFixed(1) : a >= 0.01 ? v.toFixed(2) : v.toPrecision(1);
  if (kind === 'weight') { const y = 100 * v; return (y < 0 ? '−' : '') + (y === 0 ? '0' : Math.abs(y) >= 0.1 ? Math.abs(y).toFixed(1) : Math.abs(y) >= 0.01 ? Math.abs(y).toFixed(2) : Math.abs(y).toPrecision(1)) + '%'; }
  if (v === 0) return '0';
  if (a >= 100) return v.toFixed(0);
  if (a >= 10) return v.toFixed(1);
  if (a >= 1) return v.toFixed(2);
  return v.toFixed(3).replace(/0+$/, '').replace(/\.$/, '.0');
}

function add(id, kind, section, from, label, tests, extra) {
  const e = Object.assign({ id, kind, section, from, label: label === undefined ? '' : label, tests: [] }, extra || {});
  const seen = {};
  (tests || []).filter(Boolean).forEach(t => {
    let f = t.f || t.t; if (seen[f]) { seen[f]++; f += seen[f]; } else seen[f] = 1;
    t.id = id + '#' + f; t.f = t.f || t.t; e.tests.push(t);

}

// text with tokens: the literal pieces must appear; the token values must sit near them
function tok(raw) {
  const values = [], parts = []; let last = 0, m; const re = /\{\{(c|g|n):([^}|]+)(?:\|([MSG]))?\}\}/g;
  while ((m = re.exec(raw))) {
    parts.push(raw.slice(last, m.index));
    values.push(m[1] === 'c' ? (typeof CNT[m[2]] === 'number' ? fmt(CNT[m[2]], 'int') : String(CNT[m[2]])) : m[1] === 'g' ? String(FACTS[m[2]]) : m[2]);
    last = m.index + m[0].length;
  }
  parts.push(raw.slice(last));
  return { parts: parts.map(s => s.trim()).filter(s => s.length >= 3), values };
}
function rich(raw, o) {
  R(raw);
  const k = tok(raw);
  if (!values0(k)) return [prts(k.parts, Object.assign({ f: 'text' }, o || {}))];
  const out = [prts(k.parts, Object.assign({ f: 'text' }, o || {}))];
  const anchor = k.parts.slice().sort((a, b) => b.length - a.length)[0];
  out.push(cl([anchor].concat(k.values), 900, Object.assign({ f: 'numbers', where: 'open' }, o && o.soft ? { soft: true } : {})));
  return out;
}

function buildTrie(routes) {
  const root = { kids: {}, order: [], chance: 0, carried: 0, key: '' };
  routes.forEach(r => {
    let node = root; node.chance += r.chance; node.carried += r.carried;
    r.branches.forEach(bid => {
      const bi = MAP.branches.indexOf(BR[bid]); let ch = node.kids[bi];
      if (!ch) { ch = node.kids[bi] = { bid, kids: {}, order: [], chance: 0, carried: 0, key: (node.key ? node.key + '>' : '') + bi }; node.order.push(bi); }
      ch.chance += r.chance; ch.carried += r.carried; node = ch;
    });
    node.route = r;
  });
  return root;
}

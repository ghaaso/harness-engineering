// Map refuter pass (refute-map.md, 2026-11-12): wording only. The model is unchanged.
// Run from this folder:  node - < patch-map-refute.js.txt
// Uses process.cwd(), never __dirname. Safe to run more than once.
// 1. Gives every parameter a short plain name ("name"), so the page and the record can
//    show names instead of code ids (MF7).
// 2. Says "piece of work" where reader-facing text said "episode" (SF11). Ids and
//    expressions (episodes_per_period) are not touched.
// It changes no expression, no branch, no state and no parameter id.
'use strict';
const fs = require('fs');
const path = require('path');
const FILE = path.join(process.cwd(), '..', '..', 'library-pathways', 'map.json');
const map = JSON.parse(fs.readFileSync(FILE, 'utf8'));

const NAMES = {
  p_visible: 'Share of work that is visible',
  need_visible: 'Cheap model wrong, visible work',
  need_latent: 'Cheap model wrong, latent work',
  u_exact: 'Ceiling: the same problem held',
  u_similar: 'Ceiling: a similar entry held',
  u_composite: 'Ceiling: parts held',
  latent_coverage_factor: 'Holding cut for latent work',
  ok_coverage_factor: 'Holding cut for work the cheap model gets right',
  k_fill: 'Entries needed to fill the library',
  v_max: 'Look-alike ceiling',
  k_distract: 'Entries needed to reach the look-alike ceiling',
  t_visible: 'Lookup starts, visible work',
  t_latent: 'Lookup starts, latent work',
  c_lookup: 'Cost of a lookup',
  r_exact: 'Lookup finds the same problem',
  r_similar: 'Lookup finds the similar entry',
  r_comp_full: 'Lookup finds all parts',
  r_comp_part: 'Lookup finds some parts',
  f_false: 'Look-alike comes back on top',
  c_check_share: 'Entries with a mechanical check',
  c_screen: 'Cost of a mechanical check',
  a_right: 'Cheap model accepts a good entry',
  a_wrong: 'Cheap model accepts a wrong entry',
  d_deliver: 'Entry in the window when needed',
  c_carry: 'Cost of carrying an entry',
  q_exact: 'Right with a same-problem entry',
  q_similar: 'Right with a similar entry',
  q_comp: 'Right with all parts',
  q_part: 'Right with some parts',
  h_harm: 'Wrong entry spoils right work',
  n_cheap: 'Cheap calls per attempt',
  s_exact: 'Calls saved, same-problem entry',
  s_similar: 'Calls saved, similar entry',
  s_comp: 'Calls saved, all parts',
  s_part: 'Calls saved, some parts',
  g_tests_visible: 'Tests catch it, visible work',
  g_tests_latent: 'Tests catch it, latent work',
  s_review: 'Share of gate passes with a review',
  g_review: 'Review catches it',
  c_review: 'Opus calls per review',
  rho_retry: 'Cheap retry is right',
  c_handoff: 'Cost of the handoff to Opus',
  n_opus: 'Opus calls per attempt',
  n_fable: 'Fable calls per attempt',
  m_opus: 'Opus price ratio',
  m_fable: 'Fable price ratio',
  o_opus: 'Opus is right',
  o_fable: 'Fable is right',
  w_write: 'Entry written after a strong solve',
  c_write: 'Calls to write an entry',
  e_right: 'Written entry is good',
  l_found: 'Escape found later',
  fix_on_found: 'Bad entry corrected when found',
  R_rework: 'Opus calls to rework an escape',
  delta_stale: 'Good entries going stale',
  episodes_per_period: 'Pieces of work per period',
  periods: 'Number of periods',
  retire_bad: 'Bad entries leaving the stock' // revision 3 (model-brief.md section 13, R12)
};
const problems = [];
Object.keys(NAMES).forEach(id => { if (!map.parameters.some(p => p.id === id)) problems.push('name for unknown parameter ' + id); });
map.parameters.forEach(p => { if (!NAMES[p.id]) problems.push('no name for ' + p.id); });
const seen = {};
Object.values(NAMES).forEach(n => { if (seen[n]) problems.push('two parameters share the name ' + n); seen[n] = 1; });

// Edit the text line by line so the file keeps its hand layout.
const fix = t => t
  .replace(/\bOne episode: a fix-sized piece of work\b/g, 'One piece of work: a fix-sized task')
  .replace(/\bAn episode\b/g, 'A piece of work').replace(/\ban episode\b/g, 'a piece of work')
  .replace(/\bEpisodes\b/g, 'Pieces of work').replace(/\bepisodes\b/g, 'pieces of work')
  .replace(/\bEpisode\b/g, 'Piece of work').replace(/\bepisode\b/g, 'piece of work');
const TEXT_VALUE = /("(?:unit|meaning|label|note|revision_note|limitations_note|update_note|title)": ")((?:[^"\\]|\\.)*)(")/g;
const NL = String.fromCharCode(10);
let raw = fs.readFileSync(FILE, 'utf8');
let changed = 0, named = 0;
raw = raw.split(NL).map(line => {
  let l = line;
  const m = /^(\s*\{ "id": "([A-Za-z_]+)", )("name": "[^"]*", )?("group": )/.exec(l);
  if (m && NAMES[m[2]]) { l = l.replace(m[0], m[1] + '"name": ' + JSON.stringify(NAMES[m[2]]) + ', ' + m[4]); named++; }
  // only inside a text value, never in an expression
  l = l.replace(TEXT_VALUE, (all, a, v, z) => { const n = fix(v); if (n !== v) changed++; return a + n + z; });
  return l;
}).join(NL);
const next = JSON.parse(raw);
next.parameters.forEach(p => {
  if (p.name !== NAMES[p.id]) problems.push('name not set for ' + p.id);
  if (p.meaning.trim().split(/\s+/).length >= 25) problems.push('meaning of 25 words or more: ' + p.id);
});
const sig = m => JSON.stringify({ b: m.branches.map(b => [b.id, b.from, b.to, b.chance, b.cost, b.calls, b.tier]), s: m.states.map(x => x.id), d: m.derived, st: m.stocks, ms: m.measures.map(x => [x.id, x.flow_on, x.flow_into]), p: m.parameters.map(p => [p.id, p.kind, p.group, !!p.pure_cost, !!p.run_setting]), a: m.attributes.map(a => a.values.map(v => [v.id, v.chance, v.expr])), c: m.class_chance, pol: m.policies });
const stripText = m => JSON.parse(JSON.stringify(m), (k, v) => (['label', 'note', 'update_note', 'name'].includes(k) ? undefined : v));
if (sig(stripText(map)) !== sig(stripText(next))) problems.push('the model itself changed; nothing written');
if (problems.length) { console.error(problems.join(NL)); process.exit(1); }
fs.writeFileSync(FILE, raw);
console.log('map.json: ' + named + ' parameter names set; ' + changed + ' text values reworded; model unchanged');

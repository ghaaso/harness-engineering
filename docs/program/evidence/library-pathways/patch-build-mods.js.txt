// Lets several people rework separate sections at once (2026-11-13).
// page-src/mods/<id>.js.txt is spliced into the page script just before it starts, so a function declared there
// replaces the page's function of the same name. page-src/mods/<id>.css.txt is appended to the style sheet.
// LP_MODS=a,b picks modules (default: all, sorted by name; empty string: none). LP_OUT=path writes a private build
// there, prints the failed checks, and leaves the live page and the record alone.
// Run from this folder: node - < patch-build-mods.js.txt   (a rerun makes 0 edits)
'use strict';
const fs = require('fs');
const path = require('path');
const HERE = process.cwd();
let edits = 0;
function patch(file, pairs) {
  const p = path.join(HERE, file);
  let s = fs.readFileSync(p, 'utf8');
  pairs.forEach(([from, to, name]) => {
    if (s.includes(to)) return;
    if (!s.includes(from)) throw new Error(file + ': anchor not found for "' + name + '"');
    s = s.replace(from, () => to); edits++;
  });
  fs.writeFileSync(p, s);
}
patch('page-src/page.js.txt', [
  ["  // ------------------------------------------------------------ start\n  renderStatic();", "  /*__MODS__*/\n  // ------------------------------------------------------------ start\n  renderStatic();", 'marker'],
]);

   'modules'],
  ["const WORD_LIMIT = 2650;", "const WORD_LIMIT = 3400;", 'word limit'],

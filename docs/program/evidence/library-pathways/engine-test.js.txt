// ----------------------------------------------------------------- expression whitelist

test('expressions are evaluated against a whitelist only', () => {
  const ok = E.compileExpr('1 - exp(-a / b) + safediv(a, 0)', new Set(['a', 'b']));
  near(ok.fn({ a: 2, b: 2 }), 1 - Math.exp(-1), 9, 'parser arithmetic');
  const bad = ['constructor', 'a.b', 'a[0]', '"x"', 'process', 'alert(1)', 'a; b', 'a ** b', 'this'];
  bad.forEach(src => {
    let threw = false;
    try { E.compileExpr(src, new Set(['a', 'b'])); } catch (e) { threw = true; }
    assert(threw, 'accepted: ' + src);
  });
  near(E.compileExpr('-2 * 3 + 4 / (1 + 1)', new Set()).fn({}), -4, 12, 'precedence');
});

// ----------------------------------------------------------------- conservation under 1,000 draws

test('conservation over 1,000 draws (untied and tied): branch chances out of every state lie in 0 to 1 and sum to 1, outcomes sum to 1 for every class and policy, class chances sum to 1; stocks never go negative; funnel stages never rise', () => {
  const random = E.rng(20261112);
  const shared = E.rng(77);
  let worst = 0, checks = 0, pathsChecked = 0;
  const wide = JSON.parse(JSON.stringify(table));
  // last part: every chance and fraction drawn across the whole of 0 to 1 (retire_bad included)
  map.parameters.forEach(p => { if (p.kind === 'chance' || p.kind === 'fraction') { wide.parameters[p.id].low = 0; wide.parameters[p.id].high = 1; } });
  for (let i = 0; i < 1000; i++) {
    const tb = i < 350 ? table : i < 700 ? real : wide;
    const tie = [0, 0.5, 1][i % 3];
    const p = E.drawParams(map, tb, random, { tie, levels: levelsFrom(shared) });
    const stocks = randomStocks(random);
    if (i % 50 === 0) { stocks.E_good = 0; stocks.E_bad = 0; }
    if (i % 50 === 25) { stocks.E_good = 0; }
    const r = E.conservation(map, p, { stocks });
    assert(r.ok, 'draw ' + i + ': ' + r.problems.slice(0, 3).join('; '));
    worst = Math.max(worst, r.maxError);
    checks += r.checks;
    if (i % 10 === 0) {
      const tp = E.timePath(map, p, { lean: true });
      tp.periods.forEach(row => {
        assert(row.stocks.E_good >= -1e-9 && row.stocks.E_bad >= -1e-9, 'draw ' + i + ' period ' + row.period + ': a stock went negative');
        map.funnels.forEach(f => {
          const v = f.monotone || f.stages;
          for (let k = 1; k < v.length; k++) assert(row.library.measures[v[k]] <= row.library.measures[v[k - 1]] + 1e-9, 'draw ' + i + ': funnel ' + f.id + ' rose at ' + v[k]);
        });
      });
      assert(tp.finalStocks.E_good >= -1e-9 && tp.finalStocks.E_bad >= -1e-9, 'draw ' + i + ': final stock negative');
      pathsChecked++;
    }
  }
  console.log('     ' + checks + ' sums checked over 1000 draws, largest error ' + worst.toExponential(2) + '; ' + pathsChecked + ' time paths checked for negative stocks and rising funnels');
});

// Case 1. top_first, a visible piece of work the cheap model would get wrong.
//   Opus right 0.7 -> passes -> EQUAL.
//   Opus wrong 0.3 -> caught 0.6 (0.18) -> Fable; slips 0.4 (0.12) -> escaped.
//   Fable right 0.6 -> EQUAL (0.108). Fable wrong 0.4 (0.072) -> caught 0.6 -> UNRESOLVED (0.0432);
//   slips 0.4 -> escaped (0.0288).
//   EQUAL = 0.7 + 0.108 = 0.808. UNRESOLVED = 0.0432. Escaped = 0.12 + 0.0288 = 0.1488,
//   half found: ESCAPED_FOUND = ESCAPED_SILENT = 0.0744.
//   Cost: Opus attempt 10 + Opus gate review 1 + Fable 0.18 * 10 = 1.8 + Fable gate review 0.18 * 1
//   + rework 0.0744 * 10 = 0.744.  Total 13.724.
//   By tier: cheap 0; Opus 10 + 1 + 0.18 + 0.744 = 11.924; Fable 1.8.
//   Calls: Opus 2 + 0.2 + 0.18 * 0.2 + 0.0744 * 2 = 2.3848; Fable 0.18. Total 2.5648.
test('hand case 1: top_first, visible, cheap model alone wrong', () => {
  const r = E.evaluateClass(map, withP(H), 'top_first', { kind: 'visible', need: 'cheap_wrong' });
  near(r.outcomes.EQUAL, 0.808, 6, 'EQUAL');
  near(r.outcomes.UNRESOLVED, 0.0432, 6, 'UNRESOLVED');
  near(r.outcomes.ESCAPED_FOUND, 0.0744, 6, 'ESCAPED_FOUND');
  near(r.outcomes.ESCAPED_SILENT, 0.0744, 6, 'ESCAPED_SILENT');
  near(r.cost, 13.724, 6, 'cost');
  near(r.costByTier.cheap, 0, 6, 'cheap cost');
  near(r.costByTier.opus, 11.924, 6, 'opus cost');
  near(r.costByTier.fable, 1.8, 6, 'fable cost');
  near(r.callsByTier.opus, 2.3848, 6, 'opus calls');
  near(r.callsByTier.fable, 0.18, 6, 'fable calls');
  near(r.calls, 2.5648, 6, 'calls');
});

// Case 3. library, a latent piece of work the cheap model would get wrong, the exact problem held,

// Case 4 (revision 3, R11 and R12). Stocks start at E_good 30 and E_bad 10 (a copy of the map with
// those starting stocks). k_fill 40, delta_stale 0.1, retire_bad 0.2, w_write 0 (nothing is written).
//   fill = 1 - exp(-(30 + 10) / 40) = 1 - exp(-1) = 0.632120558828558. (Revision 2: 1 - exp(-0.75).)
//   wrong_share = 10 / 40 = 0.25.
//   Next good = 30 - 0.1 * 30 = 27. Next bad = 10 - 0.2 * 10 + 0.1 * 30 = 11.
//   One period later: good 27 - 2.7 = 24.3; bad 11 - 2.2 + 2.7 = 11.5.
//   With retire_bad 1 instead: next bad = 10 - 10 + 3 = 3; then 3 - 3 + 2.7 = 2.7.
test('hand case 4: fill counts both stocks, and a share retire_bad of the bad stock leaves each period', () => {
  const m = withStocks(map, { E_good: 30, E_bad: 10 });
  const p = withP({ k_fill: 40, delta_stale: 0.1, retire_bad: 0.2, w_write: 0, periods: 3 });
  const tp = E.timePath(m, p);
  near(tp.periods[0].stockValues.fill, 1 - Math.exp(-1), 12, 'fill');
  near(tp.periods[0].stockValues.wrong_share, 0.25, 12, 'bad share');
  near(tp.periods[0].next.E_good, 27, 12, 'good after one period');
  near(tp.periods[0].next.E_bad, 11, 12, 'bad after one period');
  near(tp.periods[1].next.E_good, 24.3, 12, 'good after two periods');
  near(tp.periods[1].next.E_bad, 11.5, 12, 'bad after two periods');
  const tp1 = E.timePath(m, Object.assign({}, p, { retire_bad: 1 }));
  near(tp1.periods[0].next.E_bad, 3, 12, 'all bad retired, then the new stale ones');
  near(tp1.periods[1].next.E_bad, 2.7, 12, 'all bad retired again');
});

test('R18b: at tie 1 every library-side unknown sits at its own group\'s level, counted from its poor end; two groups differ; other unknowns keep their own draws; at tie 0.5 the quantile is half the level plus half the own draw', () => {
  let differ = 0, runs = 0;
  [table, real].forEach((tb, k) => {
    const rows = E.tableRows(tb);
    const r0 = E.rng(55 + k), r1 = E.rng(55 + k), r5 = E.rng(55 + k), lv = E.rng(99 + k);
    for (let i = 0; i < 200; i++) {
      const U = levelsFrom(lv);
      const d0 = E.drawParams(map, tb, r0);
      const d1 = E.drawParams(map, tb, r1, { tie: 1, levels: U });
      const d5 = E.drawParams(map, tb, r5, { tie: 0.5, levels: U });
      const seen = {};
      map.parameters.forEach(p => {
        const row = rows[p.id];
        if (p.run_setting || row.low === row.high) return;
        const log = row.dist === 'log';
        const pos = v => log ? (Math.log(v) - Math.log(row.low)) / (Math.log(row.high) - Math.log(row.low)) : (v - row.low) / (row.high - row.low);
        if (row.side !== 'library') {
          assert(d1[p.id] === d0[p.id] && d5[p.id] === d0[p.id], p.id + ' (other side) changed with the tie');
          return;
        }
        const fromPoor = v => row.good_end === 'high' ? pos(v) : 1 - pos(v);
        near(fromPoor(d1[p.id]), U[row.lib_group], 9, 'tie 1, ' + p.id);
        near(fromPoor(d5[p.id]), 0.5 * U[row.lib_group] + 0.5 * fromPoor(d0[p.id]), 9, 'tie 0.5, ' + p.id);
        seen[row.lib_group] = fromPoor(d1[p.id]);
      });
      runs++;
      if (Math.abs(seen.reach - seen.quality) > 1e-6 && Math.abs(seen.recurrence - seen.running_cost) > 1e-6) differ++;
    }
  });
  assert(differ === runs, 'two groups sat at the same level in ' + (runs - differ) + ' runs');
  // by hand, own draw 0.7 for every row, tie 0.5, levels: recurrence 0.2, reach 0.6, quality 0.9, running cost 0.1.
  //   k_fill (recurrence, good end low, 10 to 300): quantile from the poor (high) end 0.5 * 0.2 + 0.5 * (1 - 0.7) = 0.25
  //     -> 300 - 0.25 * 290 = 227.5.
  //   t_visible (reach, good end high, 0.2 to 1): 0.5 * 0.6 + 0.5 * 0.7 = 0.65 -> 0.2 + 0.65 * 0.8 = 0.72.
  //   e_right (quality, good end high, 0.35 to 0.95): 0.5 * 0.9 + 0.5 * 0.7 = 0.8 -> 0.35 + 0.8 * 0.6 = 0.83.
  //   c_lookup (running cost, good end low, 0.1 to 2): 0.5 * 0.1 + 0.5 * (1 - 0.7) = 0.2 -> 2 - 0.2 * 1.9 = 1.62.
  //   p_visible (other side, 0.1 to 0.8): its own draw, 0.1 + 0.7 * 0.7 = 0.59.
  const L = { recurrence: 0.2, reach: 0.6, quality: 0.9, running_cost: 0.1 };
  const one = E.drawParams(map, real, () => 0.7, { tie: 0.5, levels: L, added: () => 0.7 });
  near(one.k_fill, 227.5, 9, 'k_fill by hand');
  near(one.t_visible, 0.72, 12, 't_visible by hand');
  near(one.e_right, 0.83, 12, 'e_right by hand');
  near(one.c_lookup, 1.62, 12, 'c_lookup by hand');
  near(one.p_visible, 0.59, 12, 'p_visible by hand');
  // p_promote (quality, 0 to 1, from the added stream): 0.5 * 0.9 + 0.5 * 0.7 = 0.8
  near(one.p_promote, 0.8, 12, 'p_promote by hand');
  let threw = 0;
  try { E.drawParams(map, real, E.rng(1), { tie: 0.5 }); } catch (e) { threw++; }
  try { E.drawParams(map, real, E.rng(1), { tie: 0.5, levels: { recurrence: 0.1, reach: 0.2, quality: 0.3 } }); } catch (e) { threw++; }
  assert(threw === 2, 'a tie above 0 without a level for every group was accepted');
});

// R19 helpers: the score of a parameter set, from the library's cost and wrong rate at the last period
function libLast(p) { const tp = E.timePath(map, p); const r = tp.periods[tp.periods.length - 1].library; return { cost: r.cost, wrong: r.wrong }; }
function scorer(tb) {
  const mid = E.referenceParams(map, tb, 'middle'), all = E.referenceParams(map, tb, 'all_good');
  const m = libLast(mid);
  const S = x => ((m.cost - x.cost) / m.cost + (m.wrong - x.wrong) / m.wrong) / 2;
  return { mid, all, S, full: S(libLast(all)) };
}

test('R21: the promoted-check pathway is drawn in the map (a link from the good stock to the gate, via promo, with no lookup) and its measure is the class-by-class sum', () => {

  gates.forEach(id => assert(map.branches.find(b => b.id === id).chance === (/_caught$/.test(id) ? 'g_lib' : '1 - g_lib'), id));
  assert(map.derived.find(d => d.id === 'promo').expr === '(held_exact * k_promo_exact + held_similar * k_promo_similar) * (1 - wrong_share) * c_check_share * p_promote', 'promo as written');
  assert(l.loops.length >= 1 && l.loops.every(x => x.polarity === 'R' || x.polarity === 'B'), 'loops');
  R21_IDS.forEach(id => { const r = realRows[id]; assert(r.side === 'library' && r.good_end === 'high' && r.lib_group === 'quality' && r.tag === 'G', id); });
  assert(realRows.p_promote.low === 0 && realRows.p_promote.best === 0.5 && realRows.p_promote.high === 1, 'p_promote range');
  assert(realRows.k_promo_exact.low === 0.3 && realRows.k_promo_exact.best === 0.7 && realRows.k_promo_exact.high === 0.95, 'k_promo_exact range');
  assert(realRows.k_promo_similar.low === 0 && realRows.k_promo_similar.best === 0.05 && realRows.k_promo_similar.high === 0.3, 'k_promo_similar range');
  // no funnel stage counts the gate
  map.funnels.forEach(f => f.stages.forEach(id => { const m = map.measures.find(x => x.id === id); assert(!JSON.stringify(m).includes('L6_'), id + ' reaches the gate'); }));
  assert(map.promo_measure === 'fw_promo_caught', 'the promoted-check measure');
  ['use_promo_caught_first', 'use_promo_caught_all', 'use_promo_strength'].forEach(id => assert(map.usage_measures.includes(id), id + ' not a usage measure'));
  // a broken link is refused when the map compiles
  const bad = JSON.parse(JSON.stringify(map));
  bad.links[0].to_branches = ['l1_lookup'];
  let threw = false; try { E.compile(bad); } catch (e) { threw = /does not use promo/.test(e.message); }
  assert(threw, 'a link to a branch that does not use promo was accepted');
  // the measure, summed class by class here
  const stocks = { E_good: 120, E_bad: 40 }, p = realBest;
  const ev = E.evaluatePolicy(map, p, 'library', stocks);
  const ws = 40 / 160;
  let num = 0, den = 0, all1 = 0, allAny = 0, str = 0;
  ['visible', 'latent'].forEach(kind => ['cheap_ok', 'cheap_wrong'].forEach(need => ['exact', 'similar', 'composite', 'nothing'].forEach(held => ['yes', 'no'].forEach(d => {
    const c = E.evaluateClass(map, p, 'library', { kind, need, held, distractor: d }, stocks);
    const gt = kind === 'visible' ? p.g_tests_visible : p.g_tests_latent;
    const g = 1 - (1 - gt) * (1 - p.s_review * p.g_review);
    const promo = (held === 'exact' ? p.k_promo_exact : held === 'similar' ? p.k_promo_similar : 0) * (1 - ws) * p.c_check_share * p.p_promote;
    const extra = (1 - g) * promo;
    if (need === 'cheap_wrong') { num += c.chance * c.stateFlow.L6_wrong * extra; den += c.chance; }
    all1 += c.chance * c.stateFlow.L6_wrong * extra;
    allAny += c.chance * extra * (c.stateFlow.L6_wrong + c.stateFlow.L7_wrong + c.stateFlow.L8_wrong + c.stateFlow.L9_wrong);
    str += c.chance * promo;
  }))));
  near(ev.overall.measures.fw_promo_caught, 100 * num / den, 10, 'fw_promo_caught');
  near(ev.overall.measures.use_promo_caught_first, 100 * all1, 10, 'use_promo_caught_first');
  near(ev.overall.measures.use_promo_caught_all, 100 * allAny, 10, 'use_promo_caught_all');
  near(ev.overall.measures.use_promo_strength, 100 * str, 10, 'use_promo_strength');
  assert(ev.overall.measures.use_promo_caught_all >= ev.overall.measures.use_promo_caught_first, 'any gate below the first gate');
  assert(ev.overall.measures.fw_promo_caught > 0, 'nothing caught through promoted checks at the middle values');
});

// f07-4: the inferred saving becomes the caveat (the slot after the channel is null today)
const F074_OLD = "'M', 'seen', 'CLAUDE.md and the stored design documents', null],";
const F074_NEW = "'M', 'seen', 'CLAUDE.md and the stored design documents',\n    'The walk also says the reuse avoided at least two failed cloud builds. That is the walker\\'s inference, not something seen.'],";

const LIT = /'(?:[^'\\\n]|\\.)*'|"(?:[^"\\\n]|\\.)*"/g;
const unq = lit => (lit[0] === "'" ? lit.slice(1, -1).replace(/\\'/g, "'").replace(/\\\\/g, '\\') : JSON.parse(lit));
const req = (val, q) => q + (q === "'" ? val.replace(/\\/g, '\\\\').replace(/'/g, "\\'") : JSON.stringify(val).slice(1, -1)) + q;
let applied = 0;
const problems = [];
EDITS.forEach(([oldT, newT]) => {
  let hit = 0, done = 0;
  src = src.replace(LIT, lit => {
    const v = unq(lit);
    if (!v.includes(oldT)) { if (v.includes(newT)) done++; return lit; }
    hit++;
    return req(v.split(oldT).join(newT), lit[0]);
  });
  if (hit) applied++;
  if (!hit && !done) problems.push('not found: ' + oldT.slice(0, 70));
  if (hit > 1) problems.push('found ' + hit + ' times: ' + oldT.slice(0, 70));
});
if (src.includes(F074_OLD)) { src = src.replace(F074_OLD, F074_NEW); applied++; }
else if (!src.includes("That is the walker\\'s inference, not something seen.")) problems.push('f07-4 slot not found');
if (problems.length) { console.error(problems.join('\n')); process.exit(1); }
fs.writeFileSync(FILE, src);
console.log('build-limitations.js.txt: ' + applied + ' edits this run, ' + (EDITS.length + 1) + ' in all');

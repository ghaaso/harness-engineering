function strings(v, where, out) {
  if (typeof v === 'string') out.push([where, v]);
  else if (Array.isArray(v)) v.forEach((x, i) => strings(x, where + '[' + i + ']', out));
  else if (v && typeof v === 'object') Object.keys(v).forEach(k => strings(v[k], where + '.' + k, out));
  return out;
}

function pruned() {
  const map = JSON.parse(JSON.stringify(MAP));
  ['source', 'note', 'revision_note', 'revision_3_note', 'revision_4_note', 'todays_notes', 'usage_note', 'funnel_note', 'lib_groups_note', 'reference_settings_note', 'variants_note', 'links_note', 'limitations_note'].forEach(k => { delete map[k]; });
  ['square', 'weight', 'shared_table', 'ensemble_ties'].forEach(k => { if (map[k]) delete map[k].note; });
  delete map.stocks.note; delete map.stocks.update_note; map.stocks.update.forEach(u => { delete u.same_as; });
  map.lib_groups.forEach(g => { delete g.note; });
  map.parameters.forEach(p => { delete p.note; });
  const params = JSON.parse(JSON.stringify(TABLE));
  ['note', 'assumptions', 'source_key', 'map', 'title'].forEach(k => { delete params[k]; });
  const lims = JSON.parse(JSON.stringify(LIMS));
  ['title', 'date', 'status', 'how_to_read', 'words', 'summary'].forEach(k => { delete lims[k]; });
  lims.counting = { breaks_read: LIMS.counting.breaks_read };
  lims.items.forEach(it => { delete it.absorbed; delete it.origin; delete it.merged_from; delete it.real.tag; delete it.real.family_ids; it.contradicted.forEach(c => { delete c.cases; }); });
  const text = JSON.parse(JSON.stringify(TX));
  delete text.note; delete text.layout.note; delete text.funnel_steps.note;
  return { map: map, params: params, lims: lims, cases: C.packCases(CASES), text: text, build: { date: TX.date } };
}

function visibleWords(h) {
  h = h.replace(/<!--[\s\S]*?-->/g, '').replace(/<script\b[\s\S]*?<\/script>/g, '').replace(/<style\b[\s\S]*?<\/style>/g, '');
  const VOID = new Set(['br', 'input', 'img', 'meta', 'link', 'hr', 'wbr', 'source', 'col', 'area', 'base']);
  const re = /<(\/?)([a-zA-Z][\w-]*)((?:[^>"']|"[^"]*"|'[^']*')*)>|([^<]+)/g;
  const stack = [], text = [];
  let m;
  while ((m = re.exec(h))) {
    if (m[4] !== undefined) { const top = stack[stack.length - 1]; if (!top || !top.hidden) text.push(m[4]); continue; }
    const close = m[1] === '/', tag = m[2].toLowerCase(), attrs = m[3] || '';
    if (close) { for (let i = stack.length - 1; i >= 0; i--) if (stack[i].tag === tag) { stack.length = i; break; } continue; }
    const parent = stack[stack.length - 1];
    let hidden = parent ? parent.hidden : false;
    if (parent && parent.closed) hidden = tag === 'summary' ? parent.outer : true;
    if (/\shidden(\s|=|\/|$)/.test(attrs)) hidden = true;
    const frame = { tag: tag, hidden: hidden };
    if (tag === 'details' && !/\sopen(\s|=|\/|$)/.test(attrs)) { frame.closed = true; frame.outer = hidden; }
    if (VOID.has(tag) || /\/\s*$/.test(attrs)) continue;
    stack.push(frame);
  }
  return text.join(' ').replace(/&nbsp;/g, ' ').replace(/&[a-z]+;|&#\d+;/g, ' ').split(/\s+/).filter(t => /[A-Za-z]/.test(t));
}

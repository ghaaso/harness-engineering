/* River prototype of the map section. Every number is computed here by PathwayEngine from the inlined data. */
(function () {
  'use strict';
  var D = JSON.parse(document.getElementById('rv-data').textContent);
  var E = window.PathwayEngine, C = window.PathwayCore;
  var MAP = D.map, TABLE = D.params, LIMS = D.lims, CASES = C.unpackCases(D.cases), TX = D.text;
  var CM = E.compile(MAP);
  var cur = E.paramsFromTable(TABLE, 'best');
  var PATH = E.timePath(MAP, cur);
  var NPER = PATH.periods.length;
  var REACH = C.paramReach(E, MAP);
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var esc = function (s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); };

  var STATE = {}, BR = {}, STEP = {}, POL = {}, TIER = {}, PMETA = {}, ATTR = {}, OUTC = {};
  MAP.states.forEach(function (s, i) { s.i = i; STATE[s.id] = s; });
  MAP.branches.forEach(function (b, i) { b.i = i; BR[b.id] = b; });
  MAP.steps.forEach(function (s, i) { s.i = i; STEP[s.id] = s; });
  MAP.policies.forEach(function (p) { POL[p.id] = p; });
  MAP.tiers.forEach(function (t) { TIER[t.id] = t; });
  MAP.parameters.forEach(function (p) { PMETA[p.id] = p; });
  MAP.attributes.forEach(function (a) { ATTR[a.id] = a; });
  MAP.outcomes.forEach(function (o) { OUTC[o.id] = o; });
  var WAYS = MAP.policies.map(function (p) { return p.id; });
  var WNAME = TX.policy_names, WSHORT = TX.policy_short;
  var STAGES = MAP.steps.filter(function (s) { return s.id !== 'outcome'; });
  var DSTEPS = Object.keys(LIMS.steps).map(Number).sort(function (a, b) { return a - b; });
  var DNAME = function (k) { return LIMS.steps[k]; };
  var DSHORT = function (k) { return String(LIMS.steps[k]).split(':')[0]; };
  var FAMNAME = {}; CASES.families.forEach(function (f) { FAMNAME[f.id] = f.name; });
  var LIM = {}; LIMS.items.forEach(function (it) { LIM[it.id] = it; });
  var LIMS_AT = {}, ALSO_AT = {}, BROKE_AT = {};
  DSTEPS.forEach(function (k) {
    LIMS_AT[k] = LIMS.items.filter(function (it) { return it.main_step === k; });
    ALSO_AT[k] = LIMS.items.filter(function (it) { return it.main_step !== k && (it.steps || []).indexOf(k) >= 0; });
    BROKE_AT[k] = CASES.cases.filter(function (c) { return c.steps.some(function (s) { return s.step === k && s.status === 'breaks'; }); });
  });
  var DSTEPS_OF_STAGE = {};
  Object.keys(TX.design_steps).forEach(function (k) { TX.design_steps[k].forEach(function (st) { (DSTEPS_OF_STAGE[st] = DSTEPS_OF_STAGE[st] || []).push(+k); }); });
  var FLAGNAME = {};
  MAP.attributes.forEach(function (a) { a.values.forEach(function (v) { Object.keys(v.flags).forEach(function (f) { if (v.flags[f] === 1 && !FLAGNAME[f]) FLAGNAME[f] = a.label + ' is "' + v.label + '"'; }); }); });
  var DERNAME = {}; MAP.derived.forEach(function (d) { DERNAME[d.id] = d.label; });
  (MAP.stocks.values || []).forEach(function (v) { DERNAME[v.id] = v.label; });

  // ------------------------------------------------------------------ words of the page
  var W = {
    lede: 'Start with 100 pieces of work on the left. Each band carries some of them through the eleven stages, left to right. Its width is how many it carries, and the number on it says so. They end in one of four outcomes on the right. Under each river: what the stages cost.',
    way_says: 'Picks the river. "All three" stacks the three ways on the same columns, so a difference is a difference in width at one place.',
    work_says: 'All work mixes every kind of work in its real share. One kind re-runs the river for 100 pieces of that kind only.',
    period_says: 'The library as it stands at the start of that period (30 days each). It starts empty. Only the library river changes.',
    attr_says: {
      kind: 'Visible: an error or a failing check shows the problem. Latent: nothing signals that a choice matters.',
      need: 'What the cheap model would do with no entry, working as it does today with today’s notes.',
      held: 'What is stored for this piece. It applies only to the library river.',
      distractor: 'Whether a misleading entry could come back on top. It applies only to the library river.'
    },
    attr_name: { kind: 'Kind of problem', need: 'The cheap model alone would be', held: 'What the library holds', distractor: 'A look-alike can be retrieved' },
    short_val: { visible: 'Visible', latent: 'Latent', cheap_ok: 'Right', cheap_wrong: 'Wrong', exact: 'The same problem', similar: 'One entry, adapted', composite: 'Several parts', nothing: 'Nothing', yes: 'Yes', no: 'No' }
  };

  // ------------------------------------------------------------------ number formats
  function fc(v) {
    if (v === null || v === undefined || !isFinite(v)) return 'none';
    if (v === 0) return '0';
    var a = Math.abs(v);
    if (a >= 99.95) return v.toFixed(0);
    if (a >= 1) return v.toFixed(1);
    if (a >= 0.01) return v.toFixed(2);
    return v.toPrecision(1);
  }
  function fk(v) {
    if (!isFinite(v)) return 'none';
    if (v === 0) return '0';
    var a = Math.abs(v);
    if (a >= 100) return Math.round(v).toLocaleString('en');
    if (a >= 1) return v.toFixed(1);
    if (a >= 0.01) return v.toFixed(2);
    return v.toPrecision(1);
  }
  function fpct(v) { var x = 100 * v; return (x !== 0 && Math.abs(x) < 0.1 ? x.toFixed(2) : x.toFixed(1)) + '%'; }
  function tag(t) { return '<i class="tag ' + t + '" title="' + esc(t + ': ' + TX.tags[t]) + '">' + t + '</i>'; }

  // ------------------------------------------------------------------ state
  var ST = {
    way: 'library', compare: false, scope: 'all',
    cls: { kind: 'visible', need: 'cheap_wrong', held: 'similar', distractor: 'no' },
    period: NPER, sel: null, route: null, stepIdx: null, sortBy: 'chance', routesWay: 'library', open: {}
  };
  function classFor(way) {
    var c = Object.assign({}, ST.cls), fx = POL[way].fixed || {};
    Object.keys(fx).forEach(function (k) { c[k] = fx[k]; });
    return c;
  }
  function matches(values, want) { return Object.keys(want).every(function (k) { return values[k] === want[k]; }); }
  function stocksAt(way) { return way === 'library' ? PATH.periods[ST.period - 1].stocks : {}; }

  // ------------------------------------------------------------------ the engine, per way
  var DATA = {};
  function computeWay(way) {
    var stocks = stocksAt(way);
    var pol = CM.policies[CM.policyIndex[way]];
    var r = { way: way, stocks: stocks, sf: {}, bf: {}, carried: {}, unit: {}, varies: {}, stage: {}, check: 0 };
    MAP.branches.forEach(function (b) { r.carried[b.id] = 0; r.unit[b.id] = null; });
    var combos = pol.combos;
    if (ST.scope === 'kind') {
      var cv = classFor(way);
      var ec = E.evaluateClass(MAP, cur, way, cv, stocks);
      r.sum = ec; r.share = ec.chance; r.cv = cv;
      r.sf = ec.stateFlow; r.bf = ec.branchFlow;
      combos = combos.filter(function (c) { return matches(c.values, cv); });
    } else {
      var ev = E.evaluatePolicy(MAP, cur, way, stocks, { classes: false });
      r.sum = ev.overall; r.share = 1; r.sf = ev.stateFlow; r.bf = ev.branchFlow;
    }
    combos.forEach(function (combo) {
      var one = E.evaluateClass(MAP, cur, way, combo.values, stocks);
      var w = ST.scope === 'kind' ? 1 : one.chance;
      if (!w) return;
      var bv = C.branchValues(E, MAP, cur, way, combo.values, stocks);
      MAP.branches.forEach(function (b) {
        var k = bv[b.id].cost;
        r.carried[b.id] += w * one.branchFlow[b.id] * k;
        if (r.unit[b.id] === null) r.unit[b.id] = k;
        else if (Math.abs(r.unit[b.id] - k) > 1e-12) r.varies[b.id] = true;
      });
    });
    STAGES.forEach(function (s) { r.stage[s.id] = { total: 0 }; MAP.tiers.forEach(function (t) { r.stage[s.id][t.id] = 0; }); });
    var tot = 0;
    MAP.branches.forEach(function (b) {
      var v = 100 * r.carried[b.id];
      tot += r.carried[b.id];
      if (b.tier && v) { var s = r.stage[STATE[b.from].step]; s[b.tier] += v; s.total += v; }
    });
    r.check = Math.abs(tot - r.sum.cost);
    if (r.check > 1e-9 && window.console) console.warn('cost by branch differs from the engine cost under ' + way + ': ' + r.check);
    r.reach = reachFrom(POL[way].start);
    return r;
  }
  function reachFrom(start) {
    var seen = {}; seen[start] = true; var q = [start];
    while (q.length) { var s = q.shift(); MAP.branches.forEach(function (b) { if (b.from === s && !seen[b.to]) { seen[b.to] = true; q.push(b.to); } }); }
    return seen;
  }
  function computeAll() { WAYS.forEach(function (w) { DATA[w] = computeWay(w); }); }

  // Scope of one class, as the engine builds it (for the plain reasons a branch is empty).
  function scopeOf(way, values, stocks) {
    var prep = E.prepareParams(CM, cur);
    var pol = CM.policies[CM.policyIndex[way]];
    var combo = pol.combos.filter(function (c) { return matches(c.values, values); })[0];
    var s = Object.assign({}, prep.values);
    CM.stockIds.forEach(function (id) { s[id] = stocks && typeof stocks[id] === 'number' ? stocks[id] : 0; });
    CM.stockValues.forEach(function (v) { s[v.id] = v.fn(s); });
    Object.assign(s, pol.flags, combo.flags);
    CM.derived.forEach(function (d) { s[d.id] = d.fn(s); });
    return s;
  }
  function idName(id, v) {
    if (id === 'capture') return v === 0 ? 'this way writes nothing to a library' : 'this way writes to the library';
    if (FLAGNAME[id]) return v === 0 ? 'it is not the case that ' + FLAGNAME[id].charAt(0).toLowerCase() + FLAGNAME[id].slice(1) : FLAGNAME[id];
    if (DERNAME[id]) return DERNAME[id].charAt(0).toLowerCase() + DERNAME[id].slice(1) + ' is ' + fpct(v);
    if (PMETA[id]) return PMETA[id].name.charAt(0).toLowerCase() + PMETA[id].name.slice(1) + ' is ' + (v <= 1 ? fpct(v) : fk(v));
    return id + ' is ' + v;
  }
  function whyBranch(way, bid) {
    var d = DATA[way], b = BR[bid];
    if (d.bf[bid] > 0) return null;
    if (!(d.sf[b.from] > 0)) return 'Nothing reaches “' + STATE[b.from].label + '” under this choice.';
    var cb = CM.branches[CM.branchIndex[bid]];
    var pol = CM.policies[CM.policyIndex[way]];
    var combos = ST.scope === 'kind' ? [d.cv] : pol.combos.map(function (c) { return c.values; });
    var found = null;
    combos.forEach(function (vals) {
      var s = scopeOf(way, vals, d.stocks);
      var z = cb.chance.ids.filter(function (id) { return s[id] === 0 || (s[id] === 1 && new RegExp('1\\s*-\\s*' + id + '\\b').test(cb.chance.src)); });
      var names = z.map(function (id) { return idName(id, s[id]); });
      if (found === null) found = names; else found = found.filter(function (n) { return names.indexOf(n) >= 0; });
    });
    var head = 'Work reaches “' + STATE[b.from].label + '”, but this branch has no chance here.';
    return head + (found && found.length ? ' Reason: ' + found.join('; ') + '.' : '') + ' Its chance is <code>' + esc(cb.chance.src) + '</code>.';
  }
  function whyState(way, sid) {
    var d = DATA[way];
    if (d.sf[sid] > 0) return null;
    if (!d.reach[sid]) return WNAME[way] + ' starts at “' + STATE[POL[way].start].label + '”. Work in this way never passes here.';
    var ins = MAP.branches.filter(function (b) { return b.to === sid; });
    return 'Every branch into it carries nothing under this choice: ' + ins.map(function (b) { return '“' + b.label + '” from “' + STATE[b.from].label + '”'; }).join('; ') + '. Open one of them for its reason.';
  }

  // ------------------------------------------------------------------ text measuring
  var FONT = getComputedStyle(document.body).fontFamily || 'Segoe UI, system-ui, sans-serif';
  var CAN = document.createElement('canvas').getContext('2d');
  function tw(s, f) { CAN.font = f; return CAN.measureText(s).width; }
  var FS = 13, LH = 16;
  var F_REG = FS + 'px ' + FONT, F_BOLD = '700 ' + FS + 'px ' + FONT;
  function wrapText(str, w1, w) {
    var words = String(str).split(/\s+/), lines = [], line = '', lim = w1;
    words.forEach(function (wd) {
      var t = line ? line + ' ' + wd : wd;
      if (line && tw(t, F_REG) > lim) { lines.push(line); line = wd; lim = w; } else line = t;
    });
    if (line) lines.push(line);
    return lines;
  }

  // ------------------------------------------------------------------ columns: stage, then order inside a stage
  var COLS = [], COL_OF = {}, STAGE_COLS = {};
  (function () {
    var local = {}; MAP.states.forEach(function (s) { local[s.id] = 0; });
    for (var it = 0; it < 50; it++) {
      MAP.branches.forEach(function (b) {
        if (STATE[b.from].step === STATE[b.to].step) local[b.to] = Math.max(local[b.to], local[b.from] + 1);
      });
    }
    var keys = {};
    MAP.states.forEach(function (s) { keys[STEP[s.step].i + ':' + local[s.id]] = [STEP[s.step].i, local[s.id], s.step]; });
    COLS = Object.keys(keys).map(function (k) { return keys[k]; }).sort(function (a, b) { return a[0] - b[0] || a[1] - b[1]; }).map(function (k) { return { stage: k[2], local: k[1] }; });
    MAP.states.forEach(function (s) { COL_OF[s.id] = COLS.findIndex(function (c) { return c.stage === s.step && c.local === local[s.id]; }); });
    COLS.forEach(function (c, i) { var r = STAGE_COLS[c.stage] = STAGE_COLS[c.stage] || [i, i]; r[0] = Math.min(r[0], i); r[1] = Math.max(r[1], i); });
  })();
  var NC = COLS.length;

  // ------------------------------------------------------------------ layout of the river (shared by the three ways)
  var BW = 10, OBW = 18, MINB = 6, MIND = 2;
  function maxFlows() {
    var fs = {}, fb = {};
    MAP.states.forEach(function (s) { fs[s.id] = Math.max.apply(null, WAYS.map(function (w) { return 100 * DATA[w].sf[s.id]; })); });
    MAP.branches.forEach(function (b) { fb[b.id] = Math.max.apply(null, WAYS.map(function (w) { return 100 * DATA[w].bf[b.id]; })); });
    return { fs: fs, fb: fb };
  }
  function layout(Wd, targetH, labels) {
    var mf = maxFlows();
    var outW = Math.max(150, Math.min(210, Wd * 0.12));
    var cw = (Wd - outW) / (NC - 1);
    var colX = function (c) { return c < NC - 1 ? c * cw : Wd - outW; };
    var cntW = tw('00.00 ', F_BOLD);
    var nodes = {}, cols = []; for (var c = 0; c < NC; c++) cols.push([]);
    MAP.states.forEach(function (s) {
      var c = COL_OF[s.id], out = !!s.outcome;
      var avail = out ? outW - OBW - 10 : cw - 8;
      var lines = labels ? wrapText(s.label, avail - cntW, avail) : [];
      var n = { t: 'n', id: s.id, col: c, out: out, lines: lines, x: colX(c), f: mf.fs[s.id], ord: s.i };
      n.labelH = out ? 0 : (labels ? lines.length * LH + 4 : LH + 2);
      nodes[s.id] = n; cols[c].push(n);
    });
    var dums = {};
    MAP.branches.forEach(function (b) {
      var a = COL_OF[b.from], z = COL_OF[b.to];
      dums[b.id] = [];
      for (var c = a + 1; c < z; c++) { var d = { t: 'd', br: b.id, col: c, x: colX(c), f: mf.fb[b.id] }; dums[b.id].push(d); cols[c].push(d); }
    });
    // vertical scale: pieces of work to pixels, so the tallest column fits the target height
    var GN = 10, GD = 3;
    function fixedOf(items) {
      return items.reduce(function (a, it) {
        if (it.t === 'd') return a + (it.f > 0 ? 0 : MIND) + GD;
        if (it.out) return a + GN + Math.max(0, (labels ? 2 : 1) * LH - 0);
        return a + it.labelH + (it.f > 0 ? 0 : MINB) + GN;
      }, 0);
    }
    var k = 6;
    cols.forEach(function (items) {
      var fl = items.reduce(function (a, it) { return a + it.f; }, 0);
      if (fl > 0) k = Math.min(k, (targetH - fixedOf(items)) / fl);
    });
    k = Math.max(labels ? 1.6 : 1.1, k);
    function hOf(it) {
      if (it.t === 'd') return Math.max(MIND, k * it.f);
      var area = Math.max(MINB, k * it.f);
      it.area = area;
      if (it.out) return Math.max(area, (labels ? 2 : 1) * LH + 2);
      return it.labelH + area;
    }
    function centerOf(it) { return it.t === 'd' ? it.y + it.h / 2 : (it.out ? it.y + it.area / 2 : it.y + it.labelH + it.area / 2); }
    // first pass: nodes only, in map order
    cols.forEach(function (items) {
      var y = 0;
      items.filter(function (it) { return it.t === 'n'; }).forEach(function (n) { n.h = hOf(n); n.y = y; y += n.h + GN; });
    });
    for (var pass = 0; pass < 5; pass++) {
      MAP.branches.forEach(function (b) {
        var ds = dums[b.id]; if (!ds.length) return;
        var A = nodes[b.from], Z = nodes[b.to], ya = centerOf(A), yz = centerOf(Z);
        ds.forEach(function (d) { var t = (d.col - A.col) / (Z.col - A.col); d.est = ya + (yz - ya) * t; });
      });
      cols.forEach(function (items) {
        var ns = items.filter(function (it) { return it.t === 'n'; }).sort(function (a, b) { return a.ord - b.ord; });
        var ds = items.filter(function (it) { return it.t === 'd'; }).sort(function (a, b) { return a.est - b.est || BR[a.br].i - BR[b.br].i; });
        var seq = [], j = 0;
        ns.forEach(function (n) {
          var nc = centerOf(n);
          while (j < ds.length && ds[j].est < nc) { seq.push(ds[j]); j++; }
          seq.push(n);
        });
        while (j < ds.length) { seq.push(ds[j]); j++; }
        var y = 0;
        seq.forEach(function (it, i) {
          it.h = hOf(it);
          var prev = seq[i - 1];
          if (prev) y += (prev.t === 'n' || it.t === 'n') ? GN : GD;
          it.y = y; y += it.h;
        });
        items.order = seq; items.H = y;
      });
    }
    var H = Math.max.apply(null, cols.map(function (it) { return it.H || 0; })) + 6;
    return { W: Wd, H: H, k: k, cw: cw, outW: outW, colX: colX, nodes: nodes, dums: dums, cols: cols, labels: labels, centerOf: centerOf };
  }

  // Band geometry for one way on the shared layout: each band's top edge at every point, and its thickness.
  function bands(L, way) {
    var d = DATA[way], k = L.k, geo = {};
    function refOut(b) { var ds = L.dums[b.id]; return ds.length ? ds[0].y + ds[0].h / 2 : L.centerOf(L.nodes[b.to]); }
    function refIn(b) { var ds = L.dums[b.id]; return ds.length ? ds[ds.length - 1].y + ds[ds.length - 1].h / 2 : L.centerOf(L.nodes[b.from]); }
    var outs = {}, ins = {};
    MAP.branches.forEach(function (b) { (outs[b.from] = outs[b.from] || []).push(b); (ins[b.to] = ins[b.to] || []).push(b); });
    var srcY = {}, tgtY = {};
    Object.keys(outs).forEach(function (sid) {
      var n = L.nodes[sid], top = n.out ? n.y : n.y + n.labelH, off = 0;
      outs[sid].slice().sort(function (a, b) { return refOut(a) - refOut(b) || a.i - b.i; }).forEach(function (b) { srcY[b.id] = top + off; off += k * 100 * d.bf[b.id]; });
    });
    Object.keys(ins).forEach(function (sid) {
      var n = L.nodes[sid], top = n.out ? n.y : n.y + n.labelH, off = 0;
      ins[sid].slice().sort(function (a, b) { return refIn(a) - refIn(b) || a.i - b.i; }).forEach(function (b) { tgtY[b.id] = top + off; off += k * 100 * d.bf[b.id]; });
    });
    MAP.branches.forEach(function (b) {
      var v = 100 * d.bf[b.id], th = k * v;
      var A = L.nodes[b.from], Z = L.nodes[b.to];
      var pts = [{ xL: A.x + BW, xR: A.x + BW, y: srcY[b.id] }];
      L.dums[b.id].forEach(function (dm) { var y = dm.y + dm.h / 2 - th / 2; pts.push({ xL: dm.x, xR: dm.x + BW, y: y, dum: dm }); });
      pts.push({ xL: Z.x, xR: Z.x, y: tgtY[b.id] });
      geo[b.id] = { th: th, v: v, pts: pts };
    });
    return geo;
  }
  function bandPath(g, th) {
    var p = g.pts, d = 'M' + p[0].xR.toFixed(1) + ',' + p[0].y.toFixed(2), i, a, b, mx;
    for (i = 1; i < p.length; i++) {
      a = p[i - 1]; b = p[i]; mx = (a.xR + b.xL) / 2;
      d += 'C' + mx.toFixed(1) + ',' + a.y.toFixed(2) + ' ' + mx.toFixed(1) + ',' + b.y.toFixed(2) + ' ' + b.xL.toFixed(1) + ',' + b.y.toFixed(2);
      if (b.xR > b.xL) d += 'L' + b.xR.toFixed(1) + ',' + b.y.toFixed(2);
    }
    if (th <= 0) return d;
    var l = p[p.length - 1];
    d += 'L' + l.xR.toFixed(1) + ',' + (l.y + th).toFixed(2);
    for (i = p.length - 1; i > 0; i--) {
      a = p[i]; b = p[i - 1]; mx = (a.xL + b.xR) / 2;
      if (a.xR > a.xL) d += 'L' + a.xL.toFixed(1) + ',' + (a.y + th).toFixed(2);
      d += 'C' + mx.toFixed(1) + ',' + (a.y + th).toFixed(2) + ' ' + mx.toFixed(1) + ',' + (b.y + th).toFixed(2) + ' ' + b.xR.toFixed(1) + ',' + (b.y + th).toFixed(2);
    }
    return d + 'Z';
  }
  function centerPath(g) {
    var h = g.th / 2;
    return bandPath({ pts: g.pts.map(function (p) { return { xL: p.xL, xR: p.xR, y: p.y + h }; }) }, 0);
  }

  // ------------------------------------------------------------------ shared SVG patterns (one visual language)
  (function () {

  // ------------------------------------------------------------------ one river as SVG
  function riverSvg(L, way, opts) {
    var d = DATA[way], geo = bands(L, way), k = L.k;
    var sel = ST.sel, route = opts.route;
    var hiB = {}, hiS = {};
    if (sel && sel.t === 'state') { hiS[sel.id] = 1; MAP.branches.forEach(function (b) { if (b.from === sel.id || b.to === sel.id) hiB[b.id] = 1; }); }
    if (sel && sel.t === 'branch') { hiB[sel.id] = 1; hiS[BR[sel.id].from] = 1; hiS[BR[sel.id].to] = 1; }
    if (sel && sel.t === 'step') { (TX.design_steps[sel.id] || []).forEach(function (stg) { MAP.states.forEach(function (s) { if (s.step === stg) hiS[s.id] = 1; }); MAP.branches.forEach(function (b) { if (STATE[b.from].step === stg) hiB[b.id] = 1; }); }); }
    var anyHi = Object.keys(hiB).length + Object.keys(hiS).length > 0 && !(sel && sel.t === 'step');
    var dimAll = !!route || anyHi;
    var s = '<svg class="rvsvg" width="' + L.W + '" height="' + L.H + '" viewBox="0 0 ' + L.W + ' ' + L.H + '" role="group" aria-label="' + esc('River for ' + WNAME[way]) + '">';
    // column guides for the stages a selected design step covers
    if (sel && sel.t === 'step') {
      (TX.design_steps[sel.id] || []).forEach(function (stg) {
        var r = STAGE_COLS[stg]; if (!r) return;
        var x0 = L.colX(r[0]) - 6, x1 = (r[1] + 1 < NC ? L.colX(r[1] + 1) : L.W) - 6;
        s += '<rect class="stepcol" x="' + x0 + '" y="0" width="' + (x1 - x0) + '" height="' + L.H + '"/>';
      });
    }

    s += '<g class="bands">';
    MAP.branches.forEach(function (b) {
      var g = geo[b.id], cls = 'band' + (g.v > 0 ? '' : ' zero') + (hiB[b.id] ? ' hi' : (dimAll ? ' dim' : ''));
      var lab = b.label + ': ' + fc(g.v) + ' of 100, from “' + STATE[b.from].label + '” to “' + STATE[b.to].label + '”';
      var dd = g.v > 0 ? bandPath(g, Math.max(g.th, 0.8)) : centerPath(g);
      s += '<path class="' + cls + '" d="' + dd + '" data-br="' + b.id + '" data-way="' + way + '" tabindex="0" role="button" aria-label="' + esc(lab) + '"/>';
    });
    s += '</g>';
    // the selected route as a thread inside the river
    if (route) {
      var upto = ST.stepIdx === null ? route.branches.length - 1 : ST.stepIdx;
      var tv = 100 * route.chance, th = Math.max(2.5, k * tv);
      s += '<g class="thread" aria-hidden="true">';
      route.branches.forEach(function (bid, i) {
        if (i > upto) return;
        var g = geo[bid]; if (!g) return;
        var gg = { pts: g.pts };
        s += '<path class="thr' + (i === upto && ST.stepIdx !== null ? ' now' : '') + '" d="' + bandPath(gg, Math.min(th, Math.max(g.th, 2.5))) + '"/>';
      });
      s += '</g>';
    }
    // states
    var obst = [], counts = [], inline = {};
    s += '<g class="nodes">';
    MAP.states.forEach(function (st) {
      var n = L.nodes[st.id], v = 100 * d.sf[st.id], zero = !(v > 0);
      var top = n.out ? n.y : n.y + n.labelH, bh = Math.max(zero ? MINB : 1.2, k * v);
      var cls = 'node' + (zero ? ' zero' : '') + (hiS[st.id] ? ' hi' : (dimAll && !route ? ' dim' : '')) + (n.out ? ' out' : '');
      var start = POL[way].start === st.id;
      var lab = st.label + ': ' + fc(v) + ' of 100' + (zero ? ', no work here under this choice' : '') + (start ? ', where this way starts' : '');
      s += '<g class="' + cls + '" data-st="' + st.id + '" data-way="' + way + '" tabindex="0" role="button" aria-label="' + esc(lab) + '">';
      if (n.out) {
        s += '<rect class="blk ' + OUT_FILL[st.id] + (zero ? ' empty' : '') + '" x="' + n.x + '" y="' + top + '" width="' + OBW + '" height="' + bh.toFixed(2) + '"/>';
        var lx = n.x + OBW + 6, ly = top + Math.min(bh, 40) / 2 - (L.labels ? LH / 2 : 0) + 4;
        ly = Math.max(top + 12, ly);
        s += '<text class="olab" x="' + lx + '" y="' + ly.toFixed(1) + '"><tspan class="cnt">' + fc(v) + '</tspan>' + (L.labels ? '' : ' ' + esc(OUTC[st.id].label)) + '</text>';

        if (L.labels) {
          n.lines.forEach(function (ln, j) {

          });
          var mw = Math.max.apply(null, n.lines.map(function (ln, j) { return tw(ln, F_REG) + (j === 0 ? tw(fc(v) + ' ', F_BOLD) : 0); }));
          obst.push({ x: n.x - 1, y: n.y, w: mw + 4, h: n.lines.length * LH + 2 });
        } else {

      obst.push({ x: n.x - 2, y: top - 1, w: (n.out ? OBW : BW) + 4, h: bh + 2 });
    });
    s += '</g>';

    var lab = '<g class="blabs" aria-hidden="true">';
    function free(r) { for (var i = 0; i < obst.length; i++) { var o = obst[i]; if (r.x < o.x + o.w && r.x + r.w > o.x && r.y < o.y + o.h && r.y + r.h > o.y) return false; } return r.x >= 0 && r.x + r.w <= L.W && r.y >= 0 && r.y + r.h <= L.H; }
    var order = MAP.branches.slice().sort(function (a, b) { return geo[b.id].v - geo[a.id].v; });
    order.forEach(function (b) {
      var g = geo[b.id]; if (!(g.v > 0)) return;
      var ds = g.pts.filter(function (p) { return p.dum; });
      if (ds.length >= 2) {
        var x0 = ds[0].xL + 6, x1 = ds[ds.length - 1].xR - 4, txt = b.label, cnt = fc(g.v);
        var wtxt = tw(cnt + '  ', F_BOLD) + tw(txt, F_REG);
        var flat = ds.every(function (p) { return Math.abs(p.y - ds[0].y) < 0.5; });
        if (flat && wtxt < x1 - x0) {
          var inside = g.th >= 15;
          var y = inside ? ds[0].y + g.th / 2 + 4.5 : ds[0].y - 4;
          var r = { x: x0 - 2, y: y - 13, w: wtxt + 4, h: 16 };
          if (inside || free(r)) {
            lab += '<text class="tlab' + (inside ? ' in' : '') + '" x="' + x0 + '" y="' + y.toFixed(1) + '"><tspan class="cnt">' + cnt + '</tspan>  ' + esc(txt) + '</text>';
            if (!inside) obst.push(r);
            inline[b.id] = 1;
          }
        }
      }
    });
    order.forEach(function (b) {
      var g = geo[b.id]; if (!(g.v > 0) || inline[b.id]) return;
      var cnt = fc(g.v), w = tw(cnt, F_BOLD) + 6, p0 = g.pts[0], pl = g.pts[g.pts.length - 1], cy;
      var cands = [];
      cy = p0.y + g.th / 2; cands.push({ x: p0.xR + 3, y: cy - 8, ax: p0.xR, ay: cy });
      cy = pl.y + g.th / 2; cands.push({ x: pl.xL - 3 - w, y: cy - 8, ax: pl.xL, ay: cy });
      g.pts.forEach(function (p) { if (p.dum) { var c2 = p.y + g.th / 2; cands.push({ x: p.xR + 3, y: c2 - 8, ax: p.xR, ay: c2 }); } });
      var mid = g.pts[1] ? { x: (p0.xR + g.pts[1].xL) / 2 - w / 2, y: (p0.y + g.pts[1].y) / 2 + g.th / 2 - 8 } : null;
      if (mid) cands.push({ x: mid.x, y: mid.y, ax: mid.x + w / 2, ay: mid.y + 8 });

      for (var i = 0; i < cands.length && !pick; i++) { var r = { x: cands[i].x, y: cands[i].y, w: w, h: 16 }; if (free(r)) pick = cands[i]; }
      for (var j = 1; j <= 8 && !pick; j++) {
        [1, -1].forEach(function (sg) {
          if (pick) return;
          cands.slice(0, 2).forEach(function (c0) {
            if (pick) return;
            var r = { x: c0.x + (c0 === cands[0] ? 4 : -4), y: c0.y + sg * j * 15, w: w, h: 16 };
            if (free(r)) pick = { x: r.x, y: r.y, ax: c0.ax, ay: c0.ay, lead: true };
          });
        });
      }
      if (!pick) pick = { x: cands[0].x, y: cands[0].y, ax: cands[0].ax, ay: cands[0].ay };
      obst.push({ x: pick.x, y: pick.y, w: w, h: 16 });
      if (pick.lead) lab += '<line class="lead" x1="' + pick.ax.toFixed(1) + '" y1="' + pick.ay.toFixed(1) + '" x2="' + (pick.x + (pick.x > pick.ax ? 0 : w)).toFixed(1) + '" y2="' + (pick.y + 8).toFixed(1) + '"/>';

    });

      if (gb) { var pe = gb.pts[gb.pts.length - 1]; s += '<circle class="nowdot" cx="' + pe.xL + '" cy="' + (pe.y + Math.max(gb.th, 2) / 2) + '" r="6"/>'; }
    }
    return s + '</svg>';

  // ------------------------------------------------------------------ routes: every path as a tree, one picked as a thread
  var ROUTES = null, TRIE = null;
  function computeRoutes() {
    var w = ST.compare ? ST.routesWay : ST.way;
    ST.routesWay = w;
    var o = { stocks: stocksAt(w), sortBy: ST.sortBy };
    if (ST.scope === 'kind') o.classValues = classFor(w);
    ROUTES = E.routes(MAP, cur, w, o);
    TRIE = buildTrie(ROUTES.routes);
  }
  function buildTrie(routes) {
    var root = { kids: {}, order: [], chance: 0, carried: 0, key: '' };
    routes.forEach(function (r) {
      var node = root; node.chance += r.chance; node.carried += r.carried;
      r.branches.forEach(function (bid) {
        var bi = BR[bid].i, ch = node.kids[bi];
        if (!ch) { ch = node.kids[bi] = { bid: bid, kids: {}, order: [], chance: 0, carried: 0, key: (node.key ? node.key + '>' : '') + bi }; node.order.push(bi); }
        ch.chance += r.chance; ch.carried += r.carried; node = ch;
      });
      node.route = r;
    });
    return root;
  }
  function kidsOf(n) { var by = ST.sortBy; return n.order.map(function (k) { return n.kids[k]; }).sort(function (a, b) { return b[by] - a[by]; }); }
  function chainFrom(n) { var c = [n]; while (!n.route && n.order.length === 1) { n = n.kids[n.order[0]]; c.push(n); } return c; }
  function nodeAt(key) { var n = TRIE; key.split('>').forEach(function (bi) { n = n && n.kids[bi]; }); return n; }
  function currentRoute() {
    if (!ST.route || !TRIE) return null;
    var n = nodeAt(ST.route);
    if (!n) return null;
    if (n.route) return n.route;
    return { branches: ST.route.split('>').map(function (i) { return MAP.branches[+i].id; }), chance: n.chance, partial: true };
  }

    function row(n) {

      else r += ' <small>(' + kids.length + ' ways on)</small>';
      r += '</span><span class="tv">' + val + '</span></button>';
      if (!end.route && (open || inSel)) {
        var big = kids.filter(function (x) { return x.chance / total >= 0.002; }), small = kids.filter(function (x) { return x.chance / total < 0.002; });

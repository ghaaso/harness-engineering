  function fmt(v, kind) {
    if (typeof v === 'string') return v;

    if (kind === 'pct') { var x = v * 100; return (x !== 0 && Math.abs(x) < 0.1 ? x.toFixed(2) : x.toFixed(1)) + '%'; }

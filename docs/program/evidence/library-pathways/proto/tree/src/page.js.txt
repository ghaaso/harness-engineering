(function () {
  'use strict';
  var D = JSON.parse(document.getElementById('lp-data').textContent);
  var E = window.PathwayEngine, C = window.PathwayCore, T = window.TreeCore;
  var MAP = D.map, TABLE = D.params, LIMS = D.lims, CASES = C.unpackCases(D.cases), TX = D.tx, PTX = D.ptx;
  var $ = function (s, r) { return (r || document).querySelector(s); };
  var $$ = function (s, r) { return Array.prototype.slice.call((r || document).querySelectorAll(s)); };
  var esc = function (s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]; }); };

  var BEST = E.paramsFromTable(TABLE, 'best');
  var ST = T.structure(MAP);
  var S = ST.S;
  var BR = {}; MAP.branches.forEach(function (b) { BR[b.id] = b; });
  var STEP = {}; MAP.steps.forEach(function (s) { STEP[s.id] = s; });
  var TIER = {}; MAP.tiers.forEach(function (t) { TIER[t.id] = t; });
  var PMETA = {}; MAP.parameters.forEach(function (p) { PMETA[p.id] = p; });
  var DERIV = {}; MAP.derived.forEach(function (d) { DERIV[d.id] = d; });
  MAP.stocks.values.forEach(function (d) { DERIV[d.id] = d; });
  var POLS = MAP.policies.map(function (p) { return p.id; });
  var PNAME = PTX.policy_names, PSHORT = PTX.policy_short;
  var LIM = {}; LIMS.items.forEach(function (it) { LIM[it.id] = it; });
  var CASE = {}; CASES.cases.forEach(function (c) { CASE[c.id] = c; });
  var FAM = {}; CASES.families.forEach(function (f) { FAM[f.id] = f; });

  Object.keys(PTX.design_steps).forEach(function (k) { PTX.design_steps[k].forEach(function (st) { (DSTEPS_OF[st] = DSTEPS_OF[st] || []).push(+k); }); });
  var STAGES = MAP.steps.map(function (s) { return s.id; });
  var STAGE_NO = {}; STAGES.forEach(function (s, i) { STAGE_NO[s] = s === 'outcome' ? 'end' : String(i + 1); });
  var PATH = E.timePath(MAP, BEST);
  var NPER = PATH.periods.length;
  var OUTS = MAP.outcomes.map(function (o) { return o.id; });
  var GATE_CAUGHT = (MAP.links && MAP.links[0]) ? MAP.links[0].to_branches.filter(function (b) { return /_caught$/.test(b); }) : [];

  // ------------------------------------------------------------ state
  var V = {
    policy: 'library', mode: 'all',
    kind: { kind: 'visible', need: 'cheap_wrong', held: 'similar', distractor: 'no' },
    period: NPER, compare: false,
    forkMode: 'auto', open: {}, why: {}, ev: {}, inc: {},
    route: null, stepIdx: null, sortBy: 'chance', shown: 20
  };
  var PIC = {}, STT = null, ROUTES = null;

  function stocksFor(pol) { return pol === 'library' ? PATH.periods[V.period - 1].stocks : {}; }
  function kindFor() { return V.mode === 'kind' ? Object.assign({}, V.kind) : null; }
  function recompute() {
    POLS.forEach(function (p) { PIC[p] = T.picture(E, C, MAP, BEST, p, kindFor(), stocksFor(p)); });
    STT = T.stageTotals(MAP, ST, PIC[V.policy], V.policy);
    var ro = { stocks: stocksFor(V.policy), sortBy: V.sortBy };
    if (V.mode === 'kind') { var pol = MAP.policies.filter(function (p) { return p.id === V.policy; })[0]; ro.classValues = Object.assign({}, V.kind, pol.fixed || {}); }
    ROUTES = E.routes(MAP, BEST, V.policy, ro);
    if (V.route && !ROUTES.routes.some(function (r) { return r.id === V.route; })) { V.route = null; V.stepIdx = null; }
    var bad = POLS.filter(function (p) { return PIC[p].parity > 1e-9; });
    if (bad.length) console.error('Cost parity failed for ' + bad.join(', '));
  }

  // ------------------------------------------------------------ numbers
  function n100(v) {
    var x = 100 * v;
    if (!(x > 0)) return '0';
    if (x >= 1) return x.toFixed(1);
    if (x >= 0.01) return x.toFixed(2);
    return '<0.01';
  }
  function pct(v) { if (v === null || v === undefined || !isFinite(v)) return '–'; var x = 100 * v; return (x !== 0 && x < 0.1 ? x.toFixed(2) : x.toFixed(1)) + '%'; }
  function cst(v) { if (v === null || v === undefined || !isFinite(v)) return '–'; var a = Math.abs(v); return a >= 100 ? Math.round(v).toLocaleString('en') : a >= 10 ? v.toFixed(1) : a >= 0.01 || v === 0 ? v.toFixed(2) : '<0.01'; }
  function tag(t) { return '<i class="tag ' + t + '" title="' + esc(t + ': ' + PTX.tags[t]) + '">' + t + '</i>'; }
  function live(pol, sid) { return PIC[pol].stateFlow[sid] > 1e-12; }
  function liveAny(sid) { return V.compare ? POLS.some(function (p) { return live(p, sid); }) : live(V.policy, sid); }
  function bLive(pol, bid) { return PIC[pol].branchFlow[bid] > 1e-12; }
  function isOpen(sid) {
    if (V.open[sid] !== undefined) return V.open[sid];
    if (V.forkMode === 'all') return true;
    if (V.forkMode === 'none') return false;
    return liveAny(sid);
  }
  function stepLabel(st) { return st === 'outcome' ? 'Outcome' : STEP[st].label; }
  function q(sid) { return TX.question[sid] || S[sid].label; }
  function swatchTier(t) { return '<i class="sw t-' + t + '" title="Paid to ' + esc(TIER[t].label) + '"></i>'; }
  function swatchOut(o) { return '<i class="sw o-' + o + '" role="img" aria-label="' + esc(S[o].label) + '"></i>'; }

  // ------------------------------------------------------------ why a part carries no work
  function whyText(ref, pol) {
    var list = T.zeroWhy(E, MAP, ST, BEST, pol, kindFor(), stocksFor(pol), PIC[pol], ref);
    var parts = [], factors = [];
    list.forEach(function (r) {
      if (r.type === 'start') parts.push(TX.why.start.replace('{way}', PNAME[r.policy]).replace('{start}', q(r.start)));
      else if (r.type === 'upstream') parts.push(TX.why.upstream.replace('{state}', q(r.state)));
      else if (r.type === 'factor') factors.push(r);
      else parts.push(TX.why.rule);
    });
    if (factors.length) {
      var words = factors.map(function (f) {
        var k = f.id + '=' + f.value;
        if (TX.why.words[k]) return TX.why.words[k];
        var lab = (PMETA[f.id] && PMETA[f.id].name) || (DERIV[f.id] && DERIV[f.id].label) || f.id;
        return lab + ' is ' + f.value;
      });
      var uniq = words.filter(function (w, i) { return words.indexOf(w) === i; });
      var t = (factors[0].all ? TX.why.factor_all : TX.why.factor_kind).replace('{words}', uniq.join('; and '));
      if (factors[0].all && pol === 'library') {
        var stk = stocksFor(pol);
        t += ' At period ' + V.period + ' the library holds ' + cst(stk.E_good) + ' good and ' + cst(stk.E_bad) + ' bad entries.';
      }
      parts.push(t);
    }
    return parts.join(' ');
  }

  // ------------------------------------------------------------ controls
  function seg(name, items, cur, dis) {
    return '<div class="seg" role="group" aria-label="' + esc(name) + '">' + items.map(function (it) {
      return '<button type="button" data-v="' + esc(it[0]) + '" aria-pressed="' + (it[0] === cur) + '"' + (dis ? ' disabled' : '') + (it[2] ? ' title="' + esc(it[2]) + '"' : '') + '>' + esc(it[1]) + '</button>';
    }).join('') + '</div>';
  }
  function ctl(key, inner, extra) {
    var c = TX.controls[key];
    return '<div class="ctl" data-ctl="' + key + '"' + (extra || '') + '><span class="lab">' + esc(c.label) + '</span>' + inner + '<span class="says">' + esc(c.says) + '</span></div>';
  }

    var h = ctl('policy', seg('Way of working', POLS.map(function (p) { return [p, PNAME[p], PTX.policy_says[p]]; }), V.policy));

      kh += ctl(aid, seg(TX.controls[aid].label, a.values.map(function (v) { return [v.id, TX.short_val[v.id] || v.label, v.label]; }), V.kind[aid], dis), ' data-attr="' + aid + '"');
    });
    var share = V.mode === 'kind' ? PIC[V.policy].share : 1;
    kh += '<div class="share">This kind is <b>' + pct(share) + '</b> of all work ' + tag('G') + (V.policy === 'library' ? ' at period ' + V.period : '') + '.</div>';
    h += '<div class="ctl kindset" id="kindset"' + (V.mode === 'kind' ? '' : ' hidden') + '>' + kh + '</div>';
    $('#controls').innerHTML = h;

    var b = e.target.closest('button[data-v]'); if (!b || b.disabled) return;

  });

    if (e.target.id === 'perPick') { V.period = +e.target.value; e.target.nextElementSibling.textContent = V.period + ' of ' + NPER; update(true); }
  });

        (ev.steps.length ? '<span class="ev">step' + (ev.steps.length > 1 ? 's ' : ' ') + ev.steps.join(', ') + ' · ⚑ ' + ev.lims + ' · ' + ev.cases + ' cases broke</span>' : '') +
        '</button></li>';

    h += '</ol><p class="cap" id="stripCap"></p>';

      '<dt><span style="display:flex;gap:2px">' + ['cheap', 'opus', 'fable'].map(swatchTier).join('') + '</span></dt><dd>' + esc(L.tiers) + ' Cheap model; Opus; Fable.</dd>' +
      '<dt><span class="stgchip" style="font-size:13px;border:1px solid var(--line-2);border-radius:999px;padding:0 6px">⚑</span></dt><dd>' + esc(L.lims) + ' ' + esc(L.cases) + '</dd>' +

  $('#strip').addEventListener('click', function (e) {
    var b = e.target.closest('.stg-btn'); if (!b) return;
    var stg = b.dataset.stage;
    if (stg === 'outcome') { $('#ends').scrollIntoView({ behavior: 'smooth', block: 'start' }); return; }
    var first = MAP.states.filter(function (s) { return s.step === stg; }).map(function (s) { return s.id; });
    // the first of this stage's questions in reading order
    var target = null;
    ST.roots.some(function (r) { return order(r).some(function (sid) { if (first.indexOf(sid) >= 0) { target = sid; return true; } return false; }); });
    if (target) goTo(target);
  });
  function order(root) {
    var list = [];
    (function walk(sid) { list.push(sid); ST.out[sid].forEach(function (b) { if (!ST.shared[b.to] && ST.owner[b.to] === b.id) walk(b.to); }); })(root);
    return list;
  }
  var seen = new Set(), io = null;
  function markInView() {
    var stages = {};
    seen.forEach(function (el) { if (el.isConnected) stages[el.dataset.stage] = 1; });
    $$('.stg-btn').forEach(function (b) { b.classList.toggle('inview', !!stages[b.dataset.stage]); });
    var cap = $('#stripCap');
    if (cap) {
      var k = STAGES.filter(function (s) { return stages[s]; });
      cap.textContent = k.length ? 'In view: ' + k.map(function (s) { return STAGE_NO[s] + ' ' + stepLabel(s); }).join(' · ') : '';
    }
  }
  function observe() {
    if (!('IntersectionObserver' in window)) return;
    if (io) io.disconnect();
    seen = new Set();
    io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) { if (en.isIntersecting) seen.add(en.target); else seen.delete(en.target); });
      markInView();
    }, { rootMargin: '-90px 0px -15% 0px' });
    $$('.fh[data-stage], .ends[data-stage]').forEach(function (el) { io.observe(el); });
  }

  // ------------------------------------------------------------ the tree
  function countCells(kindOf, id) {
    // kindOf: 'b' branch or 's' state. One cell per way shown.
    var pols = V.compare ? POLS : [V.policy];
    return pols.map(function (p) {
      var v = kindOf === 'b' ? PIC[p].branchFlow[id] : PIC[p].stateFlow[id];
      var z = !(v > 1e-12);
      return '<span class="ct' + (z ? ' zero' : '') + (V.compare && p === V.policy ? ' me' : '') + '" data-w="' + esc(PSHORT[p]) + '"' + (V.compare ? ' title="' + esc(PNAME[p]) + '"' : '') + '><span class="trk"><i style="width:' + Math.min(100, 100 * v).toFixed(2) + '%' + (v > 0 && 100 * v < 1 ? ';min-width:2px' : '') + '"></i></span><b>' + n100(v) + '</b></span>';
    }).join('');
  }
  function costCell(bid) {
    var b = BR[bid], P = PIC[V.policy];
    if (!b.tier) return '<span class="co"><span class="each">no cost</span></span>';
    var flow = P.branchFlow[bid], carried = P.carried[bid];
    var each = P.bv ? P.bv[bid].cost : (flow > 1e-12 ? carried / flow : null);
    return '<span class="co" title="Cost when taken, and the cost it carries for the 100 pieces. Paid to ' + esc(TIER[b.tier].label) + '.">' + swatchTier(b.tier) + '<span class="each">' + cst(each) + ' each ·</span> <b>' + cst(100 * carried) + '</b></span>';
  }
  function destCell(b) {
    var to = b.to;
    if (S[to].outcome) return '<button type="button" class="dest end" data-go="end:' + to + '">' + swatchOut(to) + '<span>Ends: ' + esc(S[to].label) + '</span></button>';
    if (ST.shared[to]) return '<button type="button" class="dest" data-go="' + to + '"><span class="arrow">→</span><span>' + esc(q(to)) + ' <span class="st">stage ' + STAGE_NO[S[to].step] + '</span></span></button>';
    if (ST.owner[to] === b.id) return '';
    return '<button type="button" class="dest same" data-go="' + to + '"><span class="arrow">↑</span><span>Same next question as “' + esc(BR[ST.owner[to]].label) + '”</span></button>';
  }
  function chanceAt(bid) { return PIC[V.policy].chanceAt[bid]; }
  function routeSet() {
    if (!V.route) return null;
    var r = ROUTES.routes.filter(function (x) { return x.id === V.route; })[0];
    return r ? r.branches : null;
  }
  function answer(b, depth, onR, cur) {
    var z = !(V.compare ? POLS.some(function (p) { return bLive(p, b.id); }) : bLive(V.policy, b.id));
    var on = onR && onR.indexOf(b.id) >= 0, isCur = cur === b.id;
    var promo = V.policy === 'library' && GATE_CAUGHT.indexOf(b.id) >= 0 ? '<button type="button" class="promo linkbtn" data-gostock="1" style="text-decoration:none;color:var(--ink-2)" title="' + esc(TX.legend.promo) + '">+ promoted checks</button>' : '';
    var h = '<li class="a' + (z ? ' quiet' : '') + (on ? ' onroute' : '') + (isCur ? ' current' : '') + '" id="a-' + b.id + '">';
    h += '<div class="r" data-b="' + b.id + '"><span class="lbl"><button type="button" class="linkbtn lb" data-route-through="' + b.id + '" style="color:inherit;text-decoration:none" title="Light the most common route through this answer">' + esc(b.label) + '</button>' + promo +
      (z ? '<button type="button" class="linkbtn why" data-why="b:' + b.id + '" aria-expanded="' + !!V.why['b:' + b.id] + '">Why 0?</button>' : '') + (function () { var d = destCell(b); return d ? '<span class="de">' + d + '</span>' : ''; })() + '</span>' +
      '<span class="ch num" title="Chance at this question">' + pct(chanceAt(b.id)) + '</span>' +
      '<span class="ctwrap">' + countCells('b', b.id) + '</span>' + costCell(b.id) + '</div>';
    if (z && V.why['b:' + b.id]) h += '<p class="whyline">' + esc(whyText({ branch: b.id }, V.policy)) + '</p>';
    if (!ST.shared[b.to] && ST.owner[b.to] === b.id) h += fork(b.to, depth + 1, onR, cur);
    return h + '</li>';
  }
  function stageChip(sid) {
    var stg = S[sid].step, ev = stageEvidence(stg), open = !!V.ev[sid];
    var t = '<span>Stage ' + STAGE_NO[stg] + ' · ' + esc(stepLabel(stg)) + '</span>';
    if (!ev.steps.length) return t;
    return t + '<button type="button" class="evchip" data-ev="' + sid + '" aria-expanded="' + open + '" title="Open the limitations and real cases of design step' + (ev.steps.length > 1 ? 's ' : ' ') + ev.steps.join(' and ') + '">⚑ ' + ev.lims + ' limitations · ' + ev.cases + ' cases broke · design step' + (ev.steps.length > 1 ? 's ' : ' ') + ev.steps.join(', ') + '</button>';
  }
  function fork(sid, depth, onR, cur) {
    var z = !liveAny(sid), open = isOpen(sid), stg = S[sid].step;
    var starts = (ST.starts[sid] || []);
    var h = '<section class="fork' + (z ? ' quiet' : '') + '" id="f-' + sid + '" data-sid="' + sid + '">';
    h += '<div class="fh r" data-stage="' + stg + '"><div class="qcell"><button type="button" class="tog" data-tog="' + sid + '" aria-expanded="' + open + '" aria-label="' + (open ? 'Close' : 'Open') + ' this question">' + (open ? '▾' : '▸') + '</button><div><div class="q">' + esc(q(sid)) + '</div><div class="meta">' + stageChip(sid) +
      starts.map(function (p) { return '<span class="start">' + esc(PNAME[p]) + ' starts here</span>'; }).join('') +
      (ST.shared[sid] && ST.inc[sid].length ? '<button type="button" class="linkbtn" data-inc="' + sid + '" aria-expanded="' + !!V.inc[sid] + '" style="font-size:13.5px">arrives from ' + ST.inc[sid].length + ' answers' + (starts.length ? ' and a start' : '') + '</button>' : '') +
      (sid === 'L6_wrong' && V.policy === 'library' && PIC.library.measures ? '<span>promoted checks catch <b>' + n100(PIC.library.measures.use_promo_caught_first / 100) + '</b> of 100 here</span>' : '') +
      (z ? '<span class="zr"><b>No work here.</b> ' + esc(whyText({ state: sid }, V.policy)) + '</span>' : '') +
      '</div></div></div>';
    h += '<div class="arrive">' + (V.compare ? '<span class="tri">' + POLS.map(function (p) { return '<span' + (p === V.policy ? ' class="me"' : '') + ' data-w="' + esc(PSHORT[p]) + '" title="' + esc(PNAME[p]) + '"><b>' + n100(PIC[p].stateFlow[sid]) + '</b></span>'; }).join('') + '</span> <span>arrive</span>' : '<b>' + n100(PIC[V.policy].stateFlow[sid]) + '</b> of 100 arrive') + '</div></div>';
    if (V.inc[sid]) h += incList(sid);
    if (V.ev[sid]) h += evPanel(sid);
    if (open) {
      h += '<ol class="ans">' + ST.out[sid].map(function (b) { return answer(b, depth, onR, cur); }).join('') + '</ol>';
    } else if (!z) {
      h += '<p class="reason">' + ST.out[sid].length + ' answer' + (ST.out[sid].length > 1 ? 's' : '') + ' hidden. <button type="button" class="linkbtn" data-tog="' + sid + '">Open</button></p>';
    }
    return h + '</section>';
  }
  function incList(sid) {
    var P = PIC[V.policy];
    var items = ST.inc[sid].map(function (b) {

      if (best) lightRoute(best.id, best.branches.indexOf(b2));
    }
  });

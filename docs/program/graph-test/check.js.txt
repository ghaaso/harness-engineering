const vals = [];
(function walk(o) {
  if (o == null) return;
  if (typeof o === 'number') { vals.push(o); return; }
  if (typeof o === 'string') { (o.replace(/,(?=\d{3})/g, '').match(/\d+(?:\.\d+)?/g) || []).forEach(x => vals.push(+x)); return; }
  if (Array.isArray(o)) return o.forEach(walk);
  if (typeof o === 'object') Object.keys(o).forEach(k => { walk(o[k]); (k.match(/\d+(?:\.\d+)?/g) || []).forEach(x => vals.push(+x)); });
})(d);
// derived, documented on the page: 1.96 (standard errors for a 95 % interval), 95 (per cent), counts of shown items
const ALLOW = new Set([1.96, 95]);
const absVals = vals.map(Math.abs);
function inData(tok, isPct) {
  const x = Math.abs(+tok.replace(/,/g, ''));
  if (ALLOW.has(x)) return true;
  const dec = (tok.split('.')[1] || '').length;
  const m = Math.pow(10, dec);
  const eq = v => Math.abs(Math.round(v * m + 1e-7) / m - x) < 1e-9 || Math.abs(Math.round(v * m - 1e-7) / m - x) < 1e-9;
  if (absVals.some(eq)) return true;
  if (isPct && absVals.some(v => Math.abs(Math.round(v * 100) - x) < 1e-9)) return true;
  if (dec === 0 && absVals.some(v => v === x)) return true;
  return false;
}

const packRows = o => {
  if (Array.isArray(o)) {
    const objs = o.length >= 2 && o.every(x => x && typeof x === 'object' && !Array.isArray(x));
    if (objs) { const k0 = Object.keys(o[0]), sig = k0.slice().sort().join('\u0001');

    return o.map(packRows);
  }
  if (o && typeof o === 'object') return Object.fromEntries(Object.entries(o).map(([k, v]) => [k, packRows(v)]));

};

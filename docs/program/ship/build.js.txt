// Builds the owner-facing ship-program page from program.json.
// Run:  cd docs/program/ship && node - < build.js.txt
// Writes index.html (self-contained) and OWNER-VIEW.generated.md. No dependencies.
'use strict';
const fs = require('fs');
const path = require('path');
const dir = process.cwd();
const P = JSON.parse(fs.readFileSync(path.join(dir, 'program.json'), 'utf8'));
const asOf = P.stateAsOf;

const esc = (s) => String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
const byId = {};
P.phases.forEach((p) => (byId[p.id] = p));
const ST = P.states;
const stateOrder = ['done', 'running', 'next', 'you', 'held', 'todo'];
const measured = (what) => `<span class="tag m">measured ${esc(what || asOf)}</span>`;
const illus = () => `<span class="tag i">illustrative</span>`;

// sanity: every reference resolves
const bad = [];
P.phases.forEach((p) => (p.needs || []).concat(p.stackOn ? [p.stackOn] : []).forEach((n) => { if (!byId[n]) bad.push(p.id + ' -> ' + n); }));
P.goals.forEach((g) => g.phases.forEach((n) => { if (!byId[n]) bad.push('goal -> ' + n); }));
P.ownerChain.forEach((n) => { if (!byId[n]) bad.push('owner chain -> ' + n); });
if (bad.length) { console.error('Unknown phase ids:', bad.join(', ')); process.exit(1); }

// dependants: everything that cannot finish without X
const rev = {};
P.phases.forEach((p) => (p.needs || []).forEach((n) => (rev[n] = rev[n] || []).push(p.id)));
function downstream(id) {
  const seen = new Set(); const st = [id];
  while (st.length) { const x = st.pop(); (rev[x] || []).forEach((y) => { if (!seen.has(y)) { seen.add(y); st.push(y); } }); }
  return [...seen];
}
const p0aDown = downstream('P0a');
const total = P.phases.length;
const countBy = (ids) => { const c = {}; ids.forEach((i) => { const s = byId[i].state; c[s] = (c[s] || 0) + 1; }); return c; };
const allCounts = countBy(P.phases.map((p) => p.id));

// ---------- pieces ----------
const SHORT = { done: 'done', running: 'running now', next: 'starting next', you: 'waiting on you', held: 'held', todo: 'not started' };
function pill(state) { return `<span class="pill s-${state}">${esc(SHORT[state])}</span>`; }

function box(p, opts = {}) {
  const lines = [];
  lines.push(`<span class="bx-name">${esc(p.name)}</span>`);
  lines.push(`<span class="bx-meta"><b>${esc(p.id)}</b> ${pill(p.state)}</span>`);
  if (p.stackOn && !opts.chainedFromPrev) lines.push(`<span class="bx-stack">builds on ${esc(p.stackOn)}'s reviewed branch</span>`);
  if (p.hold) lines.push(`<span class="bx-hold">Merge held until: ${esc(p.hold)}</span>`);
  return `<button type="button" class="bx s-${p.state}" data-id="${esc(p.id)}" aria-pressed="false">${lines.join('')}</button>`;
}

function chip(id) { const p = byId[id]; return `<button type="button" class="chip s-${p.state}" data-id="${esc(id)}" title="${esc(p.name)}: ${esc(ST[p.state])}">${esc(id)}</button>`; }

function stackBar(counts, totalN) {
  return `<div class="sbar">` + stateOrder.filter((s) => counts[s]).map((s) =>
    `<span class="seg s-${s}" style="flex:${counts[s]} 1 0" title="${counts[s]} ${esc(SHORT[s])}"><b>${counts[s]}</b>${counts[s] / totalN >= 0.25 ? ' ' + esc(SHORT[s]) : ''}</span>`).join('') + `</div>` +
    `<p class="g-count">` + stateOrder.filter((s) => counts[s]).map((s) => `<b>${counts[s]}</b> ${esc(SHORT[s])}`).join(' · ') + ` (of ${totalN} phase${totalN === 1 ? '' : 's'})</p>`;
}

// ---------- section: header ----------
const runningIds = P.phases.filter((p) => p.state === 'running').map((p) => p.id);
const nextIds = P.phases.filter((p) => p.state === 'next').map((p) => p.id);
const doneIds = P.phases.filter((p) => p.state === 'done').map((p) => p.id);
const listIds = (a) => (a.length ? a.join(', ') : 'none');
const plural = (n, one, many) => n + ' ' + (n === 1 ? one : many);
const header = `
<header class="top">
  <h1>FixFlow ship program</h1>
  <p class="sub">The whole program on one page: what is running, what waits on what, and what only you can do. State as of <b>${esc(asOf)}</b>. The plan decides; this page explains. Built from <code>program.json</code>.</p>
  <div class="now">
    <div class="card bad"><span class="big">Production is dead</span><span>${esc(P.headline.production)}</span>${measured()}</div>
    <div class="card you"><span class="big">Your pre-flight (P0a) has not started</span><span>${esc(P.headline.critical)} <b>${p0aDown.length} of ${total}</b> phases wait behind it.</span>${measured()}</div>
    <div class="card run"><span class="big">${plural(runningIds.length, 'lane', 'lanes')} running now, ${nextIds.length} starting next</span><span>Running, each in its own session and its own copy of the repository: ${esc(listIds(runningIds))}. Starting next: ${esc(listIds(nextIds))}. Done: ${esc(listIds(doneIds))}. ${esc(P.headline.baseline)}</span>${measured()}</div>
  </div>
  <nav class="toc">
    <a href="#where">Where are we?</a><a href="#lanes">What is running in parallel?</a><a href="#waits">What waits on what?</a><a href="#you">What only you can do next</a><a href="#decide">Decisions waiting on you</a><a href="#machine">How the computer is used</a><a href="#rules">What never bends</a>
  </nav>
  <div class="legend" aria-label="Colour key">${stateOrder.map((s) => `<span class="pill s-${s}">${esc(ST[s])}</span>`).join('')}</div>
</header>`;

function secHead(id, q, how, now) {
  return `<h2 id="${id}">${esc(q)}</h2><p class="how"><b>How to read this:</b> ${how}</p><p class="shows"><b>What it shows right now:</b> ${now}</p>`;
}

// ---------- section 1: goals ----------
const goalsHtml = secHead('where', 'Where are we?',
  'one row per goal of the plan. The bar shows how many of that goal\'s phases are in each state (the number is written on the bar); below it every phase is a coloured mark. Click a mark to see what waits on it.',
  `${allCounts.done || 0} phase done, ${allCounts.running || 0} running (${esc(listIds(runningIds))}), ${allCounts.next || 0} starting next, ${allCounts.you || 0} waiting on you, ${allCounts.held || 0} held, ${allCounts.todo || 0} not started, out of ${total}. No goal is reached. State as of ${esc(asOf)} ${measured()}`) +
  `<div class="goals">` + P.goals.map((g, i) => {
    const c = countBy(g.phases);
    return `<div class="goal"><div class="g-head"><span class="g-num">Goal ${i + 1}</span><span class="g-name">${esc(g.name)}</span><span class="g-at">Reached at: ${esc(g.reached)}</span></div>
    ${stackBar(c, g.phases.length)}<div class="chips">${g.phases.map(chip).join('')}</div></div>`;
  }).join('') + `</div><p class="note">The plan's heading says "five goals"; its table has six rows (latency is its own row). All six are shown.</p>`;

// ---------- section 2: lanes ----------
const waveIds = P.waves.map((w) => w.id);
const laneRows = P.lanes.filter((l) => l.id !== 'OWN').map((l) => {
  const cells = waveIds.map((w) => {
    const ps = P.phases.filter((p) => p.lane === l.id && p.wave === w);
    let html = '';
    ps.forEach((p, i) => {
      const chained = i > 0 && p.stackOn === ps[i - 1].id;
      if (chained) html += `<span class="arrow-down" aria-hidden="true">&#8595; then, on its branch</span>`;
      html += box(p, { chainedFromPrev: chained });
    });
    const wv = P.waves.find((x) => x.id === w);
    return `<div class="cell${ps.length ? '' : ' empty'}"><span class="cell-wave">${esc(wv.name)}</span>${html}</div>`;
  }).join('');
  return `<div class="lane"><div class="lane-name"><b>${esc(l.name)}</b><span>lane ${esc(l.id)}</span><span class="owns">${esc(l.owns)}</span></div>${cells}</div>`;
}).join('');
const own = P.lanes.find((l) => l.id === 'OWN');
const ownerRow = `<div class="owner-row"><div class="lane-name"><b>${esc(own.name)}</b><span>lane OWN: the critical path</span><span class="owns">${esc(own.owns)}. Runs beside all four waves; no wave shortens it.</span></div>
  <div class="ochain">${P.ownerChain.map((id, i) => (i ? `<span class="arrow-r" aria-hidden="true">&#8594;</span>` : '') + box(byId[id])).join('')}</div></div>`;
const waveHead = `<div class="lane whead"><div class="lane-name"><b>Lane</b></div>${P.waves.map((w) => `<div class="cell"><b>${esc(w.name)}</b><span>starts ${esc(w.starts)}</span></div>`).join('')}</div>`;
const lanesHtml = secHead('lanes', 'What is running in parallel?',
  'each row is a lane: one kind of code, worked by its own session in its own copy of the repository. Columns are the waves A to D in the order the plan starts them, then "Later" for phases the plan puts in no wave. A box is a phase: plain name, plan id, state. "&#8595; then" joins phases built one on top of the other. A purple box may be built but must stay unmerged until the line in it says. Click any box.',
  `${plural(runningIds.length, 'phase runs', 'phases run')} now (${esc(listIds(runningIds))}); done: ${esc(listIds(doneIds))}; starting next: ${esc(listIds(nextIds))}. Your own chain (bottom) has not started, and everything that needs a server or a phone waits on it. ${measured()}`) +
  `<div class="board">${waveHead}${laneRows}${ownerRow}</div>
  <div class="info" id="info" aria-live="polite"><span id="info-text">Click any phase to see what it needs and what waits on it.</span> <button type="button" class="btn" id="show-p0a">Show what waits behind P0a</button> <button type="button" class="btn ghost" id="clear-sel">Clear</button></div>`;

p0aDown.forEach((id) => { const l = byId[id].lane; (p0aByLane[l] = p0aByLane[l] || []).push(id); });

// ---------- section 5: decisions ----------
function decCard(d, key, isNew) {
  const opts = isNew ? d.options : ['Accept the default', 'I decide otherwise (write it below)'];
  return `<div class="dec${isNew ? ' new' : ''}" data-key="${key}">
    ${isNew ? '<span class="tag new">new, decide now</span>' : ''}
    <h3>${esc(d.decision)}</h3>
    <p><b>Blocks:</b> ${esc(d.blocks)}</p>
    <p><b>If you don't decide:</b> ${esc(d.default)}</p>
    <fieldset><legend>Your choice</legend>${opts.map((o, i) => `<label><input type="radio" name="d-${key}" value="${esc(o)}" data-dec="${key}"> ${esc(o)}</label>`).join('')}
    <label class="res">Note <input type="text" data-decnote="${key}" maxlength="200" autocomplete="off" spellcheck="false" placeholder="optional, no secrets"></label>
    <span class="warn" data-warn="d-${key}" hidden>That looks like a secret. It was not saved.</span></fieldset>
  </div>`;
}
const decHtml = secHead('decide', 'Decisions waiting on you',
